# ── تعريفات الترجمة ───────────────────────────────────────────────────────────
add_compile_definitions(
    SAD_WASM_BUILD=1        # تعريف خاص للـ WASM
    HAS_GRAPHICS=0          # disable gettext: no graphics
    SAD_NO_FILESYSTEM=1     # no filesystem access
    SAD_NO_NETWORK=1        # no network
    SAD_NO_SDL2=1           # no SDL2
    UNICODE=1
)

# ── الهدف ─────────────────────────────────────────────────────────────────────
add_executable(sad_wasm ${WASM_SOURCES})

set_target_properties(sad_wasm PROPERTIES
    OUTPUT_NAME "sad"
    SUFFIX ".js"
)

# ── خيارات Emscripten ─────────────────────────────────────────────────────────
if(EMSCRIPTEN)
    target_link_options(sad_wasm PRIVATE
        -O2
        -sWASM=1
        -sALLOW_MEMORY_GROWTH=1
        -sMAXIMUM_MEMORY=256MB
        -sMODULARIZE=1
        -sEXPORT_NAME=SadWasm
        "-sEXPORTED_FUNCTIONS=[\"_sad_execute\",\"_sad_version\",\"_malloc\",\"_free\"]"
        "-sEXPORTED_RUNTIME_METHODS=[\"cwrap\",\"ccall\",\"UTF8ToString\",\"stringToUTF8\",\"lengthBytesUTF8\"]"
        -sNODERAWFS=0
        --no-entry
    )
endif()

# نسخ للموقع
set(WASM_DEST "${ROOT}/website/docs/public")
add_custom_command(TARGET sad_wasm POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E make_directory "${WASM_DEST}"
    COMMAND ${CMAKE_COMMAND} -E copy "$<TARGET_FILE_DIR:sad_wasm>/sad.js"   "${WASM_DEST}/sad.js"
    COMMAND ${CMAKE_COMMAND} -E copy "$<TARGET_FILE_DIR:sad_wasm>/sad.wasm" "${WASM_DEST}/sad.wasm"
    COMMENT "نسخ sad.js + sad.wasm إلى website/docs/public/"
    VERBATIM
)
