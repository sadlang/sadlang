# Advanced Examples - Android Bridges

## 1. تطبيق دردشة متكامل

```sad
# ملف: chat_example.ص

استورد "android/communication/sms"
استورد "android/communication/contacts"
استورد "android/communication/notifications"
استورد "android/demo/chat"

# === تهيئة التطبيق ===

دالة هيئ_التطبيق(السياق)
    SadSMSBridge.هيئ(السياق)
    SadContactsBridge.هيئ(السياق)
    SadNotificationBridge.هيئ(السياق)
    تطبيق_الدردشة.هيئ(السياق)
    
    # الاستماع للرسائل الجديدة
    SadSMSBridge.عند_استلام(معالج_الرسالة_الجديدة)
نهاية

# === معالجة الرسائل ===

دالة معالج_الرسالة_الجديدة(رسالة)
    # البحث عن جهة الاتصال
    متغير جهة = SadContactsBridge.ابحث_بالرقم(رسالة.المرسل)
    متغير الاسم = جهة?.الاسم أو رسالة.المرسل
    
    # إضافة للمحادثة
    متغير محادثة = تطبيق_الدردشة.احصل_أو_أنشئ_محادثة(رسالة.المرسل)
    تطبيق_الدردشة.أضف_رسالة(محادثة.المعرف، رسالة.المحتوى)
    
    # إرسال إشعار
    SadNotificationBridge.أرسل(
        العنوان: الاسم،
        المحتوى: رسالة.المحتوى،
        القناة: "messages"،
        الأهمية: عالية،
        الإجراءات: [
            إجراء("رد"، "reply"، نوع: DIRECT_REPLY)،
            إجراء("تعليم كمقروء"، "mark_read")
        ]
    )
نهاية

# === إرسال رسالة ===

دالة أرسل_رسالة(معرف_المحادثة، النص)
    متغير محادثة = تطبيق_الدردشة.احصل_على_محادثة(معرف_المحادثة)
    إذا (محادثة == لاشيء)
        ارجع خطأ
    نهاية
    
    # إرسال SMS
    متغير الرقم = محادثة.المشاركون[0].الهاتف
    متغير نتيجة = SadSMSBridge.أرسل(الرقم، النص)
    
    إذا (نتيجة.نجح)
        # تحديث المحادثة
        تطبيق_الدردشة.أرسل_رسالة(معرف_المحادثة، النص)
        ارجع صحيح
    وإلا
        اطبع_سطر("فشل الإرسال: " + نتيجة.الخطأ)
        ارجع خطأ
    نهاية
نهاية
```

## 2. ويدجت إنتاجية متقدم

```sad
# ملف: productivity_widget.ص

استورد "android/widgets/core"
استورد "android/widgets/glance"
استورد "android/widgets/data"
استورد "android/demo/productivity"

# === تكوين الويدجت ===

ثابت تكوين_الويدجت = تكوين_ويدجت(
    الاسم: "productivity_widget"،
    الأحجام: [SMALL، MEDIUM، LARGE]،
    التحديث_دقائق: 15،
    يدعم_التكوين: صحيح
)

# === إنشاء الويدجت ===

دالة أنشئ_الويدجت()
    SadGlanceBridge.أنشئ(
        التكوين: تكوين_الويدجت،
        المحتوى: لامدا(الحجم)
            طابق الحجم
                عندما SMALL
                    ارجع محتوى_صغير()
                عندما MEDIUM
                    ارجع محتوى_متوسط()
                عندما LARGE
                    ارجع محتوى_كبير()
            نهاية
        نهاية
    )
نهاية

# === المحتوى حسب الحجم ===

دالة محتوى_صغير()
    متغير إحصائيات = تطبيق_الإنتاجية.إحصائيات_اليوم()
    
    ارجع عمود(المحاذاة: CENTER)
        نص(
            إحصائيات.المهام_المكتملة + "/" + إحصائيات.إجمالي_المهام،
            الحجم: 32،
            الوزن: BOLD
        )
        نص("مهام مكتملة"، الحجم: 12)
    نهاية_عمود
نهاية

دالة محتوى_متوسط()
    متغير المهام = تطبيق_الإنتاجية.مهام_اليوم()
    متغير إحصائيات = تطبيق_الإنتاجية.إحصائيات_اليوم()
    
    ارجع عمود()
        # الرأس
        صف(التوزيع: SPACE_BETWEEN)
            نص("مهام اليوم"، الوزن: BOLD)
            نص(إحصائيات.المهام_المكتملة + "/" + إحصائيات.إجمالي_المهام)
        نهاية_صف
        
        # قائمة المهام
        لكل مهمة في المهام.أول(3)
            عنصر_مهمة(مهمة)
        نهاية
        
        # رابط للتطبيق
        زر("عرض الكل"، فعل: افتح_التطبيق)
    نهاية_عمود
نهاية

دالة محتوى_كبير()
    متغير المهام = تطبيق_الإنتاجية.مهام_اليوم()
    متغير الأحداث = تطبيق_الإنتاجية.أحداث_اليوم()
    متغير العادات = تطبيق_الإنتاجية.العادات()
    
    ارجع عمود()
        # المهام
        قسم("المهام")
            لكل مهمة في المهام.أول(4)
                عنصر_مهمة(مهمة)
            نهاية
        نهاية_قسم
        
        # الأحداث
        قسم("الأحداث")
            لكل حدث في الأحداث.أول(2)
                عنصر_حدث(حدث)
            نهاية
        نهاية_قسم
        
        # العادات
        قسم("العادات")
            صف()
                لكل عادة في العادات
                    رمز_عادة(عادة)
                نهاية
            نهاية_صف
        نهاية_قسم
    نهاية_عمود
نهاية

# === مكونات مساعدة ===

دالة عنصر_مهمة(مهمة)
    ارجع صف(
        الهامش: 4،
        عند_النقر: لامدا() أكمل_المهمة(مهمة.المعرف) نهاية
    )
        مربع_اختيار(مهمة.مكتملة)
        عمود()
            نص(مهمة.العنوان)
            إذا (مهمة.الموعد != لاشيء)
                نص(تنسيق_الموعد(مهمة.الموعد)، الحجم: 10)
            نهاية
        نهاية_عمود
        دائرة_أولوية(مهمة.الأولوية)
    نهاية_صف
نهاية

دالة عنصر_حدث(حدث)
    ارجع صف()
        خط_عمودي(اللون: حدث.اللون)
        عمود()
            نص(حدث.العنوان، الوزن: BOLD)
            نص(تنسيق_الوقت(حدث.البداية) + " - " + تنسيق_الوقت(حدث.النهاية))
        نهاية_عمود
    نهاية_صف
نهاية

دالة رمز_عادة(عادة)
    متغير مكتملة = عادة.مكتملة_اليوم
    ارجع دائرة(
        اللون: إذا (مكتملة) عادة.اللون وإلا GRAY نهاية،
        عند_النقر: لامدا() سجل_عادة(عادة.المعرف) نهاية
    )
        نص(عادة.الرمز)
    نهاية_دائرة
نهاية

# === الإجراءات ===

دالة أكمل_المهمة(المعرف)
    تطبيق_الإنتاجية.أكمل_مهمة(المعرف)
    SadGlanceBridge.حدّث("productivity_widget")
نهاية

دالة سجل_عادة(المعرف)
    تطبيق_الإنتاجية.سجل_عادة(المعرف)
    SadGlanceBridge.حدّث("productivity_widget")
نهاية
```

## 3. تطبيق تلفزيون للبث

```sad
# ملف: tv_streaming.ص

استورد "android/tv/core"
استورد "android/tv/browse"
استورد "android/tv/details"
استورد "android/tv/playback"
استورد "android/tv/player"
استورد "android/tv/channels"
استورد "android/tv/recommendations"
استورد "android/tv/navigation"
استورد "android/tv/voice"
استورد "android/tv/session"

# === تهيئة التطبيق ===

دالة هيئ_التطبيق(السياق)
    SadTVCoreBridge.هيئ(السياق، تكوين(
        اسم_التطبيق: "بث لغة ص"،
        اللغة: "ar"،
        الاتجاه: RTL
    ))
    
    # إعداد جلسة الوسائط
    SadTVMediaSessionBridge.هيئ()
    
    # إعداد الأوامر الصوتية
    SadTVVoiceBridge.سجل_أوامر([
        أمر_صوتي("شغل"، لامدا(س) ابحث_وشغل(س) نهاية)،
        أمر_صوتي("أوقف"، لامدا() أوقف_التشغيل() نهاية)،
        أمر_صوتي("التالي"، لامدا() المحتوى_التالي() نهاية)
    ])
نهاية

# === الصفحة الرئيسية ===

دالة أنشئ_الصفحة_الرئيسية()
    متغير الصفوف = [
        صف_تصفح(
            المعرف: "continue"،
            العنوان: "متابعة المشاهدة"،
            العناصر: احصل_على_متابعة_المشاهدة()
        )،
        صف_تصفح(
            المعرف: "trending"،
            العنوان: "الأكثر مشاهدة"،
            العناصر: احصل_على_الأكثر_مشاهدة()
        )،
        صف_تصفح(
            المعرف: "new"،
            العنوان: "جديد هذا الأسبوع"،
            العناصر: احصل_على_الجديد()
        )،
        صف_تصفح(
            المعرف: "categories"،
            العنوان: "الفئات"،
            العناصر: احصل_على_الفئات()،
            نوع_البطاقة: ICON_CARD
        )
    ]
    
    SadTVBrowseBridge.عرض(صفوف، تكوين_التصفح(
        لون_الخلفية: 0x1A1A2E،
        لون_العنوان: 0xFFFFFF،
        العنوان: "بث لغة ص"،
        الشعار: "logo.png"
    ))
    
    # الأحداث
    SadTVBrowseBridge.عند_الاختيار(لامدا(عنصر)
        اعرض_التفاصيل(عنصر)
    نهاية)
نهاية

# === صفحة التفاصيل ===

دالة اعرض_التفاصيل(محتوى)
    متغير التفاصيل = تفاصيل_المحتوى(
        المعرف: محتوى.المعرف،
        العنوان: محتوى.العنوان،
        الوصف: محتوى.الوصف،
        الصورة: محتوى.الصورة_الكبيرة،
        المدة: محتوى.المدة،
        السنة: محتوى.السنة،
        التقييم: محتوى.التقييم،
        الفئات: محتوى.الفئات
    )
    
    متغير الإجراءات = [
        إجراء_تفاصيل(
            المعرف: "play"،
            النص: "تشغيل"،
            الأيقونة: "ic_play"،
            الأساسي: صحيح
        )،
        إجراء_تفاصيل(
            المعرف: "trailer"،
            النص: "المقدمة"،
            الأيقونة: "ic_trailer"
        )،
        إجراء_تفاصيل(
            المعرف: "add_list"،
            النص: "إضافة لقائمتي"،
            الأيقونة: "ic_add"
        )
    ]
    
    # المحتوى المرتبط
    متغير مرتبط = احصل_على_المحتوى_المرتبط(محتوى.الفئات)
    
    SadTVDetailsBridge.عرض(التفاصيل، الإجراءات، مرتبط)
    
    SadTVDetailsBridge.عند_الإجراء(لامدا(معرف_الإجراء)
        طابق معرف_الإجراء
            عندما "play"
                شغل_المحتوى(محتوى)
            عندما "trailer"
                شغل_المقدمة(محتوى.رابط_المقدمة)
            عندما "add_list"
                أضف_لقائمتي(محتوى)
        نهاية
    نهاية)
نهاية

# === المشغل ===

دالة شغل_المحتوى(محتوى)
    متغير المصدر = مصدر_فيديو(
        الرابط: محتوى.رابط_الفيديو،
        النوع: إذا (محتوى.مشفر) HLS_DRM وإلا HLS نهاية،
        مفتاح_DRM: محتوى.مفتاح_DRM
    )
    
    متغير الترجمات = محتوى.الترجمات.حول(لامدا(ت)
        ارجع مسار_ترجمة(
            اللغة: ت.اللغة،
            الرابط: ت.الرابط
        )
    نهاية)
    
    SadTVVideoPlayerBridge.شغل(المصدر، تكوين_المشغل(
        العنوان: محتوى.العنوان،
        الصورة: محتوى.الصورة،
        الترجمات: الترجمات،
        موضع_البدء: محتوى.آخر_موضع أو 0،
        جودة_افتراضية: AUTO
    ))
    
    # تحديث جلسة الوسائط
    SadTVMediaSessionBridge.حدث_البيانات(
        العنوان: محتوى.العنوان،
        الصورة: محتوى.الصورة،
        المدة: محتوى.المدة
    )
    
    # أحداث المشغل
    SadTVVideoPlayerBridge.عند_التقدم(لامدا(الموضع)
        احفظ_التقدم(محتوى.المعرف، الموضع)
    نهاية)
    
    SadTVVideoPlayerBridge.عند_الانتهاء(لامدا()
        احذف_التقدم(محتوى.المعرف)
        اعرض_التالي(محتوى)
    نهاية)
نهاية

# === القنوات والتوصيات ===

دالة حدث_قنوات_المعاينة()
    # قناة متابعة المشاهدة
    متغير قناة_المتابعة = قناة_معاينة(
        المعرف: "continue_watching"،
        النوع: WATCH_NEXT،
        العنوان: "متابعة المشاهدة"،
        الشعار: "logo.png"
    )
    
    SadTVPreviewBridge.حدث_القناة(قناة_المتابعة، احصل_على_متابعة_المشاهدة())
    
    # قناة التوصيات
    متغير قناة_التوصيات = قناة_معاينة(
        المعرف: "recommendations"،
        النوع: PREVIEW_TYPE_CHANNEL،
        العنوان: "موصى به لك"،
        الشعار: "logo.png"
    )
    
    SadTVPreviewBridge.حدث_القناة(قناة_التوصيات، احصل_على_التوصيات())
نهاية

# === الأوامر الصوتية ===

دالة ابحث_وشغل(نص_البحث)
    متغير نتائج = SadTVSearchBridge.ابحث(نص_البحث)
    
    إذا (نتائج.غير_فارغة)
        متغير أول_نتيجة = نتائج[0]
        شغل_المحتوى(أول_نتيجة)
    وإلا
        SadTVVoiceBridge.تحدث("لم أجد نتائج لـ " + نص_البحث)
    نهاية
نهاية

# === التشغيل في الخلفية (PiP) ===

دالة فعل_صورة_داخل_صورة()
    SadTVPiPBridge.فعل(تكوين_pip(
        نسبة_العرض: 16،
        نسبة_الارتفاع: 9،
        الإجراءات: [
            إجراء_pip("تشغيل/إيقاف"، PLAY_PAUSE)،
            إجراء_pip("إغلاق"، CLOSE)
        ]
    ))
نهاية
```

## 4. معالجة الأخطاء المتقدمة

```sad
# ملف: error_handling.ص

# === معالج الأخطاء العام ===

صنف معالج_الأخطاء
    ساكن دالة عالج(الخطأ، السياق)
        طابق نوع(الخطأ)
            عندما خطأ_شبكة
                اعرض_خطأ_شبكة(الخطأ)
            عندما خطأ_صلاحية
                اطلب_صلاحية(الخطأ.الصلاحية)
            عندما خطأ_مصادقة
                أعد_المصادقة()
            عندما خطأ_DRM
                اعرض_خطأ_drm(الخطأ)
            افتراضي
                اعرض_خطأ_عام(الخطأ)
        نهاية
    نهاية
    
    ساكن دالة اعرض_خطأ_شبكة(الخطأ)
        SadTVErrorBridge.عرض(خطأ_عرض(
            النوع: NETWORK_ERROR،
            العنوان: "مشكلة في الاتصال"،
            الرسالة: "تحقق من اتصالك بالإنترنت"،
            الإجراء: إجراء("إعادة المحاولة"، لامدا()
                أعد_المحاولة()
            نهاية)
        ))
    نهاية
    
    ساكن دالة اعرض_خطأ_drm(الخطأ)
        SadTVErrorBridge.عرض(خطأ_عرض(
            النوع: DRM_ERROR،
            العنوان: "خطأ في التشغيل"،
            الرسالة: "لا يمكن تشغيل هذا المحتوى"،
            التفاصيل: الخطأ.الرسالة
        ))
    نهاية
نهاية

# === الاستخدام ===

حاول
    شغل_المحتوى(محتوى)
امسك خطأ
    معالج_الأخطاء.عالج(خطأ، "تشغيل المحتوى")
نهاية
```
