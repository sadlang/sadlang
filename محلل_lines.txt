284:             إذا (اقرأ_حالي() == "\\")
285:                 # تسلسل هروب داخل الفئة
286:                 تقدم(1)
287:                 إذا (نهاية_النمط())
288:                     ارمي("خطأ في النمط: '\' في نهاية الفئة")
289:                 نهاية
290:                 متغير ح_هروب = استهلك()
291:                 إذا (ح_هروب == "d" أو ح_هروب == "D" أو ح_هروب == "w" أو ح_هروب == "W" أو ح_هروب == "s" أو ح_هروب == "S")
292:                     أضف(عقدة_ف.العناصر، ["خ", ح_هروب])
293:                 وإلا إذا (ح_هروب == "n")
294:                     أضف(عقدة_ف.العناصر، ["ح", "\n"])
295:                 وإلا إذا (ح_هروب == "t")
296:                     أضف(عقدة_ف.العناصر، ["ح", "\t"])
297:                 وإلا إذا (ح_هروب == "r")
298:                     أضف(عقدة_ف.العناصر، ["ح", "\r"])

300:                     أضف(عقدة_ف.العناصر، ["ح", ح_هروب])
301:                 نهاية
302:             وإلا
303:                 # حرف عادي — قد يكون بداية نطاق
304:                 متغير ح_الفئة = استهلك()
305:                 إذا (ليس نهاية_النمط() و اقرأ_حالي() == "-" و اقرأ_تالي() != "]")
306:                     # نطاق: a-z
307:                     تقدم(1)  # استهلك '-'

310:                         أضف(عقدة_ف.العناصر، ["ح", "-"])
311:                     وإلا
312:                         متغير ح_نهاية = استهلك()
313:                         أضف(عقدة_ف.العناصر، ["ن", ح_الفئة, ح_نهاية])
314:                     نهاية
315:                 وإلا
316:                     أضف(عقدة_ف.العناصر، ["ح", ح_الفئة])
317:                 نهاية
318:             نهاية

390:     ## يُطبَّق على الذرة المعطاة

392:     دالة تحليل_كمية(ذرة)
393:         متغير ح = استهلك()

400:             أدنى_ك = 0
401:             أقصى_ك = -1
402:         وإلا إذا (ح == "+")
403:             أدنى_ك = 1
404:             أقصى_ك = -1
405:         وإلا إذا (ح == "?")
406:             أدنى_ك = 0
407:             أقصى_ك = 1
408:         وإلا إذا (ح == "{")
409:             # قراءة {n} أو {n,} أو {n,m}
410:             متغير أدنى_نص = ""
411:             بينما (ليس نهاية_النمط() و اقرأ_حالي() != "," و اقرأ_حالي() != "}")
412:                 أدنى_نص = أدنى_نص + استهلك()

414: 
415:             إذا (طول(أدنى_نص) == 0)
416:                 ارمي("خطأ في النمط: الحد الأدنى في {} فارغ")
417:             نهاية
418: 
419:             أدنى_ك = حول_إلى_رقم(أدنى_نص)
420: 
421:             إذا (ليس نهاية_النمط() و اقرأ_حالي() == ",")
422:                 تقدم(1)  # استهلك ','
423:                 متغير أقصى_نص = ""
424:                 بينما (ليس نهاية_النمط() و اقرأ_حالي() != "}")
425:                     أقصى_نص = أقصى_نص + استهلك()
426:                 نهاية
427:                 إذا (طول(أقصى_نص) == 0)
428:                     أقصى_ك = -1  # {n,} بلا حد أقصى
429:                 وإلا
430:                     أقصى_ك = حول_إلى_رقم(أقصى_نص)
431:                 نهاية
432:             وإلا
433:                 أقصى_ك = أدنى_ك  # {n} بالضبط
434:             نهاية
435: 
436:             توقع("}")
437:         نهاية

440:         إذا (ليس نهاية_النمط() و اقرأ_حالي() == "?")
