# الدوال في لغة ص - الدليل الشامل

## ما هي الدوال؟

الدوال (Methods) هي الوظائف التي تُعرَّف داخل الصنف وتُحدد سلوكيات الكائن. إذا كانت الحقول تُخزّن "ماذا يكون" الكائن، فإن الدوال تُحدد "ماذا يستطيع أن يفعل". الدوال يمكنها الوصول إلى حقول الكائن وتعديلها، وهي الواجهة التي يتفاعل من خلالها المستخدمون مع الكائنات.

لفهم الفرق بشكل عملي، تخيّل صنف `سيارة`: الحقول هي اللون والسرعة والموديل، أما الدوال فهي "تسريع"، "إبطاء"، "تشغيل"، "إيقاف" - أي الأفعال التي تستطيع السيارة القيام بها.

## تعريف الدوال في لغة ص

لتعريف دالة داخل صنف، نستخدم الكلمة المفتاحية `دالة` متبوعة بنوع القيمة المُرجعة (إن وُجدت)، ثم اسم الدالة، ثم المعاملات بين قوسين:

```
صنف اسم_الصنف
    دالة اسم_الدالة()
        # كود الدالة
    نهاية
    
    دالة نوع_القيمة اسم_دالة_مع_إرجاع()
        # كود الدالة
        ارجع قيمة
    نهاية
    
    دالة اسم_دالة_مع_معاملات(نوع معامل1، نوع معامل2)
        # كود الدالة
    نهاية
نهاية
```

## أنواع الدوال

### 1. دوال بدون معاملات ولا إرجاع

أبسط أنواع الدوال، تُنفّذ عملية ما دون الحاجة لمدخلات ولا تُرجع قيمة:

```
صنف طابعة
    نص الاسم
    
    باني(نص اسم)
        الاسم = اسم
    نهاية
    
    دالة تشغيل()
        اطبع_سطر("تم تشغيل الطابعة: " + الاسم)
    نهاية
    
    دالة إيقاف()
        اطبع_سطر("تم إيقاف الطابعة: " + الاسم)
    نهاية
    
    دالة طباعة_صفحة_اختبار()
        اطبع_سطر("═══════════════════════════")
        اطبع_سطر("   صفحة اختبار الطابعة    ")
        اطبع_سطر("   الطابعة: " + الاسم)
        اطبع_سطر("   الحالة: تعمل بشكل جيد  ")
        اطبع_سطر("═══════════════════════════")
    نهاية
نهاية

دالة رئيسية()
    طابعة ط = جديد طابعة("HP LaserJet")
    ط.تشغيل()
    ط.طباعة_صفحة_اختبار()
    ط.إيقاف()
    ارجع 0
نهاية
```

### 2. دوال مع معاملات

دوال تأخذ بيانات من المستدعي لتُستخدم في تنفيذ العملية:

```
صنف آلة_حاسبة
    رقم آخر_نتيجة
    
    باني()
        آخر_نتيجة = 0
    نهاية
    
    دالة جمع(رقم أ، رقم ب)
        آخر_نتيجة = أ + ب
        اطبع_سطر(أ + " + " + ب + " = " + آخر_نتيجة)
    نهاية
    
    دالة طرح(رقم أ، رقم ب)
        آخر_نتيجة = أ - ب
        اطبع_سطر(أ + " - " + ب + " = " + آخر_نتيجة)
    نهاية
    
    دالة ضرب(رقم أ، رقم ب)
        آخر_نتيجة = أ * ب
        اطبع_سطر(أ + " × " + ب + " = " + آخر_نتيجة)
    نهاية
    
    دالة قسمة(رقم أ، رقم ب)
        إذا ب == 0
            اطبع_سطر("خطأ: لا يمكن القسمة على صفر!")
        وإلا
            آخر_نتيجة = أ / ب
            اطبع_سطر(أ + " ÷ " + ب + " = " + آخر_نتيجة)
        نهاية
    نهاية
نهاية

دالة رئيسية()
    آلة_حاسبة حاسبة = جديد آلة_حاسبة()
    
    حاسبة.جمع(10، 5)
    حاسبة.طرح(10، 5)
    حاسبة.ضرب(10، 5)
    حاسبة.قسمة(10، 5)
    حاسبة.قسمة(10، 0)
    
    ارجع 0
نهاية
```

### 3. دوال مع قيمة مُرجعة

دوال تُنفّذ عملية وتُرجع نتيجة للمستدعي:

```
صنف مستطيل
    رقم العرض
    رقم الارتفاع
    
    باني(رقم ع، رقم ا)
        العرض = ع
        الارتفاع = ا
    نهاية
    
    دالة رقم المساحة()
        ارجع العرض * الارتفاع
    نهاية
    
    دالة رقم المحيط()
        ارجع 2 * (العرض + الارتفاع)
    نهاية
    
    دالة منطقي مربع()
        ارجع العرض == الارتفاع
    نهاية
    
    دالة نص الوصف()
        إذا مربع()
            ارجع "مربع بضلع " + العرض
        نهاية
        ارجع "مستطيل " + العرض + " × " + الارتفاع
    نهاية
نهاية

دالة رئيسية()
    مستطيل م1 = جديد مستطيل(10، 5)
    مستطيل م2 = جديد مستطيل(7، 7)
    
    اطبع_سطر("الشكل الأول: " + م1.الوصف())
    اطبع_سطر("المساحة: " + م1.المساحة())
    اطبع_سطر("المحيط: " + م1.المحيط())
    
    اطبع_سطر("")
    
    اطبع_سطر("الشكل الثاني: " + م2.الوصف())
    اطبع_سطر("المساحة: " + م2.المساحة())
    اطبع_سطر("المحيط: " + م2.المحيط())
    
    ارجع 0
نهاية
```

### 4. دوال مع معاملات وقيمة مُرجعة

الدوال الأكثر شمولاً تأخذ مدخلات وتُرجع مخرجات:

```
صنف محول_درجات_حرارة
    دالة عشري سيلزيوس_إلى_فهرنهايت(عشري س)

    دالة عشري فهرنهايت_إلى_سيلزيوس(عشري ف)

    نهاية
    
    دالة عشري سيلزيوس_إلى_كلفن(عشري س)
        ارجع س + 273.15
    نهاية
    
    دالة عشري كلفن_إلى_سيلزيوس(عشري ك)
        ارجع ك - 273.15
    نهاية
    
    دالة نص تقييم_درجة_سيلزيوس(عشري درجة)
        إذا درجة < 0
            ارجع "تجمد ❄️"
        نهاية
        إذا درجة < 15
            ارجع "بارد 🌬️"
        نهاية
        إذا درجة < 25
            ارجع "معتدل 🌤️"
        نهاية
        إذا درجة < 35
            ارجع "حار ☀️"
        نهاية
        ارجع "شديد الحرارة 🔥"

دالة رئيسية()

    عشري درجة_سيلزيوس = 25.0

    اطبع_سطر("بالكلفن: " + محول.سيلزيوس_إلى_كلفن(درجة_سيلزيوس) + "K")
    اطبع_سطر("التقييم: " + محول.تقييم_درجة_سيلزيوس(درجة_سيلزيوس))
    اطبع_سطر("═══════════════════════════════════")
    
    ارجع 0

```

## الدوال والوصول للحقول

الدوال داخل الصنف تستطيع الوصول مباشرة لجميع حقول الكائن:

```
صنف حساب_توفير
    نص رقم_الحساب
    نص اسم_المالك
    عشري الرصيد
    عشري نسبة_الفائدة
    
    باني(نص رقم، نص اسم، عشري نسبة)
        رقم_الحساب = رقم
        اسم_المالك = اسم
        الرصيد = 0.0
        نسبة_الفائدة = نسبة
    نهاية
    
    # دالة تقرأ الحقول
    دالة عرض_الرصيد()
        اطبع_سطر("رصيد الحساب " + رقم_الحساب + ": " + الرصيد + " ريال")
    نهاية
    
    # دالة تُعدّل الحقول
    دالة إيداع(عشري مبلغ)
        إذا مبلغ > 0
            الرصيد = الرصيد + مبلغ
            اطبع_سطر("تم إيداع " + مبلغ + " ريال")
            عرض_الرصيد()
        وإلا
            اطبع_سطر("خطأ: المبلغ يجب أن يكون موجباً")
        نهاية
    نهاية
    
    # دالة تُعدّل الحقول مع التحقق
    دالة منطقي سحب(عشري مبلغ)
        إذا مبلغ <= 0
            اطبع_سطر("خطأ: المبلغ يجب أن يكون موجباً")
            ارجع خطأ
        نهاية
        
        إذا مبلغ > الرصيد
            اطبع_سطر("خطأ: الرصيد غير كافٍ")
            ارجع خطأ
        نهاية
        
        الرصيد = الرصيد - مبلغ
        اطبع_سطر("تم سحب " + مبلغ + " ريال")
        عرض_الرصيد()
        ارجع صحيح
    نهاية
    
    # دالة تقرأ وتُعدّل الحقول
    دالة تطبيق_الفائدة()
        عشري الفائدة = الرصيد * نسبة_الفائدة / 100
        الرصيد = الرصيد + الفائدة
        اطبع_سطر("تم إضافة فائدة: " + الفائدة + " ريال")
        عرض_الرصيد()
    نهاية
نهاية
```

## استدعاء دوال من دوال أخرى

يمكن للدوال داخل نفس الصنف استدعاء بعضها البعض:

```
صنف محلل_نص
    نص النص
    
    باني(نص نص)
        النص = نص
    نهاية
    
    دالة رقم عدد_الأحرف()
        ارجع طول(النص)
    نهاية
    
    دالة رقم عدد_الكلمات()
        رقم عداد = 1
        رقم ف = 0
        طالما ف < طول(النص)
            إذا حرف(النص، ف) == " "
                عداد = عداد + 1
            نهاية
            ف = ف + 1
        نهاية
        ارجع عداد
    نهاية
    
    دالة عشري متوسط_طول_الكلمة()
        رقم الأحرف = عدد_الأحرف()
        رقم الكلمات = عدد_الكلمات()
        رقم المسافات = الكلمات - 1
        ارجع (الأحرف - المسافات) / الكلمات
    نهاية
    
    دالة تقرير_كامل()
        اطبع_سطر("═══════════════════════════════════")
        اطبع_سطر("         تقرير تحليل النص          ")
        اطبع_سطر("═══════════════════════════════════")
        اطبع_سطر("النص: " + النص)
        اطبع_سطر("───────────────────────────────────")
        اطبع_سطر("عدد الأحرف: " + عدد_الأحرف())
        اطبع_سطر("عدد الكلمات: " + عدد_الكلمات())
        اطبع_سطر("متوسط طول الكلمة: " + متوسط_طول_الكلمة())
        اطبع_سطر("═══════════════════════════════════")
    نهاية
نهاية
```

## دوال الوصول (Getters) ودوال التعديل (Setters)

نمط شائع جداً في البرمجة الكائنية هو استخدام دوال خاصة للوصول للحقول وتعديلها:

```
صنف شخص
    نص الاسم_الخاص
    رقم العمر_الخاص
    نص البريد_الخاص
    
    باني(نص اسم، رقم عمر)
        الاسم_الخاص = اسم
        العمر_الخاص = عمر
        البريد_الخاص = ""
    نهاية
    
    # دوال الوصول (Getters)
    دالة نص الاسم()
        ارجع الاسم_الخاص
    نهاية
    
    دالة رقم العمر()
        ارجع العمر_الخاص
    نهاية
    
    دالة نص البريد()
        ارجع البريد_الخاص
    نهاية
    
    # دوال التعديل (Setters) مع التحقق
    دالة تعيين_الاسم(نص اسم_جديد)
        إذا طول(اسم_جديد) < 2
            اطبع_سطر("خطأ: الاسم يجب أن يكون حرفين على الأقل")
        وإلا
            الاسم_الخاص = اسم_جديد
            اطبع_سطر("تم تغيير الاسم إلى: " + اسم_جديد)
        نهاية
    نهاية
    
    دالة تعيين_العمر(رقم عمر_جديد)
        إذا عمر_جديد < 0 أو عمر_جديد > 150
            اطبع_سطر("خطأ: العمر غير صالح")
        وإلا
            العمر_الخاص = عمر_جديد
            اطبع_سطر("تم تغيير العمر إلى: " + عمر_جديد)
        نهاية
    نهاية
    
    دالة تعيين_البريد(نص بريد_جديد)
        # التحقق البسيط من صحة البريد
        إذا طول(بريد_جديد) > 0
            البريد_الخاص = بريد_جديد
            اطبع_سطر("تم تعيين البريد: " + بريد_جديد)
        وإلا
            اطبع_سطر("خطأ: البريد لا يمكن أن يكون فارغاً")
        نهاية
    نهاية
نهاية

دالة رئيسية()
    شخص ش = جديد شخص("أحمد"، 25)
    
    # استخدام دوال الوصول
    اطبع_سطر("الاسم: " + ش.الاسم())
    اطبع_سطر("العمر: " + ش.العمر())
    
    # استخدام دوال التعديل
    ش.تعيين_البريد("ahmed@example.com")
    اطبع_سطر("البريد: " + ش.البريد())
    
    # محاولة تعيين قيم غير صالحة
    ش.تعيين_العمر(-5)
    ش.تعيين_الاسم("أ")
    
    ارجع 0
نهاية
```

## مثال شامل: نظام إدارة المهام

```
صنف مهمة
    رقم الرقم
    نص العنوان
    نص الوصف
    نص الحالة
    رقم الأولوية
    رقم نسبة_الإنجاز
    
    باني(رقم رقم، نص عنوان، نص وصف، رقم أولوية)
        الرقم = رقم
        العنوان = عنوان
        الوصف = وصف
        الأولوية = أولوية
        الحالة = "قيد الانتظار"
        نسبة_الإنجاز = 0
    نهاية
    
    # دوال تغيير الحالة
    دالة بدء()
        إذا الحالة == "قيد الانتظار"
            الحالة = "قيد التنفيذ"
            نسبة_الإنجاز = 10
            اطبع_سطر("تم بدء المهمة: " + العنوان)
        وإلا
            اطبع_سطر("لا يمكن بدء المهمة - الحالة الحالية: " + الحالة)
        نهاية
    نهاية
    
    دالة تحديث_التقدم(رقم نسبة)
        إذا الحالة != "قيد التنفيذ"
            اطبع_سطر("خطأ: يجب بدء المهمة أولاً")
            ارجع
        نهاية
        
        إذا نسبة < 0 أو نسبة > 100
            اطبع_سطر("خطأ: النسبة يجب أن تكون بين 0 و 100")
            ارجع
        نهاية
        
        نسبة_الإنجاز = نسبة
        
        إذا نسبة == 100
            إكمال()
        وإلا
            اطبع_سطر("تم تحديث التقدم إلى: " + نسبة + "%")
        نهاية
    نهاية
    
    دالة إكمال()
        الحالة = "مكتملة"
        نسبة_الإنجاز = 100
        اطبع_سطر("✓ تم إكمال المهمة: " + العنوان)
    نهاية
    
    دالة إلغاء()
        الحالة = "ملغاة"
        اطبع_سطر("✗ تم إلغاء المهمة: " + العنوان)
    نهاية
    
    # دوال المعلومات
    دالة منطقي مكتملة()
        ارجع الحالة == "مكتملة"
    نهاية
    
    دالة منطقي نشطة()
        ارجع الحالة == "قيد التنفيذ"
    نهاية
    
    دالة نص نص_الأولوية()
        إذا الأولوية == 1
            ارجع "🔴 عالية"
        نهاية
        إذا الأولوية == 2
            ارجع "🟡 متوسطة"
        نهاية
        ارجع "🟢 منخفضة"
    نهاية
    
    دالة شريط_التقدم()
        نص شريط = "["
        رقم مملوء = نسبة_الإنجاز / 10
        رقم ف = 0
        طالما ف < 10
            إذا ف < مملوء
                شريط = شريط + "█"
            وإلا
                شريط = شريط + "░"
            نهاية
            ف = ف + 1
        نهاية
        شريط = شريط + "] " + نسبة_الإنجاز + "%"
        ارجع شريط
    نهاية
    
    دالة عرض()
        اطبع_سطر("┌─────────────────────────────────────┐")
        اطبع_سطر("│ المهمة #" + الرقم + ": " + العنوان)
        اطبع_سطر("├─────────────────────────────────────┤")
        اطبع_سطر("│ الوصف: " + الوصف)
        اطبع_سطر("│ الأولوية: " + نص_الأولوية())
        اطبع_سطر("│ الحالة: " + الحالة)
        اطبع_سطر("│ التقدم: " + شريط_التقدم())
        اطبع_سطر("└─────────────────────────────────────┘")
    نهاية
نهاية

دالة رئيسية()

```

## نصائح لكتابة دوال فعّالة

**اجعل كل دالة تفعل شيئاً واحداً**: دالة `حساب_المجموع` يجب أن تحسب المجموع فقط، لا تحسبه وتطبعه وتحفظه.

**استخدم أسماء واضحة**: `حساب_الضريبة` أفضل من `ح_ض`.

**تحقق من المدخلات**: لا تفترض أن المستدعي سيُمرر قيماً صالحة.

**أرجع قيماً ذات معنى**: إذا كانت الدالة تُرجع نجاح/فشل، استخدم `منطقي`.

**وثّق الدوال المعقدة**: أضف تعليقات تشرح ما تفعله الدالة.

## الخلاصة

الدوال هي قلب السلوك في البرمجة الكائنية. كتابة دوال جيدة ومنظمة يجعل الكود أسهل في الفهم والصيانة. في القسم التالي، سنتعرف على الوراثة وكيف تُمكّننا من إعادة استخدام الكود بين الأصناف.
