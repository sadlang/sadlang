# 🎭 تعدد الأشكال (Polymorphism)

## مقدمة

تعدد الأشكال (Polymorphism) هو مفهوم أساسي في البرمجة الكائنية يتيح معاملة كائنات من أصناف مختلفة بنفس الطريقة. الكلمة مشتقة من اليونانية وتعني "أشكال متعددة". في البرمجة، يعني أن نفس العملية يمكن أن تتصرف بشكل مختلف على أنواع مختلفة من الكائنات.

## الفكرة الأساسية

```s
صنف شكل
    دالة ارسم()
        اطبع("رسم شكل عام")
    نهاية
نهاية

صنف دائرة يرث شكل
    دالة ارسم()
        اطبع("⭕ رسم دائرة")
    نهاية
نهاية

صنف مربع يرث شكل
    دالة ارسم()
        اطبع("⬛ رسم مربع")
    نهاية
نهاية

صنف مثلث يرث شكل
    دالة ارسم()
        اطبع("🔺 رسم مثلث")
    نهاية
نهاية

دالة رئيسية()
    # قائمة من أشكال مختلفة
    قائمة أشكال = [
        جديد دائرة()،
        جديد مربع()،
        جديد مثلث()،
        جديد دائرة()
    ]
    
    # نفس الاستدعاء، سلوكيات مختلفة
    لكل شكل في أشكال
        شكل.ارسم()  # كل شكل يرسم نفسه بطريقته
    نهاية
نهاية

# المخرجات:
# ⭕ رسم دائرة
# ⬛ رسم مربع
# 🔺 رسم مثلث
# ⭕ رسم دائرة
```

## أنواع تعدد الأشكال

### 1. تعدد الأشكال بالوراثة (Runtime Polymorphism)

```s
صنف حيوان
    نص الاسم
    
    منشئ(نص اسم)
        هذا.الاسم = اسم
    نهاية
    
    دالة أصدر_صوت()
        اطبع("...")
    نهاية
    
    دالة تحرك()
        اطبع(هذا.الاسم + " يتحرك")
    نهاية
نهاية

صنف قط يرث حيوان
    منشئ(نص اسم)
        أبي(اسم)
    نهاية
    
    دالة أصدر_صوت()
        اطبع(هذا.الاسم + " يقول: مياو! 🐱")
    نهاية
نهاية

صنف كلب يرث حيوان
    منشئ(نص اسم)
        أبي(اسم)
    نهاية
    
    دالة أصدر_صوت()
        اطبع(هذا.الاسم + " يقول: هاو! 🐕")
    نهاية
نهاية

صنف عصفور يرث حيوان
    منشئ(نص اسم)
        أبي(اسم)
    نهاية
    
    دالة أصدر_صوت()
        اطبع(هذا.الاسم + " يقول: زقزقة! 🐦")
    نهاية
    
    # إعادة تعريف التحرك أيضاً
    دالة تحرك()
        اطبع(هذا.الاسم + " يطير ✈️")
    نهاية
نهاية

دالة رئيسية()
    قائمة حيوانات = [
        جديد قط("ميشو")،
        جديد كلب("بادي")،
        جديد عصفور("تويتي")،
        جديد قط("نمر")
    ]
    
    اطبع("=== أصوات الحيوانات ===")
    لكل حيوان في حيوانات
        حيوان.أصدر_صوت()
    نهاية
    
    اطبع("\n=== حركة الحيوانات ===")
    لكل حيوان في حيوانات
        حيوان.تحرك()
    نهاية
نهاية
```

### 2. تعدد الأشكال بالواجهات (Interface Polymorphism)

```s
# واجهة للكائنات القابلة للرسم
واجهة قابل_للرسم
    دالة ارسم()
    دالة لون_التعبئة()
نهاية

# واجهة للكائنات القابلة للتحريك
واجهة قابل_للتحريك
    دالة تحرك(رقم س، رقم ص)
    دالة الموقع()
نهاية

# صنف يطبق كلا الواجهتين
صنف شخصية_لعبة ينفذ قابل_للرسم، قابل_للتحريك
    رقم س = 0
    رقم ص = 0
    نص اللون
    نص الاسم
    
    منشئ(نص اسم، نص لون)
        هذا.الاسم = اسم
        هذا.اللون = لون
    نهاية
    
    دالة ارسم()
        اطبع("🎮 رسم " + هذا.الاسم + " في (" + هذا.س + "، " + هذا.ص + ")")
    نهاية
    
    دالة لون_التعبئة()
        إرجاع هذا.اللون
    نهاية
    
    دالة تحرك(رقم س_جديد، رقم ص_جديد)
        هذا.س = س_جديد
        هذا.ص = ص_جديد
        اطبع(هذا.الاسم + " انتقل إلى (" + س_جديد + "، " + ص_جديد + ")")
    نهاية
    
    دالة الموقع()
        إرجاع (هذا.س، هذا.ص)
    نهاية
نهاية

صنف عنصر_واجهة ينفذ قابل_للرسم
    نص النص
    نص اللون = "رمادي"
    
    منشئ(نص نص)
        هذا.النص = نص
    نهاية
    
    دالة ارسم()
        اطبع("📱 عرض: [" + هذا.النص + "]")
    نهاية
    
    دالة لون_التعبئة()
        إرجاع هذا.اللون
    نهاية
نهاية

دالة ارسم_الكل(قائمة عناصر)
    # هذه الدالة تعمل مع أي شيء قابل للرسم
    لكل عنصر في عناصر
        عنصر.ارسم()
    نهاية
نهاية

دالة رئيسية()
    قائمة قابلة_للرسم = [
        جديد شخصية_لعبة("البطل"، "أزرق")،
        جديد عنصر_واجهة("ابدأ اللعبة")،
        جديد شخصية_لعبة("العدو"، "أحمر")،
        جديد عنصر_واجهة("خروج")
    ]
    
    ارسم_الكل(قابلة_للرسم)
نهاية
```

نهاية

    متجر.أتمم_عملية_شراء(200، باي_بال)

### نظام الملفات

```s
صنف عنصر_ملف
    نص الاسم
    
    منشئ(نص اسم)
        هذا.الاسم = اسم
    نهاية
    
    دالة الحجم()
        إرجاع 0
    نهاية
    
    دالة اعرض(رقم مستوى = 0)
        نص مسافة = "  " * مستوى
        اطبع(مسافة + هذا.الاسم)
    نهاية
نهاية

صنف ملف يرث عنصر_ملف
    رقم حجم_البايتات
    نص النوع
    
    منشئ(نص اسم، رقم حجم، نص نوع)
        أبي(اسم)
        هذا.حجم_البايتات = حجم
        هذا.النوع = نوع
    نهاية
    
    دالة الحجم()
        إرجاع هذا.حجم_البايتات
    نهاية
    
    دالة اعرض(رقم مستوى = 0)
        نص مسافة = "  " * مستوى
        نص أيقونة = هذا.أيقونة_النوع()
        اطبع(مسافة + أيقونة + " " + هذا.الاسم + " (" + هذا.حجم_منسق() + ")")
    نهاية
    
    دالة أيقونة_النوع()
        طابق هذا.النوع
            حالة "txt" | "doc" | "pdf":
                إرجاع "📄"
            حالة "jpg" | "png" | "gif":
                إرجاع "🖼️"
            حالة "mp3" | "wav":
                إرجاع "🎵"
            حالة "mp4" | "avi":
                إرجاع "🎬"
            حالة _:
                إرجاع "📁"
        نهاية
    نهاية
    
    دالة حجم_منسق()
        إذا (هذا.حجم_البايتات < 1024)
            إرجاع هذا.حجم_البايتات + " B"
        وإلا إذا (هذا.حجم_البايتات < 1024 * 1024)
            إرجاع قرب(هذا.حجم_البايتات / 1024، 1) + " KB"
        وإلا
            إرجاع قرب(هذا.حجم_البايتات / 1024 / 1024، 1) + " MB"
        نهاية
    نهاية
نهاية

صنف مجلد يرث عنصر_ملف
    قائمة المحتويات = []
    
    منشئ(نص اسم)
        أبي(اسم)
    نهاية
    
    دالة أضف(عنصر_ملف عنصر)
        هذا.المحتويات.أضف(عنصر)
    نهاية
    
    دالة الحجم()
        # حجم المجلد = مجموع أحجام محتوياته
        رقم الإجمالي = 0
        لكل عنصر في هذا.المحتويات
            الإجمالي = الإجمالي + عنصر.الحجم()
        نهاية
        إرجاع الإجمالي
    نهاية
    
    دالة اعرض(رقم مستوى = 0)
        نص مسافة = "  " * مستوى
        اطبع(مسافة + "📁 " + هذا.الاسم + "/")
        
        لكل عنصر في هذا.المحتويات
            عنصر.اعرض(مستوى + 1)
        نهاية
    نهاية
نهاية

دالة رئيسية()
    # إنشاء هيكل ملفات
    متغير جذر = جديد مجلد("المستندات")
    
    متغير صور = جديد مجلد("صور")
    صور.أضف(جديد ملف("صورة1.jpg"، 2500000، "jpg"))
    صور.أضف(جديد ملف("صورة2.png"، 1800000، "png"))
    
    متغير مستندات = جديد مجلد("مستندات نصية")
    مستندات.أضف(جديد ملف("تقرير.pdf"، 500000، "pdf"))
    مستندات.أضف(جديد ملف("ملاحظات.txt"، 2048، "txt"))
    
    جذر.أضف(صور)
    جذر.أضف(مستندات)
    جذر.أضف(جديد ملف("فيديو.mp4"، 150000000، "mp4"))
    
    # عرض الهيكل
    جذر.اعرض()
    
    اطبع("\nالحجم الإجمالي: " + جذر.الحجم() + " بايت")
نهاية
```

## الخلاصة

تعدد الأشكال في لغة ص:
- نفس الاستدعاء، سلوكيات مختلفة
- يتحقق عبر الوراثة والواجهات
- يتيح كتابة كود مرن وقابل للتوسعة
- يسهل الاختبار باستخدام كائنات وهمية
- يفصل الاهتمامات ويقلل الترابط
- أساس للعديد من أنماط التصميم
