# 🧬 الوراثة في لغة ص

## مقدمة

الوراثة (Inheritance) هي أحد أركان البرمجة الكائنية الأساسية. تتيح لك إنشاء صنف جديد يرث خصائص وسلوكيات من صنف موجود. الصنف الجديد يُسمى "الصنف الابن" أو "الصنف المشتق"، والصنف الأصلي يُسمى "الصنف الأب" أو "الصنف الأساسي".

## لماذا الوراثة؟

### المشكلة بدون وراثة

```s
# بدون وراثة - تكرار الكود
صنف قط
    نص الاسم
    رقم العمر
    نص اللون
    
    دالة تنفس()
        اطبع(هذا.الاسم + " يتنفس")
    نهاية
    
    دالة نم()
        اطبع(هذا.الاسم + " ينام")
    نهاية
    
    دالة موء()
        اطبع("مياو!")
    نهاية
نهاية

صنف كلب
    نص الاسم
    رقم العمر
    نص اللون
    
    # نفس الدوال مكررة!
    دالة تنفس()
        اطبع(هذا.الاسم + " يتنفس")
    نهاية
    
    دالة نم()
        اطبع(هذا.الاسم + " ينام")
    نهاية
    
    دالة انبح()
        اطبع("هاو هاو!")
    نهاية
نهاية
```

## استدعاء منشئ الصنف الأب

```s
صنف موظف
    نص الاسم
    رقم الرقم_الوظيفي
    رقم الراتب
    
    منشئ(نص اسم، رقم رقم، رقم راتب)
        هذا.الاسم = اسم
        هذا.الرقم_الوظيفي = رقم
        هذا.الراتب = راتب
    نهاية
    
    دالة معلومات()
        اطبع("الموظف: " + هذا.الاسم)
        اطبع("الرقم: " + هذا.الرقم_الوظيفي)
        اطبع("الراتب: " + هذا.الراتب)
    نهاية
نهاية

صنف مدير يرث موظف
    نص القسم
    قائمة الموظفين_تحت_إدارته = []
    
    منشئ(نص اسم، رقم رقم، رقم راتب، نص قسم)
        # استدعاء منشئ الصنف الأب
        أبي(اسم، رقم، راتب)
        هذا.القسم = قسم
    نهاية
    
    دالة أضف_موظف(موظف م)
        هذا.الموظفين_تحت_إدارته.أضف(م)
    نهاية
    
    دالة معلومات()
        # استدعاء دالة الأب أولاً
        أبي.معلومات()
        اطبع("القسم: " + هذا.القسم)
        اطبع("عدد الموظفين: " + طول(هذا.الموظفين_تحت_إدارته))
    نهاية
نهاية

صنف مبرمج يرث موظف
    قائمة اللغات = []
    
    منشئ(نص اسم، رقم رقم، رقم راتب، قائمة لغات)
        أبي(اسم، رقم، راتب)
        هذا.اللغات = لغات
    نهاية
    
    دالة أضف_لغة(نص لغة)
        هذا.اللغات.أضف(لغة)
    نهاية
    
    دالة معلومات()
        أبي.معلومات()
        اطبع("اللغات: " + هذا.اللغات.اربط("، "))
    نهاية
نهاية

دالة رئيسية()
    متغير مدير1 = جديد مدير("أحمد"، 1001، 15000، "التقنية")
    متغير مبرمج1 = جديد مبرمج("محمد"، 1002، 10000، ["بايثون"، "جافا"])
    
    مدير1.أضف_موظف(مبرمج1)
    
    اطبع("=== معلومات المدير ===")
    مدير1.معلومات()
    
    اطبع("\n=== معلومات المبرمج ===")
    مبرمج1.معلومات()
نهاية
```

## إعادة تعريف الدوال (Override)

يمكن للصنف الابن إعادة تعريف دالة موجودة في الصنف الأب لتغيير سلوكها:

```s
صنف شكل
    نص الاسم
    
    منشئ(نص اسم)
        هذا.الاسم = اسم
    نهاية
    
    دالة المساحة()
        # الصنف الأساسي لا يعرف كيف يحسب المساحة
        إرجاع 0
    نهاية
    
    دالة المحيط()
        إرجاع 0
    نهاية
    
    دالة وصف()
        اطبع("هذا شكل: " + هذا.الاسم)
    نهاية
نهاية

صنف مستطيل يرث شكل
    رقم الطول
    رقم العرض
    
    منشئ(رقم طول، رقم عرض)
        أبي("مستطيل")
        هذا.الطول = طول
        هذا.العرض = عرض
    نهاية
    
    # إعادة تعريف دالة المساحة
    دالة المساحة()
        إرجاع هذا.الطول * هذا.العرض
    نهاية
    
    # إعادة تعريف دالة المحيط
    دالة المحيط()
        إرجاع 2 * (هذا.الطول + هذا.العرض)
    نهاية
    
    # إعادة تعريف مع استدعاء الأب
    دالة وصف()
        أبي.وصف()
        اطبع("الأبعاد: " + هذا.الطول + " × " + هذا.العرض)
    نهاية
نهاية

صنف دائرة يرث شكل
    رقم نصف_القطر
    
    منشئ(رقم نصف_قطر)
        أبي("دائرة")
        هذا.نصف_القطر = نصف_قطر
    نهاية
    
    دالة المساحة()
        إرجاع 3.14159 * هذا.نصف_القطر ** 2
    نهاية
    
    دالة المحيط()
        إرجاع 2 * 3.14159 * هذا.نصف_القطر
    نهاية
    
    دالة وصف()
        أبي.وصف()
        اطبع("نصف القطر: " + هذا.نصف_القطر)
    نهاية
نهاية

صنف مثلث يرث شكل
    رقم أ
    رقم ب
    رقم ج
    
    منشئ(رقم ضلع1، رقم ضلع2، رقم ضلع3)
        أبي("مثلث")
        هذا.أ = ضلع1
        هذا.ب = ضلع2
        هذا.ج = ضلع3
    نهاية
    
    دالة المساحة()
        # صيغة هيرون
        رقم ن = هذا.المحيط() / 2  # نصف المحيط
        إرجاع جذر(ن * (ن - هذا.أ) * (ن - هذا.ب) * (ن - هذا.ج))
    نهاية
    
    دالة المحيط()
        إرجاع هذا.أ + هذا.ب + هذا.ج
    نهاية
نهاية

دالة رئيسية()
    قائمة أشكال = [
        جديد مستطيل(5، 3)،
        جديد دائرة(4)،
        جديد مثلث(3، 4، 5)
    ]
    
    لكل شكل في أشكال
        شكل.وصف()
        اطبع("المساحة: " + شكل.المساحة())
        اطبع("المحيط: " + شكل.المحيط())
        اطبع("---")
    نهاية
نهاية
```

## سلسلة الوراثة

يمكن أن يرث صنف من صنف يرث من صنف آخر:

```s
صنف كائن_حي
    منطقي حي = صحيح
    
    دالة تنفس()
        اطبع("يتنفس...")
    نهاية
نهاية

صنف حيوان يرث كائن_حي
    نص الاسم
    
    منشئ(نص اسم)
        هذا.الاسم = اسم
    نهاية
    
    دالة تحرك()
        اطبع(هذا.الاسم + " يتحرك")
    نهاية
نهاية

صنف ثديي يرث حيوان
    منشئ(نص اسم)
        أبي(اسم)
    نهاية
    
    دالة أرضع()
        اطبع(هذا.الاسم + " يُرضع صغاره")
    نهاية
نهاية

صنف قط يرث ثديي
    منشئ(نص اسم)
        أبي(اسم)
    نهاية
    
    دالة موء()
        اطبع(هذا.الاسم + " يقول: مياو!")
    نهاية
نهاية

دالة رئيسية()
    متغير قطتي = جديد قط("ميشو")
    
    # كل الدوال متاحة من سلسلة الوراثة
    قطتي.تنفس()   # من كائن_حي
    قطتي.تحرك()   # من حيوان
    قطتي.أرضع()   # من ثديي
    قطتي.موء()    # من قط
نهاية
```

## التحقق من النوع

```s
دالة رئيسية()
    متغير حيوانات = [
        جديد قط("ميشو"، 3، "أبيض")،
        جديد كلب("بادي"، 5، "بني")،
        جديد قط("نمر"، 2، "برتقالي")
    ]
    
    لكل حيوان في حيوانات
        # التحقق من النوع
        إذا (حيوان هو قط)
            حيوان.موء()
        وإلا إذا (حيوان هو كلب)
            حيوان.انبح()
        نهاية
    نهاية
    
    # عد أنواع الحيوانات
    رقم قطط = 0
    رقم كلاب = 0
    
    لكل حيوان في حيوانات
        إذا (حيوان هو قط)
            قطط = قطط + 1
        وإلا إذا (حيوان هو كلب)
            كلاب = كلاب + 1
        نهاية
    نهاية
    
    اطبع("عدد القطط: " + قطط)
    اطبع("عدد الكلاب: " + كلاب)
نهاية
```

## الخلاصة

الوراثة في لغة ص:
- `يرث` لتحديد الصنف الأب
- `أبي()` لاستدعاء منشئ الأب
- `أبي.دالة()` لاستدعاء دالة الأب
- يمكن إعادة تعريف الدوال (Override)
- سلاسل الوراثة متعددة المستويات ممكنة
- `هو` للتحقق من نوع الكائن
- تجنب الوراثة العميقة جداً
