# 🔒 التغليف (Encapsulation)

## مقدمة

التغليف (Encapsulation) هو مبدأ أساسي في البرمجة الكائنية يتعلق بإخفاء التفاصيل الداخلية للكائن وتوفير واجهة عامة للتفاعل معه. فكر فيه كصندوق أسود: تعرف ما يدخل وما يخرج، لكن لا تحتاج لمعرفة كيف يعمل من الداخل.

## لماذا التغليف؟

### المشكلة بدون تغليف

```s
صنف حساب_بنكي
    نص رقم_الحساب
    رقم الرصيد
    
    منشئ(نص رقم، رقم رصيد)
        هذا.رقم_الحساب = رقم
        هذا.الرصيد = رصيد
    نهاية
نهاية

دالة رئيسية()
    متغير حساب = جديد حساب_بنكي("SA001"، 1000)
    
    # المشكلة: يمكن تعديل الرصيد مباشرة!
    حساب.الرصيد = -5000  # رصيد سالب!
    حساب.الرصيد = 999999999  # رصيد غير واقعي!
    
    # لا يوجد تحقق أو تسجيل للعمليات
نهاية
```

### الحل بالتغليف

```s
صنف حساب_بنكي
    خاص نص _رقم_الحساب
    خاص رقم _الرصيد
    خاص قائمة _سجل_العمليات = []
    
    منشئ(نص رقم، رقم رصيد_ابتدائي)
        هذا._رقم_الحساب = رقم
        هذا._الرصيد = رصيد_ابتدائي
        هذا._سجل("إنشاء الحساب"، رصيد_ابتدائي)
    نهاية
    
    # قراءة الرصيد فقط
    عام دالة الرصيد()
        إرجاع هذا._الرصيد
    نهاية
    
    # قراءة رقم الحساب فقط
    عام دالة رقم_الحساب()
        إرجاع هذا._رقم_الحساب
    نهاية
    
    عام دالة إيداع(رقم مبلغ)
        إذا (مبلغ <= 0)
            ارفع خطأ("المبلغ يجب أن يكون موجباً")
        نهاية
        
        هذا._الرصيد = هذا._الرصيد + مبلغ
        هذا._سجل("إيداع"، مبلغ)
        إرجاع صحيح
    نهاية
    
    عام دالة سحب(رقم مبلغ)
        إذا (مبلغ <= 0)
            ارفع خطأ("المبلغ يجب أن يكون موجباً")
        نهاية
        
        إذا (مبلغ > هذا._الرصيد)
            ارفع خطأ("الرصيد غير كافٍ")
        نهاية
        
        هذا._الرصيد = هذا._الرصيد - مبلغ
        هذا._سجل("سحب"، مبلغ)
        إرجاع صحيح
    نهاية
    
    # دالة خاصة للتسجيل
    خاص دالة _سجل(نص العملية، رقم المبلغ)
        هذا._سجل_العمليات.أضف({
            "العملية": العملية،
            "المبلغ": المبلغ،
            "الرصيد_بعدها": هذا._الرصيد،
            "التاريخ": تاريخ_الآن()
        })
    نهاية
    
    عام دالة كشف_حساب()
        إرجاع هذا._سجل_العمليات.نسخة()
    نهاية
نهاية

دالة رئيسية()
    متغير حساب = جديد حساب_بنكي("SA001"، 1000)
    
    # الآن لا يمكن الوصول للرصيد مباشرة
    # حساب._الرصيد = 5000  # خطأ! خاصية خاصة
    
    # يجب استخدام الدوال العامة
    حساب.إيداع(500)
    حساب.سحب(200)
    
    اطبع("الرصيد: " + حساب.الرصيد())
نهاية
```

## مستويات الوصول

### خاص (Private)

لا يمكن الوصول إليه إلا من داخل الصنف:

```s
صنف مثال
    خاص رقم _قيمة_سرية = 42
    
    خاص دالة _عملية_داخلية()
        إرجاع هذا._قيمة_سرية * 2
    نهاية
    
    عام دالة احصل_على_النتيجة()
        # يمكن الوصول للخاص من داخل الصنف
        إرجاع هذا._عملية_داخلية()
    نهاية
نهاية

دالة رئيسية()
    متغير م = جديد مثال()
    
    # م._قيمة_سرية  # خطأ!
    # م._عملية_داخلية()  # خطأ!
    
    اطبع(م.احصل_على_النتيجة())  # يعمل: 84
نهاية
```

### محمي (Protected)

يمكن الوصول إليه من الصنف والأصناف الوارثة:

```s
صنف حيوان
    محمي نص _الاسم
    محمي رقم _العمر
    
    منشئ(نص اسم، رقم عمر)
        هذا._الاسم = اسم
        هذا._العمر = عمر
    نهاية
    
    محمي دالة _تحقق_من_العمر()
        إرجاع هذا._العمر > 0
    نهاية
نهاية

صنف قط يرث حيوان
    منشئ(نص اسم، رقم عمر)
        أبي(اسم، عمر)
    نهاية
    
    عام دالة معلومات()
        # يمكن الوصول للمحمي من الصنف الابن
        إذا (هذا._تحقق_من_العمر())
            إرجاع هذا._الاسم + " (" + هذا._العمر + " سنوات)"
        نهاية
        إرجاع "بيانات غير صالحة"
    نهاية
نهاية

دالة رئيسية()
    متغير ق = جديد قط("ميشو"، 3)
    اطبع(ق.معلومات())  # يعمل
    
    # ق._الاسم  # خطأ! محمي
نهاية
```

### عام (Public)

يمكن الوصول إليه من أي مكان:

```s
صنف شخص
    عام نص الاسم  # أو بدون كلمة "عام"
    
    عام دالة قل_مرحبا()
        اطبع("مرحباً، أنا " + هذا.الاسم)
    نهاية
نهاية

دالة رئيسية()
    متغير ش = جديد شخص()
    ش.الاسم = "أحمد"  # يعمل
    ش.قل_مرحبا()      # يعمل
نهاية
```

## خصائص القراءة والكتابة (Getters & Setters)

```s
صنف درجة_الحرارة
    خاص رقم _سلزيوس
    
    منشئ(رقم درجة = 0)
        هذا._سلزيوس = درجة
    نهاية
    
    # Getter للسلزيوس
    عام دالة سلزيوس()
        إرجاع هذا._سلزيوس
    نهاية
    
    # Setter للسلزيوس مع تحقق
    عام دالة عين_سلزيوس(رقم درجة)
        إذا (درجة < -273.15)
            ارفع خطأ("لا يمكن أن تكون درجة الحرارة أقل من الصفر المطلق")
        نهاية
        هذا._سلزيوس = درجة
    نهاية
    
    # Getter محسوب للفهرنهايت
    عام دالة فهرنهايت()
        إرجاع هذا._سلزيوس * 9 / 5 + 32
    نهاية
    
    # Setter للفهرنهايت
    عام دالة عين_فهرنهايت(رقم درجة)
        هذا.عين_سلزيوس((درجة - 32) * 5 / 9)
    نهاية
    
    # Getter محسوب للكلفن
    عام دالة كلفن()
        إرجاع هذا._سلزيوس + 273.15
    نهاية
نهاية

دالة رئيسية()
    متغير حرارة = جديد درجة_الحرارة(25)
    
    اطبع("سلزيوس: " + حرارة.سلزيوس())     # 25
    اطبع("فهرنهايت: " + حرارة.فهرنهايت())  # 77
    اطبع("كلفن: " + حرارة.كلفن())         # 298.15
    
    حرارة.عين_فهرنهايت(100)
    اطبع("سلزيوس: " + حرارة.سلزيوس())     # 37.78
نهاية
```

## أمثلة عملية

### التحقق من صحة البيانات

```s
صنف مستخدم
    خاص نص _اسم_المستخدم
    خاص نص _البريد
    خاص نص _كلمة_السر_مشفرة
    خاص رقم _العمر
    
    منشئ(نص اسم، نص بريد، نص كلمة_سر، رقم عمر)
        هذا.عين_اسم_المستخدم(اسم)
        هذا.عين_البريد(بريد)
        هذا.عين_كلمة_السر(كلمة_سر)
        هذا.عين_العمر(عمر)
    نهاية
    
    # ===== اسم المستخدم =====
    عام دالة اسم_المستخدم()
        إرجاع هذا._اسم_المستخدم
    نهاية
    
    عام دالة عين_اسم_المستخدم(نص اسم)
        إذا (طول(اسم) < 3)
            ارفع خطأ("اسم المستخدم يجب أن يكون 3 أحرف على الأقل")
        نهاية
        إذا (طول(اسم) > 20)
            ارفع خطأ("اسم المستخدم يجب أن يكون 20 حرفاً كحد أقصى")
        نهاية
        إذا (!يطابق_نمط(اسم، "^[a-zA-Z0-9_]+$"))
            ارفع خطأ("اسم المستخدم يجب أن يحتوي على حروف وأرقام و _ فقط")
        نهاية
        هذا._اسم_المستخدم = اسم
    نهاية
    
    # ===== البريد =====
    عام دالة البريد()
        إرجاع هذا._البريد
    نهاية
    
    عام دالة عين_البريد(نص بريد)
        إذا (!هذا._تحقق_من_البريد(بريد))
            ارفع خطأ("بريد إلكتروني غير صالح")
        نهاية
        هذا._البريد = بريد
    نهاية
    
    خاص دالة _تحقق_من_البريد(نص بريد)
        إرجاع يطابق_نمط(بريد، "^[^@]+@[^@]+\.[^@]+$")
    نهاية
    
    # ===== كلمة السر =====
    عام دالة عين_كلمة_السر(نص كلمة_سر)
        إذا (طول(كلمة_سر) < 8)
            ارفع خطأ("كلمة السر يجب أن تكون 8 أحرف على الأقل")
        نهاية
        إذا (!يحتوي_على_رقم(كلمة_سر))
            ارفع خطأ("كلمة السر يجب أن تحتوي على رقم")
        نهاية
        # تشفير كلمة السر
        هذا._كلمة_السر_مشفرة = شفر(كلمة_سر)
    نهاية
    
    عام دالة تحقق_من_كلمة_السر(نص كلمة_سر)
        إرجاع شفر(كلمة_سر) == هذا._كلمة_السر_مشفرة
    نهاية
    
    # ===== العمر =====
    عام دالة العمر()
        إرجاع هذا._العمر
    نهاية
    
    عام دالة عين_العمر(رقم عمر)
        إذا (عمر < 13)
            ارفع خطأ("يجب أن يكون العمر 13 سنة على الأقل")
        نهاية
        إذا (عمر > 120)
            ارفع خطأ("عمر غير صالح")
        نهاية
        هذا._العمر = عمر
    نهاية
نهاية
```

### إخفاء التنفيذ

```s
صنف ذاكرة_تخزين
    خاص قاموس _البيانات = {}
    خاص رقم _الحد_الأقصى
    خاص قائمة _ترتيب_الاستخدام = []
    
    منشئ(رقم حد = 100)
        هذا._الحد_الأقصى = حد
    نهاية
    
    عام دالة احفظ(نص مفتاح، متغير قيمة)
        # التنفيذ الداخلي مخفي
        هذا._تحديث_الترتيب(مفتاح)
        هذا._البيانات[مفتاح] = قيمة
        
        إذا (طول(هذا._البيانات) > هذا._الحد_الأقصى)
            هذا._إزالة_الأقدم()
        نهاية
    نهاية
    
    عام دالة احصل(نص مفتاح)
        إذا (مفتاح في هذا._البيانات)
            هذا._تحديث_الترتيب(مفتاح)
            إرجاع هذا._البيانات[مفتاح]
        نهاية
        إرجاع عدم
    نهاية
    
    عام دالة احذف(نص مفتاح)
        إذا (مفتاح في هذا._البيانات)
            احذف هذا._البيانات[مفتاح]
            هذا._ترتيب_الاستخدام.أزل(مفتاح)
            إرجاع صحيح
        نهاية
        إرجاع خطأ
    نهاية
    
    خاص دالة _تحديث_الترتيب(نص مفتاح)
        إذا (مفتاح في هذا._ترتيب_الاستخدام)
            هذا._ترتيب_الاستخدام.أزل(مفتاح)
        نهاية
        هذا._ترتيب_الاستخدام.أضف(مفتاح)
    نهاية
    
    خاص دالة _إزالة_الأقدم()
        إذا (طول(هذا._ترتيب_الاستخدام) > 0)
            نص أقدم = هذا._ترتيب_الاستخدام[0]
            هذا._ترتيب_الاستخدام.أزل_من_البداية()
            احذف هذا._البيانات[أقدم]
        نهاية
    نهاية
نهاية

دالة رئيسية()
    # المستخدم يتعامل مع واجهة بسيطة
    متغير ذاكرة = جديد ذاكرة_تخزين(5)
    
    ذاكرة.احفظ("اسم"، "أحمد")
    ذاكرة.احفظ("عمر"، 25)
    
    اطبع(ذاكرة.احصل("اسم"))  # أحمد
    
    # لا يحتاج لمعرفة كيف تعمل خوارزمية LRU
نهاية
```

### بناء واجهة API مستقرة

```s
صنف محرك_بحث
    خاص نص _فهرس_المسار
    خاص قاموس _الفهرس = {}
    خاص رقم _إصدار_الفهرس = 0
    
    منشئ(نص مسار_الفهرس)
        هذا._فهرس_المسار = مسار_الفهرس
        هذا._حمل_الفهرس()
    نهاية
    
    # واجهة عامة مستقرة
    عام دالة ابحث(نص استعلام)
        قائمة كلمات = هذا._حلل_الاستعلام(استعلام)
        قائمة نتائج = هذا._نفذ_البحث(كلمات)
        إرجاع هذا._رتب_النتائج(نتائج)
    نهاية
    
    عام دالة أضف_مستند(نص معرف، نص محتوى)
        قائمة كلمات = هذا._استخرج_الكلمات(محتوى)
        هذا._فهرس_المستند(معرف، كلمات)
        هذا._إصدار_الفهرس = هذا._إصدار_الفهرس + 1
    نهاية
    
    # التنفيذ الداخلي يمكن تغييره بحرية
    خاص دالة _حمل_الفهرس()
        # تنفيذ تحميل الفهرس
    نهاية
    
    خاص دالة _حلل_الاستعلام(نص استعلام)
        # تنفيذ تحليل الاستعلام
        إرجاع استعلام.قسم(" ")
    نهاية
    
    خاص دالة _نفذ_البحث(قائمة كلمات)
        # تنفيذ البحث
        إرجاع []
    نهاية
    
    خاص دالة _رتب_النتائج(قائمة نتائج)
        # تنفيذ الترتيب
        إرجاع نتائج
    نهاية
    
    خاص دالة _استخرج_الكلمات(نص محتوى)
        إرجاع محتوى.قسم(" ")
    نهاية
    
    خاص دالة _فهرس_المستند(نص معرف، قائمة كلمات)
        # تنفيذ الفهرسة
    نهاية
نهاية
```

## أنماط التغليف الشائعة

### نمط البناء (Builder Pattern)

```s
صنف بناء_سيارة
    خاص نص _الماركة
    خاص نص _الموديل
    خاص نص _اللون = "أبيض"
    خاص رقم _المحرك = 2000
    خاص منطقي _فتحة_سقف = خطأ
    
    عام دالة الماركة(نص ماركة)
        هذا._الماركة = ماركة
        إرجاع هذا  # للسلسلة
    نهاية
    
    عام دالة الموديل(نص موديل)
        هذا._الموديل = موديل
        إرجاع هذا
    نهاية
    
    عام دالة اللون(نص لون)
        هذا._اللون = لون
        إرجاع هذا
    نهاية
    
    عام دالة المحرك(رقم سعة)
        هذا._المحرك = سعة
        إرجاع هذا
    نهاية
    
    عام دالة فتحة_سقف(منطقي نعم)
        هذا._فتحة_سقف = نعم
        إرجاع هذا
    نهاية
    
    عام دالة ابني()
        إذا (!هذا._الماركة أو !هذا._الموديل)
            ارفع خطأ("الماركة والموديل مطلوبان")
        نهاية
        
        إرجاع {
            "الماركة": هذا._الماركة،
            "الموديل": هذا._الموديل،
            "اللون": هذا._اللون،
            "المحرك": هذا._المحرك،
            "فتحة_سقف": هذا._فتحة_سقف
        }
    نهاية
نهاية

دالة رئيسية()
    متغير سيارة = جديد بناء_سيارة()
        .الماركة("تويوتا")
        .الموديل("كامري")
        .اللون("أسود")
        .المحرك(2500)
        .فتحة_سقف(صحيح)
        .ابني()
    
    اطبع(سيارة)
نهاية
```

## الخلاصة

التغليف في لغة ص:
- إخفاء التفاصيل الداخلية
- توفير واجهة عامة مستقرة
- التحقق من صحة البيانات
- حماية حالة الكائن
- مستويات وصول: عام، محمي، خاص
- Getters و Setters للتحكم
- يسهل تغيير التنفيذ دون كسر الكود
