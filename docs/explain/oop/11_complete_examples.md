# أمثلة عملية شاملة - البرمجة الكائنية في لغة ص

## مقدمة

هذا القسم يُقدّم مجموعة من المشاريع العملية الكاملة التي تُطبّق جميع مفاهيم البرمجة الكائنية التي تعلمناها. كل مثال يُركّز على سيناريو واقعي يمكنك الاستفادة منه في مشاريعك الخاصة.

## المشروع الأول: نظام إدارة المكتبة

### الوصف
نظام متكامل لإدارة المكتبة يشمل الكتب، الأعضاء، عمليات الاستعارة والإرجاع.

```
# ═══════════════════════════════════════════════════════════════
# نظام إدارة المكتبة - مثال شامل
# ═══════════════════════════════════════════════════════════════

صنف كتاب
    رقم الرقم
    نص العنوان
    نص المؤلف
    نص الفئة
    رقم سنة_النشر
    منطقي متاح
    نص المستعير
    
    باني(رقم رقم، نص عنوان، نص مؤلف، نص فئة، رقم سنة)
        الرقم = رقم
        العنوان = عنوان
        المؤلف = مؤلف
        الفئة = فئة
        سنة_النشر = سنة
        متاح = صحيح
        المستعير = ""
    نهاية
    
    دالة منطقي استعارة(نص اسم_المستعير)
        إذا !متاح
            اطبع_سطر("❌ الكتاب غير متاح - مُستعار بواسطة: " + المستعير)
            ارجع خطأ
        نهاية
        
        متاح = خطأ
        المستعير = اسم_المستعير
        اطبع_سطر("✓ تمت استعارة \"" + العنوان + "\" بواسطة " + اسم_المستعير)
        ارجع صحيح
    نهاية
    
    دالة منطقي إرجاع()
        إذا متاح
            اطبع_سطر("⚠️ الكتاب ليس مُستعاراً")
            ارجع خطأ
        نهاية
        
        اطبع_سطر("✓ تم إرجاع \"" + العنوان + "\" من " + المستعير)
        متاح = صحيح
        المستعير = ""
        ارجع صحيح
    نهاية
    
    دالة عرض()
        اطبع_سطر("┌────────────────────────────────────────┐")
        اطبع_سطر("│ 📖 " + العنوان)
        اطبع_سطر("├────────────────────────────────────────┤")
        اطبع_سطر("│ الرقم: " + الرقم)
        اطبع_سطر("│ المؤلف: " + المؤلف)
        اطبع_سطر("│ الفئة: " + الفئة)
        اطبع_سطر("│ سنة النشر: " + سنة_النشر)
        إذا متاح
            اطبع_سطر("│ الحالة: متاح ✅")
        وإلا
            اطبع_سطر("│ الحالة: مُستعار ❌")
            اطبع_سطر("│ المستعير: " + المستعير)
        نهاية
        اطبع_سطر("└────────────────────────────────────────┘")
    نهاية
نهاية

صنف عضو
    رقم الرقم
    نص الاسم
    نص البريد
    نص الهاتف
    رقم عدد_الكتب_المستعارة
    رقم الحد_الأقصى
    
    باني(رقم رقم، نص اسم، نص بريد، نص هاتف)
        الرقم = رقم
        الاسم = اسم
        البريد = بريد
        الهاتف = هاتف
        عدد_الكتب_المستعارة = 0
        الحد_الأقصى = 5
    نهاية
    
    دالة منطقي يمكن_الاستعارة()
        ارجع عدد_الكتب_المستعارة < الحد_الأقصى
    نهاية
    
    دالة استعارة_كتاب(كتاب ك)
        إذا !يمكن_الاستعارة()
            اطبع_سطر("❌ وصلت للحد الأقصى من الاستعارات (" + الحد_الأقصى + " كتب)")
            ارجع
        نهاية
        
        إذا ك.استعارة(الاسم)
            عدد_الكتب_المستعارة = عدد_الكتب_المستعارة + 1
        نهاية
    نهاية
    
    دالة إرجاع_كتاب(كتاب ك)
        إذا ك.إرجاع()
            عدد_الكتب_المستعارة = عدد_الكتب_المستعارة - 1
        نهاية
    نهاية
    
    دالة عرض()
        اطبع_سطر("┌────────────────────────────────────────┐")
        اطبع_سطر("│ 👤 " + الاسم)
        اطبع_سطر("├────────────────────────────────────────┤")
        اطبع_سطر("│ رقم العضوية: " + الرقم)
        اطبع_سطر("│ البريد: " + البريد)
        اطبع_سطر("│ الهاتف: " + الهاتف)
        اطبع_سطر("│ الكتب المستعارة: " + عدد_الكتب_المستعارة + "/" + الحد_الأقصى)
        اطبع_سطر("└────────────────────────────────────────┘")
    نهاية
نهاية

دالة رئيسية()
    اطبع_سطر("╔════════════════════════════════════════════╗")
    اطبع_سطر("║         نظام إدارة المكتبة                ║")
    اطبع_سطر("╚════════════════════════════════════════════╝\n")
    
    # إنشاء الكتب
    كتاب ك1 = جديد كتاب(1، "البرمجة بلغة ص"، "أحمد محمد"، "برمجة"، 2024)
    كتاب ك2 = جديد كتاب(2، "الذكاء الاصطناعي"، "سارة علي"، "تقنية"، 2023)
    كتاب ك3 = جديد كتاب(3، "تعلم الخوارزميات"، "خالد العمر"، "برمجة"، 2022)
    
    # إنشاء الأعضاء
    عضو ع1 = جديد عضو(101، "محمد أحمد"، "mohammed@email.com"، "0501234567")
    عضو ع2 = جديد عضو(102، "نورة السالم"، "noura@email.com"، "0507654321")
    
    # عرض الكتب
    اطبع_سطر("═══════════════════════════════════════════")
    اطبع_سطر("              الكتب المتاحة                ")
    اطبع_سطر("═══════════════════════════════════════════\n")
    ك1.عرض()
    ك2.عرض()
    ك3.عرض()
    
    # عرض الأعضاء
    اطبع_سطر("\n═══════════════════════════════════════════")
    اطبع_سطر("               الأعضاء                     ")
    اطبع_سطر("═══════════════════════════════════════════\n")
    ع1.عرض()
    ع2.عرض()
    
    # عمليات الاستعارة
    اطبع_سطر("\n═══════════════════════════════════════════")
    اطبع_سطر("            عمليات الاستعارة               ")
    اطبع_سطر("═══════════════════════════════════════════\n")
    
    ع1.استعارة_كتاب(ك1)
    ع1.استعارة_كتاب(ك2)
    ع2.استعارة_كتاب(ك1)  # سيفشل - الكتاب مُستعار
    ع2.استعارة_كتاب(ك3)
    
    # عرض الحالة بعد الاستعارة
    اطبع_سطر("\n═══════════════════════════════════════════")
    اطبع_سطر("         الحالة بعد الاستعارة              ")
    اطبع_سطر("═══════════════════════════════════════════\n")
    ك1.عرض()
    ع1.عرض()
    
    # إرجاع كتاب
    اطبع_سطر("\n═══════════════════════════════════════════")
    اطبع_سطر("            عمليات الإرجاع                 ")
    اطبع_سطر("═══════════════════════════════════════════\n")
    
    ع1.إرجاع_كتاب(ك1)
    ك1.عرض()
    
    ارجع 0
نهاية
```

## المشروع الثاني: نظام إدارة المطعم

### الوصف
نظام لإدارة طلبات المطعم يشمل القائمة، الطلبات، الفواتير.

```
# ═══════════════════════════════════════════════════════════════
# نظام إدارة المطعم - مثال شامل
# ═══════════════════════════════════════════════════════════════

صنف صنف_طعام
    رقم الرقم
    نص الاسم
    نص الوصف
    عشري السعر
    نص الفئة  # "مقبلات"، "أطباق رئيسية"، "مشروبات"، "حلويات"
    منطقي متاح
    
    باني(رقم رقم، نص اسم، نص وصف، عشري سعر، نص فئة)
        الرقم = رقم
        الاسم = اسم
        الوصف = وصف
        السعر = سعر
        الفئة = فئة
        متاح = صحيح
    نهاية
    
    دالة عرض()
        نص الحالة = ""
        إذا متاح
            الحالة = "✅"
        وإلا
            الحالة = "❌ غير متاح"
        نهاية
        اطبع_سطر("  " + الرقم + ". " + الاسم + " - " + السعر + " ريال " + الحالة)
        اطبع_سطر("     " + الوصف)
    نهاية
نهاية

صنف عنصر_طلب
    صنف_طعام الصنف
    رقم الكمية
    نص ملاحظات
    
    باني(صنف_طعام صنف، رقم كمية)
        الصنف = صنف
        الكمية = كمية
        ملاحظات = ""
    نهاية
    
    دالة إضافة_ملاحظة(نص ملاحظة)
        ملاحظات = ملاحظة
    نهاية
    
    دالة عشري الإجمالي()
        ارجع الصنف.السعر * الكمية
    نهاية
    
    دالة عرض()
        اطبع_سطر("  • " + الصنف.الاسم + " × " + الكمية + " = " + الإجمالي() + " ريال")
        إذا طول(ملاحظات) > 0
            اطبع_سطر("    📝 " + ملاحظات)
        نهاية
    نهاية
نهاية

صنف طلب
    رقم رقم_الطلب
    رقم رقم_الطاولة
    نص الحالة  # "جديد"، "قيد التحضير"، "جاهز"، "مُقدّم"، "مدفوع"
    عشري المجموع
    عشري الضريبة
    رقم عدد_العناصر
    
    باني(رقم رقم، رقم طاولة)
        رقم_الطلب = رقم
        رقم_الطاولة = طاولة
        الحالة = "جديد"
        المجموع = 0
        الضريبة = 0
        عدد_العناصر = 0
    نهاية
    
    دالة إضافة_عنصر(عنصر_طلب عنصر)
        إذا !عنصر.الصنف.متاح
            اطبع_سطر("❌ عذراً، " + عنصر.الصنف.الاسم + " غير متاح حالياً")
            ارجع
        نهاية
        
        المجموع = المجموع + عنصر.الإجمالي()
        عدد_العناصر = عدد_العناصر + 1
        اطبع_سطر("✓ تم إضافة " + عنصر.الصنف.الاسم + " × " + عنصر.الكمية)
    نهاية
    
    دالة تحديث_الحالة(نص حالة_جديدة)
        الحالة = حالة_جديدة
        اطبع_سطر("📋 طلب #" + رقم_الطلب + " - الحالة: " + حالة_جديدة)
    نهاية
    
    دالة عشري حساب_الضريبة()
        الضريبة = المجموع * 0.15
        ارجع الضريبة
    نهاية
    
    دالة عشري الإجمالي_النهائي()
        ارجع المجموع + حساب_الضريبة()
    نهاية
    
    دالة طباعة_الفاتورة()
        اطبع_سطر("\n╔════════════════════════════════════════╗")
        اطبع_سطر("║            فاتورة المطعم               ║")
        اطبع_سطر("╠════════════════════════════════════════╣")
        اطبع_سطر("║ رقم الطلب: " + رقم_الطلب)
        اطبع_سطر("║ رقم الطاولة: " + رقم_الطاولة)
        اطبع_سطر("╠════════════════════════════════════════╣")
        اطبع_سطر("║ عدد العناصر: " + عدد_العناصر)
        اطبع_سطر("║ ────────────────────────────────────── ")
        اطبع_سطر("║ المجموع الفرعي: " + المجموع + " ريال")
        اطبع_سطر("║ ضريبة القيمة المضافة (15%): " + الضريبة + " ريال")
        اطبع_سطر("║ ════════════════════════════════════════")
        اطبع_سطر("║ الإجمالي: " + الإجمالي_النهائي() + " ريال")
        اطبع_سطر("╚════════════════════════════════════════╝")
        اطبع_سطر("           شكراً لزيارتكم! 🙏            ")
    نهاية
نهاية

دالة رئيسية()
    اطبع_سطر("╔════════════════════════════════════════════╗")
    اطبع_سطر("║        🍽️  مطعم لغة ص  🍽️               ║")
    اطبع_سطر("╚════════════════════════════════════════════╝\n")
    
    # إنشاء القائمة
    صنف_طعام ط1 = جديد صنف_طعام(1، "سلطة سيزر"، "سلطة طازجة مع صوص سيزر"، 25، "مقبلات")
    صنف_طعام ط2 = جديد صنف_طعام(2، "شوربة عدس"، "شوربة عدس ساخنة"، 18، "مقبلات")
    صنف_طعام ط3 = جديد صنف_طعام(3، "ستيك لحم"، "ستيك لحم مشوي مع الخضار"، 120، "أطباق رئيسية")
    صنف_طعام ط4 = جديد صنف_طعام(4، "دجاج مشوي"، "نصف دجاجة مشوية مع الأرز"، 65، "أطباق رئيسية")
    صنف_طعام ط5 = جديد صنف_طعام(5، "عصير برتقال"، "عصير برتقال طبيعي"، 15، "مشروبات")
    صنف_طعام ط6 = جديد صنف_طعام(6، "كنافة"، "كنافة نابلسية ساخنة"، 30، "حلويات")
    
    # عرض القائمة
    اطبع_سطر("═══════════════════════════════════════════")
    اطبع_سطر("              📜 قائمة الطعام              ")
    اطبع_سطر("═══════════════════════════════════════════\n")
    
    اطبع_سطر("🥗 المقبلات:")
    ط1.عرض()
    ط2.عرض()
    
    اطبع_سطر("\n🍖 الأطباق الرئيسية:")
    ط3.عرض()
    ط4.عرض()
    
    اطبع_سطر("\n🥤 المشروبات:")
    ط5.عرض()
    
    اطبع_سطر("\n🍰 الحلويات:")
    ط6.عرض()
    
    # إنشاء طلب جديد
    اطبع_سطر("\n═══════════════════════════════════════════")
    اطبع_سطر("              🛒 طلب جديد                  ")
    اطبع_سطر("═══════════════════════════════════════════\n")
    
    طلب طلب1 = جديد طلب(1001، 5)
    
    # إضافة عناصر للطلب
    عنصر_طلب ع1 = جديد عنصر_طلب(ط1، 1)
    عنصر_طلب ع2 = جديد عنصر_طلب(ط3، 2)
    ع2.إضافة_ملاحظة("متوسط النضج")
    عنصر_طلب ع3 = جديد عنصر_طلب(ط5، 3)
    عنصر_طلب ع4 = جديد عنصر_طلب(ط6، 2)
    
    طلب1.إضافة_عنصر(ع1)
    طلب1.إضافة_عنصر(ع2)
    طلب1.إضافة_عنصر(ع3)
    طلب1.إضافة_عنصر(ع4)
    
    # تحديث حالة الطلب
    اطبع_سطر("\n═══════════════════════════════════════════")
    اطبع_سطر("            📊 تتبع الطلب                  ")
    اطبع_سطر("═══════════════════════════════════════════\n")
    
    طلب1.تحديث_الحالة("قيد التحضير")
    طلب1.تحديث_الحالة("جاهز")
    طلب1.تحديث_الحالة("مُقدّم")
    
    # طباعة الفاتورة
    طلب1.طباعة_الفاتورة()
    
    ارجع 0
نهاية

## المشروع الثالث: نظام التجارة الإلكترونية

### الوصف
نظام متكامل للتجارة الإلكترونية يشمل المنتجات، سلة التسوق، الطلبات.

```
# ═══════════════════════════════════════════════════════════════
# نظام التجارة الإلكترونية - مثال شامل
# ═══════════════════════════════════════════════════════════════

صنف منتج
    رقم الرقم
    نص الاسم
    نص الوصف
    عشري السعر
    رقم الكمية_المتاحة
    نص الفئة
    عشري التقييم
    
    باني(رقم رقم، نص اسم، نص وصف، عشري سعر، رقم كمية، نص فئة)
        الرقم = رقم
        الاسم = اسم
        الوصف = وصف
        السعر = سعر
        الكمية_المتاحة = كمية
        الفئة = فئة
        التقييم = 0
    نهاية
    
    دالة منطقي متاح()
        ارجع الكمية_المتاحة > 0
    نهاية
    
    دالة نقص_الكمية(رقم كمية)
        إذا كمية > الكمية_المتاحة
            اطبع_سطر("❌ الكمية المطلوبة غير متوفرة")
            ارجع
        نهاية
        الكمية_المتاحة = الكمية_المتاحة - كمية
    نهاية
    
    دالة نص نجوم_التقييم()
        رقم نجوم = التقييم
        نص نتيجة = ""
        رقم ف = 0
        طالما ف < 5
            إذا ف < نجوم
                نتيجة = نتيجة + "⭐"
            وإلا
                نتيجة = نتيجة + "☆"
            نهاية
            ف = ف + 1
        نهاية
        ارجع نتيجة
    نهاية
    
    دالة عرض()
        اطبع_سطر("┌────────────────────────────────────────┐")
        اطبع_سطر("│ 📦 " + الاسم)
        اطبع_سطر("├────────────────────────────────────────┤")
        اطبع_سطر("│ " + الوصف)
        اطبع_سطر("│ السعر: " + السعر + " ريال")
        اطبع_سطر("│ المتاح: " + الكمية_المتاحة + " قطعة")
        اطبع_سطر("│ التقييم: " + نجوم_التقييم())
        اطبع_سطر("└────────────────────────────────────────┘")
    نهاية
نهاية

صنف عنصر_سلة
    منتج المنتج
    رقم الكمية
    
    باني(منتج م، رقم كمية)
        المنتج = م
        الكمية = كمية
    نهاية
    
    دالة عشري الإجمالي()
        ارجع المنتج.السعر * الكمية
    نهاية
نهاية

صنف سلة_تسوق
    رقم عدد_العناصر
    عشري المجموع
    نص كود_الخصم
    عشري نسبة_الخصم
    
    باني()
        عدد_العناصر = 0
        المجموع = 0
        كود_الخصم = ""
        نسبة_الخصم = 0
    نهاية
    
    دالة إضافة(منتج م، رقم كمية)
        إذا !م.متاح()
            اطبع_سطر("❌ المنتج غير متاح")
            ارجع
        نهاية
        
        إذا كمية > م.الكمية_المتاحة
            اطبع_سطر("❌ الكمية المطلوبة غير متوفرة")
            ارجع
        نهاية
        
        عنصر_سلة عنصر = جديد عنصر_سلة(م، كمية)
        المجموع = المجموع + عنصر.الإجمالي()
        عدد_العناصر = عدد_العناصر + 1
        اطبع_سطر("✓ تم إضافة " + م.الاسم + " × " + كمية + " للسلة")
    نهاية
    
    دالة منطقي تطبيق_كود_خصم(نص كود)
        إذا كود == "WELCOME10"
