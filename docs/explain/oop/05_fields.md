# الحقول في لغة ص - الدليل الشامل

## ما هي الحقول؟

الحقول (Fields) هي المتغيرات التي تُعرَّف داخل الصنف وتُخزّن حالة الكائن وبياناته. كل كائن يُنشأ من الصنف يمتلك نسخته الخاصة من هذه الحقول، مما يعني أن تغيير قيمة حقل في كائن لا يؤثر على نفس الحقل في كائنات أخرى من نفس الصنف.

يمكنك تشبيه الحقول بالخصائص الفيزيائية للأشياء. مثلاً، كل سيارة لها لون وسرعة وعدد أبواب - هذه هي "حقول" السيارة. كل سيارة محددة (كائن) لها قيمها الخاصة لهذه الحقول.

## تعريف الحقول في لغة ص

لتعريف حقل في صنف، نكتب نوع البيانات متبوعاً باسم الحقل:

```
صنف اسم_الصنف
    نوع_البيانات اسم_الحقل
نهاية
```

### أنواع البيانات المدعومة للحقول

```
صنف مثال_الأنواع
    # الأنواع الأساسية
    رقم عدد_صحيح
    عشري عدد_عشري
    نص سلسلة_نصية
    منطقي قيمة_منطقية
    
    # الأنواع المركبة (كائنات من أصناف أخرى)
    نقطة موقع
    تاريخ تاريخ_الإنشاء
نهاية
```

## مثال أساسي: حقول بسيطة

```
صنف سيارة
    نص الماركة
    نص الموديل
    رقم سنة_الصنع
    نص اللون
    رقم عدد_الأبواب
    عشري السعر
نهاية

دالة رئيسية()
    سيارة سيارتي = جديد سيارة()
    
    # تعيين قيم الحقول
    سيارتي.الماركة = "تويوتا"
    سيارتي.الموديل = "كامري"
    سيارتي.سنة_الصنع = 2024
    سيارتي.اللون = "أبيض"
    سيارتي.عدد_الأبواب = 4
    سيارتي.السعر = 120000.50
    
    # قراءة قيم الحقول
    اطبع_سطر("السيارة: " + سيارتي.الماركة + " " + سيارتي.الموديل)
    اطبع_سطر("سنة الصنع: " + سيارتي.سنة_الصنع)
    اطبع_سطر("اللون: " + سيارتي.اللون)
    اطبع_سطر("عدد الأبواب: " + سيارتي.عدد_الأبواب)
    اطبع_سطر("السعر: " + سيارتي.السعر + " ريال")
    
    ارجع 0
نهاية
```

## استقلالية الحقول بين الكائنات

كل كائن يمتلك نسخته الخاصة من الحقول:

```
صنف عداد
    رقم القيمة
    
    باني()
        القيمة = 0
    نهاية
    
    دالة زيادة()
        القيمة = القيمة + 1
    نهاية
    
    دالة عرض()
        اطبع_سطر("القيمة: " + القيمة)
    نهاية
نهاية

دالة رئيسية()
    عداد ع1 = جديد عداد()
    عداد ع2 = جديد عداد()
    
    # زيادة العداد الأول 3 مرات
    ع1.زيادة()
    ع1.زيادة()
    ع1.زيادة()
    
    # زيادة العداد الثاني مرة واحدة
    ع2.زيادة()
    
    # كل عداد يحتفظ بقيمته الخاصة
    اطبع_سطر("العداد 1:")
    ع1.عرض()  # سيطبع: 3
    
    اطبع_سطر("العداد 2:")
    ع2.عرض()  # سيطبع: 1
    
    ارجع 0
نهاية
```

## تهيئة الحقول

### التهيئة في الباني (الطريقة المُفضّلة)

```
صنف منتج
    نص الاسم
    عشري السعر
    رقم الكمية
    نص الفئة
    
    باني(نص اسم، عشري سعر، رقم كمية، نص فئة)
        الاسم = اسم
        السعر = سعر
        الكمية = كمية
        الفئة = فئة
    نهاية
نهاية

دالة رئيسية()
    منتج م = جديد منتج("لابتوب"، 4500.0، 10، "إلكترونيات")
    اطبع_سطر(م.الاسم + " - " + م.السعر + " ريال")
    ارجع 0
نهاية
```

### التهيئة الجزئية في الباني

```
صنف مستخدم
    نص الاسم
    نص البريد
    نص كلمة_المرور
    تاريخ تاريخ_التسجيل
    رقم عدد_التسجيلات
    منطقي نشط
    
    باني(نص اسم، نص بريد، نص كلمة_سر)
        # القيم الإلزامية من المعاملات
        الاسم = اسم
        البريد = بريد
        كلمة_المرور = كلمة_سر
        
        # القيم الافتراضية
        تاريخ_التسجيل = جديد تاريخ(1، 1، 2024)
        عدد_التسجيلات = 1
        نشط = صحيح
    نهاية
نهاية
```

## أنماط الحقول المختلفة

### 1. حقول البيانات الأساسية

```
صنف طالب
    # بيانات تعريفية
    رقم الرقم_الجامعي
    نص الاسم
    
    # بيانات أكاديمية
    عشري المعدل
    رقم الساعات_المكتسبة
    نص التخصص
    
    # بيانات التواصل
    نص البريد_الجامعي
    نص الهاتف
نهاية
```

### 2. حقول الحالة (Status Fields)

```
صنف طلب
    رقم رقم_الطلب
    نص الحالة  # "جديد"، "قيد المعالجة"، "مكتمل"، "ملغي"
    منطقي مدفوع
    منطقي مُرسَل
    
    باني(رقم رقم)
        رقم_الطلب = رقم
        الحالة = "جديد"
        مدفوع = خطأ
        مُرسَل = خطأ
    نهاية
    
    دالة دفع()
        مدفوع = صحيح
        الحالة = "قيد المعالجة"
        اطبع_سطر("تم الدفع بنجاح")
    نهاية
    
    دالة إرسال()
        إذا مدفوع
            مُرسَل = صحيح
            الحالة = "مكتمل"
            اطبع_سطر("تم الإرسال بنجاح")
        وإلا
            اطبع_سطر("خطأ: يجب الدفع أولاً")
        نهاية
    نهاية
نهاية
```

### 3. حقول مُحسَبة (Calculated Fields)

```
صنف مثلث
    عشري الضلع_أ
    عشري الضلع_ب
    عشري الضلع_ج
    عشري المحيط
    عشري المساحة
    
    باني(عشري أ، عشري ب، عشري ج)
        الضلع_أ = أ
        الضلع_ب = ب
        الضلع_ج = ج
        
        # حساب المحيط
        المحيط = أ + ب + ج
        
        # حساب المساحة باستخدام صيغة هيرون
        عشري نصف_المحيط = المحيط / 2
        المساحة = جذر(نصف_المحيط * (نصف_المحيط - أ) * (نصف_المحيط - ب) * (نصف_المحيط - ج))
    نهاية
    
    دالة عرض_المعلومات()
        اطبع_سطر("الأضلاع: " + الضلع_أ + "، " + الضلع_ب + "، " + الضلع_ج)
        اطبع_سطر("المحيط: " + المحيط)
        اطبع_سطر("المساحة: " + المساحة)
    نهاية
نهاية
```

### 4. حقول كائنية (Object Fields)

```
صنف عنوان
    نص المدينة
    نص الحي
    نص الشارع
    رقم رقم_المبنى
    
    باني(نص مدينة، نص حي، نص شارع، رقم رقم)
        المدينة = مدينة
        الحي = حي
        الشارع = شارع
        رقم_المبنى = رقم
    نهاية
    
    دالة نص صيغة_كاملة()
        ارجع رقم_المبنى + "، " + الشارع + "، " + الحي + "، " + المدينة
    نهاية
نهاية

صنف شركة
    نص الاسم
    عنوان المقر_الرئيسي
    عنوان فرع_ثانوي
    رقم عدد_الموظفين
    
    باني(نص اسم، عنوان مقر)
        الاسم = اسم
        المقر_الرئيسي = مقر
        عدد_الموظفين = 0
    نهاية
    
    دالة إضافة_فرع(عنوان فرع)
        فرع_ثانوي = فرع
        اطبع_سطر("تم إضافة فرع جديد في: " + فرع.المدينة)
    نهاية
نهاية
```

## مثال شامل: نظام إدارة المخزون

```
صنف فئة_منتج
    رقم الرقم
    نص الاسم
    نص الوصف
    
    باني(رقم رقم، نص اسم، نص وصف)
        الرقم = رقم
        الاسم = اسم
        الوصف = وصف
    نهاية
نهاية

صنف مورد
    رقم الرقم
    نص الاسم
    نص الهاتف
    نص البريد
    
    باني(رقم رقم، نص اسم، نص هاتف)
        الرقم = رقم
        الاسم = اسم
        الهاتف = هاتف
        البريد = ""
    نهاية
نهاية

صنف منتج
    # معلومات أساسية
    رقم رقم_المنتج
    نص الاسم
    نص الوصف
    نص الباركود
    
    # معلومات السعر
    عشري سعر_الشراء
    عشري سعر_البيع
    عشري نسبة_الربح
    
    # معلومات المخزون
    رقم الكمية_الحالية
    رقم الحد_الأدنى
    منطقي متاح
    
    # علاقات
    فئة_منتج الفئة
    مورد المورد
    
    باني(رقم رقم، نص اسم، عشري سعر_شراء، عشري سعر_بيع، رقم كمية)
        رقم_المنتج = رقم
        الاسم = اسم
        الوصف = ""
        الباركود = ""
        
        سعر_الشراء = سعر_شراء
        سعر_البيع = سعر_بيع
        نسبة_الربح = ((سعر_بيع - سعر_شراء) / سعر_شراء) * 100
        
        الكمية_الحالية = كمية
        الحد_الأدنى = 10
        متاح = كمية > 0
    نهاية
    
    دالة تعيين_الفئة(فئة_منتج فئة)
        الفئة = فئة
        اطبع_سطر("تم تعيين فئة: " + فئة.الاسم)
    نهاية
    
    دالة تعيين_المورد(مورد م)
        المورد = م
        اطبع_سطر("تم تعيين المورد: " + م.الاسم)
    نهاية
    
    دالة إضافة_كمية(رقم كمية)
        الكمية_الحالية = الكمية_الحالية + كمية
        متاح = صحيح
        اطبع_سطر("تم إضافة " + كمية + " وحدة. الكمية الحالية: " + الكمية_الحالية)
    نهاية
    
    دالة منطقي بيع(رقم كمية)
        إذا كمية > الكمية_الحالية
            اطبع_سطر("خطأ: الكمية المطلوبة غير متوفرة")
            ارجع خطأ
        نهاية
        
        الكمية_الحالية = الكمية_الحالية - كمية
        
        إذا الكمية_الحالية == 0
            متاح = خطأ
        نهاية
        
        إذا الكمية_الحالية < الحد_الأدنى
            اطبع_سطر("تحذير: الكمية أقل من الحد الأدنى!")
        نهاية
        
        اطبع_سطر("تم البيع بنجاح. الكمية المتبقية: " + الكمية_الحالية)
        ارجع صحيح
    نهاية
    
    دالة عرض_معلومات_كاملة()
        اطبع_سطر("══════════════════════════════════════")
        اطبع_سطر("         معلومات المنتج              ")
        اطبع_سطر("══════════════════════════════════════")
        اطبع_سطر("رقم المنتج: " + رقم_المنتج)
        اطبع_سطر("الاسم: " + الاسم)
        اطبع_سطر("──────────────────────────────────────")
        اطبع_سطر("سعر الشراء: " + سعر_الشراء + " ريال")
        اطبع_سطر("سعر البيع: " + سعر_البيع + " ريال")
        اطبع_سطر("نسبة الربح: " + نسبة_الربح + "%")
        اطبع_سطر("──────────────────────────────────────")
        اطبع_سطر("الكمية الحالية: " + الكمية_الحالية)
        اطبع_سطر("الحد الأدنى: " + الحد_الأدنى)
        إذا متاح
            اطبع_سطر("الحالة: متاح ✓")
        وإلا
            اطبع_سطر("الحالة: غير متاح ✗")
        نهاية
        اطبع_سطر("══════════════════════════════════════")
    نهاية
نهاية

دالة رئيسية()
    # إنشاء فئة
    فئة_منتج إلكترونيات = جديد فئة_منتج(1، "إلكترونيات"، "الأجهزة الإلكترونية والكهربائية")
    
    # إنشاء مورد
    مورد شركة_توريد = جديد مورد(101، "شركة التقنية للتوريد"، "0500000000")
    
    # إنشاء منتج
    منتج لابتوب = جديد منتج(1001، "لابتوب HP"، 3000.0، 4500.0، 25)
    لابتوب.تعيين_الفئة(إلكترونيات)
    لابتوب.تعيين_المورد(شركة_توريد)
    
    # عرض معلومات المنتج
    لابتوب.عرض_معلومات_كاملة()
    
    # عمليات على المنتج
    اطبع_سطر("\n=== عمليات البيع ===")
    لابتوب.بيع(5)
    لابتوب.بيع(15)
    لابتوب.بيع(10)  # سيظهر تحذير الحد الأدنى
    
    # إعادة التوريد
    اطبع_سطر("\n=== إعادة التوريد ===")
    لابتوب.إضافة_كمية(30)
    
    # عرض المعلومات النهائية
    اطبع_سطر("\n=== المعلومات النهائية ===")
    لابتوب.عرض_معلومات_كاملة()
    
    ارجع 0
نهاية
```

## نصائح للعمل مع الحقول

**اختر أسماء واضحة ووصفية**: `الرصيد_الحالي` أفضل من `ر` أو `رصيد`.

**جمّع الحقول المتعلقة**: ضع الحقول ذات الصلة متجاورة في تعريف الصنف.

**استخدم التعليقات**: أضف تعليقات لشرح الحقول المعقدة أو غير الواضحة.

**هيّئ جميع الحقول**: تأكد من تهيئة كل حقل في الباني لتجنب القيم غير المتوقعة.

## الخلاصة

الحقول هي الأساس الذي تُبنى عليه الكائنات. فهم كيفية تعريفها واستخدامها وتهيئتها بشكل صحيح هو مفتاح كتابة برامج كائنية فعّالة. في القسم التالي، سنتعرف على الدوال وكيفية استخدامها للتعامل مع الحقول.
