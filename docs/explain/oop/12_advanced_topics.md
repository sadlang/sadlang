# مواضيع متقدمة في البرمجة الكائنية - لغة ص

## مقدمة

بعد أن أتقنت أساسيات البرمجة الكائنية، حان الوقت للتعمق في بعض المفاهيم والتقنيات المتقدمة التي ستُساعدك على كتابة كود أكثر احترافية ومرونة.

---

## 1. أنماط التصميم (Design Patterns)

أنماط التصميم هي حلول مجربة لمشاكل شائعة في تصميم البرمجيات. إليك بعض الأنماط المهمة:

### نمط المصنع (Factory Pattern)

يُستخدم لإنشاء كائنات دون تحديد الصنف الدقيق:

```
صنف شكل
    دالة رسم()
        اطبع_سطر("رسم شكل...")
    نهاية
نهاية

صنف دائرة يرث شكل
    دالة رسم()
        اطبع_سطر("رسم دائرة ○")
    نهاية
نهاية

صنف مربع يرث شكل
    دالة رسم()
        اطبع_سطر("رسم مربع □")
    نهاية
نهاية

صنف مثلث يرث شكل
    دالة رسم()
        اطبع_سطر("رسم مثلث △")
    نهاية
نهاية

صنف مصنع_الأشكال
    دالة شكل إنشاء(نص نوع)
        إذا نوع == "دائرة"
            ارجع جديد دائرة()
        نهاية
        
        إذا نوع == "مربع"
            ارجع جديد مربع()
        نهاية
        
        إذا نوع == "مثلث"
            ارجع جديد مثلث()
        نهاية
        
        اطبع_سطر("خطأ: نوع غير معروف")
        ارجع جديد شكل()
    نهاية
نهاية

دالة رئيسية()
    مصنع_الأشكال مصنع = جديد مصنع_الأشكال()
    
    شكل ش1 = مصنع.إنشاء("دائرة")
    شكل ش2 = مصنع.إنشاء("مربع")
    شكل ش3 = مصنع.إنشاء("مثلث")
    
    ش1.رسم()
    ش2.رسم()
    ش3.رسم()
    
    ارجع 0
نهاية
```

### نمط المفرد (Singleton Pattern)

يضمن وجود نسخة واحدة فقط من الصنف:

```
صنف إعدادات
    نص لغة_النظام
    نص المنطقة_الزمنية
    منطقي الوضع_المظلم
    
    باني()
        لغة_النظام = "العربية"
        المنطقة_الزمنية = "Asia/Riyadh"
        الوضع_المظلم = خطأ
    نهاية
    
    دالة عرض()
        اطبع_سطر("═══════════════════════════════════")
        اطبع_سطر("         إعدادات النظام            ")
        اطبع_سطر("═══════════════════════════════════")
        اطبع_سطر("اللغة: " + لغة_النظام)
        اطبع_سطر("المنطقة الزمنية: " + المنطقة_الزمنية)
        إذا الوضع_المظلم
            اطبع_سطر("الوضع: مظلم 🌙")
        وإلا
            اطبع_سطر("الوضع: فاتح ☀️")
        نهاية
    نهاية
نهاية

# في تطبيق حقيقي، نستخدم نمطاً لضمان نسخة واحدة
صنف مدير_الإعدادات
    إعدادات _الإعدادات
    منطقي _تم_التهيئة
    
    باني()
        _تم_التهيئة = خطأ
    نهاية
    
    دالة إعدادات الحصول_على_الإعدادات()
        إذا !_تم_التهيئة
            _الإعدادات = جديد إعدادات()
            _تم_التهيئة = صحيح
        نهاية
        ارجع _الإعدادات
    نهاية
نهاية
```

### نمط المراقب (Observer Pattern)

يُمكّن الكائنات من الاشتراك في أحداث كائن آخر:

```
صنف حدث
    نص النوع
    نص البيانات
    
    باني(نص نوع، نص بيانات)
        النوع = نوع
        البيانات = بيانات
    نهاية
نهاية

صنف مراقب
    نص الاسم
    
    باني(نص اسم)
        الاسم = اسم
    نهاية
    
    دالة عند_الحدث(حدث ح)
        اطبع_سطر("📢 " + الاسم + " استلم حدث: " + ح.النوع)
        اطبع_سطر("   البيانات: " + ح.البيانات)
    نهاية
نهاية

صنف مُرسِل_الأحداث
    رقم عدد_المراقبين
    
    باني()
        عدد_المراقبين = 0
    نهاية
    
    دالة إضافة_مراقب(مراقب م)
        عدد_المراقبين = عدد_المراقبين + 1
        اطبع_سطر("✓ تم تسجيل المراقب: " + م.الاسم)
    نهاية
    
    دالة إرسال_حدث(حدث ح)
        اطبع_سطر("\n📤 إرسال حدث: " + ح.النوع)
        # في التطبيق الحقيقي، نُخطر جميع المراقبين المسجلين
    نهاية
نهاية

دالة رئيسية()
    مُرسِل_الأحداث مُرسِل = جديد مُرسِل_الأحداث()
    
    مراقب م1 = جديد مراقب("لوحة التحكم")
    مراقب م2 = جديد مراقب("نظام الإشعارات")
    مراقب م3 = جديد مراقب("سجل الأحداث")
    
    مُرسِل.إضافة_مراقب(م1)
    مُرسِل.إضافة_مراقب(م2)
    مُرسِل.إضافة_مراقب(م3)
    
    حدث ح = جديد حدث("طلب_جديد"، "طلب رقم 1001")
    مُرسِل.إرسال_حدث(ح)
    
    # محاكاة استلام الحدث
    م1.عند_الحدث(ح)
    م2.عند_الحدث(ح)
    م3.عند_الحدث(ح)
    
    ارجع 0
نهاية
```

## 2. التركيب المتقدم (Advanced Composition)

### بناء أنظمة معقدة من مكونات بسيطة

```
صنف مكون_واجهة
    نص الاسم
    رقم العرض
    رقم الارتفاع
    رقم س
    رقم ص
    
    باني(نص اسم، رقم عرض، رقم ارتفاع)
        الاسم = اسم
        العرض = عرض
        الارتفاع = ارتفاع
        س = 0
        ص = 0
    نهاية
    
    دالة تحريك(رقم س_جديد، رقم ص_جديد)
        س = س_جديد
        ص = ص_جديد
    نهاية
    
    دالة رسم()
        اطبع_سطر("رسم " + الاسم + " في الموقع (" + س + "، " + ص + ")")
    نهاية
نهاية

صنف زر يرث مكون_واجهة
    نص النص
    نص اللون
    
    باني(نص نص، رقم عرض، رقم ارتفاع)
        الاسم = "زر"
        العرض = عرض
        الارتفاع = ارتفاع
        النص = نص
        اللون = "أزرق"
    نهاية
    
    دالة رسم()
        اطبع_سطر("🔘 [" + النص + "] في (" + س + "، " + ص + ")")
    نهاية
    
    دالة نقر()
        اطبع_سطر("تم النقر على الزر: " + النص)
    نهاية
نهاية

صنف حقل_نص يرث مكون_واجهة
    نص القيمة
    نص التلميح
    
    باني(نص تلميح، رقم عرض)
        الاسم = "حقل نص"
        العرض = عرض
        الارتفاع = 30
        التلميح = تلميح
        القيمة = ""
    نهاية
    
    دالة رسم()
        إذا طول(القيمة) == 0
            اطبع_سطر("📝 [" + التلميح + "] في (" + س + "، " + ص + ")")
        وإلا
            اطبع_سطر("📝 [" + القيمة + "] في (" + س + "، " + ص + ")")
        نهاية
    نهاية
    
    دالة تعيين_القيمة(نص قيمة)
        القيمة = قيمة
    نهاية
نهاية

صنف نافذة
    نص العنوان
    رقم العرض
    رقم الارتفاع
    رقم عدد_المكونات
    
    باني(نص عنوان، رقم عرض، رقم ارتفاع)
        العنوان = عنوان
        العرض = عرض
        الارتفاع = ارتفاع
        عدد_المكونات = 0
    نهاية
    
    دالة إضافة_مكون(مكون_واجهة مكون)
        عدد_المكونات = عدد_المكونات + 1
        اطبع_سطر("✓ تم إضافة " + مكون.الاسم + " للنافذة")
    نهاية
    
    دالة رسم()
        اطبع_سطر("\n╔════════════════════════════════════╗")
        اطبع_سطر("║ " + العنوان)
        اطبع_سطر("╠════════════════════════════════════╣")
        اطبع_سطر("║ عدد المكونات: " + عدد_المكونات)
        اطبع_سطر("╚════════════════════════════════════╝")
    نهاية
نهاية

دالة رئيسية()
    اطبع_سطر("═══════════════════════════════════════")
    اطبع_سطر("       نظام بناء واجهات المستخدم       ")
    اطبع_سطر("═══════════════════════════════════════\n")
    
    # إنشاء نافذة
    نافذة نموذج_تسجيل = جديد نافذة("تسجيل الدخول"، 400، 300)
    
    # إنشاء المكونات
    حقل_نص حقل_اسم = جديد حقل_نص("اسم المستخدم"، 200)
    حقل_نص حقل_كلمة_سر = جديد حقل_نص("كلمة المرور"، 200)
    زر زر_دخول = جديد زر("تسجيل الدخول"، 100، 40)
    زر زر_إلغاء = جديد زر("إلغاء"، 100، 40)
    
    # تحديد المواقع
    حقل_اسم.تحريك(100، 50)
    حقل_كلمة_سر.تحريك(100، 100)
    زر_دخول.تحريك(100، 150)
    زر_إلغاء.تحريك(210، 150)
    
    # إضافة المكونات للنافذة
    نموذج_تسجيل.إضافة_مكون(حقل_اسم)
    نموذج_تسجيل.إضافة_مكون(حقل_كلمة_سر)
    نموذج_تسجيل.إضافة_مكون(زر_دخول)
    نموذج_تسجيل.إضافة_مكون(زر_إلغاء)
    
    # رسم النافذة ومكوناتها
    نموذج_تسجيل.رسم()
    
    اطبع_سطر("\nمكونات النموذج:")
    حقل_اسم.رسم()
    حقل_كلمة_سر.رسم()
    زر_دخول.رسم()
    زر_إلغاء.رسم()
    
    # محاكاة التفاعل
    اطبع_سطر("\n--- محاكاة التفاعل ---")
    حقل_اسم.تعيين_القيمة("admin")
    حقل_كلمة_سر.تعيين_القيمة("****")
    زر_دخول.نقر()
    
    ارجع 0
نهاية
```

## 3. سلاسل المسؤولية (Chain of Responsibility)

```
صنف معالج_طلب
    نص الاسم
    رقم الحد_الأقصى
    
    باني(نص اسم، رقم حد)
        الاسم = اسم
        الحد_الأقصى = حد
    نهاية
    
    دالة منطقي يمكن_المعالجة(عشري مبلغ)
        ارجع مبلغ <= الحد_الأقصى
    نهاية
    
    دالة معالجة(عشري مبلغ)
        إذا يمكن_المعالجة(مبلغ)
            اطبع_سطر("✓ " + الاسم + " وافق على طلب بقيمة " + مبلغ + " ريال")
        وإلا
            اطبع_سطر("⚠️ " + الاسم + ": المبلغ يتجاوز صلاحياتي")
        نهاية
    نهاية
نهاية

صنف موظف يرث معالج_طلب
    باني()
        الاسم = "موظف"
        الحد_الأقصى = 1000
    نهاية
نهاية

صنف مشرف يرث معالج_طلب
    باني()
        الاسم = "مشرف"
        الحد_الأقصى = 5000
    نهاية
نهاية

صنف مدير يرث معالج_طلب
    باني()
        الاسم = "مدير"
        الحد_الأقصى = 20000
    نهاية
نهاية

صنف مدير_عام يرث معالج_طلب
    باني()
        الاسم = "مدير عام"
        الحد_الأقصى = 100000
    نهاية
نهاية

صنف سلسلة_الموافقات
    موظف المستوى_الأول
    مشرف المستوى_الثاني
    مدير المستوى_الثالث
    مدير_عام المستوى_الرابع
    
    باني()
        المستوى_الأول = جديد موظف()
        المستوى_الثاني = جديد مشرف()
        المستوى_الثالث = جديد مدير()
        المستوى_الرابع = جديد مدير_عام()
    نهاية
    
    دالة معالجة_طلب(عشري مبلغ)
        اطبع_سطر("\n📋 معالجة طلب بقيمة " + مبلغ + " ريال:")
        اطبع_سطر("────────────────────────────────────")
        
        إذا المستوى_الأول.يمكن_المعالجة(مبلغ)
            المستوى_الأول.معالجة(مبلغ)
            ارجع
        نهاية
        المستوى_الأول.معالجة(مبلغ)
        
        إذا المستوى_الثاني.يمكن_المعالجة(مبلغ)
            المستوى_الثاني.معالجة(مبلغ)
            ارجع
        نهاية
        المستوى_الثاني.معالجة(مبلغ)
        
        إذا المستوى_الثالث.يمكن_المعالجة(مبلغ)
            المستوى_الثالث.معالجة(مبلغ)
            ارجع
        نهاية
        المستوى_الثالث.معالجة(مبلغ)
        
        إذا المستوى_الرابع.يمكن_المعالجة(مبلغ)
            المستوى_الرابع.معالجة(مبلغ)
            ارجع
        نهاية
        المستوى_الرابع.معالجة(مبلغ)
        اطبع_سطر("❌ الطلب يتجاوز جميع الصلاحيات!")
    نهاية
نهاية

دالة رئيسية()
    اطبع_سطر("═══════════════════════════════════════════")
    اطبع_سطر("       نظام الموافقات على المشتريات       ")
    اطبع_سطر("═══════════════════════════════════════════")
    
    سلسلة_الموافقات نظام = جديد سلسلة_الموافقات()
    
    نظام.معالجة_طلب(500)
    نظام.معالجة_طلب(3000)
    نظام.معالجة_طلب(15000)
    نظام.معالجة_طلب(75000)
    نظام.معالجة_طلب(150000)
    
    ارجع 0
نهاية
```

## 4. الحالة (State Pattern)

```
صنف حالة_المهمة
    نص الاسم
    
    باني(نص اسم)
        الاسم = اسم
    نهاية
    
    دالة منطقي يمكن_البدء()
        ارجع خطأ
    نهاية
    
    دالة منطقي يمكن_الإيقاف()
        ارجع خطأ
    نهاية
    
    دالة منطقي يمكن_الإكمال()
        ارجع خطأ
    نهاية
    
    دالة منطقي يمكن_الإلغاء()
        ارجع خطأ
    نهاية
نهاية

صنف حالة_جديدة يرث حالة_المهمة
    باني()
        الاسم = "جديدة"
    نهاية
    
    دالة منطقي يمكن_البدء()
        ارجع صحيح
    نهاية
    
    دالة منطقي يمكن_الإلغاء()
        ارجع صحيح
    نهاية
نهاية

صنف حالة_قيد_التنفيذ يرث حالة_المهمة
    باني()
        الاسم = "قيد التنفيذ"
    نهاية
    
    دالة منطقي يمكن_الإيقاف()
        ارجع صحيح
    نهاية
    
    دالة منطقي يمكن_الإكمال()
        ارجع صحيح
    نهاية
نهاية

صنف حالة_متوقفة يرث حالة_المهمة
    باني()
        الاسم = "متوقفة"
    نهاية
    
    دالة منطقي يمكن_البدء()
        ارجع صحيح
    نهاية
    
    دالة منطقي يمكن_الإلغاء()
        ارجع صحيح
    نهاية
نهاية

صنف حالة_مكتملة يرث حالة_المهمة
    باني()
        الاسم = "مكتملة"
    نهاية
نهاية

صنف حالة_ملغاة يرث حالة_المهمة
    باني()
        الاسم = "ملغاة"
    نهاية
نهاية

صنف مهمة
    نص العنوان
    حالة_المهمة الحالة
    
    باني(نص عنوان)
        العنوان = عنوان
        الحالة = جديد حالة_جديدة()
    نهاية
    
    دالة بدء()
        إذا !الحالة.يمكن_البدء()
            اطبع_سطر("❌ لا يمكن البدء - الحالة الحالية: " + الحالة.الاسم)
            ارجع
        نهاية
        الحالة = جديد حالة_قيد_التنفيذ()
        اطبع_سطر("▶️ تم بدء المهمة: " + العنوان)
    نهاية
    
    دالة إيقاف()
        إذا !الحالة.يمكن_الإيقاف()
            اطبع_سطر("❌ لا يمكن الإيقاف - الحالة الحالية: " + الحالة.الاسم)
            ارجع
        نهاية
        الحالة = جديد حالة_متوقفة()
        اطبع_سطر("⏸️ تم إيقاف المهمة: " + العنوان)
    نهاية
    
    دالة إكمال()
        إذا !الحالة.يمكن_الإكمال()
            اطبع_سطر("❌ لا يمكن الإكمال - الحالة الحالية: " + الحالة.الاسم)
            ارجع
        نهاية
        الحالة = جديد حالة_مكتملة()
        اطبع_سطر("✅ تم إكمال المهمة: " + العنوان)
    نهاية
    
    دالة إلغاء()
        إذا !الحالة.يمكن_الإلغاء()
            اطبع_سطر("❌ لا يمكن الإلغاء - الحالة الحالية: " + الحالة.الاسم)
            ارجع
        نهاية
        الحالة = جديد حالة_ملغاة()
        اطبع_سطر("🚫 تم إلغاء المهمة: " + العنوان)
    نهاية
    
    دالة عرض()
        اطبع_سطر("📋 " + العنوان + " [" + الحالة.الاسم + "]")
    نهاية
نهاية

دالة رئيسية()
    اطبع_سطر("═══════════════════════════════════════════")
    اطبع_سطر("         نمط الحالة - إدارة المهام         ")
    اطبع_سطر("═══════════════════════════════════════════\n")
    
    مهمة م = جديد مهمة("تطوير واجهة المستخدم")
    م.عرض()
    
    اطبع_سطر("\n--- تجربة العمليات ---")
    
    م.إكمال()  # سيفشل - لم تبدأ بعد
    م.بدء()
    م.عرض()
    
    م.بدء()    # سيفشل - بدأت بالفعل
    م.إيقاف()
    م.عرض()
    
    م.بدء()    # استئناف
    م.إكمال()
    م.عرض()
    
    م.بدء()    # سيفشل - مكتملة
    م.إلغاء()  # سيفشل - مكتملة
    
    ارجع 0
نهاية
```

## 5. بناء سلسلة (Builder Pattern)

```
صنف تقرير
    نص العنوان
    نص المقدمة
    نص المحتوى
    نص الخاتمة
    نص التاريخ
    نص المؤلف
    منطقي مع_ترقيم_صفحات
    منطقي مع_جدول_محتويات
    
    باني()
        العنوان = ""
        المقدمة = ""
        المحتوى = ""
        الخاتمة = ""
        التاريخ = ""
        المؤلف = ""
        مع_ترقيم_صفحات = خطأ
        مع_جدول_محتويات = خطأ
    نهاية
    
    دالة طباعة()
        اطبع_سطر("\n╔════════════════════════════════════════════╗")
        
        إذا مع_جدول_محتويات
            اطبع_سطر("║           📑 جدول المحتويات              ║")
            اطبع_سطر("╠════════════════════════════════════════════╣")
        نهاية
        
        اطبع_سطر("║ 📄 " + العنوان)
        اطبع_سطر("║ المؤلف: " + المؤلف)
        اطبع_سطر("║ التاريخ: " + التاريخ)
        اطبع_سطر("╠════════════════════════════════════════════╣")
        
        إذا طول(المقدمة) > 0
            اطبع_سطر("║ 📌 المقدمة:")
            اطبع_سطر("║ " + المقدمة)
            اطبع_سطر("║ ──────────────────────────────────────────")
        نهاية
        
        اطبع_سطر("║ 📝 المحتوى:")
        اطبع_سطر("║ " + المحتوى)
        
        إذا طول(الخاتمة) > 0
            اطبع_سطر("║ ──────────────────────────────────────────")
            اطبع_سطر("║ 📌 الخاتمة:")
            اطبع_سطر("║ " + الخاتمة)
        نهاية
        
        إذا مع_ترقيم_صفحات
            اطبع_سطر("╠════════════════════════════════════════════╣")
            اطبع_سطر("║                    - 1 -                   ║")
        نهاية
        
        اطبع_سطر("╚════════════════════════════════════════════╝")
    نهاية
نهاية

صنف باني_التقرير
    تقرير _التقرير
    
    باني()
        _التقرير = جديد تقرير()
    نهاية
    
    دالة باني_التقرير العنوان(نص عنوان)
        _التقرير.العنوان = عنوان
        ارجع ذاتي
    نهاية
    
    دالة باني_التقرير المؤلف(نص مؤلف)
        _التقرير.المؤلف = مؤلف
        ارجع ذاتي
    نهاية
    
    دالة باني_التقرير التاريخ(نص تاريخ)
        _التقرير.التاريخ = تاريخ
        ارجع ذاتي
    نهاية
    
    دالة باني_التقرير المقدمة(نص مقدمة)
        _التقرير.المقدمة = مقدمة
        ارجع ذاتي
    نهاية
    
    دالة باني_التقرير المحتوى(نص محتوى)
        _التقرير.المحتوى = محتوى
        ارجع ذاتي
    نهاية
    
    دالة باني_التقرير الخاتمة(نص خاتمة)
        _التقرير.الخاتمة = خاتمة
        ارجع ذاتي
    نهاية
    
    دالة باني_التقرير مع_ترقيم()
        _التقرير.مع_ترقيم_صفحات = صحيح
        ارجع ذاتي
    نهاية
    
    دالة باني_التقرير مع_جدول_محتويات()
        _التقرير.مع_جدول_محتويات = صحيح
        ارجع ذاتي
    نهاية
    
    دالة تقرير بناء()
        ارجع _التقرير
    نهاية
نهاية

دالة رئيسية()
    اطبع_سطر("═══════════════════════════════════════════")
    اطبع_سطر("         نمط البناء - إنشاء التقارير       ")
    اطبع_سطر("═══════════════════════════════════════════")
    
    # بناء تقرير بسيط
    باني_التقرير باني = جديد باني_التقرير()
    تقرير تقرير1 = باني.العنوان("تقرير المبيعات")
                        .المؤلف("أحمد محمد")
                        .التاريخ("2024/01/15")
                        .المحتوى("المبيعات زادت بنسبة 15% هذا الشهر")
                        .بناء()
    
    تقرير1.طباعة()
    
    # بناء تقرير كامل
    باني_التقرير باني2 = جديد باني_التقرير()
    تقرير تقرير2 = باني2.العنوان("التقرير السنوي للشركة")
                         .المؤلف("إدارة التخطيط")
                         .التاريخ("2024/12/31")
                         .المقدمة("هذا التقرير يُلخص إنجازات الشركة خلال العام")
                         .المحتوى("تم تحقيق نمو بنسبة 25% في الإيرادات...")
                         .الخاتمة("نتطلع لعام أفضل")
                         .مع_جدول_محتويات()
                         .مع_ترقيم()
                         .بناء()
    
    تقرير2.طباعة()
    
    ارجع 0
نهاية
```

## الخلاصة

أنماط التصميم والتقنيات المتقدمة هي أدوات قوية في صندوق أدوات المبرمج المحترف. إتقان هذه المفاهيم يُمكّنك من:

- كتابة كود أكثر مرونة وقابلية للتوسع
- حل المشاكل المعقدة بطرق منظمة ومجربة
- التواصل بفعالية مع المبرمجين الآخرين باستخدام لغة مشتركة
- بناء أنظمة كبيرة ومعقدة بثقة

**نصيحة أخيرة:** لا تُفرط في استخدام أنماط التصميم. استخدمها عندما تحل مشكلة حقيقية، وليس لمجرد استخدامها. البساطة دائماً أفضل من التعقيد غير الضروري.
