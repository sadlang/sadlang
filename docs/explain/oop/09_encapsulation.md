# التغليف في لغة ص - الدليل الشامل

## ما هو التغليف؟

التغليف (Encapsulation) هو أحد المبادئ الأساسية في البرمجة الكائنية، ويعني إخفاء التفاصيل الداخلية للكائن وتقديم واجهة بسيطة وآمنة للتعامل معه. بمعنى آخر، التغليف يجمع البيانات (الحقول) والسلوكيات (الدوال) في وحدة واحدة، ويتحكم في كيفية الوصول إليها.

لفهم التغليف، تخيّل هاتفك الذكي: أنت لا تحتاج لمعرفة كيف تعمل الدوائر الإلكترونية داخله، كل ما تحتاجه هو الشاشة والأزرار للتفاعل معه. الهاتف "يُغلّف" تعقيداته الداخلية ويُقدّم لك واجهة بسيطة. هذا بالضبط ما يفعله التغليف في البرمجة.

## فوائد التغليف

**حماية البيانات**: يمنع التعديل غير المتحكم به على البيانات الداخلية.

**إخفاء التعقيد**: يُبسّط الواجهة بإخفاء التفاصيل غير الضرورية.

**سهولة الصيانة**: يمكن تغيير التنفيذ الداخلي دون التأثير على المستخدمين.

**التحقق من صحة البيانات**: يُمكّن من التحقق قبل تعيين القيم.

**المرونة**: يُتيح تغيير طريقة تخزين أو حساب البيانات داخلياً.

## تطبيق التغليف في لغة ص

### النمط الأساسي: دوال الوصول والتعديل

```
صنف حساب_بنكي
    # حقول خاصة (تُسمّى أحياناً بالبادئة _)
    نص _رقم_الحساب
    نص _اسم_المالك
    عشري _الرصيد
    
    باني(نص رقم، نص اسم)
        _رقم_الحساب = رقم
        _اسم_المالك = اسم
        _الرصيد = 0.0
    نهاية
    
    # دوال القراءة (Getters)
    دالة نص رقم_الحساب()
        ارجع _رقم_الحساب
    نهاية
    
    دالة نص اسم_المالك()
        ارجع _اسم_المالك
    نهاية
    
    دالة عشري الرصيد()
        ارجع _الرصيد
    نهاية
    
    # دوال التعديل (Setters) مع التحقق
    دالة منطقي إيداع(عشري مبلغ)
        إذا مبلغ <= 0
            اطبع_سطر("خطأ: المبلغ يجب أن يكون موجباً")
            ارجع خطأ
        نهاية
        
        _الرصيد = _الرصيد + مبلغ
        اطبع_سطر("تم إيداع " + مبلغ + " ريال بنجاح")
        ارجع صحيح
    نهاية
    
    دالة منطقي سحب(عشري مبلغ)
        إذا مبلغ <= 0
            اطبع_سطر("خطأ: المبلغ يجب أن يكون موجباً")
            ارجع خطأ
        نهاية
        
        إذا مبلغ > _الرصيد
            اطبع_سطر("خطأ: الرصيد غير كافٍ")
            ارجع خطأ
        نهاية
        
        _الرصيد = _الرصيد - مبلغ
        اطبع_سطر("تم سحب " + مبلغ + " ريال بنجاح")
        ارجع صحيح
    نهاية
نهاية

دالة رئيسية()
    حساب_بنكي حسابي = جديد حساب_بنكي("SA123456789"، "محمد أحمد")
    
    # لا يمكن الوصول المباشر للرصيد وتعديله
    # حسابي._الرصيد = 1000000  # هذا ليس آمناً!
    
    # بدلاً من ذلك، نستخدم الواجهة الآمنة
    حسابي.إيداع(5000)
    حسابي.إيداع(3000)
    حسابي.سحب(2000)
    حسابي.سحب(100000)  # سيفشل - رصيد غير كاف
    
    اطبع_سطر("\nالرصيد النهائي: " + حسابي.الرصيد() + " ريال")
    
    ارجع 0
نهاية

## أمثلة على حماية البيانات

### مثال 1: التحقق من صحة البيانات

```
صنف شخص
    نص _الاسم
    رقم _العمر
    نص _البريد
    
    باني(نص اسم، رقم عمر)
        تعيين_الاسم(اسم)
        تعيين_العمر(عمر)
        _البريد = ""
    نهاية
    
    # قراءة الاسم
    دالة نص الاسم()
        ارجع _الاسم
    نهاية
    
    # تعيين الاسم مع التحقق
    دالة منطقي تعيين_الاسم(نص اسم)
        إذا طول(اسم) < 2
            اطبع_سطر("خطأ: الاسم يجب أن يكون حرفين على الأقل")
            ارجع خطأ
        نهاية
        
        إذا طول(اسم) > 50
            اطبع_سطر("خطأ: الاسم طويل جداً")
            ارجع خطأ
        نهاية
        
        _الاسم = اسم
        ارجع صحيح
    نهاية
    
    # قراءة العمر
    دالة رقم العمر()
        ارجع _العمر
    نهاية
    
    # تعيين العمر مع التحقق
    دالة منطقي تعيين_العمر(رقم عمر)
        إذا عمر < 0
            اطبع_سطر("خطأ: العمر لا يمكن أن يكون سالباً")
            ارجع خطأ
        نهاية
        
        إذا عمر > 150
            اطبع_سطر("خطأ: العمر غير واقعي")
            ارجع خطأ
        نهاية
        
        _العمر = عمر
        ارجع صحيح
    نهاية
    
    # قراءة البريد
    دالة نص البريد()
        ارجع _البريد
    نهاية
    
    # تعيين البريد مع التحقق
    دالة منطقي تعيين_البريد(نص بريد)
        # تحقق بسيط من وجود @
        منطقي يحتوي_علامة = خطأ
        رقم ف = 0
        طالما ف < طول(بريد)
            إذا حرف(بريد، ف) == "@"
                يحتوي_علامة = صحيح
            نهاية
            ف = ف + 1
        نهاية
        
        إذا !يحتوي_علامة
            اطبع_سطر("خطأ: البريد الإلكتروني غير صالح")
            ارجع خطأ
        نهاية
        
        _البريد = بريد
        ارجع صحيح
    نهاية
نهاية

دالة رئيسية()
    شخص ش = جديد شخص("أحمد"، 25)
    
    اطبع_سطر("=== اختبار التحقق من البيانات ===\n")

    ش.تعيين_الاسم("أ")        # سيفشل

### مثال 2: إخفاء تعقيد الحسابات

```
صنف سلة_تسوق
    # بيانات داخلية
    عشري _مجموع_الأسعار
    رقم _عدد_المنتجات
    عشري _نسبة_الخصم
    عشري _نسبة_الضريبة
    
    باني()
        _مجموع_الأسعار = 0
        _عدد_المنتجات = 0
        _نسبة_الخصم = 0
        _نسبة_الضريبة = 15  # ضريبة القيمة المضافة
    نهاية
    
    # واجهة بسيطة للمستخدم
    دالة إضافة_منتج(نص اسم، عشري سعر، رقم كمية)
        عشري القيمة = سعر * كمية
        _مجموع_الأسعار = _مجموع_الأسعار + القيمة
        _عدد_المنتجات = _عدد_المنتجات + كمية
        اطبع_سطر("✓ تم إضافة: " + اسم + " × " + كمية + " = " + القيمة + " ريال")
    نهاية
    
    دالة تطبيق_خصم(عشري نسبة)
        إذا نسبة < 0 أو نسبة > 50
            اطبع_سطر("خطأ: نسبة الخصم يجب أن تكون بين 0 و 50%")
            ارجع
        نهاية
        _نسبة_الخصم = نسبة
        اطبع_سطر("✓ تم تطبيق خصم " + نسبة + "%")
    نهاية
    
    # الحسابات الداخلية مخفية
    دالة عشري _حساب_الخصم()
        ارجع _مجموع_الأسعار * _نسبة_الخصم / 100
    نهاية
    
    دالة عشري _حساب_المبلغ_بعد_الخصم()
        ارجع _مجموع_الأسعار - _حساب_الخصم()
    نهاية
    
    دالة عشري _حساب_الضريبة()
        ارجع _حساب_المبلغ_بعد_الخصم() * _نسبة_الضريبة / 100
    نهاية
    
    # واجهة بسيطة للحصول على الإجمالي
    دالة عشري الإجمالي()
        ارجع _حساب_المبلغ_بعد_الخصم() + _حساب_الضريبة()
    نهاية
    
    دالة رقم عدد_المنتجات()
        ارجع _عدد_المنتجات
    نهاية
    
    # عرض الفاتورة
    دالة طباعة_الفاتورة()
        اطبع_سطر("\n╔═══════════════════════════════════════╗")
        اطبع_سطر("║            فاتورة المشتريات          ║")
        اطبع_سطر("╠═══════════════════════════════════════╣")
        اطبع_سطر("║ عدد المنتجات: " + _عدد_المنتجات)
        اطبع_سطر("║ ───────────────────────────────────── ")
        اطبع_سطر("║ المجموع الفرعي: " + _مجموع_الأسعار + " ريال")
        
        إذا _نسبة_الخصم > 0
            اطبع_سطر("║ الخصم (" + _نسبة_الخصم + "%): -" + _حساب_الخصم() + " ريال")
            اطبع_سطر("║ بعد الخصم: " + _حساب_المبلغ_بعد_الخصم() + " ريال")
        نهاية
        
        اطبع_سطر("║ ضريبة القيمة المضافة (" + _نسبة_الضريبة + "%): +" + _حساب_الضريبة() + " ريال")
        اطبع_سطر("║ ═════════════════════════════════════ ")
        اطبع_سطر("║ الإجمالي النهائي: " + الإجمالي() + " ريال")
        اطبع_سطر("╚═══════════════════════════════════════╝")
    نهاية
نهاية

    اطبع_سطر("═══════════════════════════════════════")

## مثال متقدم: نظام إدارة الموظفين

```
صنف موظف
    # بيانات خاصة
    رقم _الرقم_الوظيفي
    نص _الاسم
    عشري _الراتب_الأساسي
    عشري _البدلات
    عشري _الاستقطاعات
    نص _القسم
    تاريخ _تاريخ_التعيين
    
    باني(رقم رقم، نص اسم، عشري راتب، نص قسم)
        _الرقم_الوظيفي = رقم
        _الاسم = اسم
        _القسم = قسم
        _البدلات = 0
        _الاستقطاعات = 0
        
        # التحقق من الراتب الأساسي
        إذا راتب < 3000
            اطبع_سطر("تحذير: الراتب أقل من الحد الأدنى، تم تعديله")
            _الراتب_الأساسي = 3000
        وإلا
            _الراتب_الأساسي = راتب
        نهاية
    نهاية
    
    # دوال القراءة
    دالة رقم الرقم_الوظيفي()
        ارجع _الرقم_الوظيفي
    نهاية
    
    دالة نص الاسم()
        ارجع _الاسم
    نهاية
    
    دالة نص القسم()
        ارجع _القسم
    نهاية
    
    # دوال التعديل مع التحقق
    دالة منطقي تغيير_القسم(نص قسم_جديد)
        إذا طول(قسم_جديد) == 0
            اطبع_سطر("خطأ: اسم القسم لا يمكن أن يكون فارغاً")
            ارجع خطأ
        نهاية
        
        نص القسم_القديم = _القسم
        _القسم = قسم_جديد
        اطبع_سطر("تم نقل " + _الاسم + " من " + القسم_القديم + " إلى " + قسم_جديد)
        ارجع صحيح
    نهاية
    
    دالة منطقي زيادة_الراتب(عشري نسبة)
        إذا نسبة <= 0 أو نسبة > 50
            اطبع_سطر("خطأ: نسبة الزيادة يجب أن تكون بين 0 و 50%")
            ارجع خطأ
        نهاية
        
        عشري الزيادة = _الراتب_الأساسي * نسبة / 100
        _الراتب_الأساسي = _الراتب_الأساسي + الزيادة
        اطبع_سطر("تمت زيادة راتب " + _الاسم + " بمقدار " + الزيادة + " ريال")
        ارجع صحيح
    نهاية
    
    دالة إضافة_بدل(نص نوع، عشري مبلغ)
        إذا مبلغ <= 0
            اطبع_سطر("خطأ: مبلغ البدل يجب أن يكون موجباً")
            ارجع
        نهاية
        
        _البدلات = _البدلات + مبلغ
        اطبع_سطر("تم إضافة بدل " + نوع + " بقيمة " + مبلغ + " ريال")
    نهاية
    
    دالة إضافة_استقطاع(نص نوع، عشري مبلغ)
        إذا مبلغ <= 0
            اطبع_سطر("خطأ: مبلغ الاستقطاع يجب أن يكون موجباً")
            ارجع
        نهاية
        
        _الاستقطاعات = _الاستقطاعات + مبلغ
        اطبع_سطر("تم إضافة استقطاع " + نوع + " بقيمة " + مبلغ + " ريال")
    نهاية
    
    # الحسابات الداخلية
    دالة عشري _حساب_التأمينات()
        ارجع _الراتب_الأساسي * 0.1  # 10% تأمينات
    نهاية
    
    دالة عشري _حساب_صافي_الراتب()
        عشري الإجمالي = _الراتب_الأساسي + _البدلات
        عشري الخصومات = _الاستقطاعات + _حساب_التأمينات()
        ارجع الإجمالي - الخصومات
    نهاية
    
    # واجهة آمنة للحصول على الراتب
    دالة عشري صافي_الراتب()
        ارجع _حساب_صافي_الراتب()
    نهاية
    
    # كشف الراتب التفصيلي
    دالة طباعة_كشف_الراتب()
        اطبع_سطر("\n┌─────────────────────────────────────────┐")
        اطبع_سطر("│            كشف الراتب الشهري            │")
        اطبع_سطر("├─────────────────────────────────────────┤")
        اطبع_سطر("│ الرقم الوظيفي: " + _الرقم_الوظيفي)
        اطبع_سطر("│ الاسم: " + _الاسم)
        اطبع_سطر("│ القسم: " + _القسم)
        اطبع_سطر("├─────────────────────────────────────────┤")
        اطبع_سطر("│ المستحقات:")
        اطبع_سطر("│   الراتب الأساسي: " + _الراتب_الأساسي + " ريال")
        اطبع_سطر("│   البدلات: " + _البدلات + " ريال")
        اطبع_سطر("│   ─────────────────────────────────────")
        اطبع_سطر("│   إجمالي المستحقات: " + (_الراتب_الأساسي + _البدلات) + " ريال")
        اطبع_سطر("├─────────────────────────────────────────┤")
        اطبع_سطر("│ الاستقطاعات:")
        اطبع_سطر("│   التأمينات الاجتماعية: " + _حساب_التأمينات() + " ريال")
        اطبع_سطر("│   استقطاعات أخرى: " + _الاستقطاعات + " ريال")
        اطبع_سطر("│   ─────────────────────────────────────")
        اطبع_سطر("│   إجمالي الاستقطاعات: " + (_حساب_التأمينات() + _الاستقطاعات) + " ريال")
        اطبع_سطر("├─────────────────────────────────────────┤")
        اطبع_سطر("│ صافي الراتب: " + صافي_الراتب() + " ريال")
        اطبع_سطر("└─────────────────────────────────────────┘")
    نهاية

## نصائح للتغليف الجيد

**اجعل الحقول خاصة**: دائماً ابدأ بجعل الحقول خاصة، وأضف دوال وصول فقط عند الحاجة.

**تحقق من المدخلات**: في دوال التعديل، تحقق دائماً من صحة القيم قبل تعيينها.

**أخفِ تفاصيل التنفيذ**: لا تكشف عن كيفية عمل الأشياء داخلياً، فقط وفّر واجهة بسيطة.

**وفّر رسائل خطأ واضحة**: عندما يحاول المستخدم فعل شيء غير صالح، أخبره بوضوح.

## الخلاصة

التغليف هو مفتاح كتابة كود آمن وقابل للصيانة. بإخفاء التفاصيل الداخلية وتوفير واجهات محكومة، تضمن أن الكائنات تُستخدم بالطريقة الصحيحة. في القسم التالي، سنتعرف على أفضل الممارسات في البرمجة الكائنية.
