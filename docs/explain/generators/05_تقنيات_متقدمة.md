# 🚀 تقنيات متقدمة في المولدات

## المولدات المتداخلة

المولدات المتداخلة هي مولدات تستدعي مولدات أخرى داخلها. هذه التقنية مفيدة جداً لتقسيم المنطق المعقد إلى وحدات أصغر وأكثر قابلية للإدارة.

### مثال: معالجة متعددة المراحل

```s
# المرحلة الأولى: قراءة البيانات الخام
دالة مولد قراءة_البيانات_الخام(نص المصدر)
    متغير بيانات = احصل_على_البيانات(المصدر)
    لكل عنصر في بيانات
        أنتج عنصر
    نهاية
نهاية

# المرحلة الثانية: التنظيف
دالة مولد تنظيف_البيانات(متغير المدخلات)
    لكل عنصر في المدخلات
        إذا (عنصر != لاشيء و طول(عنصر) > 0)
            أنتج ازالة_المسافات(عنصر)
        نهاية
    نهاية
نهاية

# المرحلة الثالثة: التحويل
دالة مولد تحويل_البيانات(متغير المدخلات)
    لكل عنصر في المدخلات
        أنتج {
            "القيمة": عنصر،
            "الطول": طول(عنصر)،
            "الوقت": الوقت_الحالي()
        }
    نهاية
نهاية

# المرحلة الرابعة: التصفية النهائية
دالة مولد تصفية_النتائج(متغير المدخلات، رقم الحد_الأدنى_للطول)
    لكل عنصر في المدخلات
        إذا (عنصر.الطول >= الحد_الأدنى_للطول)
            أنتج عنصر
        نهاية
    نهاية
نهاية

# المولد الرئيسي الذي يجمع كل المراحل
دالة مولد خط_المعالجة_الكامل(نص المصدر، رقم الحد_الأدنى)
    متغير خام = قراءة_البيانات_الخام(المصدر)
    متغير نظيف = تنظيف_البيانات(خام)
    متغير محول = تحويل_البيانات(نظيف)
    متغير مصفى = تصفية_النتائج(محول، الحد_الأدنى)
    
    أنتج من مصفى
نهاية

دالة رئيسية()
    لكل نتيجة في خط_المعالجة_الكامل("بيانات.txt"، 5)
        اطبع(نتيجة)
    نهاية
نهاية
```

## مولدات مع حالة (Stateful Generators)

المولدات يمكن أن تحتفظ بحالة معقدة عبر الإنتاج المتعدد.

### مولد مع ذاكرة

```s
دالة مولد مع_ذاكرة_القيم_السابقة(متغير المدخلات، رقم حجم_الذاكرة)
    متغير الذاكرة = []
    
    لكل قيمة في المدخلات
        الذاكرة.اضف(قيمة)
        
        إذا (طول(الذاكرة) > حجم_الذاكرة)
            الذاكرة.احذف_الأول()
        نهاية
        
        أنتج {
            "القيمة_الحالية": قيمة،
            "القيم_السابقة": نسخة(الذاكرة)،
            "عدد_القيم_المحفوظة": طول(الذاكرة)
        }
    نهاية
نهاية

دالة رئيسية()
    متغير أرقام = [10، 20، 30، 40، 50]
    
    لكل حالة في مع_ذاكرة_القيم_السابقة(أرقام، 3)
        اطبع("الحالية: " + حالة.القيمة_الحالية)
        اطبع("السابقة: " + حالة.القيم_السابقة)
        اطبع("---")
    نهاية
نهاية
```

### مولد مع عدادات وإحصائيات

```s
دالة مولد مع_إحصائيات(متغير المدخلات)
    رقم العدد = 0
    رقم المجموع = 0
    رقم الأدنى = لانهاية
    رقم الأقصى = سالب_لانهاية
    
    لكل قيمة في المدخلات
        العدد = العدد + 1
        المجموع = المجموع + قيمة
        
        إذا (قيمة < الأدنى)
            الأدنى = قيمة
        نهاية
        
        إذا (قيمة > الأقصى)
            الأقصى = قيمة
        نهاية
        
        أنتج {
            "القيمة": قيمة،
            "إحصائيات": {
                "العدد": العدد،
                "المجموع": المجموع،
                "المتوسط": المجموع / العدد،
                "الأدنى": الأدنى،
                "الأقصى": الأقصى
            }
        }
    نهاية
نهاية

دالة رئيسية()
    متغير قيم = [5، 10، 3، 8، 12، 1، 9]
    
    لكل نتيجة في مع_إحصائيات(قيم)
        اطبع("القيمة: " + نتيجة.القيمة)
        اطبع("المتوسط الحالي: " + نتيجة.إحصائيات.المتوسط)
        اطبع("المدى: " + نتيجة.إحصائيات.الأدنى + " - " + نتيجة.إحصائيات.الأقصى)
        اطبع("---")
    نهاية
نهاية
```

## مولدات قابلة للإيقاف والاستئناف

```s
دالة مولد قابل_للإيقاف(متغير المدخلات)
    لكل عنصر في المدخلات
        # نقطة توقف محتملة
        إذا (يجب_الإيقاف())
            أنتج {
                "نوع": "توقف"،
                "آخر_عنصر": عنصر
            }
            
            # انتظار إشارة الاستئناف
            بينما (يجب_الإيقاف())
                انتظر(100)  # انتظار 100 مللي ثانية
            نهاية
        نهاية
        
        أنتج {
            "نوع": "قيمة"،
            "البيانات": عنصر
        }
    نهاية
نهاية
```

## مولدات مع معالجة الأخطاء

```s
دالة مولد آمن(متغير المدخلات)
    لكل عنصر في المدخلات
        حاول
            متغير نتيجة = معالجة_معقدة(عنصر)
            أنتج {
                "نجاح": صحيح،
                "البيانات": نتيجة
            }
        التقط خطأ كـ خ
            أنتج {
                "نجاح": خطأ،
                "الخطأ": خ.الرسالة،
                "العنصر_الأصلي": عنصر
            }
        نهاية
    نهاية
نهاية

دالة رئيسية()
    متغير بيانات = ["صالح1"، "غير_صالح"، "صالح2"، "خطأ"، "صالح3"]
    
    رقم نجاحات = 0
    رقم إخفاقات = 0
    
    لكل نتيجة في آمن(بيانات)
        إذا (نتيجة.نجاح)
            اطبع("✅ " + نتيجة.البيانات)
            نجاحات = نجاحات + 1
        وإلا
            اطبع("❌ خطأ: " + نتيجة.الخطأ)
            إخفاقات = إخفاقات + 1
        نهاية
    نهاية
    
    اطبع("النتيجة: " + نجاحات + " نجاح، " + إخفاقات + " إخفاق")
نهاية
```

## الدمج بين مولدات متعددة (Zip/Merge)

### دمج متزامن (Zip)

```s
دالة مولد دمج_متزامن(متغير المولد1، متغير المولد2)
    متغير قيم1 = قائمة_من(المولد1)
    متغير قيم2 = قائمة_من(المولد2)
    
    رقم الطول = أدنى(طول(قيم1)، طول(قيم2))
    
    رقم ع = 0
    بينما (ع < الطول)
        أنتج [قيم1[ع]، قيم2[ع]]
        ع = ع + 1
    نهاية
نهاية

دالة رئيسية()
    متغير أسماء = ["أحمد"، "سارة"، "محمد"]
    متغير درجات = [95، 88، 92]
    
    لكل زوج في دمج_متزامن(أسماء، درجات)
        اطبع(زوج[0] + ": " + زوج[1])
    نهاية
    # الإخراج:
    # أحمد: 95
    # سارة: 88
    # محمد: 92
نهاية
```

### دمج متتابع (Chain)

```s
دالة مولد سلسلة(متغير قائمة_المولدات)
    لكل مولد في قائمة_المولدات
        أنتج من مولد
    نهاية
نهاية

دالة رئيسية()
    متغير م1 = نطاق(1، 3)
    متغير م2 = ["أ"، "ب"، "ج"]
    متغير م3 = نطاق(100، 102)
    
    لكل قيمة في سلسلة([م1، م2، م3])
        اطبع(قيمة)
    نهاية
    # الإخراج: 1، 2، 3، أ، ب، ج، 100، 101، 102
نهاية
```

### دمج تبادلي (Interleave)

```s
دالة مولد تبادل(متغير المولد1، متغير المولد2)
    متغير قيم1 = قائمة_من(المولد1)
    متغير قيم2 = قائمة_من(المولد2)
    
    رقم ع1 = 0
    رقم ع2 = 0
    
    بينما (ع1 < طول(قيم1) أو ع2 < طول(قيم2))
        إذا (ع1 < طول(قيم1))
            أنتج قيم1[ع1]
            ع1 = ع1 + 1
        نهاية
        
        إذا (ع2 < طول(قيم2))
            أنتج قيم2[ع2]
            ع2 = ع2 + 1
        نهاية
    نهاية
نهاية

دالة رئيسية()
    متغير أرقام = [1، 2، 3]
    متغير حروف = ["أ"، "ب"، "ج"، "د"، "هـ"]
    
    لكل قيمة في تبادل(أرقام، حروف)
        اطبع(قيمة)
    نهاية
    # الإخراج: 1، أ، 2، ب، 3، ج، د، هـ
نهاية
```

## مولدات لا نهائية مع حدود

```s
# مولد لا نهائي للأرقام
دالة مولد أرقام_لانهائية()
    رقم ع = 0
    بينما (صحيح)
        أنتج ع
        ع = ع + 1
    نهاية
نهاية

# أخذ عدد محدد من مولد لا نهائي
دالة مولد خذ(متغير المولد، رقم الكمية)
    رقم عداد = 0
    لكل قيمة في المولد
        إذا (عداد >= الكمية)
            اخرج
        نهاية
        أنتج قيمة
        عداد = عداد + 1
    نهاية
نهاية

# تجاوز عدد محدد ثم البدء
دالة مولد تجاوز(متغير المولد، رقم الكمية)
    رقم عداد = 0
    لكل قيمة في المولد
        إذا (عداد >= الكمية)
            أنتج قيمة
        نهاية
        عداد = عداد + 1
    نهاية
نهاية

# أخذ بينما الشرط صحيح
دالة مولد خذ_بينما(متغير المولد، دالة الشرط)
    لكل قيمة في المولد
        إذا (ليس الشرط(قيمة))
            اخرج
        نهاية
        أنتج قيمة
    نهاية
نهاية

دالة رئيسية()
    # أول 10 أرقام
    اطبع("أول 10 أرقام:")
    لكل رقم في خذ(أرقام_لانهائية()، 10)
        اطبع(رقم)
    نهاية
    
    # تجاوز أول 5 ثم أخذ 3
    اطبع("تجاوز 5 ثم أخذ 3:")
    لكل رقم في خذ(تجاوز(أرقام_لانهائية()، 5)، 3)
        اطبع(رقم)
    نهاية
    # الإخراج: 5، 6، 7
نهاية
```

## مولدات مع التخزين المؤقت (Caching)

```s
دالة مولد مع_تخزين_مؤقت(متغير المولد_الأصلي)
    متغير ذاكرة_تخزين = []
    متغير انتهى = خطأ
    
    دالة مولد مولد_مغلف()
        # أولاً، أنتج القيم المخزنة
        لكل قيمة في ذاكرة_تخزين
            أنتج قيمة
        نهاية
        
        # ثم، استمر في المولد الأصلي
        إذا (ليس انتهى)
            لكل قيمة في المولد_الأصلي
                ذاكرة_تخزين.اضف(قيمة)
                أنتج قيمة
            نهاية
            انتهى = صحيح
        نهاية
    نهاية
    
    إرجاع مولد_مغلف
نهاية

دالة رئيسية()
    # مولد بطيء (يحاكي عملية مكلفة)
    دالة مولد بطيء()
        لكل ع في نطاق(1، 5)
            انتظر(1000)  # تأخير ثانية واحدة
            أنتج ع
        نهاية
    نهاية
    
    # إنشاء نسخة مخزنة مؤقتاً
    متغير مخزن = مع_تخزين_مؤقت(بطيء())
    
    # المرة الأولى - بطيئة
    اطبع("المرة الأولى:")
    لكل قيمة في مخزن()
        اطبع(قيمة)
    نهاية
    
    # المرة الثانية - سريعة (من التخزين المؤقت)
    اطبع("المرة الثانية:")
    لكل قيمة في مخزن()
        اطبع(قيمة)
    نهاية
نهاية
```

## الخلاصة

التقنيات المتقدمة في المولدات تفتح آفاقاً واسعة لكتابة كود أكثر كفاءة وأناقة:

1. **المولدات المتداخلة** - لتقسيم المنطق المعقد
2. **المولدات مع حالة** - للاحتفاظ بمعلومات عبر التكرارات
3. **معالجة الأخطاء** - لبناء تطبيقات متينة
4. **الدمج بين المولدات** - لتجميع مصادر البيانات
5. **التحكم في المولدات اللانهائية** - باستخدام `خذ` و`تجاوز`
6. **التخزين المؤقت** - لتحسين الأداء

إتقان هذه التقنيات يجعلك مبرمجاً أكثر فعالية في لغة ص.
