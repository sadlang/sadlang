## مثال 1: نظام إدارة صلاحيات ملفات

برنامج كامل يستخدم الأرقام الثنائية لإدارة صلاحيات الملفات (مثل Unix):

```s
# تعريف الصلاحيات بالنظام الثنائي
رقم قراءة = 0b001         # 1
رقم كتابة = 0b010         # 2  
رقم تنفيذ = 0b100         # 4

# البادئة العربية للثنائي
رقم قراءة_عربي = 0ثن001
رقم كتابة_عربي = 0ثن010
رقم تنفيذ_عربي = 0ثن100

# دالة للتحقق من الصلاحية
دالة لديه_صلاحية(صلاحيات: رقم، صلاحية: رقم) -> منطقي {
    ارجع (صلاحيات & صلاحية) != 0
}

# صلاحيات للمالك: قراءة + كتابة + تنفيذ
رقم صلاحيات_مالك = 0b111  # 7
# صلاحيات للمجموعة: قراءة + تنفيذ
رقم صلاحيات_مجموعة = 0b101  # 5
# صلاحيات للآخرين: قراءة فقط
رقم صلاحيات_آخرون = 0b001  # 1

اطبع("=== نظام الصلاحيات ===")
اطبع("المالك (111): يمكنه القراءة؟", لديه_صلاحية(صلاحيات_مالك، قراءة))
اطبع("المالك (111): يمكنه الكتابة؟", لديه_صلاحية(صلاحيات_مالك، كتابة))
اطبع("المجموعة (101): يمكنها الكتابة؟", لديه_صلاحية(صلاحيات_مجموعة، كتابة))
اطبع("الآخرون (001): يمكنهم التنفيذ؟", لديه_صلاحية(صلاحيات_آخرون، تنفيذ))

# تمثيل كامل بالثماني (مثل chmod)
رقم صلاحيات_كاملة = 0o751  # 7 للمالك، 5 للمجموعة، 1 للآخرين
اطبع("الصلاحيات بالثماني: 0o751 =", صلاحيات_كاملة)
```

## مثال 2: نظام ألوان RGB

برنامج يستخدم الأرقام الست عشرية لإدارة الألوان:

```s
# تعريف ألوان أساسية بالست عشري
رقم أحمر_نقي = 0xFF0000    # Red
رقم أخضر_نقي = 0x00FF00    # Green  
رقم أزرق_نقي = 0x0000FF    # Blue

# البادئة العربية
رقم برتقالي = 0سFF5733
رقم أرجواني = 0س9B59B6
رقم سماوي = 0س3498DB

# دالة لاستخراج مكونات RGB
دالة احصل_على_أحمر(لون: رقم) -> رقم {
    ارجع (لون >> 16) & 0xFF
}

دالة احصل_على_أخضر(لون: رقم) -> رقم {
    ارجع (لون >> 8) & 0xFF
}

دالة احصل_على_أزرق(لون: رقم) -> رقم {
    ارجع لون & 0xFF
}

# اختبار
رقم لون_اختبار = 0xFF5733

اطبع("=== نظام الألوان ===")
اطبع("اللون:", لون_اختبار)
اطبع("الأحمر:", احصل_على_أحمر(لون_اختبار))
اطبع("الأخضر:", احصل_على_أخضر(لون_اختبار))
اطبع("الأزرق:", احصل_على_أزرق(لون_اختبار))
```

## مثال 3: نظام تقارير مالية مع تنسيق

برنامج يستخدم الفواصل البصرية والأرقام العربية:

```s
# الميزانية بملايين (مع فواصل للوضوح)
رقم ميزانية_2024 = 5_000_000
رقم إيرادات = 4_250_000
رقم مصروفات = 3_800_000

# حساب الأرباح
رقم أرباح = إيرادات - مصروفات
عشري نسبة_الربح = (أرباح * 100.0) / إيرادات

# عرض بأرقام عربية
رقم نسبة_عربية = ٨٥

اطبع("=== تقرير مالي ===")
اطبع("الميزانية:", ميزانية_2024, "ريال")
اطبع("الإيرادات:", إيرادات, "ريال")
اطبع("المصروفات:", مصروفات, "ريال")  
اطبع("الأرباح:", أرباح, "ريال")
اطبع("نسبة الربح:", نسبة_الربح, "%")

# حالة الأنظمة (ثنائي)
رقم حالة_الأنظمة = 0b1111  # كل الأنظمة تعمل
اطبع("حالة الأنظمة:", حالة_الأنظمة)
```

## مثال 4: معالج مسارات ملفات

برنامج يستخدم النصوص الخام لمعالجة مسارات Windows:

```s
# مسارات بدون مشاكل الشرطة المائلة
نص مسار_التطبيق = r"C:\Program Files\MyApp"
نص مسار_البيانات = r"C:\Users\Public\Documents\MyApp\Data"
نص مسار_النسخ_الاحتياطي = r"\\Server\Backups\Daily"

# دالة لبناء مسار كامل
دالة بناء_مسار(قاعدة: نص، ملف: نص) -> نص {
    ارجع قاعدة + r"\" + ملف
}

نص ملف_التكوين = بناء_مسار(مسار_التطبيق، "config.ini")
نص ملف_السجل = بناء_مسار(مسار_التطبيق، "app.log")

اطبع("=== إدارة المسارات ===")
اطبع("التطبيق:", مسار_التطبيق)
اطبع("التكوين:", ملف_التكوين)
اطبع("السجل:", ملف_السجل)

# تعبيرات نمطية للتحقق من صحة البيانات
نص نمط_رقم_هاتف = r"\+966\d{9}"
نص نمط_بريد = r"\w+@\w+\.\w+"

اطبع("نمط رقم الهاتف:", نمط_رقم_هاتف)
اطبع("نمط البريد:", نمط_بريد)
```

## مثال 5: نظام شامل يجمع كل شيء

برنامج كبير يستخدم كل الميزات المتقدمة:

```s
# === التكوين ===
نص اسم_التطبيق = "نظام إدارة متكامل"
نص مسار_قاعدة_البيانات = r"C:\ProgramData\MySystem\database.db"

# الإعدادات بالثنائي
رقم إعدادات = 0b00001111  # 4 إعدادات مفعلة
رقم تسجيل_دخول = 0b0001
رقم تشفير = 0b0010  
رقم نسخ_احتياطي = 0b0100
رقم تنبيهات = 0b1000

# الميزانية بالملايين
رقم ميزانية = 10_000_000
رقم مصروف_حالي = 7_500_000

# ألوان الواجهة بالست عشري
رقم لون_أساسي = 0x2196F3
رقم لون_ثانوي = 0x4CAF50
رقم لون_خطر = 0xF44336

# === الدوال ===
دالة إعداد_مفعل(إعدادات: رقم، إعداد: رقم) -> منطقي {
    ارجع (إعدادات & إعداد) != 0
}

دالة نسبة_الصرف(مصروف: رقم، ميزانية: رقم) -> عشري {
    ارجع (مصروف * 100.0) / ميزانية  
}

# === التقرير ===
اطبع("=" * 50)
اطبع("         ", اسم_التطبيق)
اطبع("=" * 50)

اطبع("\n--- الإعدادات ---")
اطبع("تسجيل الدخول:", إعداد_مفعل(إعدادات، تسجيل_دخول))
اطبع("التشفير:", إعداد_مفعل(إعدادات، تشفير))
اطبع("النسخ الاحتياطي:", إعداد_مفعل(إعدادات، نسخ_احتياطي))
اطبع("التنبيهات:", إعداد_مفعل(إعدادات، تنبيهات))

اطبع("\n--- المالية ---")
اطبع("الميزانية:", ميزانية, "ريال")
اطبع("المصروف:", مصروف_حالي, "ريال")
اطبع("نسبة الصرف:", نسبة_الصرف(مصروف_حالي، ميزانية), "%")

اطبع("\n--- الملفات ---")
اطبع("قاعدة البيانات:", مسار_قاعدة_البيانات)

اطبع("\n--- الألوان ---")
اطبع("اللون الأساسي: 0x", لون_أساسي)
اطبع("اللون الثانوي: 0x", لون_ثانوي)
اطبع("=" * 50)
```

## الخلاصة

هذه الأمثلة توضح كيف تعمل كل الميزات المتقدمة معاً:

- ✅ **الأرقام الثنائية** (0b، 0ثن): للصلاحيات والأعلام
- ✅ **الأرقام الثمانية** (0o، 0ع): لصلاحيات Unix
- ✅ **الأرقام الست عشرية** (0x، 0س): للألوان والبيانات الثنائية
- ✅ **الفواصل البصرية** (_): للأرقام الكبيرة
- ✅ **الأرقام العربية** (٠-٩): للعرض
- ✅ **النصوص الخام** (r، ح): لمسارات Windows والتعبيرات النمطية
