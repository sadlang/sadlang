# البرمجة الكائنية في لغة ص - مقدمة شاملة

## ما هي البرمجة الكائنية؟

البرمجة الكائنية (Object-Oriented Programming - OOP) هي نمط برمجي يُنظّم الكود حول "الكائنات" بدلاً من الدوال والمنطق فقط. الكائن هو وحدة تجمع البيانات (الخصائص) والسلوكيات (الدوال) معاً في كيان واحد.

تخيل أنك تريد برمجة نظام لإدارة مكتبة. في البرمجة التقليدية، ستُنشئ متغيرات منفصلة لاسم الكتاب ومؤلفه وسنة نشره، ودوال منفصلة للاستعارة والإرجاع. لكن مع البرمجة الكائنية، تُنشئ "صنف كتاب" يحتوي على كل هذه البيانات والدوال معاً!

## المفاهيم الأساسية

### 1. الصنف (Class)

الصنف هو القالب أو المخطط الذي يُحدد شكل الكائنات. فكر فيه كمخطط معماري للبناء - المخطط ليس البناء نفسه، لكنه يُحدد كيف سيكون البناء.

```س
صنف سيارة
    # الخصائص (البيانات)
    خاص:
        نص الماركة
        نص الموديل
        رقم السنة
        رقم السرعة_الحالية
    
    # السلوكيات (الدوال)
    عام:
        دالة منشئ(نص ماركة، نص موديل، رقم سنة)
            الماركة = ماركة
            الموديل = موديل
            السنة = سنة
            السرعة_الحالية = 0
        نهاية
        
        دالة تسريع(رقم مقدار)
            السرعة_الحالية = السرعة_الحالية + مقدار
            اطبع("السرعة الحالية: ")
            اطبع(السرعة_الحالية)
        نهاية
        
        دالة فرملة(رقم مقدار)
            السرعة_الحالية = السرعة_الحالية - مقدار
            إذا(السرعة_الحالية < 0)
                السرعة_الحالية = 0
            نهاية
            اطبع("السرعة الحالية: ")
            اطبع(السرعة_الحالية)
        نهاية
نهاية
```

### 2. الكائن (Object)

الكائن هو نسخة فعلية من الصنف. إذا كان الصنف هو المخطط، فالكائن هو البناء الحقيقي المُنشأ من هذا المخطط.

```س
دالة رئيسية() رقم
    # إنشاء كائنات من صنف سيارة
    متغير سيارتي = سيارة("تويوتا"، "كامري"، 2023)
    متغير سيارة_صديقي = سيارة("هوندا"، "أكورد"، 2022)
    
    # استخدام الكائنات
    سيارتي.تسريع(60)
    سيارتي.فرملة(20)
    
    ارجع 0
نهاية
```

### 3. الخصائص (Properties/Attributes)

الخصائص هي البيانات التي يحتويها الكائن. في مثال السيارة، الخصائص هي: الماركة، الموديل، السنة، السرعة_الحالية.

### 4. الدوال (Methods)

الدوال هي السلوكيات التي يمكن للكائن تنفيذها. في مثال السيارة، الدوال هي: تسريع، فرملة.

## الأركان الأربعة للبرمجة الكائنية

### 1. التغليف (Encapsulation)

التغليف يعني إخفاء التفاصيل الداخلية للكائن وتوفير واجهة محددة للتعامل معه. هذا يحمي البيانات من التعديل غير المقصود.

```س
صنف حساب_بنكي
    خاص:
        عشري الرصيد        # لا يمكن الوصول إليه مباشرة من الخارج
        نص رقم_الحساب
    
    عام:
        دالة منشئ(نص رقم، عشري رصيد_ابتدائي)
            رقم_الحساب = رقم
            إذا(رصيد_ابتدائي >= 0)
                الرصيد = رصيد_ابتدائي
            وإلا
                الرصيد = 0
                اطبع("تحذير: الرصيد الابتدائي لا يمكن أن يكون سالباً")
            نهاية
        نهاية
        
        # دالة للوصول للرصيد (Getter)
        دالة احصل_الرصيد() عشري
            ارجع الرصيد
        نهاية
        
        # إيداع مبلغ
        دالة إيداع(عشري مبلغ) منطقي
            إذا(مبلغ > 0)
                الرصيد = الرصيد + مبلغ
                اطبع("تم إيداع ")
                اطبع(مبلغ)
                اطبع(" بنجاح")
                ارجع صحيح
            نهاية
            اطبع("خطأ: المبلغ يجب أن يكون موجباً")
            ارجع خطأ
        نهاية
        
        # سحب مبلغ
        دالة سحب(عشري مبلغ) منطقي
            إذا(مبلغ > 0 و مبلغ <= الرصيد)
                الرصيد = الرصيد - مبلغ
                اطبع("تم سحب ")
                اطبع(مبلغ)
                اطبع(" بنجاح")
                ارجع صحيح
            نهاية
            
            إذا(مبلغ > الرصيد)
                اطبع("خطأ: الرصيد غير كافٍ")
            وإلا
                اطبع("خطأ: المبلغ يجب أن يكون موجباً")
            نهاية
            ارجع خطأ
        نهاية
نهاية

دالة رئيسية() رقم
    متغير حسابي = حساب_بنكي("123456"، 1000.0)
    
    # لا يمكن الوصول للرصيد مباشرة
    # حسابي.الرصيد = 1000000  # خطأ! الرصيد خاص
    
    # نستخدم الدوال المتاحة
    حسابي.إيداع(500.0)
    اطبع("الرصيد الحالي: ")
    اطبع(حسابي.احصل_الرصيد())
    
    حسابي.سحب(200.0)
    اطبع("الرصيد بعد السحب: ")
    اطبع(حسابي.احصل_الرصيد())
    
    ارجع 0
نهاية
```

التغليف يحمي البيانات ويضمن:
- لا يمكن تعيين رصيد سالب مباشرة
- كل عملية سحب يتم التحقق منها
- منطق الأعمال موجود في مكان واحد

### 2. الوراثة (Inheritance)

الوراثة تُتيح إنشاء أصناف جديدة تبني على أصناف موجودة. الصنف الجديد (الابن) يرث خصائص ودوال الصنف الأصلي (الأب) ويمكنه إضافة خصائص ودوال جديدة.

```س
# الصنف الأب
صنف حيوان
    محمي:
        نص الاسم
        رقم العمر
    
    عام:
        دالة منشئ(نص اسم، رقم عمر)
            الاسم = اسم
            العمر = عمر
        نهاية
        
        دالة صوت()
            اطبع("...")  # صوت عام
        نهاية
        
        دالة احصل_الاسم() نص
            ارجع الاسم
        نهاية
        
        دالة وصف()
            اطبع("حيوان اسمه ")
            اطبع(الاسم)
            اطبع(" عمره ")
            اطبع(العمر)
            اطبع(" سنة")
        نهاية
نهاية

# صنف ابن يرث من حيوان
صنف قطة يرث حيوان
    خاص:
        نص اللون
    
    عام:
        دالة منشئ(نص اسم، رقم عمر، نص لون)
            أب.منشئ(اسم، عمر)  # استدعاء مُنشئ الأب
            اللون = لون
        نهاية
        
        # إعادة تعريف دالة الصوت
        دالة صوت()
            اطبع("مياو مياو!")
        نهاية
        
        دالة خرخرة()
            اطبع(الاسم)
            اطبع(" يخرخر...")
        نهاية
نهاية

# صنف ابن آخر
صنف كلب يرث حيوان
    خاص:
        نص السلالة
    
    عام:
        دالة منشئ(نص اسم، رقم عمر، نص سلالة)
            أب.منشئ(اسم، عمر)
            السلالة = سلالة
        نهاية
        
        # إعادة تعريف دالة الصوت
        دالة صوت()
            اطبع("هاو هاو!")
        نهاية
        
        دالة هز_الذيل()
            اطبع(الاسم)
            اطبع(" يهز ذيله بسعادة!")
        نهاية
نهاية

دالة رئيسية() رقم
    متغير قطتي = قطة("ميمي"، 3، "أبيض")
    متغير كلبي = كلب("بادي"، 5، "جيرمن شيبرد")
    
    قطتي.وصف()        # موروثة من حيوان
    قطتي.صوت()        # معاد تعريفها
    قطتي.خرخرة()      # خاصة بالقطة
    
    كلبي.وصف()        # موروثة من حيوان
    كلبي.صوت()        # معاد تعريفها
    كلبي.هز_الذيل()   # خاصة بالكلب
    
    ارجع 0
نهاية
```

### 3. تعدد الأشكال (Polymorphism)

تعدد الأشكال يعني أن نفس الدالة يمكن أن تتصرف بطرق مختلفة حسب نوع الكائن. هذا يُتيح كتابة كود عام يعمل مع أنواع مختلفة.

```س
صنف شكل
    عام:
        دالة افتراضية احسب_المساحة() عشري
            ارجع 0.0
        نهاية
        
        دالة افتراضية احسب_المحيط() عشري
            ارجع 0.0
        نهاية
        
        دالة افتراضية اطبع_المعلومات()
            اطبع("شكل هندسي")
        نهاية
نهاية

صنف مستطيل يرث شكل
    خاص:
        عشري الطول
        عشري العرض
    
    عام:
        دالة منشئ(عشري طول، عشري عرض)
            الطول = طول
            العرض = عرض
        نهاية
        
        دالة احسب_المساحة() عشري
            ارجع الطول * العرض
        نهاية
        
        دالة احسب_المحيط() عشري
            ارجع 2 * (الطول + العرض)
        نهاية
        
        دالة اطبع_المعلومات()
            اطبع("مستطيل: طول=")
            اطبع(الطول)
            اطبع("، عرض=")
            اطبع(العرض)
        نهاية
نهاية

صنف دائرة يرث شكل
    خاص:
        عشري نصف_القطر
        ثابت عشري باي = 3.14159
    
    عام:
        دالة منشئ(عشري نق)
            نصف_القطر = نق
        نهاية
        
        دالة احسب_المساحة() عشري
            ارجع باي * نصف_القطر * نصف_القطر
        نهاية
        
        دالة احسب_المحيط() عشري
            ارجع 2 * باي * نصف_القطر
        نهاية
        
        دالة اطبع_المعلومات()
            اطبع("دائرة: نصف القطر=")
            اطبع(نصف_القطر)
        نهاية
نهاية

صنف مثلث يرث شكل
    خاص:
        عشري القاعدة
        عشري الارتفاع
        عشري الضلع1
        عشري الضلع2
        عشري الضلع3
    
    عام:
        دالة منشئ(عشري قاعدة، عشري ارتفاع، عشري ض1، عشري ض2، عشري ض3)
            القاعدة = قاعدة
            الارتفاع = ارتفاع
            الضلع1 = ض1
            الضلع2 = ض2
            الضلع3 = ض3
        نهاية
        
        دالة احسب_المساحة() عشري
            ارجع 0.5 * القاعدة * الارتفاع
        نهاية
        
        دالة احسب_المحيط() عشري
            ارجع الضلع1 + الضلع2 + الضلع3
        نهاية
        
        دالة اطبع_المعلومات()
            اطبع("مثلث: قاعدة=")
            اطبع(القاعدة)
            اطبع("، ارتفاع=")
            اطبع(الارتفاع)
        نهاية
نهاية

# دالة تعمل مع أي شكل (تعدد الأشكال)
دالة اطبع_تفاصيل_شكل(شكل ش)
    ش.اطبع_المعلومات()
    اطبع("المساحة: ")
    اطبع(ش.احسب_المساحة())
    اطبع("المحيط: ")
    اطبع(ش.احسب_المحيط())
نهاية

دالة رئيسية() رقم
    متغير م = مستطيل(10.0، 5.0)
    متغير د = دائرة(7.0)
    متغير ث = مثلث(6.0، 4.0، 5.0، 5.0، 6.0)
    
    # نفس الدالة تعمل مع أشكال مختلفة!
    اطبع("=== المستطيل ===")
    اطبع_تفاصيل_شكل(م)
    
    اطبع("=== الدائرة ===")
    اطبع_تفاصيل_شكل(د)
    
    اطبع("=== المثلث ===")
    اطبع_تفاصيل_شكل(ث)
    
    ارجع 0
نهاية
```

### 4. التجريد (Abstraction)

التجريد يعني إخفاء التعقيد وعرض الواجهة الأساسية فقط. المستخدم لا يحتاج لمعرفة كيف تعمل السيارة من الداخل، يحتاج فقط لمعرفة كيف يقودها.

```س
# صنف مجرد - لا يمكن إنشاء كائنات منه مباشرة
صنف مجرد قاعدة_بيانات
    عام:
        دالة مجردة اتصل(نص عنوان) منطقي
        دالة مجردة استعلم(نص استعلام) نص
        دالة مجردة أغلق()
نهاية

# تنفيذ لقاعدة بيانات SQL
صنف قاعدة_بيانات_SQL يرث قاعدة_بيانات
    خاص:
        منطقي متصل
        نص سلسلة_الاتصال
    
    عام:
        دالة منشئ()
            متصل = خطأ
        نهاية
        
        دالة اتصل(نص عنوان) منطقي
            اطبع("جاري الاتصال بقاعدة البيانات SQL...")
            سلسلة_الاتصال = عنوان
            # منطق الاتصال الفعلي
            متصل = صحيح
            اطبع("تم الاتصال بنجاح!")
            ارجع صحيح
        نهاية
        
        دالة استعلم(نص استعلام) نص
            إذا(!متصل)
                اطبع("خطأ: غير متصل بقاعدة البيانات")
                ارجع ""
            نهاية
            اطبع("تنفيذ استعلام SQL: ")
            اطبع(استعلام)
            # تنفيذ الاستعلام الفعلي
            ارجع "نتائج الاستعلام"
        نهاية
        
        دالة أغلق()
            إذا(متصل)
                اطبع("إغلاق الاتصال...")
                متصل = خطأ
            نهاية
        نهاية
نهاية

# تنفيذ لقاعدة بيانات NoSQL
صنف قاعدة_بيانات_NoSQL يرث قاعدة_بيانات
    خاص:
        منطقي متصل
    
    عام:
        دالة منشئ()
            متصل = خطأ
        نهاية
        
        دالة اتصل(نص عنوان) منطقي
            اطبع("جاري الاتصال بقاعدة البيانات NoSQL...")
            # منطق الاتصال المختلف
            متصل = صحيح
            اطبع("تم الاتصال!")
            ارجع صحيح
        نهاية
        
        دالة استعلم(نص استعلام) نص
            إذا(!متصل)
                ارجع ""
            نهاية
            اطبع("تنفيذ استعلام NoSQL: ")
            اطبع(استعلام)
            ارجع "نتائج المستند"
        نهاية
        
        دالة أغلق()
            متصل = خطأ
            اطبع("تم الإغلاق")
        نهاية
نهاية

# الكود العميل لا يهتم بنوع قاعدة البيانات
دالة احصل_بيانات(قاعدة_بيانات قب، نص استعلام) نص
    قب.اتصل("localhost:3306")
    متغير نتيجة = قب.استعلم(استعلام)
    قب.أغلق()
    ارجع نتيجة
نهاية

دالة رئيسية() رقم
    متغير sql = قاعدة_بيانات_SQL()
    متغير nosql = قاعدة_بيانات_NoSQL()
    
    # نفس الدالة تعمل مع أنواع مختلفة من قواعد البيانات
    احصل_بيانات(sql، "SELECT * FROM users")
    احصل_بيانات(nosql، "db.users.find()")
    
    ارجع 0
نهاية
```

## فوائد البرمجة الكائنية

### 1. تنظيم الكود
الكود يُصبح منظماً حول كائنات تُمثل مفاهيم حقيقية، مما يُسهّل فهمه وصيانته.

### 2. إعادة الاستخدام
من خلال الوراثة والتركيب، يمكن إعادة استخدام الكود بدلاً من تكراره.

### 3. التوسعية
يمكن إضافة ميزات جديدة بسهولة من خلال إنشاء أصناف جديدة أو توسيع الموجودة.

### 4. الصيانة
التغليف يجعل التعديلات محلية، فتغيير تنفيذ صنف لا يؤثر على الأصناف الأخرى.

### 5. المرونة
تعدد الأشكال يُتيح كتابة كود عام يعمل مع أنواع مختلفة.

## ملخص

البرمجة الكائنية في لغة ص تُقدم لك:

- **الأصناف والكائنات**: لتمثيل المفاهيم الحقيقية
- **التغليف**: لحماية البيانات وتوفير واجهات واضحة
- **الوراثة**: لبناء أصناف جديدة على الموجودة
- **تعدد الأشكال**: لكتابة كود عام ومرن
- **التجريد**: لإخفاء التعقيد وعرض الأساسيات

ابدأ بأمثلة بسيطة وتدرج في التعقيد. البرمجة الكائنية مهارة تُبنى مع الممارسة!
