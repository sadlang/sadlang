# الوراثة في لغة ص

## ما هي الوراثة؟

الوراثة هي آلية تُمكّن صنفاً جديداً (الصنف الابن) من أن يرث خصائص ودوال صنف موجود (الصنف الأب). هذا يُتيح إعادة استخدام الكود وبناء تسلسلات هرمية من الأصناف.

تخيل أنك تُصمم نظاماً للمركبات. كل المركبات تشترك في خصائص معينة (اللون، السرعة القصوى) وسلوكيات (تسريع، فرملة). بدلاً من تكرار هذه الخصائص في كل صنف، تُنشئ صنفاً أساسياً "مركبة" وترث منه أصنافاً محددة مثل "سيارة" و"دراجة نارية".

## الصيغة الأساسية

```س
صنف صنف_ابن يرث صنف_أب
    # خصائص ودوال إضافية
نهاية
```

## مثال توضيحي بسيط

```س
# الصنف الأب
صنف حيوان
    محمي:
        نص الاسم
        رقم العمر
    
    عام:
        دالة منشئ(نص اسم، رقم عمر)
            الاسم = اسم
            العمر = عمر
        نهاية
        
        دالة أكل()
            اطبع(الاسم)
            اطبع(" يأكل...")
        نهاية
        
        دالة نوم()
            اطبع(الاسم)
            اطبع(" ينام...")
        نهاية
        
        دالة وصف()
            اطبع("حيوان اسمه ")
            اطبع(الاسم)
            اطبع(" وعمره ")
            اطبع(العمر)
        نهاية
نهاية

# صنف ابن
صنف قطة يرث حيوان
    خاص:
        نص اللون
    
    عام:
        دالة منشئ(نص اسم، رقم عمر، نص لون)
            أب.منشئ(اسم، عمر)  # استدعاء مُنشئ الأب
            اللون = لون
        نهاية
        
        # دالة جديدة خاصة بالقطة
        دالة مواء()
            اطبع(الاسم)  # يمكن الوصول للخاصية المحمية
            اطبع(" يقول: مياو!")
        نهاية
        
        # إعادة تعريف دالة الأكل
        دالة أكل()
            اطبع(الاسم)
            اطبع(" يأكل السمك...")
        نهاية
نهاية

دالة رئيسية() رقم
    متغير ق = قطة("ميمي"، 3، "أبيض")
    
    ق.وصف()      # موروثة من حيوان
    ق.نوم()      # موروثة من حيوان
    ق.أكل()      # معاد تعريفها في قطة
    ق.مواء()     # خاصة بالقطة
    
    ارجع 0
نهاية
```

## أنواع الوراثة

### 1. الوراثة المفردة (Single Inheritance)

صنف ابن يرث من صنف أب واحد فقط:

```س
صنف أ
    عام:
        دالة دالة_أ()
            اطبع("من الصنف أ")
        نهاية
نهاية

صنف ب يرث أ
    عام:
        دالة دالة_ب()
            اطبع("من الصنف ب")
        نهاية
نهاية

دالة رئيسية() رقم
    متغير كائن = ب()
    كائن.دالة_أ()  # موروثة
    كائن.دالة_ب()  # خاصة
    ارجع 0
نهاية
```

### 2. الوراثة المتعددة المستويات (Multilevel Inheritance)

سلسلة من الوراثة: أ -> ب -> ج

```س
صنف كائن_حي
    محمي:
        منطقي حي
    
    عام:
        دالة منشئ()
            حي = صحيح
        نهاية
        
        دالة تنفس()
            اطبع("يتنفس...")
        نهاية
نهاية

صنف حيوان يرث كائن_حي
    محمي:
        نص الاسم
    
    عام:
        دالة منشئ(نص اسم)
            أب.منشئ()
            الاسم = اسم
        نهاية
        
        دالة تحرك()
            اطبع(الاسم)
            اطبع(" يتحرك...")
        نهاية
نهاية

صنف طائر يرث حيوان
    خاص:
        عشري طول_الجناح
    
    عام:
        دالة منشئ(نص اسم، عشري جناح)
            أب.منشئ(اسم)
            طول_الجناح = جناح
        نهاية
        
        دالة طير()
            اطبع(الاسم)
            اطبع(" يطير بجناحين طولهما ")
            اطبع(طول_الجناح)
            اطبع(" سم")
        نهاية
نهاية

دالة رئيسية() رقم
    متغير عصفور = طائر("العصفور"، 15.0)
    
    عصفور.تنفس()   # من كائن_حي
    عصفور.تحرك()   # من حيوان
    عصفور.طير()    # من طائر
    
    ارجع 0
نهاية
```

### 3. الوراثة الهرمية (Hierarchical Inheritance)

أصناف متعددة ترث من صنف أب واحد:

```س
صنف شكل
    محمي:
        نص اللون
    
    عام:
        دالة منشئ(نص لون)
            اللون = لون
        نهاية
        
        دالة افتراضية احسب_المساحة() عشري
            ارجع 0.0
        نهاية
        
        دالة احصل_اللون() نص
            ارجع اللون
        نهاية
نهاية

صنف مربع يرث شكل
    خاص:
        عشري الضلع
    
    عام:
        دالة منشئ(نص لون، عشري ضلع)
            أب.منشئ(لون)
            الضلع = ضلع
        نهاية
        
        دالة احسب_المساحة() عشري
            ارجع الضلع * الضلع
        نهاية
نهاية

صنف مستطيل يرث شكل
    خاص:
        عشري الطول
        عشري العرض
    
    عام:
        دالة منشئ(نص لون، عشري طول، عشري عرض)
            أب.منشئ(لون)
            الطول = طول
            العرض = عرض
        نهاية
        
        دالة احسب_المساحة() عشري
            ارجع الطول * العرض
        نهاية
نهاية

صنف دائرة يرث شكل
    خاص:
        عشري نصف_القطر
        ثابت عشري باي = 3.14159
    
    عام:
        دالة منشئ(نص لون، عشري نق)
            أب.منشئ(لون)
            نصف_القطر = نق
        نهاية
        
        دالة احسب_المساحة() عشري
            ارجع باي * نصف_القطر * نصف_القطر
        نهاية
نهاية

دالة رئيسية() رقم
    متغير م = مربع("أحمر"، 5.0)
    متغير مس = مستطيل("أزرق"، 10.0، 5.0)
    متغير د = دائرة("أخضر"، 7.0)
    
    اطبع("مساحة المربع الأحمر: ")
    اطبع(م.احسب_المساحة())
    
    اطبع("مساحة المستطيل الأزرق: ")
    اطبع(مس.احسب_المساحة())
    
    اطبع("مساحة الدائرة الخضراء: ")
    اطبع(د.احسب_المساحة())
    
    ارجع 0
نهاية
```

## استدعاء مُنشئ الصنف الأب

عند إنشاء كائن من صنف ابن، يجب استدعاء مُنشئ الصنف الأب لتهيئة الخصائص الموروثة:

```س
صنف موظف
    محمي:
        نص الاسم
        رقم الرقم_الوظيفي
        عشري الراتب_الأساسي
    
    عام:
        دالة منشئ(نص اسم، رقم رقم، عشري راتب)
            الاسم = اسم
            الرقم_الوظيفي = رقم
            الراتب_الأساسي = راتب
        نهاية
        
        دالة احصل_الاسم() نص
            ارجع الاسم
        نهاية
        
        دالة افتراضية احسب_الراتب() عشري
            ارجع الراتب_الأساسي
        نهاية
نهاية

صنف مدير يرث موظف
    خاص:
        عشري علاوة_الإدارة
        رقم عدد_المرؤوسين
    
    عام:
        دالة منشئ(نص اسم، رقم رقم، عشري راتب، عشري علاوة، رقم مرؤوسين)
            # استدعاء مُنشئ الأب أولاً
            أب.منشئ(اسم، رقم، راتب)
            
            # ثم تهيئة الخصائص الخاصة
            علاوة_الإدارة = علاوة
            عدد_المرؤوسين = مرؤوسين
        نهاية
        
        دالة احسب_الراتب() عشري
            # الراتب الأساسي + علاوة الإدارة
            ارجع الراتب_الأساسي + علاوة_الإدارة
        نهاية
        
        دالة احصل_عدد_المرؤوسين() رقم
            ارجع عدد_المرؤوسين
        نهاية
نهاية

صنف مهندس يرث موظف
    خاص:
        نص التخصص
        عشري علاوة_الخبرة
    
    عام:
        دالة منشئ(نص اسم، رقم رقم، عشري راتب، نص تخصص، عشري خبرة)
            أب.منشئ(اسم، رقم، راتب)
            التخصص = تخصص
            علاوة_الخبرة = خبرة
        نهاية
        
        دالة احسب_الراتب() عشري
            ارجع الراتب_الأساسي + علاوة_الخبرة
        نهاية
        
        دالة احصل_التخصص() نص
            ارجع التخصص
        نهاية
نهاية

دالة رئيسية() رقم
    متغير م = مدير("أحمد"، 1001، 5000.0، 2000.0، 10)
    متغير هـ = مهندس("محمد"، 1002، 4000.0، "برمجيات"، 1500.0)
    
    اطبع("المدير ")
    اطبع(م.احصل_الاسم())
    اطبع(" - الراتب: ")
    اطبع(م.احسب_الراتب())
    اطبع(" - يدير ")
    اطبع(م.احصل_عدد_المرؤوسين())
    اطبع(" موظفين")
    
    اطبع("المهندس ")
    اطبع(هـ.احصل_الاسم())
    اطبع(" - الراتب: ")
    اطبع(هـ.احسب_الراتب())
    اطبع(" - تخصص: ")
    اطبع(هـ.احصل_التخصص())
    
    ارجع 0
نهاية
```

## إعادة تعريف الدوال (Method Overriding)

الصنف الابن يمكنه إعادة تعريف دوال الصنف الأب لتغيير سلوكها:

```س
صنف طابعة
    محمي:
        نص الطراز
    
    عام:
        دالة منشئ(نص طراز)
            الطراز = طراز
        نهاية
        
        دالة اطبع(نص نص)
            اطبع("[طابعة أساسية] ")
            اطبع(نص)
        نهاية
        
        دالة معلومات()
            اطبع("طابعة طراز: ")
            اطبع(الطراز)
        نهاية
نهاية

صنف طابعة_ليزر يرث طابعة
    خاص:
        رقم سرعة_الصفحات
    
    عام:
        دالة منشئ(نص طراز، رقم سرعة)
            أب.منشئ(طراز)
            سرعة_الصفحات = سرعة
        نهاية
        
        # إعادة تعريف دالة الطباعة
        دالة اطبع(نص نص)
            اطبع("[طابعة ليزر - ")
            اطبع(سرعة_الصفحات)
            اطبع(" صفحة/دقيقة] ")
            اطبع(نص)
        نهاية
        
        # إعادة تعريف مع استدعاء دالة الأب
        دالة معلومات()
            أب.معلومات()  # استدعاء دالة الأب
            اطبع(" - ليزر - سرعة: ")
            اطبع(سرعة_الصفحات)
            اطبع(" صفحة/دقيقة")
        نهاية
نهاية

صنف طابعة_ملونة يرث طابعة
    خاص:
        منطقي لون_كامل
    
    عام:
        دالة منشئ(نص طراز، منطقي كامل)
            أب.منشئ(طراز)
            لون_كامل = كامل
        نهاية
        
        دالة اطبع(نص نص)
            إذا(لون_كامل)
                اطبع("[طابعة ملونة - لون كامل] ")
            وإلا
                اطبع("[طابعة ملونة - أساسي] ")
            نهاية
            اطبع(نص)
        نهاية
نهاية

دالة رئيسية() رقم
    متغير ط1 = طابعة("Basic-100")
    متغير ط2 = طابعة_ليزر("Laser-Pro"، 30)
    متغير ط3 = طابعة_ملونة("Color-Max"، صحيح)
    
    ط1.اطبع("مرحباً")
    ط2.اطبع("مرحباً")
    ط3.اطبع("مرحباً")
    
    ط2.معلومات()
    
    ارجع 0
نهاية
```

## الدوال الافتراضية (Virtual Functions)

الدوال الافتراضية تُمكّن تعدد الأشكال - استدعاء الدالة الصحيحة حسب نوع الكائن الفعلي:

```س
صنف حساب
    محمي:
        عشري الرصيد
        نص رقم_الحساب
    
    عام:
        دالة منشئ(نص رقم، عشري رصيد_ابتدائي)
            رقم_الحساب = رقم
            الرصيد = رصيد_ابتدائي
        نهاية
        
        # دالة افتراضية - يمكن إعادة تعريفها
        دالة افتراضية احسب_الفائدة() عشري
            ارجع 0.0  # بدون فائدة افتراضياً
        نهاية
        
        دالة احصل_الرصيد() عشري
            ارجع الرصيد
        نهاية
        
        دالة أضف_الفائدة()
            متغير فائدة = احسب_الفائدة()
            الرصيد = الرصيد + فائدة
            اطبع("تمت إضافة فائدة: ")
            اطبع(فائدة)
        نهاية
نهاية

صنف حساب_توفير يرث حساب
    خاص:
        عشري نسبة_الفائدة  # نسبة سنوية
    
    عام:
        دالة منشئ(نص رقم، عشري رصيد، عشري نسبة)
            أب.منشئ(رقم، رصيد)
            نسبة_الفائدة = نسبة
        نهاية
        
        # إعادة تعريف الدالة الافتراضية
        دالة احسب_الفائدة() عشري
            ارجع الرصيد * نسبة_الفائدة / 100.0
        نهاية
نهاية

صنف حساب_ذهبي يرث حساب
    خاص:
        عشري نسبة_الفائدة
        عشري مكافأة_ثابتة
    
    عام:
        دالة منشئ(نص رقم، عشري رصيد، عشري نسبة، عشري مكافأة)
            أب.منشئ(رقم، رصيد)
            نسبة_الفائدة = نسبة
            مكافأة_ثابتة = مكافأة
        نهاية
        
        دالة احسب_الفائدة() عشري
            # فائدة نسبية + مكافأة ثابتة
            ارجع (الرصيد * نسبة_الفائدة / 100.0) + مكافأة_ثابتة
        نهاية
نهاية

دالة اطبع_معلومات_حساب(حساب ح)
    اطبع("الرصيد: ")
    اطبع(ح.احصل_الرصيد())
    اطبع(" - الفائدة المتوقعة: ")
    اطبع(ح.احسب_الفائدة())  # تستدعي الدالة الصحيحة!
نهاية

دالة رئيسية() رقم
    متغير عادي = حساب("001"، 10000.0)
    متغير توفير = حساب_توفير("002"، 10000.0، 5.0)
    متغير ذهبي = حساب_ذهبي("003"، 10000.0، 7.0، 100.0)
    
    اطبع("=== الحساب العادي ===")
    اطبع_معلومات_حساب(عادي)
    
    اطبع("=== حساب التوفير ===")
    اطبع_معلومات_حساب(توفير)
    
    اطبع("=== الحساب الذهبي ===")
    اطبع_معلومات_حساب(ذهبي)
    
    # إضافة الفائدة
    توفير.أضف_الفائدة()
    اطبع("الرصيد الجديد لحساب التوفير: ")
    اطبع(توفير.احصل_الرصيد())
    
    ارجع 0
نهاية
```

## الأصناف المجردة (Abstract Classes)

الأصناف المجردة تُحدد واجهة لا يمكن إنشاء كائنات منها مباشرة:

```س
صنف مجرد وسيلة_نقل
    محمي:
        نص الاسم
        رقم عدد_الركاب
    
    عام:
        دالة منشئ(نص اسم، رقم ركاب)
            الاسم = اسم
            عدد_الركاب = ركاب
        نهاية
        
        # دالة مجردة - يجب تنفيذها في الأصناف الفرعية
        دالة مجردة تحرك()
        
        # دالة مجردة أخرى
        دالة مجردة احسب_وقت_الرحلة(عشري مسافة) عشري
        
        # دالة عادية
        دالة معلومات()
            اطبع("وسيلة النقل: ")
            اطبع(الاسم)
            اطبع(" - سعة: ")
            اطبع(عدد_الركاب)
            اطبع(" راكب")
        نهاية
نهاية

صنف سيارة يرث وسيلة_نقل
    خاص:
        عشري السرعة_القصوى
    
    عام:
        دالة منشئ(نص اسم، عشري سرعة)
            أب.منشئ(اسم، 5)
            السرعة_القصوى = سرعة
        نهاية
        
        # تنفيذ الدالة المجردة
        دالة تحرك()
            اطبع(الاسم)
            اطبع(" تسير على الطريق...")
        نهاية
        
        دالة احسب_وقت_الرحلة(عشري مسافة) عشري
            ارجع مسافة / السرعة_القصوى
        نهاية
نهاية

صنف طائرة يرث وسيلة_نقل
    خاص:
        عشري السرعة_القصوى
        عشري وقت_الإقلاع
    
    عام:
        دالة منشئ(نص اسم، رقم ركاب، عشري سرعة)
            أب.منشئ(اسم، ركاب)
            السرعة_القصوى = سرعة
            وقت_الإقلاع = 0.5  # نصف ساعة
        نهاية
        
        دالة تحرك()
            اطبع(الاسم)
            اطبع(" تطير في السماء...")
        نهاية
        
        دالة احسب_وقت_الرحلة(عشري مسافة) عشري
            ارجع (مسافة / السرعة_القصوى) + وقت_الإقلاع
        نهاية
نهاية

صنف قطار يرث وسيلة_نقل
    خاص:
        عشري السرعة
        رقم عدد_المحطات
    
    عام:
        دالة منشئ(نص اسم، رقم ركاب، عشري سرعة، رقم محطات)
            أب.منشئ(اسم، ركاب)
            السرعة = سرعة
            عدد_المحطات = محطات
        نهاية
        
        دالة تحرك()
            اطبع(الاسم)
            اطبع(" يسير على القضبان...")
        نهاية
        
        دالة احسب_وقت_الرحلة(عشري مسافة) عشري
            # إضافة وقت التوقف في المحطات
            ارجع (مسافة / السرعة) + (عدد_المحطات * 0.1)
        نهاية
نهاية

دالة خطط_رحلة(وسيلة_نقل و، عشري مسافة)
    و.معلومات()
    و.تحرك()
    اطبع("وقت الرحلة لمسافة ")
    اطبع(مسافة)
    اطبع(" كم: ")
```

## نصائح وأفضل الممارسات

### 1. استخدم الوراثة للعلاقة "هو-نوع-من"

```س
# صحيح - القطة هي نوع من الحيوان
صنف قطة يرث حيوان

# خطأ - المحرك ليس نوعاً من السيارة، بل جزء منها
صنف محرك يرث سيارة  # استخدم التركيب بدلاً من ذلك!
```

### 2. لا تُبالغ في عمق الوراثة

```س
# تجنب هذا - عميق جداً
صنف أ
صنف ب يرث أ
صنف ج يرث ب
صنف د يرث ج
صنف هـ يرث د

# أفضل - تسلسل أقصر وأوضح
```

### 3. استخدم `محمي` بدلاً من `عام` للخصائص الموروثة

```س
صنف أساس
    محمي:
        رقم قيمة  # الأصناف الفرعية تستطيع الوصول

صنف فرعي يرث أساس
    دالة استخدم_القيمة()
        اطبع(قيمة)  # يعمل!
    نهاية
نهاية
```

## ملخص

الوراثة في لغة ص تُقدم لك:

- **إعادة استخدام الكود**: وراثة الخصائص والدوال
- **التنظيم الهرمي**: بناء تسلسلات من الأصناف
- **إعادة التعريف**: تغيير سلوك الدوال الموروثة
- **الدوال الافتراضية**: تعدد الأشكال
- **الأصناف المجردة**: تحديد واجهات

استخدم الوراثة بحكمة لبناء كود منظم وقابل لإعادة الاستخدام!
