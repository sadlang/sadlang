# التجريد في لغة ص

## ما هو التجريد؟

التجريد (Abstraction) هو مبدأ أساسي في البرمجة الكائنية يعني إخفاء التفاصيل المعقدة وعرض الواجهة الأساسية فقط للمستخدم. التجريد يُمكّنك من التركيز على "ماذا" يفعل الكود بدلاً من "كيف" يفعله.

تخيل قيادة السيارة - أنت تعرف أن الضغط على دواسة الوقود يُسرّع السيارة، لكنك لا تحتاج لفهم كيف يعمل المحرك داخلياً. هذا هو التجريد!

## الأصناف المجردة

الصنف المجرد هو صنف لا يمكن إنشاء كائنات منه مباشرة. يُستخدم كقالب للأصناف الفرعية.

### الصيغة الأساسية

```س
صنف مجرد اسم_الصنف
    عام:
        # دالة مجردة - يجب تنفيذها في الأصناف الفرعية
        دالة مجردة اسم_الدالة()
        
        # دالة عادية - يمكن استخدامها أو إعادة تعريفها
        دالة دالة_عادية()
            # تنفيذ
        نهاية
نهاية
```

### مثال: نظام الدفع

```س
# صنف مجرد لطرق الدفع
صنف مجرد طريقة_دفع
    محمي:
        نص اسم_الطريقة
        عشري المبلغ
    
    عام:
        دالة منشئ(نص اسم)
            اسم_الطريقة = اسم
            المبلغ = 0.0
        نهاية
        
        # دوال مجردة - يجب تنفيذها
        دالة مجردة تحقق() منطقي
        دالة مجردة نفذ_الدفع(عشري مبلغ) منطقي
        دالة مجردة استرجع(عشري مبلغ) منطقي
        
        # دالة عادية
        دالة احصل_الاسم() نص
            ارجع اسم_الطريقة
        نهاية
        
        دالة اطبع_إيصال(عشري مبلغ)
            اطبع("═══════════════════════")
            اطبع("       إيصال الدفع      ")
            اطبع("═══════════════════════")
            اطبع("طريقة الدفع: ")
            اطبع(اسم_الطريقة)
            اطبع("المبلغ: ")
            اطبع(مبلغ)
            اطبع(" ريال")
            اطبع("═══════════════════════")
        نهاية
نهاية

# تنفيذ: الدفع بالبطاقة
صنف دفع_بطاقة يرث طريقة_دفع
    خاص:
        نص رقم_البطاقة
        نص تاريخ_الانتهاء
        نص CVV
    
    عام:
        دالة منشئ(نص رقم، نص تاريخ، نص cvv)
            أب.منشئ("بطاقة ائتمان")
            رقم_البطاقة = رقم
            تاريخ_الانتهاء = تاريخ
            CVV = cvv
        نهاية
        
        دالة تحقق() منطقي
            # التحقق من صحة البطاقة
            إذا(طول(رقم_البطاقة) != 16)
                اطبع("خطأ: رقم البطاقة غير صحيح")
                ارجع خطأ
            نهاية
            
            إذا(طول(CVV) != 3)
                اطبع("خطأ: رمز CVV غير صحيح")
                ارجع خطأ
            نهاية
            
            اطبع("✓ تم التحقق من البطاقة بنجاح")
            ارجع صحيح
        نهاية
        
        دالة نفذ_الدفع(عشري مبلغ) منطقي
            إذا(!تحقق())
                ارجع خطأ
            نهاية
            
            اطبع("جاري معالجة الدفع بالبطاقة...")
            اطبع("تم خصم ")
            اطبع(مبلغ)
            اطبع(" ريال من البطاقة")
            
            اطبع_إيصال(مبلغ)
            ارجع صحيح
        نهاية
        
        دالة استرجع(عشري مبلغ) منطقي
            اطبع("جاري استرجاع ")
            اطبع(مبلغ)
            اطبع(" ريال للبطاقة...")
            اطبع("تم الاسترجاع بنجاح")
            ارجع صحيح
        نهاية
نهاية

# تنفيذ: الدفع النقدي
صنف دفع_نقدي يرث طريقة_دفع
    خاص:
        عشري المبلغ_المستلم
    
    عام:
        دالة منشئ()
            أب.منشئ("دفع نقدي")
            المبلغ_المستلم = 0.0
        نهاية
        
        دالة تحقق() منطقي
            # الدفع النقدي دائماً صالح
            ارجع صحيح
        نهاية
        
        دالة نفذ_الدفع(عشري مبلغ) منطقي
            اطبع("المبلغ المطلوب: ")
            اطبع(مبلغ)
            اطبع(" ريال")
            
            # افتراض استلام المبلغ

            متغير الباقي = المبلغ_المستلم - مبلغ

            # محاكاة الاتصال بالخادم
            متصل = صحيح
            اطبع("✓ تم التحقق من الحساب")

        نهاية

            إذا(!تحقق())

        نهاية
نهاية

# نظام نقاط البيع
صنف نقطة_بيع
    عام:
        دالة معالجة_دفع(طريقة_دفع طريقة، عشري مبلغ) منطقي
            اطبع("=== بدء عملية الدفع ===")
            اطبع("طريقة الدفع: ")
            اطبع(طريقة.احصل_الاسم())
            
            متغير نتيجة = طريقة.نفذ_الدفع(مبلغ)
            
            إذا(نتيجة)
                اطبع("✓ تمت عملية الدفع بنجاح")
            وإلا
                اطبع("✗ فشلت عملية الدفع")
            نهاية
            
            ارجع نتيجة
        نهاية
نهاية

دالة رئيسية() رقم
    متغير نظام = نقطة_بيع()
    
    # دفع بالبطاقة
    اطبع("\n>>> اختبار الدفع بالبطاقة <<<")
    متغير بطاقة = دفع_بطاقة("1234567890123456"، "12/25"، "123")
    نظام.معالجة_دفع(بطاقة، 150.0)
    
    # دفع نقدي
    اطبع("\n>>> اختبار الدفع النقدي <<<")
    متغير نقد = دفع_نقدي()
    نظام.معالجة_دفع(نقد، 75.0)
    
    # دفع إلكتروني
    اطبع("\n>>> اختبار الدفع الإلكتروني <<<")
    متغير إلكتروني = دفع_إلكتروني("user@email.com"، "pass123")
    نظام.معالجة_دفع(إلكتروني، 200.0)
    
    ارجع 0
نهاية
```

## الواجهات (Interfaces)

الواجهة هي صنف مجرد بالكامل - جميع دواله مجردة. تُحدد عقداً يجب على الأصناف المُنفذة الالتزام به.

```س
# واجهة للكائنات القابلة للرسم
صنف مجرد قابل_للرسم
    عام:
        دالة مجردة ارسم()
        دالة مجردة امسح()
        دالة مجردة حرّك(رقم س، رقم ص)
نهاية

# واجهة للكائنات القابلة للتحريك
صنف مجرد قابل_للتحريك
    عام:
        دالة مجردة ابدأ_الحركة()
        دالة مجردة أوقف_الحركة()
        دالة مجردة احصل_السرعة() عشري
نهاية

# صنف يُنفذ واجهتين
صنف شخصية_لعبة يرث قابل_للرسم
    خاص:
        رقم س
        رقم ص
        عشري السرعة
        منطقي متحرك
        نص الاسم
    
    عام:
        دالة منشئ(نص اسم، رقم موقع_س، رقم موقع_ص)
            الاسم = اسم
            س = موقع_س
            ص = موقع_ص
            السرعة = 1.0
            متحرك = خطأ
        نهاية
        
        # تنفيذ واجهة قابل_للرسم
        دالة ارسم()
            اطبع("رسم الشخصية '")
            اطبع(الاسم)
            اطبع("' في الموقع (")
            اطبع(س)
            اطبع("، ")
            اطبع(ص)
            اطبع(")")
        نهاية
        
        دالة امسح()
            اطبع("مسح الشخصية '")
            اطبع(الاسم)
            اطبع("'")
        نهاية
        
        دالة حرّك(رقم جديد_س، رقم جديد_ص)
            امسح()
            س = جديد_س
            ص = جديد_ص
            ارسم()
        نهاية
نهاية

صنف زر يرث قابل_للرسم
    خاص:
        رقم س
        رقم ص
        رقم العرض
        رقم الارتفاع
        نص النص
    
    عام:
        دالة منشئ(نص نص، رقم موقع_س، رقم موقع_ص، رقم عرض، رقم ارتفاع)
            النص = نص
            س = موقع_س
            ص = موقع_ص
            العرض = عرض
            الارتفاع = ارتفاع
        نهاية
        
        دالة ارسم()
            اطبع("┌")
            متغير ع = 0
            طالما(ع < العرض)
                اطبع("─")
                ع = ع + 1
            نهاية
            اطبع("┐")
            
            اطبع("│ ")
            اطبع(النص)
            اطبع(" │")
            
            اطبع("└")
            ع = 0
            طالما(ع < العرض)
                اطبع("─")
                ع = ع + 1
            نهاية
            اطبع("┘")
        نهاية
        
        دالة امسح()
            اطبع("مسح الزر '")
            اطبع(النص)
            اطبع("'")
        نهاية
        
        دالة حرّك(رقم جديد_س، رقم جديد_ص)
            س = جديد_س
            ص = جديد_ص
        نهاية
نهاية

# دالة تعمل مع أي كائن قابل للرسم
دالة ارسم_الكل(قابل_للرسم[] عناصر، رقم عدد)
    اطبع("=== رسم جميع العناصر ===")
    متغير ع = 0
    طالما(ع < عدد)
        عناصر[ع].ارسم()
        ع = ع + 1
    نهاية
نهاية

دالة رئيسية() رقم
    متغير شخصية = شخصية_لعبة("البطل"، 100، 200)
    متغير زر1 = زر("ابدأ"، 50، 50، 10، 3)
    متغير زر2 = زر("خروج"، 50، 100، 10، 3)
    
    متغير عناصر = قابل_للرسم[3]
    عناصر[0] = شخصية
    عناصر[1] = زر1
    عناصر[2] = زر2
    
    ارسم_الكل(عناصر، 3)
    
    # تحريك الشخصية
    اطبع("\n=== تحريك الشخصية ===")
    شخصية.حرّك(150، 250)
    
    ارجع 0
نهاية
```

            اطبع(الاسم)
            مفتوح = خطأ
        نهاية
        
        دالة اقرأ() نص
            إذا(!مفتوح)
                اطبع("خطأ: الملف غير مفتوح")
                ارجع ""
            نهاية
            ارجع المحتوى
        نهاية
        
        دالة اكتب(نص محتوى) منطقي
            إذا(!مفتوح)
                اطبع("خطأ: الملف غير مفتوح")
                ارجع خطأ
            نهاية
            
            المحتوى = محتوى
            الحجم = طول(محتوى)
            اطبع("تم كتابة ")
            اطبع(الحجم)
            اطبع(" حرف في الملف")
            ارجع صحيح
        نهاية
        
        دالة احصل_عدد_الأسطر() رقم

                ع = ع + 1

        نهاية
نهاية

## فوائد التجريد

1. **تبسيط التعقيد**: إخفاء التفاصيل الداخلية
2. **تقليل الاعتماد**: الكود يعتمد على الواجهة لا التنفيذ
3. **سهولة التوسع**: إضافة تنفيذات جديدة بدون تغيير الكود
4. **تحسين الصيانة**: التغييرات محصورة في التنفيذ
5. **إعادة الاستخدام**: الواجهات قابلة للاستخدام في سياقات مختلفة

## ملخص

التجريد في لغة ص يُقدم لك:

- **الأصناف المجردة**: قوالب للأصناف الفرعية
- **الدوال المجردة**: عقود يجب تنفيذها
- **الواجهات**: تحديد السلوك المتوقع
- **إخفاء التعقيد**: واجهة بسيطة للمستخدم

استخدم التجريد لبناء أنظمة مرنة وقابلة للتوسع!
