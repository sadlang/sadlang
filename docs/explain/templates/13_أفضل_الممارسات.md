# أفضل الممارسات في البرمجة الكائنية

## مقدمة

هذا الدليل يُقدم أفضل الممارسات والأنماط التي يجب اتباعها عند كتابة كود كائني في لغة ص. اتباع هذه الممارسات يُساعد في كتابة كود نظيف، قابل للصيانة، وسهل الفهم.

## 1. مبادئ التصميم الأساسية (SOLID)

### S - مبدأ المسؤولية الواحدة (Single Responsibility)

كل صنف يجب أن يكون له سبب واحد للتغيير. الصنف يجب أن يفعل شيئاً واحداً ويفعله جيداً.

```س
# ❌ خطأ - صنف يفعل أشياء كثيرة
صنف موظف_سيء
    عام:
        نص الاسم
        
        دالة احسب_الراتب() عشري
            # حساب الراتب
        نهاية
        
        دالة احفظ_في_قاعدة_البيانات()
            # حفظ في قاعدة البيانات
        نهاية
        
        دالة اطبع_تقرير()
            # طباعة التقرير
        نهاية
        
        دالة أرسل_بريد()
            # إرسال بريد إلكتروني
        نهاية
نهاية

# ✓ صحيح - فصل المسؤوليات
صنف موظف
    عام:
        نص الاسم
        عشري الراتب
        
        دالة احصل_الاسم() نص
            ارجع الاسم
        نهاية
        
        دالة احصل_الراتب() عشري
            ارجع الراتب
        نهاية
نهاية

صنف حاسب_الرواتب
    عام:
        دالة احسب_الراتب_الشهري(موظف م) عشري
            ارجع م.احصل_الراتب()
        نهاية
        
        دالة احسب_الضريبة(موظف م) عشري
            ارجع م.احصل_الراتب() * 0.15
        نهاية
نهاية

صنف مخزن_الموظفين
    عام:
        دالة احفظ(موظف م)
            # حفظ في قاعدة البيانات
        نهاية
        
        دالة حمّل(رقم معرف) موظف
            # تحميل من قاعدة البيانات
        نهاية
نهاية

صنف منشئ_التقارير
    عام:
        دالة أنشئ_تقرير(موظف م) نص
            # إنشاء تقرير
        نهاية
نهاية
```

### O - مبدأ الفتح/الإغلاق (Open/Closed)

الأصناف يجب أن تكون مفتوحة للتوسع، مغلقة للتعديل.

```س
# ❌ خطأ - تعديل الصنف عند إضافة نوع جديد
صنف حاسب_الخصم_سيء
    عام:
        دالة احسب_الخصم(نص نوع_العميل، عشري مبلغ) عشري
            إذا(نوع_العميل == "عادي")
                ارجع 0.0
            وإلا إذا(نوع_العميل == "فضي")
                ارجع مبلغ * 0.05
            وإلا إذا(نوع_العميل == "ذهبي")
                ارجع مبلغ * 0.10
            # كل نوع جديد يتطلب تعديل هذا الصنف!
            نهاية
            ارجع 0.0
        نهاية
نهاية

# ✓ صحيح - استخدام تعدد الأشكال
صنف مجرد عميل
    محمي:
        نص الاسم
    
    عام:
        دالة مجردة احسب_الخصم(عشري مبلغ) عشري
نهاية

صنف عميل_عادي يرث عميل
    عام:
        دالة احسب_الخصم(عشري مبلغ) عشري
            ارجع 0.0
        نهاية
نهاية

صنف عميل_فضي يرث عميل
    عام:
        دالة احسب_الخصم(عشري مبلغ) عشري
            ارجع مبلغ * 0.05
        نهاية
نهاية

صنف عميل_ذهبي يرث عميل
    عام:
        دالة احسب_الخصم(عشري مبلغ) عشري
            ارجع مبلغ * 0.10
        نهاية
نهاية

# إضافة نوع جديد بدون تعديل الكود الموجود
صنف عميل_ماسي يرث عميل
    عام:
        دالة احسب_الخصم(عشري مبلغ) عشري
            ارجع مبلغ * 0.20
        نهاية
نهاية
```

### L - مبدأ استبدال ليسكوف (Liskov Substitution)

الأصناف الفرعية يجب أن تكون قابلة للاستبدال بأصنافها الأساسية.

```س
# ❌ خطأ - المربع يُكسر سلوك المستطيل
صنف مستطيل
    خاص:
        عشري الطول
        عشري العرض
    
    عام:
        دالة عيّن_الطول(عشري ط)
            الطول = ط
        نهاية
        
        دالة عيّن_العرض(عشري ع)
            العرض = ع
        نهاية
        
        دالة المساحة() عشري
            ارجع الطول * العرض
        نهاية
نهاية

صنف مربع يرث مستطيل
    عام:
        # يُكسر السلوك المتوقع!
        دالة عيّن_الطول(عشري ط)
            الطول = ط
            العرض = ط  # يُغير العرض أيضاً!
        نهاية
        
        دالة عيّن_العرض(عشري ع)
            العرض = ع
            الطول = ع  # يُغير الطول أيضاً!
        نهاية
نهاية

# ✓ صحيح - استخدام التركيب بدلاً من الوراثة
صنف مجرد شكل
    عام:
        دالة مجردة المساحة() عشري
نهاية

صنف مستطيل يرث شكل
    خاص:
        عشري الطول
        عشري العرض
    
    عام:
        دالة منشئ(عشري طول، عشري عرض)
            الطول = طول
            العرض = عرض
        نهاية
        
        دالة المساحة() عشري
            ارجع الطول * العرض
        نهاية
نهاية

صنف مربع يرث شكل
    خاص:
        عشري الضلع
    
    عام:
        دالة منشئ(عشري ضلع)
            الضلع = ضلع
        نهاية
        
        دالة المساحة() عشري
            ارجع الضلع * الضلع
        نهاية
نهاية
```

### I - مبدأ فصل الواجهات (Interface Segregation)

لا تُجبر الأصناف على تنفيذ دوال لا تحتاجها.

```س
# ❌ خطأ - واجهة كبيرة جداً
صنف مجرد عامل_سيء
    عام:
        دالة مجردة اعمل()
        دالة مجردة كل()
        دالة مجردة نم()
        دالة مجردة اجتمع()
        دالة مجردة قدّم_تقرير()
نهاية

صنف روبوت يرث عامل_سيء
    عام:
        دالة اعمل()
            # يعمل
        نهاية
        
        دالة كل()
            # الروبوت لا يأكل! لكن مُجبر على التنفيذ
        نهاية
        
        دالة نم()
            # الروبوت لا ينام! لكن مُجبر على التنفيذ
        نهاية
        
        # ... باقي الدوال التي لا يحتاجها
نهاية

# ✓ صحيح - واجهات صغيرة ومُركزة
صنف مجرد قابل_للعمل
    عام:
        دالة مجردة اعمل()
نهاية

صنف مجرد قابل_للأكل
    عام:
        دالة مجردة كل()
نهاية

صنف مجرد قابل_للنوم
    عام:
        دالة مجردة نم()
نهاية

صنف موظف_بشري يرث قابل_للعمل
    عام:
        دالة اعمل()
            # يعمل
        نهاية
        
        # يمكن تنفيذ واجهات أخرى حسب الحاجة
نهاية

صنف روبوت يرث قابل_للعمل
    عام:
        دالة اعمل()
            # يعمل فقط - لا حاجة لتنفيذ أكل أو نوم
        نهاية
نهاية
```

### D - مبدأ عكس التبعية (Dependency Inversion)

اعتمد على التجريدات، لا على التنفيذات المحددة.

```س
# ❌ خطأ - اعتماد مباشر على تنفيذ محدد
صنف تقرير_سيء
    خاص:
        قاعدة_بيانات_MySQL قب
    
    عام:
        دالة منشئ()
            قب = قاعدة_بيانات_MySQL()  # مرتبط بـ MySQL!
        نهاية
        
        دالة أنشئ_تقرير()
            متغير بيانات = قب.استعلم("SELECT * FROM sales")
            # إنشاء التقرير
        نهاية
نهاية

# ✓ صحيح - اعتماد على تجريد
صنف مجرد قاعدة_بيانات
    عام:
        دالة مجردة استعلم(نص استعلام) نص
        دالة مجردة احفظ(نص بيانات)
نهاية

صنف قاعدة_بيانات_MySQL يرث قاعدة_بيانات
    عام:
        دالة استعلم(نص استعلام) نص
            # تنفيذ MySQL
        نهاية
        
        دالة احفظ(نص بيانات)
            # تنفيذ MySQL
        نهاية
نهاية

صنف قاعدة_بيانات_PostgreSQL يرث قاعدة_بيانات
    عام:
        دالة استعلم(نص استعلام) نص
            # تنفيذ PostgreSQL
        نهاية
        
        دالة احفظ(نص بيانات)
            # تنفيذ PostgreSQL
        نهاية
نهاية

صنف تقرير
    خاص:
        قاعدة_بيانات قب  # يعتمد على التجريد
    
    عام:
        دالة منشئ(قاعدة_بيانات قاعدة)
            قب = قاعدة  # يتم تمرير التنفيذ من الخارج
        نهاية
        
        دالة أنشئ_تقرير()
            متغير بيانات = قب.استعلم("SELECT * FROM sales")
            # إنشاء التقرير
        نهاية
نهاية

دالة رئيسية() رقم
    # يمكن تغيير قاعدة البيانات بسهولة
    متغير mysql = قاعدة_بيانات_MySQL()
    متغير تقرير1 = تقرير(mysql)
    
    متغير postgres = قاعدة_بيانات_PostgreSQL()
    متغير تقرير2 = تقرير(postgres)
    
    ارجع 0
نهاية
```

## 2. أفضل ممارسات التسمية

### تسمية الأصناف

```س
# ✓ أسماء واضحة ومعبرة
صنف حساب_بنكي
صنف طالب_جامعي
صنف طلب_شراء
صنف مدير_الملفات

# ❌ أسماء غير واضحة
صنف ح
صنف data
صنف Manager  # غير عربي
صنف شيء
```

### تسمية الدوال

```س
# ✓ أفعال واضحة
دالة احسب_المجموع()
دالة احصل_الاسم()
دالة عيّن_القيمة()
دالة هل_صالح()
دالة أنشئ_تقرير()

# ❌ أسماء غير واضحة
دالة افعل()
دالة معالجة()
دالة func1()
```

### تسمية المتغيرات

```س
# ✓ أسماء وصفية
متغير عدد_الطلاب = 0
متغير إجمالي_المبيعات = 0.0
متغير قائمة_المنتجات = []
متغير هل_نشط = صحيح

# ❌ أسماء غير واضحة
متغير س = 0
متغير temp = 0
متغير data = []
متغير flag = صحيح
```

## 3. التوثيق والتعليقات

```س
# توثيق الصنف
# صنف يمثل حساباً مصرفياً
# يُوفر عمليات الإيداع والسحب والاستعلام عن الرصيد
صنف حساب_مصرفي
    خاص:
        نص رقم_الحساب    # رقم الحساب الفريد
        عشري الرصيد       # الرصيد الحالي بالريال
    
    عام:
        # إنشاء حساب جديد
        # المعاملات:
        #   رقم: رقم الحساب
        #   رصيد_ابتدائي: الرصيد الابتدائي (افتراضي 0)
        دالة منشئ(نص رقم، عشري رصيد_ابتدائي)
            رقم_الحساب = رقم
            الرصيد = رصيد_ابتدائي
        نهاية
        
        # إيداع مبلغ في الحساب
        # المعاملات:
        #   مبلغ: المبلغ المراد إيداعه (يجب أن يكون موجباً)
        # الإرجاع: صحيح إذا نجحت العملية، خطأ إذا فشلت
        دالة إيداع(عشري مبلغ) منطقي
            إذا(مبلغ <= 0)
                ارجع خطأ
            نهاية
            الرصيد = الرصيد + مبلغ
            ارجع صحيح
        نهاية
نهاية
```

## 4. معالجة الأخطاء

```س
صنف نتيجة
    عام:
        منطقي نجاح
        نص رسالة
        
        دالة ثابتة نجاح() نتيجة
            متغير ن = نتيجة()
            ن.نجاح = صحيح
            ارجع ن
        نهاية
        
        دالة ثابتة فشل(نص رسالة) نتيجة
            متغير ن = نتيجة()
            ن.نجاح = خطأ
            ن.رسالة = رسالة
            ارجع ن
        نهاية
نهاية

صنف حساب
    خاص:
        عشري الرصيد
    
    عام:
        دالة سحب(عشري مبلغ) نتيجة
            إذا(مبلغ <= 0)
                ارجع نتيجة::فشل("المبلغ يجب أن يكون موجباً")
            نهاية
            
            إذا(مبلغ > الرصيد)
                ارجع نتيجة::فشل("الرصيد غير كافٍ")
            نهاية
            
            الرصيد = الرصيد - مبلغ
            ارجع نتيجة::نجاح()
        نهاية
نهاية

دالة رئيسية() رقم
    متغير ح = حساب()
    
    متغير نتيجة = ح.سحب(100.0)
    إذا(!نتيجة.نجاح)
        اطبع("خطأ: ")
        اطبع(نتيجة.رسالة)
    نهاية
    
    ارجع 0
نهاية
```

## 6. قائمة التحقق

### عند كتابة صنف جديد:
- [ ] هل الصنف له مسؤولية واحدة؟
- [ ] هل الأسماء واضحة ومعبرة؟
- [ ] هل البيانات الحساسة محمية (خاصة)؟
- [ ] هل يوجد توثيق كافٍ؟
- [ ] هل تم التعامل مع الأخطاء؟

### عند استخدام الوراثة:
- [ ] هل العلاقة "هو نوع من"؟
- [ ] هل الصنف الفرعي يمكن استبداله بالأب؟
- [ ] هل التركيب أفضل؟

### عند كتابة دالة:
- [ ] هل الدالة تفعل شيئاً واحداً؟
- [ ] هل الاسم يصف ما تفعله؟
- [ ] هل المعاملات واضحة؟
- [ ] هل تُرجع قيمة مناسبة؟

## ملخص

اتبع هذه الممارسات لكتابة كود أفضل:

1. **SOLID**: المبادئ الخمسة للتصميم الجيد
2. **التسمية**: أسماء واضحة ومعبرة
3. **التوثيق**: تعليقات مفيدة ومحدثة
4. **معالجة الأخطاء**: تعامل مع الحالات الاستثنائية
5. **التركيب**: فضّله على الوراثة عند الإمكان

الكود الجيد هو كود يمكن للآخرين (وأنت في المستقبل) فهمه وصيانته بسهولة!
