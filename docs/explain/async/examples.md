## مقدمة

في هذا الملف، سنستعرض مجموعة من الأمثلة العملية الواقعية التي تُظهر كيفية استخدام البرمجة غير المتزامنة في لغة ص لحل مشاكل برمجية حقيقية. كل مثال مكتمل ويمكن تشغيله مباشرة، ويتضمن شروحات مفصلة لكل جزء.

## مثال 1: نظام إدارة المخزون

في هذا المثال، سنبني نظاماً بسيطاً لإدارة المخزون يستخدم الدوال غير المتزامنة للتعامل مع عمليات قاعدة البيانات:

```
# تعريف دالة للبحث عن منتج في قاعدة البيانات
غير_متزامن دالة البحث_عن_منتج(نص رمز_المنتج)
    اطبع("البحث عن المنتج: " + رمز_المنتج)
    
    # محاكاة البحث في قاعدة البيانات
    # في الحقيقة هنا سنتصل بقاعدة البيانات
    
    إذا رمز_المنتج == "P001"
        إرجاع "لابتوب ديل، الكمية: 15"
    نهاية
    
    إذا رمز_المنتج == "P002"
        إرجاع "ماوس لوجيتك، الكمية: 50"
    نهاية
    
    إرجاع "المنتج غير موجود"
نهاية

# تعريف دالة للتحقق من كمية المنتج
غير_متزامن دالة التحقق_من_الكمية(نص رمز_المنتج، رقم كمية_مطلوبة)
    اطبع("التحقق من توفر الكمية...")
    
    # جلب معلومات المنتج
    نص معلومات = انتظر البحث_عن_منتج(رمز_المنتج)
    
    إذا معلومات == "المنتج غير موجود"
        إرجاع "غير موجود"
    نهاية
    
    # في الحقيقة هنا سنستخرج الكمية من النص
    # للتبسيط، سنفترض أن الكمية كافية
    إرجاع "متوفر"
نهاية

# تعريف دالة لحجز المنتجات
غير_متزامن دالة حجز_منتجات(نص رمز_المنتج، رقم الكمية)
    اطبع("حجز " + الكمية + " وحدات من المنتج: " + رمز_المنتج)
    
    # في الحقيقة هنا سنقوم بتحديث قاعدة البيانات
    إرجاع "تم الحجز برقم: " + 1001 + رقم_المنتج
نهاية

# تعريف دالة لإنشاء فاتورة
غير_متزامن دالة إنشاء_فاتورة(نص رقم_الحجز، عدد المبلغ)
    اطبع("إنشاء فاتورة للحجز: " + رقم_الحجز)
    إرجاع "فاتورة رقم: INV-" + رقم_الحجز + "، المبلغ: " + المبلغ + " ريال"
نهاية

# الدالة الرئيسية لعملية البيع الكاملة
غير_متزامن دالة عملية_بيع_منتج(نص رمز_المنتج، رقم الكمية، عدد سعر_الوحدة)
    اطبع("=== بدء عملية بيع جديدة ===")
    اطبع("المنتج: " + رمز_المنتج + "، الكمية: " + الكمية)
    
    # الخطوة 1: البحث عن المنتج
    نص معلومات_المنتج = انتظر البحث_عن_منتج(رمز_المنتج)
    اطبع("معلومات المنتج: " + معلومات_المنتج)
    
    إذا معلومات_المنتج == "المنتج غير موجود"
        اطبع("خطأ: المنتج غير موجود في المخزون")
        إرجاع "فشل"
    نهاية
    
    # الخطوة 2: التحقق من الكمية
    نص حالة_التوفر = انتظر التحقق_من_الكمية(رمز_المنتج، الكمية)
    
    إذا حالة_التوفر != "متوفر"
        اطبع("خطأ: الكمية المطلوبة غير متوفرة")
        إرجاع "فشل"
    نهاية
    
    # الخطوة 3: حجز المنتجات
    نص رقم_الحجز = انتظر حجز_منتجات(رمز_المنتج، الكمية)
    اطبع("نجح الحجز: " + رقم_الحجز)
    
    # الخطوة 4: حساب المبلغ وإنشاء الفاتورة
    عدد المبلغ_الإجمالي = الكمية * سعر_الوحدة
    نص الفاتورة = انتظر إنشاء_فاتورة(رقم_الحجز، المبلغ_الإجمالي)
    اطبع(الفاتورة)
    
    اطبع("=== اكتملت عملية البيع بنجاح ===")
    إرجاع "نجح"
نهاية

# تشغيل البرنامج
دالة رئيسية()
    اطبع_سطر("*** نظام إدارة المخزون ***")
    
    # عملية بيع أولى
    نص نتيجة1 = عملية_بيع_منتج("P001"، 5، 3500)
    اطبع_سطر("نتيجة العملية: " + نتيجة1)
    
    اطبع_سطر("")
    اطبع_سطر("---")
    اطبع_سطر("")
    
    # عملية بيع ثانية
    نص نتيجة2 = عملية_بيع_منتج("P002"، 10، 45)
    اطبع_سطر("نتيجة العملية: " + نتيجة2)
نهاية
```

هذا المثال يُظهر عملية بيع كاملة في نظام مخزون، حيث كل خطوة تعتمد على نجاح الخطوة السابقة. استخدام البرمجة غير المتزامنة يجعل الكود قابلاً للتوسع بسهولة لتضمين عمليات أكثر تعقيداً.

## مثال 2: نظام معالجة الطلبات المتعددة

في هذا المثال، سنبني نظاماً يعالج عدة طلبات في نفس الوقت، مما يُظهر قوة البرمجة غير المتزامنة:

```
# دالة لمعالجة طلب واحد
غير_متزامن دالة معالجة_طلب_واحد(رقم رقم_الطلب، نص نوع_الطلب)
    اطبع("بدء معالجة الطلب رقم: " + رقم_الطلب + " من نوع: " + نوع_الطلب)
    
    # محاكاة وقت المعالجة المختلف حسب النوع
    إذا نوع_الطلب == "عادي"
        اطبع("معالجة عادية للطلب: " + رقم_الطلب)
    نهاية
    
    إذا نوع_الطلب == "عاجل"
        اطبع("معالجة عاجلة للطلب: " + رقم_الطلب)
    نهاية
    
    إذا نوع_الطلب == "خاص"
        اطبع("معالجة خاصة للطلب: " + رقم_الطلب)
    نهاية
    
    اطبع("اكتملت معالجة الطلب: " + رقم_الطلب)
    إرجاع "تم معالجة الطلب " + رقم_الطلب
نهاية

# دالة لمعالجة دفعة من الطلبات
غير_متزامن دالة معالجة_دفعة_طلبات()
    اطبع("=== بدء معالجة دفعة جديدة من الطلبات ===")
    
    # معالجة الطلب الأول
    نص نتيجة1 = انتظر معالجة_طلب_واحد(101، "عادي")
    اطبع("✓ " + نتيجة1)
    
    # معالجة الطلب الثاني
    نص نتيجة2 = انتظر معالجة_طلب_واحد(102، "عاجل")
    اطبع("✓ " + نتيجة2)
    
    # معالجة الطلب الثالث
    نص نتيجة3 = انتظر معالجة_طلب_واحد(103، "خاص")
    اطبع("✓ " + نتيجة3)
    
    # معالجة الطلب الرابع
    نص نتيجة4 = انتظر معالجة_طلب_واحد(104، "عادي")
    اطبع("✓ " + نتيجة4)
    
    # معالجة الطلب الخامس
    نص نتيجة5 = انتظر معالجة_طلب_واحد(105، "عاجل")
    اطبع("✓ " + نتيجة5)
    
    اطبع("=== اكتملت معالجة جميع الطلبات ===")
    إرجاع "تمت معالجة 5 طلبات"
نهاية

# دالة لعرض إحصائيات المعالجة
غير_متزامن دالة عرض_إحصائيات()
    اطبع("جاري جمع الإحصائيات...")
    
    نص نتيجة = انتظر معالجة_دفعة_طلبات()
    
    اطبع("")
    اطبع("--- إحصائيات المعالجة ---")
    اطبع(نتيجة)
    اطبع("الحالة: نجح")
    اطبع("----------------------------")
    
    إرجاع "تم"
نهاية

دالة رئيسية()
    اطبع_سطر("*** نظام معالجة الطلبات ***")
    اطبع_سطر("")
    
    عرض_إحصائيات()
نهاية
```

هذا المثال يُظهر كيف يمكن معالجة عدة طلبات بشكل منظم. في المستقبل، عندما نفعّل التنفيذ المتوازي الكامل، ستعمل هذه الطلبات بشكل متزامن فعلي، مما يسرّع المعالجة بشكل كبير.

## مثال 3: نظام تسجيل المستخدمين

نظام شامل لتسجيل مستخدم جديد مع جميع خطوات التحقق:

```
# دالة للتحقق من توفر اسم المستخدم
غير_متزامن دالة التحقق_من_اسم_المستخدم(نص اسم_المستخدم)
    اطبع("التحقق من توفر اسم المستخدم: " + اسم_المستخدم)
    
    # قائمة الأسماء المحجوزة (في الحقيقة من قاعدة بيانات)
    إذا اسم_المستخدم == "admin"
        إرجاع "محجوز"
    نهاية
    
    إذا اسم_المستخدم == "root"
        إرجاع "محجوز"
    نهاية
    
    # التحقق من الطول
    إذا اسم_المستخدم == ""
        إرجاع "قصير جداً"
    نهاية
    
    إرجاع "متوفر"
نهاية

# دالة للتحقق من قوة كلمة المرور
غير_متزامن دالة التحقق_من_قوة_كلمة_المرور(نص كلمة_المرور)
    اطبع("التحقق من قوة كلمة المرور...")
    
    # التحقق من الطول الأدنى
    إذا كلمة_المرور == ""
        إرجاع "قصيرة جداً"
    نهاية
    
    # في الحقيقة هنا سنتحقق من وجود أرقام وحروف كبيرة وصغيرة ورموز
    إرجاع "قوية"
نهاية

# دالة للتحقق من صحة البريد الإلكتروني
غير_متزامن دالة التحقق_من_البريد(نص بريد)
    اطبع("التحقق من صحة البريد: " + بريد)
    
    # التحقق من وجود @
    # في الحقيقة هنا سنستخدم تعبير نمطي للتحقق
    
    إذا بريد == ""
        إرجاع "فارغ"
    نهاية
    
    إرجاع "صحيح"
نهاية

# دالة للتحقق من عدم تكرار البريد
غير_متزامن دالة التحقق_من_تكرار_البريد(نص بريد)
    اطبع("التحقق من عدم تسجيل البريد مسبقاً...")
    
    # البحث في قاعدة البيانات
    إذا بريد == "used@example.com"
        إرجاع "مستخدم"
    نهاية
    
    إرجاع "جديد"
نهاية

# دالة لإنشاء ملف المستخدم في قاعدة البيانات
غير_متزامن دالة إنشاء_ملف_مستخدم(نص اسم، نص بريد)
    اطبع("إنشاء ملف مستخدم جديد في قاعدة البيانات...")
    
    # عملية إدخال في قاعدة البيانات
    عدد معرف_المستخدم = 1001
    
    اطبع("تم إنشاء المستخدم بمعرف: " + معرف_المستخدم)
    إرجاع معرف_المستخدم
نهاية

# دالة لإرسال بريد التأكيد
غير_متزامن دالة إرسال_بريد_التأكيد(نص بريد، عدد معرف)
    اطبع("إرسال بريد تأكيد إلى: " + بريد)
    
    # عملية إرسال البريد
    اطبع("تم إرسال رابط التفعيل للمعرف: " + معرف)
    
    إرجاع "تم الإرسال"
نهاية

# الدالة الرئيسية لعملية التسجيل الكاملة
غير_متزامن دالة تسجيل_مستخدم_جديد(نص اسم_المستخدم، نص كلمة_المرور، نص البريد)
    اطبع("=== بدء عملية تسجيل مستخدم جديد ===")
    اطبع("اسم المستخدم: " + اسم_المستخدم)
    اطبع("البريد: " + البريد)
    اطبع("")
    
    # التحقق 1: اسم المستخدم
    نص حالة_الاسم = انتظر التحقق_من_اسم_المستخدم(اسم_المستخدم)
    
    إذا حالة_الاسم != "متوفر"
        اطبع("✗ خطأ: اسم المستخدم " + حالة_الاسم)
        إرجاع "فشل: اسم المستخدم " + حالة_الاسم
    نهاية
    اطبع("✓ اسم المستخدم متوفر")
    
    # التحقق 2: قوة كلمة المرور
    نص حالة_كلمة_المرور = انتظر التحقق_من_قوة_كلمة_المرور(كلمة_المرور)
    
    إذا حالة_كلمة_المرور != "قوية"
        اطبع("✗ خطأ: كلمة المرور " + حالة_كلمة_المرور)
        إرجاع "فشل: كلمة المرور " + حالة_كلمة_المرور
    نهاية
    اطبع("✓ كلمة المرور قوية")
    
    # التحقق 3: صحة البريد
    نص حالة_البريد = انتظر التحقق_من_البريد(البريد)
    
    إذا حالة_البريد != "صحيح"
        اطبع("✗ خطأ: البريد الإلكتروني " + حالة_البريد)
        إرجاع "فشل: البريد " + حالة_البريد
    نهاية
    اطبع("✓ البريد الإلكتروني صحيح")
    
    # التحقق 4: تكرار البريد
    نص حالة_تكرار = انتظر التحقق_من_تكرار_البريد(البريد)
    
    إذا حالة_تكرار != "جديد"
        اطبع("✗ خطأ: البريد مسجل مسبقاً")
        إرجاع "فشل: البريد مستخدم"
    نهاية
    اطبع("✓ البريد غير مسجل مسبقاً")
    
    اطبع("")
    اطبع("--- جميع التحققات نجحت ---")
    اطبع("")
    
    # إنشاء الحساب
    عدد معرف_المستخدم = انتظر إنشاء_ملف_مستخدم(اسم_المستخدم، البريد)
    اطبع("✓ تم إنشاء الملف")
    
    # إرسال بريد التأكيد
    نص حالة_الإرسال = انتظر إرسال_بريد_التأكيد(البريد، معرف_المستخدم)
    اطبع("✓ تم إرسال بريد التأكيد")
    
    اطبع("")
    اطبع("=== اكتملت عملية التسجيل بنجاح ===")
    اطبع("معرف المستخدم الجديد: " + معرف_المستخدم)
    
    إرجاع "نجح"
نهاية

دالة رئيسية()
    اطبع_سطر("********** نظام تسجيل المستخدمين **********")
    اطبع_سطر("")
    
    # محاولة تسجيل مستخدم صحيح
    نص نتيجة1 = تسجيل_مستخدم_جديد("أحمد123"، "MyPass@2024"، "ahmad@example.com")
    اطبع_سطر("النتيجة: " + نتيجة1)
    
    اطبع_سطر("")
    اطبع_سطر("=====================================")
    اطبع_سطر("")
    
    # محاولة تسجيل باسم محجوز
    نص نتيجة2 = تسجيل_مستخدم_جديد("admin"، "MyPass@2024"، "test@example.com")
    اطبع_سطر("النتيجة: " + نتيجة2)
نهاية
```

هذا مثال واقعي جداً يُظهر كيف تُبنى أنظمة التسجيل في التطبيقات الحقيقية، مع جميع خطوات التحقق والمعالجة.

## مثال 4: نظام تحليل البيانات

مثال يُظهر كيفية معالجة وتحليل مجموعة من البيانات بشكل غير متزامن:

```
# دالة لقراءة بيانات من مصدر
غير_متزامن دالة قراءة_بيانات(نص مصدر)
    اطبع("قراءة البيانات من: " + مصدر)
    # محاكاة قراءة من ملف أو قاعدة بيانات
    إرجاع "100، 200، 150، 300، 250"
نهاية

# دالة لحساب المجموع
غير_متزامن دالة حساب_المجموع(نص بيانات)
    اطبع("حساب المجموع...")
    # في الحقيقة هنا سنقسم النص ونجمع الأرقام
    عدد مجموع = 1000
    إرجاع مجموع
نهاية

# دالة لحساب المتوسط
غير_متزامن دالة حساب_المتوسط(عدد مجموع، عدد عدد_العناصر)
    اطبع("حساب المتوسط...")
    عدد متوسط = مجموع / عدد_العناصر
    إرجاع متوسط
نهاية

# دالة للعثور على أعلى قيمة
غير_متزامن دالة إيجاد_الأعلى(نص بيانات)
    اطبع("البحث عن أعلى قيمة...")
    # في الحقيقة هنا سنقارن القيم
    عدد أعلى_قيمة = 300
    إرجاع أعلى_قيمة
نهاية

# دالة للعثور على أدنى قيمة
غير_متزامن دالة إيجاد_الأدنى(نص بيانات)
    اطبع("البحث عن أدنى قيمة...")
    عدد أدنى_قيمة = 100
    إرجاع أدنى_قيمة
نهاية

# دالة لإنشاء تقرير شامل
غير_متزامن دالة إنشاء_تقرير_تحليلي(نص مصدر_البيانات)
    اطبع("=== بدء تحليل البيانات ===")
    اطبع("المصدر: " + مصدر_البيانات)
    اطبع("")
    
    # الخطوة 1: قراءة البيانات
    نص البيانات = انتظر قراءة_بيانات(مصدر_البيانات)
    اطبع("البيانات: " + البيانات)
    اطبع("")
    
    # الخطوة 2: حساب المجموع
    عدد المجموع = انتظر حساب_المجموع(البيانات)
    اطبع("المجموع: " + المجموع)
    
    # الخطوة 3: حساب المتوسط
    عدد عدد_العناصر = 5
    عدد المتوسط = انتظر حساب_المتوسط(المجموع، عدد_العناصر)
    اطبع("المتوسط: " + المتوسط)
    
    # الخطوة 4: إيجاد أعلى قيمة
    عدد الأعلى = انتظر إيجاد_الأعلى(البيانات)
    اطبع("أعلى قيمة: " + الأعلى)
```
