# 🎯 مطابقة الأنماط (Pattern Matching)

## مقدمة

مطابقة الأنماط (Pattern Matching) هي ميزة قوية في لغة ص تتيح لك مقارنة قيمة مع عدة أنماط محتملة وتنفيذ كود مختلف بناءً على النمط المطابق. إنها طريقة أكثر تعبيرية وقراءة من سلسلة من جمل `إذا-وإلا إذا`.

## لماذا مطابقة الأنماط؟

### المشكلة مع if-else المتسلسلة

```s
# طريقة تقليدية - صعبة القراءة
دالة صنف_الرقم(رقم ن)
    إذا (ن == 0)
        إرجاع "صفر"
    وإلا إذا (ن == 1)
        إرجاع "واحد"
    وإلا إذا (ن == 2)
        إرجاع "اثنان"
    وإلا إذا (ن < 0)
        إرجاع "سالب"
    وإلا إذا (ن < 10)
        إرجاع "رقم أحادي"
    وإلا إذا (ن < 100)
        إرجاع "رقم عشري"
    وإلا
        إرجاع "رقم كبير"
    نهاية
نهاية
```

### الحل بمطابقة الأنماط

```s
دالة صنف_الرقم(رقم ن)
    طابق ن
        حالة 0:
            إرجاع "صفر"
        حالة 1:
            إرجاع "واحد"
        حالة 2:
            إرجاع "اثنان"
        حالة ن إذا ن < 0:
            إرجاع "سالب"
        حالة ن إذا ن < 10:
            إرجاع "رقم أحادي"
        حالة ن إذا ن < 100:
            إرجاع "رقم عشري"
        حالة _:
            إرجاع "رقم كبير"
    نهاية
نهاية
```

## الصيغة الأساسية

```s
طابق قيمة
    حالة نمط1:

        # الحالة الافتراضية (أي شيء آخر)
```

## مطابقة مع الحراسات (Guards)

الحراسة هي شرط إضافي يجب تحقيقه للمطابقة:

```s
دالة تقدير_الدرجة(رقم درجة)
    طابق درجة
        حالة د إذا د >= 90:
            إرجاع "ممتاز"
        حالة د إذا د >= 80:
            إرجاع "جيد جداً"
        حالة د إذا د >= 70:
            إرجاع "جيد"
        حالة د إذا د >= 60:
            إرجاع "مقبول"
        حالة د إذا د >= 0:
            إرجاع "راسب"
        حالة _:
            إرجاع "درجة غير صالحة"
    نهاية
نهاية

دالة رئيسية()
    اطبع(تقدير_الدرجة(95))  # ممتاز
    اطبع(تقدير_الدرجة(75))  # جيد
    اطبع(تقدير_الدرجة(55))  # راسب
نهاية
```

## مطابقة الأنواع

```s
دالة معالجة_قيمة(متغير قيمة)
    طابق قيمة
        حالة رقم ن:
            اطبع("رقم: " + ن * 2)
        حالة نص س:
            اطبع("نص: " + أحرف_كبيرة(س))
        حالة قائمة ق:
            اطبع("قائمة بطول: " + طول(ق))
        حالة قاموس م:
            اطبع("قاموس بـ " + طول(م) + " مفتاح")
        حالة _:
            اطبع("نوع غير معروف")
    نهاية
نهاية

دالة رئيسية()
    معالجة_قيمة(42)           # رقم: 84
    معالجة_قيمة("مرحبا")       # نص: مرحبا
    معالجة_قيمة([1، 2، 3])    # قائمة بطول: 3
    معالجة_قيمة({"أ": 1})     # قاموس بـ 1 مفتاح
نهاية
```

## مطابقة البنية (Structural Matching)

### مطابقة القوائم

```s
دالة وصف_القائمة(قائمة ق)
    طابق ق
        حالة []:
            إرجاع "قائمة فارغة"
        حالة [أ]:
            إرجاع "قائمة بعنصر واحد: " + أ
        حالة [أ، ب]:
            إرجاع "قائمة بعنصرين: " + أ + " و " + ب
        حالة [أ، ب، ج]:
            إرجاع "قائمة بثلاثة عناصر"
        حالة [أول، *الباقي]:
            إرجاع "العنصر الأول: " + أول + "، والباقي " + طول(الباقي) + " عناصر"
    نهاية
نهاية

دالة رئيسية()
    اطبع(وصف_القائمة([]))             # قائمة فارغة
    اطبع(وصف_القائمة([1]))            # قائمة بعنصر واحد: 1
    اطبع(وصف_القائمة([1، 2]))         # قائمة بعنصرين: 1 و 2
    اطبع(وصف_القائمة([1، 2، 3، 4، 5]))  # العنصر الأول: 1، والباقي 4 عناصر
نهاية
```

### مطابقة التوبل

```s
دالة معالجة_نقطة(توبل نقطة)
    طابق نقطة
        حالة (0، 0):
            إرجاع "الأصل"
        حالة (س، 0):
            إرجاع "على المحور السيني عند " + س
        حالة (0، ص):
            إرجاع "على المحور الصادي عند " + ص
        حالة (س، ص) إذا س == ص:
            إرجاع "على القطر الرئيسي"
        حالة (س، ص):
            إرجاع "نقطة عامة (" + س + "، " + ص + ")"
    نهاية
نهاية

دالة رئيسية()
    اطبع(معالجة_نقطة((0، 0)))   # الأصل
    اطبع(معالجة_نقطة((5، 0)))   # على المحور السيني عند 5
    اطبع(معالجة_نقطة((3، 3)))   # على القطر الرئيسي
    اطبع(معالجة_نقطة((2، 7)))   # نقطة عامة (2، 7)
نهاية
```

### مطابقة القواميس

```s
دالة معالجة_طلب(قاموس طلب)
    طابق طلب
        حالة {"نوع": "تسجيل"، "اسم": اسم، "بريد": بريد}:
            إرجاع تسجيل_مستخدم(اسم، بريد)
            
        حالة {"نوع": "دخول"، "اسم": اسم، "كلمة_سر": سر}:
            إرجاع تسجيل_دخول(اسم، سر)
            
        حالة {"نوع": "خروج"، "معرف": م}:
            إرجاع تسجيل_خروج(م)
            
        حالة {"نوع": نوع}:
            إرجاع "نوع طلب غير معروف: " + نوع
            
        حالة _:
            إرجاع "طلب غير صالح"
    نهاية
نهاية

دالة رئيسية()
    قاموس طلب1 = {"نوع": "تسجيل"، "اسم": "أحمد"، "بريد": "ahmed@test.com"}
    قاموس طلب2 = {"نوع": "دخول"، "اسم": "أحمد"، "كلمة_سر": "123456"}
    
    اطبع(معالجة_طلب(طلب1))
    اطبع(معالجة_طلب(طلب2))
نهاية
```

## مطابقة الأصناف

```s
صنف نقطة
    رقم س
    رقم ص
    
    منشئ(رقم س، رقم ص)
        هذا.س = س
        هذا.ص = ص
    نهاية
نهاية

صنف مستطيل
    نقطة ز1
    نقطة ز2
    
    منشئ(نقطة زاوية1، نقطة زاوية2)
        هذا.ز1 = زاوية1
        هذا.ز2 = زاوية2
    نهاية
نهاية

صنف دائرة
    نقطة مركز
    رقم نصف_القطر
    
    منشئ(نقطة مركز، رقم نصف_القطر)
        هذا.مركز = مركز
        هذا.نصف_القطر = نصف_القطر
    نهاية
نهاية

دالة احسب_المساحة(متغير شكل)
    طابق شكل
        حالة مستطيل(ز1، ز2):
            رقم عرض = مطلق(ز2.س - ز1.س)
            رقم ارتفاع = مطلق(ز2.ص - ز1.ص)
            إرجاع عرض * ارتفاع
            
        حالة دائرة(مركز، نق):
            إرجاع 3.14159 * نق * نق
            
        حالة _:
            ارفع خطأ("شكل غير معروف")
    نهاية
نهاية

دالة رئيسية()
    متغير م = جديد مستطيل(جديد نقطة(0، 0)، جديد نقطة(4، 3))
    متغير د = جديد دائرة(جديد نقطة(0، 0)، 5)
    
    اطبع("مساحة المستطيل: " + احسب_المساحة(م))  # 12
    اطبع("مساحة الدائرة: " + احسب_المساحة(د))   # 78.54
نهاية
```

## مطابقة متداخلة

```s
دالة حلل_تعبير(متغير تعبير)
    طابق تعبير
        حالة {"عملية": "+"، "يسار": ي، "يمين": م}:
            إرجاع حلل_تعبير(ي) + حلل_تعبير(م)
            
        حالة {"عملية": "-"، "يسار": ي، "يمين": م}:
            إرجاع حلل_تعبير(ي) - حلل_تعبير(م)
            
        حالة {"عملية": "*"، "يسار": ي، "يمين": م}:
            إرجاع حلل_تعبير(ي) * حلل_تعبير(م)
            
        حالة {"عملية": "/"، "يسار": ي، "يمين": م}:
            إرجاع حلل_تعبير(ي) / حلل_تعبير(م)
            
        حالة رقم ن:
            إرجاع ن
            
        حالة _:
            ارفع خطأ("تعبير غير صالح")
    نهاية
نهاية

دالة رئيسية()
    # التعبير: (3 + 4) * 2
    قاموس تعبير = {
        "عملية": "*"،
        "يسار": {
            "عملية": "+"،
            "يسار": 3،
            "يمين": 4
        }،
        "يمين": 2
    }
    
    اطبع(حلل_تعبير(تعبير))  # 14
نهاية
```

## النمط الافتراضي `_`

الشرطة السفلية `_` تطابق أي شيء ولا تربطه بمتغير:

```s
دالة رئيسية()
    متغير قيمة = 42
    
    طابق قيمة
        حالة 0:
            اطبع("صفر")
        حالة 1:
            اطبع("واحد")
        حالة _:  # يطابق أي شيء آخر
            اطبع("شيء آخر")
    نهاية
نهاية
```

```s
نهاية
```

### معالج الأوامر

```s
دالة نفذ_أمر(قائمة أجزاء_الأمر)
    طابق أجزاء_الأمر
        حالة ["مساعدة"]:
            اعرض_المساعدة()
            
        حالة ["مساعدة"، موضوع]:
            اعرض_المساعدة_عن(موضوع)
            
        حالة ["أنشئ"، نوع، اسم]:
            أنشئ_كائن(نوع، اسم)
            
        حالة ["احذف"، اسم]:
            احذف_كائن(اسم)
            
        حالة ["انسخ"، مصدر، وجهة]:
            انسخ_كائن(مصدر، وجهة)
            
        حالة ["قائمة"]:
            اعرض_القائمة()
            
        حالة ["قائمة"، مرشح]:
            اعرض_القائمة_مع_مرشح(مرشح)
            
        حالة [أمر، *معاملات]:
            اطبع("أمر غير معروف: " + أمر)
            
        حالة []:
            اطبع("لم يُدخل أمر")

```

## الخلاصة

مطابقة الأنماط في لغة ص:
- بديل أوضح لسلسلة if-else
- تدعم القيم الحرفية، الأنواع، والبنى
- الحراسات تضيف شروطاً إضافية
- مطابقة متداخلة للبنى المعقدة
- `_` للحالة الافتراضية
- مثالية لمعالجة الأحداث وتحليل البيانات
