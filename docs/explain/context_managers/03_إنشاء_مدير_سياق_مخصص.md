# 🏗️ إنشاء مدير سياق مخصص

## مقدمة

مديرو السياق المخصصون يتيحون لك تغليف أي منطق للإعداد والتنظيف في بنية قابلة لإعادة الاستخدام. في لغة ص، يمكنك إنشاء مدير سياق خاص بك عن طريق تعريف صنف يحتوي على دالتين خاصتين: `__دخول__` و `__خروج__`.

## البنية الأساسية

### الهيكل العام

```s
صنف اسم_مدير_السياق
    # خصائص الصنف
    
    منشئ(معاملات)
        # تهيئة الخصائص
    نهاية
    
    دالة __دخول__()
        # الكود الذي يُنفذ عند الدخول في كتلة باستخدام
        # أي إعداد أو تحضير
        إرجاع قيمة  # هذه القيمة تُسند لـ "كـ متغير"
    نهاية
    
    دالة __خروج__()
        # الكود الذي يُنفذ عند الخروج من كتلة باستخدام
        # التنظيف وتحرير الموارد
    نهاية
نهاية
```

### مثال بسيط: مسجل العمليات

```s
صنف مسجل_العمليات
    نص اسم_العملية
    رقم وقت_البدء
    
    منشئ(نص اسم)
        هذا.اسم_العملية = اسم
    نهاية
    
    دالة __دخول__()
        هذا.وقت_البدء = الوقت_الحالي()
        اطبع("🔵 بدء: " + هذا.اسم_العملية)
        إرجاع هذا
    نهاية
    
    دالة __خروج__()
        رقم المدة = الوقت_الحالي() - هذا.وقت_البدء
        اطبع("🟢 انتهاء: " + هذا.اسم_العملية + " (" + المدة + " ثانية)")
    نهاية
نهاية

# الاستخدام
دالة رئيسية()
    باستخدام جديد مسجل_العمليات("تحميل البيانات") كـ مسجل
        # عمليات التحميل
        انتظر(2)
    نهاية_استخدام
نهاية
```

## أمثلة عملية متقدمة

### مدير اتصال قاعدة بيانات

```s
صنف اتصال_قاعدة_بيانات
    نص المضيف
    نص اسم_القاعدة
    نص المستخدم
    نص كلمة_السر
    متغير الاتصال = عدم
    
    منشئ(نص مضيف، نص قاعدة، نص مستخدم = ""، نص سر = "")
        هذا.المضيف = مضيف
        هذا.اسم_القاعدة = قاعدة
        هذا.المستخدم = مستخدم
        هذا.كلمة_السر = سر
    نهاية
    
    دالة __دخول__()
        اطبع("📡 الاتصال بقاعدة البيانات: " + هذا.اسم_القاعدة)
        هذا.الاتصال = أنشئ_اتصال_قاعدة(
            هذا.المضيف،
            هذا.اسم_القاعدة،
            هذا.المستخدم،
            هذا.كلمة_السر
        )
        إرجاع هذا
    نهاية
    
    دالة __خروج__()
        إذا (هذا.الاتصال != عدم)
            هذا.الاتصال.أغلق()
            اطبع("📴 تم إغلاق الاتصال بقاعدة البيانات")
        نهاية
    نهاية
    
    دالة استعلام(نص sql)
        إرجاع هذا.الاتصال.نفذ(sql)
    نهاية
    
    دالة استعلام_معاملات(نص sql، قائمة معاملات)
        إرجاع هذا.الاتصال.نفذ_مع_معاملات(sql، معاملات)
    نهاية
    
    دالة التزم()
        هذا.الاتصال.التزم()
    نهاية
    
    دالة تراجع()
        هذا.الاتصال.تراجع()
    نهاية
نهاية

# الاستخدام
دالة احصل_على_جميع_المنتجات()
    باستخدام جديد اتصال_قاعدة_بيانات("localhost"، "متجر") كـ قب
        متغير منتجات = قب.استعلام("SELECT * FROM منتجات")
        إرجاع منتجات
    نهاية_استخدام
نهاية
```

### مدير معاملات قاعدة البيانات

```s
صنف معاملة_قاعدة_بيانات
    متغير الاتصال
    منطقي نجحت = خطأ
    
    منشئ(متغير اتصال)
        هذا.الاتصال = اتصال
    نهاية
    
    دالة __دخول__()
        هذا.الاتصال.نفذ("BEGIN TRANSACTION")
        اطبع("🔄 بدء معاملة قاعدة البيانات")
        إرجاع هذا
    نهاية
    
    دالة __خروج__()
        إذا (هذا.نجحت)
            هذا.الاتصال.نفذ("COMMIT")
            اطبع("✅ تم الالتزام بالمعاملة")
        وإلا
            هذا.الاتصال.نفذ("ROLLBACK")
            اطبع("❌ تم التراجع عن المعاملة")
        نهاية
    نهاية
    
    دالة نجاح()
        هذا.نجحت = صحيح
    نهاية
نهاية

# الاستخدام
دالة تحويل_أموال(رقم من، رقم إلى، رقم مبلغ)
    باستخدام جديد اتصال_قاعدة_بيانات("localhost"، "بنك") كـ قب
        باستخدام جديد معاملة_قاعدة_بيانات(قب) كـ معاملة
            حاول
                قب.استعلام_معاملات(
                    "UPDATE حسابات SET الرصيد = الرصيد - ? WHERE id = ?"،
                    [مبلغ، من]
                )
                قب.استعلام_معاملات(
                    "UPDATE حسابات SET الرصيد = الرصيد + ? WHERE id = ?"،
                    [مبلغ، إلى]
                )
                معاملة.نجاح()
                إرجاع صحيح
            التقط خطأ كـ خ
                اطبع("خطأ في التحويل: " + خ.الرسالة)
                إرجاع خطأ
            نهاية
        نهاية_استخدام
    نهاية_استخدام
نهاية
```

### مدير ملف مؤقت

```s
صنف ملف_مؤقت
    نص المسار
    نص الامتداد
    منطقي احتفظ = خطأ
    
    منشئ(نص امتداد = ".tmp"، منطقي احتفظ_بعد_الانتهاء = خطأ)
        هذا.الامتداد = امتداد
        هذا.احتفظ = احتفظ_بعد_الانتهاء
    نهاية
    
    دالة __دخول__()
        نص اسم_فريد = "مؤقت_" + وقت_يونكس() + "_" + رقم_عشوائي(1000، 9999)
        هذا.المسار = مجلد_المؤقتات() + "/" + اسم_فريد + هذا.الامتداد
        
        # إنشاء الملف الفارغ
        باستخدام افتح(هذا.المسار، "كتابة") كـ ف
            # ملف فارغ
        نهاية_استخدام
        
        اطبع("📄 تم إنشاء ملف مؤقت: " + هذا.المسار)
        إرجاع هذا.المسار
    نهاية
    
    دالة __خروج__()
        إذا (!هذا.احتفظ)
            إذا (ملف_موجود(هذا.المسار))
                احذف_ملف(هذا.المسار)
                اطبع("🗑️ تم حذف الملف المؤقت")
            نهاية
        نهاية
    نهاية
نهاية

# الاستخدام
دالة معالجة_بيانات()
    باستخدام جديد ملف_مؤقت(".json") كـ مسار_مؤقت
        # اكتب بيانات مؤقتة
        باستخدام افتح(مسار_مؤقت، "كتابة") كـ ف
            ف.اكتب('{"حالة": "معالجة"}')
        نهاية_استخدام
        
        # استخدم الملف المؤقت
        معالجة_خارجية(مسار_مؤقت)
    نهاية_استخدام
    # الملف يُحذف تلقائياً هنا
نهاية
```

### مدير تغيير الإعدادات

```s
صنف إعدادات_مؤقتة

        # حفظ الإعدادات الأصلية
        لكل مفتاح في هذا.الإعدادات_الجديدة.المفاتيح()

            عين_إعداد(مفتاح، هذا.الإعدادات_الجديدة[مفتاح])

        اطبع("⚙️ تم تطبيق إعدادات مؤقتة")
        إرجاع هذا.الإعدادات_الجديدة
    نهاية
    
    دالة __خروج__()
        # استعادة الإعدادات الأصلية
        لكل مفتاح في هذا.الإعدادات_الأصلية.المفاتيح()

        نهاية
        
        اطبع("⚙️ تم استعادة الإعدادات الأصلية")
    نهاية
نهاية

# الاستخدام
دالة اختبار_مع_إعدادات_مختلفة()
```

```s
            إذا (ت < مملوء)
                شريط = شريط + "█"
            وإلا
                شريط = شريط + "░"
            نهاية
        نهاية
        شريط = شريط + "] " + قرب(نسبة) + "%"
        
        اطبع_بدون_سطر("\r" + شريط)
    نهاية
نهاية
```

## نصائح لإنشاء مديري سياق فعالين

### 1. دائماً نظف في __خروج__

```s
صنف مورد_آمن
    متغير المورد
    
    دالة __دخول__()
        هذا.المورد = احصل_على_مورد()
        إرجاع هذا.المورد
    نهاية
    
    دالة __خروج__()
        # تأكد من التنظيف حتى لو كان هناك خطأ
        حاول
            إذا (هذا.المورد != عدم)
                هذا.المورد.أغلق()
            نهاية
        التقط خطأ
            # تجاهل أخطاء الإغلاق
        نهاية
    نهاية
نهاية
```

### 2. أعد قيمة مفيدة من __دخول__

```s
# جيد - يعيد شيئاً مفيداً
دالة __دخول__()
    إرجاع هذا.الملف  # المستخدم يمكنه استخدامه مباشرة
نهاية

# أقل فائدة - يعيد الكائن نفسه
دالة __دخول__()
    إرجاع هذا
نهاية
```

### 3. استخدم حالة داخلية للتتبع

```s
صنف مورد_مُتتبَع
    منطقي مُفتَح = خطأ
    
    دالة __دخول__()
        هذا.مُفتَح = صحيح
        # ...
    نهاية
    
    دالة __خروج__()
        هذا.مُفتَح = خطأ
        # ...
    نهاية
    
    دالة تحقق()
        إذا (!هذا.مُفتَح)
            ارفع خطأ("المورد غير مفتوح")
        نهاية
    نهاية
نهاية
```

## الخلاصة

إنشاء مديري سياق مخصصين في لغة ص:
- يتطلب صنفاً مع `__دخول__` و `__خروج__`
- `__دخول__` للإعداد والتحضير
- `__خروج__` للتنظيف والإغلاق
- القيمة المُرجعة من `__دخول__` تُسند لـ `كـ متغير`
- التنظيف مضمون حتى مع الأخطاء
