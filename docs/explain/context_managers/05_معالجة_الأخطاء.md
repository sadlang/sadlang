# ⚠️ معالجة الأخطاء في مديري السياق

## مقدمة

أحد أقوى مزايا مديري السياق هي ضمان التنظيف حتى عند حدوث أخطاء. هذا الفصل يشرح بالتفصيل كيف تتعامل مديري السياق مع الأخطاء والاستثناءات.

## السلوك الافتراضي

### الأخطاء داخل كتلة باستخدام

```s
دالة مثال_خطأ_داخلي()
    اطبع("قبل باستخدام")
    
    حاول
        باستخدام جديد مسجل("العملية") كـ م
            اطبع("داخل باستخدام")
            ارفع خطأ("خطأ متعمد!")
            اطبع("هذا لن يُطبع")
        نهاية_استخدام
    التقط خطأ كـ خ
        اطبع("التقطنا: " + خ.الرسالة)
    نهاية
    
    اطبع("بعد باستخدام")
نهاية

# المخرجات:
# قبل باستخدام
# 🔵 دخول: العملية
# داخل باستخدام
# 🟢 خروج: العملية  ← التنظيف يحدث قبل التقاط الخطأ!
# التقطنا: خطأ متعمد!
# بعد باستخدام
```

### التنظيف مضمون

```s
صنف ملف_مُتتبَع
    نص المسار
    منطقي مفتوح = خطأ
    
    منشئ(نص مسار)
        هذا.المسار = مسار
    نهاية
    
    دالة __دخول__()
        اطبع("📂 فتح: " + هذا.المسار)
        هذا.مفتوح = صحيح
        إرجاع هذا
    نهاية
    
    دالة __خروج__()
        اطبع("📁 إغلاق: " + هذا.المسار)
        هذا.مفتوح = خطأ
    نهاية
نهاية

دالة اختبار_التنظيف()
    متغير ملف = جديد ملف_مُتتبَع("بيانات.txt")
    
    حاول
        باستخدام ملف كـ ف
            اطبع("الملف مفتوح: " + ف.مفتوح)  # صحيح
            ارفع خطأ("عطل!")
        نهاية_استخدام
    التقط خطأ
        اطبع("الملف مفتوح بعد الخطأ: " + ملف.مفتوح)  # خطأ ← أُغلق!
    نهاية
نهاية
```

## أنواع الأخطاء

### أخطاء في __دخول__

```s
صنف مورد_قد_يفشل_فتحه
    دالة __دخول__()
        اطبع("محاولة الفتح...")
        إذا (رقم_عشوائي(0، 1) == 0)
            ارفع خطأ("فشل الاتصال")
        نهاية
        اطبع("تم الفتح بنجاح")
        إرجاع هذا
    نهاية
    
    دالة __خروج__()
        اطبع("تم الإغلاق")
    نهاية
نهاية

دالة رئيسية()
    حاول
        باستخدام جديد مورد_قد_يفشل_فتحه() كـ مورد
            اطبع("أستخدم المورد")
        نهاية_استخدام
    التقط خطأ كـ خ
        اطبع("فشل: " + خ.الرسالة)
        # ملاحظة: __خروج__ لم تُستدعَ لأن __دخول__ فشلت
    نهاية
نهاية
```

### أخطاء في الكود الداخلي

```s
دالة قسمة_على_صفر()
    باستخدام افتح("نتائج.txt"، "كتابة") كـ ملف
        رقم أ = 10
        رقم ب = 0
        رقم نتيجة = أ / ب  # خطأ قسمة على صفر!
        ملف.اكتب(نتيجة)
    نهاية_استخدام
نهاية

دالة رئيسية()
    حاول
        قسمة_على_صفر()
    التقط خطأ كـ خ
        اطبع("خطأ: " + خ.الرسالة)
        # الملف أُغلق بأمان رغم الخطأ
    نهاية
نهاية
```

### أخطاء في __خروج__

```s
صنف مورد_خروج_يفشل
    دالة __دخول__()
        اطبع("دخول")
        إرجاع هذا
    نهاية
    
    دالة __خروج__()
        اطبع("محاولة خروج...")
        ارفع خطأ("فشل الإغلاق!")
    نهاية
نهاية

دالة رئيسية()
    حاول
        باستخدام جديد مورد_خروج_يفشل()
            اطبع("داخل الكتلة")
        نهاية_استخدام
    التقط خطأ كـ خ
        اطبع("التقط: " + خ.الرسالة)  # فشل الإغلاق!
    نهاية
نهاية
```

## التعامل المتقدم مع الأخطاء

### استلام معلومات الخطأ في __خروج__

```s
صنف مدير_مع_معلومات_خطأ
    دالة __دخول__()
        إرجاع هذا
    نهاية
    
    # في بعض التنفيذات، __خروج__ تستلم معلومات الخطأ
    دالة __خروج__(متغير نوع_الخطأ = عدم، متغير قيمة_الخطأ = عدم)
        إذا (نوع_الخطأ != عدم)
            اطبع("حدث خطأ من نوع: " + نوع_الخطأ)
            اطبع("رسالة الخطأ: " + قيمة_الخطأ)
        وإلا
            اطبع("تم الخروج بنجاح")
        نهاية
    نهاية
نهاية
```

### كبت الأخطاء

```s
صنف كابت_الأخطاء
    قائمة أنواع_مكبوتة
    
    منشئ(قائمة أنواع)
        هذا.أنواع_مكبوتة = أنواع
    نهاية
    
    دالة __دخول__()
        إرجاع هذا
    نهاية
    
    دالة __خروج__(متغير نوع_الخطأ = عدم، متغير قيمة = عدم)
        إذا (نوع_الخطأ != عدم)
            لكل نوع في هذا.أنواع_مكبوتة
                إذا (نوع_الخطأ == نوع)
                    اطبع("تم كبت خطأ: " + قيمة)
                    إرجاع صحيح  # كبت الخطأ، لا ترفعه
                نهاية
            نهاية
        نهاية
        إرجاع خطأ  # لا تكبت، أعد رفع الخطأ
    نهاية
نهاية

# الاستخدام
دالة رئيسية()
    باستخدام جديد كابت_الأخطاء(["خطأ_قسمة"، "خطأ_فهرس"])
        رقم نتيجة = 10 / 0  # سيُكبت هذا الخطأ
    نهاية_استخدام
    
    اطبع("استمررنا بعد الخطأ المكبوت!")
نهاية
```

### تسجيل الأخطاء قبل إعادة رفعها

```s
صنف مسجل_الأخطاء
    نص مسار_السجل
    
    منشئ(نص مسار)
        هذا.مسار_السجل = مسار
    نهاية
    
    دالة __دخول__()
        إرجاع هذا
    نهاية
    
    دالة __خروج__(متغير نوع = عدم، متغير قيمة = عدم)
        إذا (نوع != عدم)
            باستخدام افتح(هذا.مسار_السجل، "إلحاق") كـ سجل
                نص سطر = "[" + الوقت_الحالي_منسق() + "] "
                سطر = سطر + "خطأ: " + نوع + " - " + قيمة
                سجل.اكتب_سطر(سطر)
            نهاية_استخدام
        نهاية
        إرجاع خطأ  # لا تكبت الخطأ، دعه ينتشر
    نهاية
نهاية

# الاستخدام
دالة عملية_محفوفة_بالمخاطر()
    باستخدام جديد مسجل_الأخطاء("أخطاء.log")
        # إذا حدث خطأ، سيُسجل ثم يُرفع
        عملية_قد_تفشل()
    نهاية_استخدام
نهاية
```

## أنماط التعامل مع الأخطاء

### نمط المحاولة المتكررة

```s
صنف محاول_متكرر
    رقم أقصى_محاولات
    رقم تأخير_بين_المحاولات
    متغير المدير_الفعلي
    
    منشئ(متغير مدير، رقم محاولات = 3، رقم تأخير = 1000)
        هذا.المدير_الفعلي = مدير
        هذا.أقصى_محاولات = محاولات
        هذا.تأخير_بين_المحاولات = تأخير
    نهاية
    
    دالة __دخول__()
        رقم محاولة = 0
        
        بينما (محاولة < هذا.أقصى_محاولات)
            حاول
                إرجاع هذا.المدير_الفعلي.__دخول__()
            التقط خطأ كـ خ
                محاولة = محاولة + 1
                اطبع("المحاولة " + محاولة + " فشلت: " + خ.الرسالة)
                
                إذا (محاولة < هذا.أقصى_محاولات)
                    اطبع("الانتظار " + هذا.تأخير_بين_المحاولات + " مللي ثانية...")
                    انتظر(هذا.تأخير_بين_المحاولات)
                وإلا
                    ارفع خ  # استنفدنا المحاولات
                نهاية
            نهاية
        نهاية
    نهاية
    
    دالة __خروج__()
        هذا.المدير_الفعلي.__خروج__()
    نهاية
نهاية

# الاستخدام
دالة اتصل_بخادم()
    متغير اتصال = جديد اتصال_شبكة("الخادم")
    
    باستخدام جديد محاول_متكرر(اتصال، 5، 2000) كـ خادم
        # يحاول الاتصال حتى 5 مرات
        # مع انتظار 2 ثانية بين المحاولات
        خادم.أرسل_طلب()
    نهاية_استخدام
نهاية
```

### نمط القيمة الاحتياطية

```s
صنف مع_قيمة_احتياطية
    متغير المدير_الأساسي
    متغير المدير_الاحتياطي
    متغير المدير_المستخدم
    
    منشئ(متغير أساسي، متغير احتياطي)
        هذا.المدير_الأساسي = أساسي
        هذا.المدير_الاحتياطي = احتياطي
    نهاية
    
    دالة __دخول__()
        حاول
            هذا.المدير_المستخدم = هذا.المدير_الأساسي
            إرجاع هذا.المدير_الأساسي.__دخول__()
        التقط خطأ
            اطبع("⚠️ المدير الأساسي فشل، استخدام الاحتياطي")
            هذا.المدير_المستخدم = هذا.المدير_الاحتياطي
            إرجاع هذا.المدير_الاحتياطي.__دخول__()
        نهاية
    نهاية
    
    دالة __خروج__()
        هذا.المدير_المستخدم.__خروج__()
    نهاية
نهاية

# الاستخدام
دالة احصل_على_إعدادات()
    متغير ملف_أساسي = جديد ملف_مدار("إعدادات.json")
    متغير ملف_احتياطي = جديد ملف_مدار("إعدادات_افتراضية.json")
    
    باستخدام جديد مع_قيمة_احتياطية(ملف_أساسي، ملف_احتياطي) كـ ملف
        إرجاع تحليل_json(ملف.اقرأ())
    نهاية_استخدام
نهاية
```

### نمط تجميع الأخطاء

```s
صنف مجمع_الأخطاء
    قائمة الأخطاء = []
    
    دالة __دخول__()
        إرجاع هذا
    نهاية
    
    دالة __خروج__()
        إذا (طول(هذا.الأخطاء) > 0)
            نص رسالة = "حدثت " + طول(هذا.الأخطاء) + " أخطاء:\n"
            لكل خطأ في هذا.الأخطاء
                رسالة = رسالة + "  - " + خطأ + "\n"
            نهاية
            ارفع خطأ(رسالة)
        نهاية
    نهاية
    
    دالة أضف_خطأ(نص رسالة)
        هذا.الأخطاء.أضف(رسالة)
    نهاية
    
    دالة نفذ_مع_التقاط(دالة عملية)
        حاول
            عملية()
        التقط خطأ كـ خ
            هذا.أضف_خطأ(خ.الرسالة)
        نهاية
    نهاية
نهاية

# الاستخدام
دالة معالجة_دفعة()
    باستخدام جديد مجمع_الأخطاء() كـ مجمع
        لكل ملف في قائمة_الملفات
            مجمع.نفذ_مع_التقاط(دالة ()
                معالجة_ملف(ملف)
            نهاية)
        نهاية
    نهاية_استخدام
    # إذا كانت هناك أخطاء، سترفع الآن مجمعة
نهاية
```

## التعامل مع أخطاء التنظيف

### تجاهل أخطاء التنظيف

```s
صنف تنظيف_آمن
    متغير المورد
    
    دالة __دخول__()
        هذا.المورد = احصل_على_مورد()
        إرجاع هذا.المورد
    نهاية
    
    دالة __خروج__()
        حاول
            هذا.المورد.أغلق()
        التقط خطأ كـ خ
            # سجل الخطأ لكن لا ترفعه
            اطبع("تحذير: فشل التنظيف - " + خ.الرسالة)
        نهاية
    نهاية
نهاية
```

### إعطاء الأولوية للخطأ الأصلي

```s
صنف أولوية_الخطأ_الأصلي
    متغير خطأ_أصلي = عدم
    
    دالة __دخول__()
        إرجاع هذا
    نهاية
    
    دالة __خروج__(متغير نوع = عدم، متغير قيمة = عدم)
        إذا (نوع != عدم)
            هذا.خطأ_أصلي = قيمة
        نهاية
        
        حاول
            تنظيف_قد_يفشل()
        التقط خطأ كـ خطأ_تنظيف
            إذا (هذا.خطأ_أصلي != عدم)
                # الخطأ الأصلي أهم، سجل خطأ التنظيف فقط
                اطبع("تحذير: خطأ تنظيف إضافي - " + خطأ_تنظيف.الرسالة)
            وإلا
                # لا يوجد خطأ أصلي، ارفع خطأ التنظيف
                ارفع خطأ_تنظيف
            نهاية
        نهاية
    نهاية
نهاية
```

## نصائح للتعامل مع الأخطاء

### 1. التنظيف يجب أن ينجح دائماً

```s
دالة __خروج__()
    حاول
        إذا (هذا.الاتصال != عدم)
            هذا.الاتصال.أغلق()
        نهاية
    التقط خطأ
        # لا تفعل شيئاً - التنظيف فشل لكن لا نريد إخفاء خطأ آخر
    نهاية
نهاية
```

### 2. سجل الأخطاء للتصحيح

```s
دالة __خروج__(متغير نوع = عدم، متغير قيمة = عدم)
    إذا (نوع != عدم)
        سجل_خطأ("خطأ في السياق: " + نوع + " - " + قيمة)
    نهاية
    
    # التنظيف العادي
    هذا.تنظيف()
نهاية
```

### 3. لا تبتلع الأخطاء بصمت

```s
# سيء ❌
دالة __خروج__()
    حاول
        عملية()
    التقط خطأ
        # لا شيء
    نهاية
نهاية

# جيد ✅
دالة __خروج__()
    حاول
        عملية()
    التقط خطأ كـ خ
        سجل_تحذير("فشل في __خروج__: " + خ.الرسالة)
    نهاية
نهاية
```

## الخلاصة

معالجة الأخطاء في مديري السياق:
- التنظيف مضمون حتى مع الأخطاء
- أخطاء __دخول__ تمنع تنفيذ __خروج__
- أخطاء الكود الداخلي لا تمنع التنظيف
- يمكن كبت الأخطاء أو تسجيلها
- أنماط متقدمة: المحاولة المتكررة، القيمة الاحتياطية، تجميع الأخطاء
- التنظيف يجب أن يكون آمناً قدر الإمكان
