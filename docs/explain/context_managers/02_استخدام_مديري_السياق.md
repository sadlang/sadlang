## التعامل مع الملفات

### قراءة ملف نصي

```s
دالة قراءة_ملف_كامل(نص المسار)
    باستخدام افتح(المسار، "قراءة") كـ ملف
        إرجاع ملف.اقرأ()
    نهاية_استخدام
نهاية

دالة رئيسية()
    نص المحتوى = قراءة_ملف_كامل("مستند.txt")
    اطبع(المحتوى)
نهاية
```

### قراءة سطراً سطراً

```s
دالة عرض_أسطر_الملف(نص المسار)
    رقم رقم_السطر = 1
    
    باستخدام افتح(المسار، "قراءة") كـ ملف
        لكل سطر في ملف.الأسطر()
            اطبع(رقم_السطر + ": " + سطر)
            رقم_السطر = رقم_السطر + 1
        نهاية
    نهاية_استخدام
نهاية

دالة رئيسية()
    عرض_أسطر_الملف("قصيدة.txt")
نهاية
```

### كتابة ملف جديد

```s
دالة إنشاء_تقرير(نص المسار، متغير البيانات)
    باستخدام افتح(المسار، "كتابة") كـ ملف
        ملف.اكتب_سطر("=== تقرير البيانات ===")
        ملف.اكتب_سطر("التاريخ: " + تاريخ_اليوم())
        ملف.اكتب_سطر("")
        
        لكل عنصر في البيانات
            ملف.اكتب_سطر("• " + عنصر)
        نهاية
        
        ملف.اكتب_سطر("")
        ملف.اكتب_سطر("=== نهاية التقرير ===")
    نهاية_استخدام
    
    اطبع("تم إنشاء التقرير: " + المسار)
نهاية

دالة رئيسية()
    متغير عناصر = ["العنصر الأول"، "العنصر الثاني"، "العنصر الثالث"]
    إنشاء_تقرير("تقرير.txt"، عناصر)
نهاية
```

### إلحاق بملف موجود

```s
دالة إضافة_سجل(نص مسار_السجل، نص الرسالة)
    باستخدام افتح(مسار_السجل، "إلحاق") كـ ملف
        نص الطابع_الزمني = الوقت_الحالي_منسق()
        ملف.اكتب_سطر("[" + الطابع_الزمني + "] " + الرسالة)
    نهاية_استخدام
نهاية

دالة رئيسية()
    إضافة_سجل("سجل.txt"، "بدء التطبيق")
    إضافة_سجل("سجل.txt"، "تحميل الإعدادات")
    إضافة_سجل("سجل.txt"، "جاهز للعمل")
نهاية
```

### نسخ ملف

```s
دالة نسخ_ملف(نص المصدر، نص الوجهة)
    باستخدام افتح(المصدر، "قراءة") كـ ملف_المصدر
        باستخدام افتح(الوجهة، "كتابة") كـ ملف_الوجهة
            # نسخ بالدفعات للملفات الكبيرة
            رقم حجم_الدفعة = 8192
            
            بينما (صحيح)
                نص جزء = ملف_المصدر.اقرأ(حجم_الدفعة)
                إذا (طول(جزء) == 0)
                    اخرج
                نهاية
                ملف_الوجهة.اكتب(جزء)
            نهاية
        نهاية_استخدام
    نهاية_استخدام
    
    اطبع("تم نسخ " + المصدر + " إلى " + الوجهة)
نهاية
```

### قراءة وكتابة ملف ثنائي

```s
دالة نسخ_صورة(نص المصدر، نص الوجهة)
    باستخدام افتح(المصدر، "قراءة_ثنائية") كـ ملف_المصدر
        باستخدام افتح(الوجهة، "كتابة_ثنائية") كـ ملف_الوجهة
            متغير بيانات = ملف_المصدر.اقرأ_ثنائي()
            ملف_الوجهة.اكتب_ثنائي(بيانات)
        نهاية_استخدام
    نهاية_استخدام
نهاية
```

## التعامل مع قواعد البيانات

### استعلام بسيط

```s
دالة احصل_على_المستخدمين()
    باستخدام اتصال_قاعدة_بيانات("المضيف"، "القاعدة"، "المستخدم"، "كلمة_السر") كـ قب
        متغير نتائج = قب.استعلام("SELECT * FROM مستخدمين WHERE نشط = 1")
        إرجاع نتائج
    نهاية_استخدام
نهاية
```

### استعلام مع معاملات

```s
دالة ابحث_عن_مستخدم(نص اسم_المستخدم)
    باستخدام اتصال_قاعدة_بيانات("المضيف"، "القاعدة") كـ قب
        متغير نتيجة = قب.استعلام_معاملات(
            "SELECT * FROM مستخدمين WHERE اسم_المستخدم = ?"،
            [اسم_المستخدم]
        )
        إرجاع نتيجة
    نهاية_استخدام
نهاية
```

### إدراج بيانات

```s
دالة أضف_مستخدم(نص الاسم، نص البريد)
    باستخدام اتصال_قاعدة_بيانات("المضيف"، "القاعدة") كـ قب
        قب.نفذ(
            "INSERT INTO مستخدمين (الاسم، البريد، تاريخ_الإنشاء) VALUES (?, ?, NOW())"،
            [الاسم، البريد]
        )
        قب.التزم()
        إرجاع قب.آخر_معرف()
    نهاية_استخدام
نهاية
```

### معاملات قاعدة البيانات

```s
دالة تحويل_أموال(رقم من_الحساب، رقم إلى_الحساب، رقم المبلغ)
    باستخدام اتصال_قاعدة_بيانات("المضيف"، "البنك") كـ قب
        حاول
            قب.ابدأ_معاملة()
            
            # خصم من الحساب المصدر
            قب.نفذ(
                "UPDATE حسابات SET الرصيد = الرصيد - ? WHERE رقم_الحساب = ?"،
                [المبلغ، من_الحساب]
            )
            
            # إضافة للحساب الهدف
            قب.نفذ(
                "UPDATE حسابات SET الرصيد = الرصيد + ? WHERE رقم_الحساب = ?"،
                [المبلغ، إلى_الحساب]
            )
            
            قب.التزم()
            إرجاع صحيح
            
        التقط خطأ كـ خ
            قب.تراجع()
            اطبع("فشل التحويل: " + خ.الرسالة)
            إرجاع خطأ
        نهاية
    نهاية_استخدام
نهاية
```

## التعامل مع الشبكات

### طلب HTTP

```s
دالة احصل_على_بيانات_API(نص الرابط)
    باستخدام طلب_HTTP(الرابط) كـ استجابة
        إذا (استجابة.الحالة == 200)
            إرجاع استجابة.json()
        وإلا
            ارفع خطأ("فشل الطلب: " + استجابة.الحالة)
        نهاية
    نهاية_استخدام
نهاية
```

### اتصال Socket

```s
دالة أرسل_رسالة(نص المضيف، رقم المنفذ، نص الرسالة)
    باستخدام اتصال_شبكة(المضيف، المنفذ) كـ مقبس
        مقبس.أرسل(الرسالة)
        نص رد = مقبس.استقبل(1024)
        إرجاع رد
    نهاية_استخدام
نهاية
```

## قياس الوقت

```s
صنف مقياس_الوقت
    رقم وقت_البدء
    نص الوصف
    
    منشئ(نص وصف)
        هذا.الوصف = وصف
    نهاية
    
    # يُستدعى عند الدخول في باستخدام
    دالة __دخول__()
        هذا.وقت_البدء = الوقت_الحالي_بالمللي()
        اطبع("⏱ بدء: " + هذا.الوصف)
        إرجاع هذا
    نهاية
    
    # يُستدعى عند الخروج من باستخدام
    دالة __خروج__()
        رقم المدة = الوقت_الحالي_بالمللي() - هذا.وقت_البدء
        اطبع("⏱ انتهاء: " + هذا.الوصف + " - " + المدة + " مللي ثانية")
    نهاية
نهاية

دالة رئيسية()
    باستخدام جديد مقياس_الوقت("عملية الفرز") كـ مقياس
        متغير قائمة = أرقام_عشوائية(10000)
        فرز(قائمة)
    نهاية_استخدام
    # سيطبع: ⏱ بدء: عملية الفرز
    # ثم: ⏱ انتهاء: عملية الفرز - 45 مللي ثانية
نهاية
```

## التحكم في الأقفال

```s
صنف قفل
    منطقي مقفل = خطأ
    
    دالة __دخول__()
        بينما (هذا.مقفل)
            انتظر(10)
        نهاية
        هذا.مقفل = صحيح
        إرجاع هذا
    نهاية
    
    دالة __خروج__()
        هذا.مقفل = خطأ
    نهاية
نهاية

متغير قفل_البيانات = جديد قفل()

دالة عملية_متزامنة(رقم رقم_العملية)
    باستخدام قفل_البيانات
        اطبع("العملية " + رقم_العملية + " بدأت")
        # عمليات حساسة هنا
        انتظر(100)
        اطبع("العملية " + رقم_العملية + " انتهت")
    نهاية_استخدام
نهاية
```

## تغيير مجلد العمل مؤقتاً

```s
صنف مغير_المجلد
    نص المجلد_الأصلي
    نص المجلد_الجديد
    
    منشئ(نص مجلد)
        هذا.المجلد_الجديد = مجلد
    نهاية
    
    دالة __دخول__()
        هذا.المجلد_الأصلي = المجلد_الحالي()
        غير_المجلد(هذا.المجلد_الجديد)
        إرجاع هذا.المجلد_الجديد
    نهاية
    
    دالة __خروج__()
        غير_المجلد(هذا.المجلد_الأصلي)
    نهاية
نهاية

دالة رئيسية()
    اطبع("المجلد الحالي: " + المجلد_الحالي())
    
    باستخدام جديد مغير_المجلد("/tmp/عمل_مؤقت") كـ مجلد
        اطبع("المجلد المؤقت: " + مجلد)
        # قم بعمليات في المجلد المؤقت
    نهاية_استخدام
    
    اطبع("عدنا إلى: " + المجلد_الحالي())
نهاية
```

## تعيين قيمة مؤقتة

```s
صنف قيمة_مؤقتة
    متغير الكائن
    نص الخاصية
    متغير القيمة_الأصلية
    متغير القيمة_الجديدة
    
    منشئ(متغير كائن، نص خاصية، متغير قيمة_جديدة)
        هذا.الكائن = كائن
        هذا.الخاصية = خاصية
        هذا.القيمة_الجديدة = قيمة_جديدة
    نهاية
    
    دالة __دخول__()
        هذا.القيمة_الأصلية = هذا.الكائن[هذا.الخاصية]
        هذا.الكائن[هذا.الخاصية] = هذا.القيمة_الجديدة
        إرجاع هذا.القيمة_الجديدة
    نهاية
    
    دالة __خروج__()
        هذا.الكائن[هذا.الخاصية] = هذا.القيمة_الأصلية
    نهاية
نهاية

دالة رئيسية()
    متغير إعدادات = {"مستوى_التسجيل": "معلومات"}
    
    اطبع("المستوى: " + إعدادات.مستوى_التسجيل)  # معلومات
    
    باستخدام جديد قيمة_مؤقتة(إعدادات، "مستوى_التسجيل"، "تصحيح")
        اطبع("المستوى المؤقت: " + إعدادات.مستوى_التسجيل)  # تصحيح
        # تصحيح مفصل هنا
    نهاية_استخدام
    
    اطبع("المستوى: " + إعدادات.مستوى_التسجيل)  # معلومات (استُعيد)
نهاية
```

## الخلاصة

مديرو السياق أداة متعددة الاستخدامات في لغة ص:
- إدارة الملفات بأمان
- التعامل مع قواعد البيانات
- اتصالات الشبكة
- القفل والتزامن
- قياس الأداء
- أي مورد يحتاج تنظيفاً

المفتاح هو: أي شيء تفتحه، يجب أن تغلقه - ومدير السياق يضمن ذلك تلقائياً.
