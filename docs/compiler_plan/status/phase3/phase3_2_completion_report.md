```cpp
// Arithmetic: %3 = ADD %1, %2
void emitArithmetic(SIR::SIRInstruction* inst) {
    emitOperand(operands[0]);  // GET_LOCAL %1
    emitOperand(operands[1]);  // GET_LOCAL %2
    module_->emitByte(OP_ADD); // ADD
    emitResult(result);        // SET_LOCAL %3
}

// Conditional Jump: BR %cond, label1
void emitJump(SIR::SIRInstruction* inst) {
    emitOperand(operands[0]);  // GET_LOCAL %cond
    emitJumpToLabel(OP_JMP_IF, "label1"); // JMP_IF +offset
}

// Function Call: %ret = CALL func, %arg1, %arg2
void emitCall(SIR::SIRInstruction* inst) {
    for (arg : args) emitOperand(arg);  // Push arguments
    module_->emitByte(OP_CALL);         // CALL
    module_->emitU32(funcIndex);        // Function index
    module_->emitByte(argc);            // Argument count
    emitResult(result);                 // Store return value
}
```

**Operand Handling**:

```cpp
void emitOperand(const SIR::SIROperand& operand) {
    if (operand.isImmediate()) {
        // Optimize immediate encoding
        if (int8 range)  PUSH_I8
        elif (int16 range) PUSH_I16
        elif (int32 range) PUSH_I32
        else PUSH_I64
        
        // Or: PUSH_F64, PUSH_TRUE/FALSE, PUSH_NULL, PUSH_STR
    } else if (operand.isRegister()) {
        // Load from variable
        isGlobal ? GET_GLOBAL : GET_LOCAL
    }
}
```

**Label Resolution**:

```cpp
// Forward reference handling
void emitJumpToLabel(VM::Opcode jumpOp, const std::string& label) {
    module_->emitByte(jumpOp);
    uint32_t patchPos = getCurrentOffset();
    labels_[label].references.push_back(patchPos); // Save for patching
    module_->emitU32(0xFFFFFFFF);                  // Placeholder
}

void resolveLabels() {
    for (auto& [name, label] : labels_) {
        for (uint32_t ref : label.references) {
            module_->patchU32(ref, label.offset); // Patch all references
        }
    }
}
```

**Variable Management**:

```cpp
// Automatic local/global tracking
uint32_t registerLocal(const std::string& name) {
    VariableInfo info;
    info.index = localCount_++;
    info.isGlobal = false;
    variables_[name] = info;
    return info.index;
}

// Smart variable lookup
uint32_t getVariableIndex(const std::string& name) {
    auto it = variables_.find(name);
    if (it != variables_.end()) return it->second.index;
    return registerLocal(name); // Auto-register new variables
}
```

## Technical Architecture | البنية التقنية

### Translation Pipeline | خط أنابيب الترجمة

```
SIR Module
    ↓
[Bytecode Emitter]
    ├─ Globals Registration
    ├─ Function-by-Function Translation
    │   ├─ Parameters → Locals
    │   ├─ Block-by-Block Emission
    │   │   ├─ Label Definition
    │   │   ├─ Instruction-by-Instruction Translation
    │   │   │   ├─ Operand Emission (PUSH/GET)
    │   │   │   ├─ Operation Emission (OP_xxx)
    │   │   │   └─ Result Storage (SET)
    │   │   └─ Line Info Tracking
    │   └─ Function Metadata Storage
    ├─ Label Resolution (Patching)
    └─ Optional Optimization
    ↓
Bytecode Module (.sbc ready)
```

## Known Limitations | القيود المعروفة

5. **No Closure Support**: Upvalue handling not implemented yet
