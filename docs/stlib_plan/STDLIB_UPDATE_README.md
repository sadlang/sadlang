## 💡 أمثلة متقدمة / Advanced Examples

### مثال: حساب الانحراف المعياري
```s
مصفوفة بيانات = [12, 15, 18, 20, 22, 25, 28]

// المتوسط
رقم_حقيقي مجموع = 0
لكل قيمة في بيانات {
    مجموع = مجموع + قيمة
}
رقم_حقيقي متوسط = مجموع / حجم(بيانات)

// التباين
رقم_حقيقي تباين = 0
لكل قيمة في بيانات {
    رقم_حقيقي فرق = قيمة - متوسط
    تباين = تباين + مربع(فرق)
}
تباين = تباين / حجم(بيانات)

// الانحراف المعياري
رقم_حقيقي انحراف = جذر(تباين)
اطبع_سطر("الانحراف المعياري:", تقريب(انحراف * 100) / 100)
```

### مثال: محاكاة قذيفة
```s
// المعطيات
رقم_حقيقي السرعة_الأولية = 50.0    // م/ث
رقم_حقيقي الزاوية_درجة = 45.0       // درجة
رقم_حقيقي الجاذبية = 9.8            // م/ث²

// تحويل لراديان
رقم_حقيقي باي = 3.14159
رقم_حقيقي الزاوية = الزاوية_درجة * باي / 180.0

// المكونات
رقم_حقيقي سرعة_س = السرعة_الأولية * جيب_تمام(الزاوية)
رقم_حقيقي سرعة_ص = السرعة_الأولية * جيب(الزاوية)

// زمن الطيران
رقم_حقيقي زمن_الطيران = 2 * سرعة_ص / الجاذبية

// المدى
رقم_حقيقي المدى = سرعة_س * زمن_الطيران

// أقصى ارتفاع
رقم_حقيقي ارتفاع_أقصى = أس(سرعة_ص, 2) / (2 * الجاذبية)

اطبع_سطر("المدى:", تقريب(المدى), "متر")
اطبع_سطر("أقصى ارتفاع:", تقريب(ارتفاع_أقصى), "متر")
```
