## 💡 أمثلة عملية / Practical Examples

### مثال 1: معالجة قائمة درجات الطلاب

```s
// إنشاء قائمة درجات
مصفوفة درجات = [85, 92, 78, 95, 88, 73, 90]

// الحصول على الإحصائيات
رقم العدد = حجم(درجات)
رقم الأعلى = آخر(فرز(درجات, صحيح))
رقم الأدنى = أول(فرز(درجات, صحيح))
رقم_حقيقي المتوسط = جمع(درجات) / العدد

اطبع_سطر("عدد الطلاب:", العدد)
اطبع_سطر("أعلى درجة:", الأعلى)
اطبع_سطر("أدنى درجة:", الأدنى)
اطبع_سطر("المتوسط:", تقريب(المتوسط))

// الطلاب المتفوقون (85+)
مصفوفة مميزون = []
لكل درجة في درجات {
    إذا درجة >= 85 {
        مميزون = إضافة(مميزون, درجة)
    }
}
اطبع_سطر("عدد المتفوقين:", حجم(مميزون))
```

### مثال 2: حساب مسافة نقطتين

```s
// نقطة A
رقم_حقيقي س1 = 3.0
رقم_حقيقي ص1 = 4.0

// نقطة B
رقم_حقيقي س2 = 7.0
رقم_حقيقي ص2 = 1.0

// حساب المسافة: √[(x2-x1)² + (y2-y1)²]
رقم_حقيقي فرق_س = مطلق(س2 - س1)
رقم_حقيقي فرق_ص = مطلق(ص2 - ص1)
رقم_حقيقي المسافة = جذر(مربع(فرق_س) + مربع(فرق_ص))

اطبع_سطر("المسافة بين النقطتين:", تقريب(المسافة * 100) / 100)
// النتيجة: 5.0
```

### مثال 3: محاكاة لعبة بسيطة

```s
// قائمة اللاعبين ونقاطهم
مصفوفة لاعبون = ["أحمد", "فاطمة", "محمد", "سارة"]
مصفوفة نقاط = [150, 280, 195, 310]

// إيجاد الفائز
رقم أكبر_نقاط = أكبر(...نقاط)
رقم موقع_الفائز = فهرس(نقاط, أكبر_نقاط)
نص الفائز = لاعبون[موقع_الفائز]

اطبع_سطر("🏆 الفائز:", الفائز)
اطبع_سطر("النقاط:", أكبر_نقاط)

// ترتيب اللاعبين
نقاط = فرز(نقاط, خطأ)  // تنازلي
اطبع_سطر("الترتيب النهائي:")
اطبع_سطر("🥇 المركز الأول:", آخر(نقاط))
اطبع_سطر("🥈 المركز الثاني:", نقاط[حجم(نقاط) - 2])
```
