## 1. استيراد بسيط / Simple Import

### المثال الأول / Example 1

**ملف: `stdlib/رياضيات.sad`**
```sad
// (AR) مكتبة الرياضيات الأساسية
// (EN) Basic math library

صدّر دالة رقم جذر(رقم ع) {
    // تنفيذ حساب الجذر التربيعي
    إرجاع ع ** 0.5
}

صدّر دالة رقم قوة(رقم أساس، رقم أس) {
    إرجاع أساس ** أس
}

صدّر ثابت PI = 3.14159265359
صدّر ثابت E = 2.71828182846
```

**ملف: `main.sad`**
```sad
// (AR) برنامج رئيسي يستخدم مكتبة الرياضيات
// (EN) Main program using math library

استورد رياضيات

دالة رقم رئيسية() {
    // استخدام الدوال من الوحدة
    رقم س = رياضيات.جذر(16)
    اطبع("الجذر التربيعي لـ 16 = " + س)  // 4
    
    رقم ص = رياضيات.قوة(2، 3)
    اطبع("2 أس 3 = " + ص)  // 8
    
    اطبع("قيمة π = " + رياضيات.PI)  // 3.14159...
    
    إرجاع 0
}
```

**المخرجات المتوقعة / Expected Output:**
```
الجذر التربيعي لـ 16 = 4
2 أس 3 = 8
قيمة π = 3.14159265359
```

## 2. استيراد مع اسم مستعار / Import with Alias

### المثال الثاني / Example 2

**ملف: `main.sad`**
```sad
// (AR) استيراد مع اسم مستعار للاختصار
// (EN) Import with alias for brevity

استورد رياضيات كـ م

دالة رقم رئيسية() {
    // استخدام الاسم المستعار "م" بدلاً من "رياضيات"
    رقم نتيجة = م.جذر(25) + م.قوة(3، 2)
    اطبع("5 + 9 = " + نتيجة)  // 14
    
    رقم محيط_دائرة = 2 * م.PI * 10
    اطبع("محيط دائرة نصف قطرها 10 = " + محيط_دائرة)
    
    إرجاع 0
}
```

**الفائدة / Benefit:**
- كود أقصر وأسهل قراءة
- Shorter, more readable code
- تجنب التكرار
- Avoid repetition

## 3. استيراد انتقائي / Selective Import

### المثال الثالث / Example 3

**ملف: `main.sad`**
```sad
// (AR) استيراد دوال محددة فقط
// (EN) Import only specific functions

من رياضيات استورد جذر، قوة، PI

دالة رقم رئيسية() {
    // استخدام الدوال مباشرة بدون بادئة
    رقم أ = جذر(36)
    رقم ب = قوة(2، 4)
    
    اطبع("جذر 36 = " + أ)      // 6
    اطبع("2 أس 4 = " + ب)      // 16
    اطبع("π = " + PI)          // 3.14159...
    
    إرجاع 0
}
```

**مع أسماء مستعارة / With Aliases:**
```sad
من رياضيات استورد جذر كـ ج، قوة كـ أس

دالة رقم رئيسية() {
    اطبع(ج(49))      // 7
    اطبع(أس(5، 2))   // 25
    
    إرجاع 0
}
```

## 4. استيراد Wildcard / Wildcard Import

### المثال الرابع / Example 4

**ملف: `main.sad`**
```sad
// (AR) استيراد كل الرموز من الوحدة
// (EN) Import all symbols from module

من رياضيات استورد *

دالة رقم رئيسية() {
    // جميع الدوال والثوابت متاحة مباشرة
    رقم مساحة_دائرة = PI * قوة(5، 2)
    اطبع("مساحة دائرة نصف قطرها 5 = " + مساحة_دائرة)
    
    رقم قطر = 2 * جذر(مساحة_دائرة / PI)
    اطبع("القطر = " + قطر)
    
    إرجاع 0
}
```

**⚠️ تحذير / Warning:**
```sad
// تجنب wildcard imports في الملفات الكبيرة
// Avoid wildcard imports in large files

// قد تتسبب في تضارب الأسماء
// May cause name conflicts

من مكتبة1 استورد *
من مكتبة2 استورد *  // ⚠️ قد يكون هناك دوال بنفس الاسم!
```

## 5. وحدات متداخلة / Nested Modules

### المثال الخامس / Example 5

**البنية / Structure:**
```
مكتبتي/
├── __init__.sad
├── أدوات/
│   ├── __init__.sad
│   ├── رياضية.sad
│   └── نصية.sad
└── واجهة/
    └── رسومات.sad
```

**ملف: `مكتبتي/أدوات/رياضية.sad`**
```sad
صدّر دالة رقم ضاعف(رقم س) {
    إرجاع س * 2
}

صدّر دالة رقم نصف(رقم س) {
    إرجاع س / 2
}
```

**ملف: `main.sad`**
```sad
// (AR) استيراد من وحدة متداخلة
// (EN) Import from nested module

استورد مكتبتي.أدوات.رياضية كـ أدوات

دالة رقم رئيسية() {
    اطبع(أدوات.ضاعف(10))  // 20
    اطبع(أدوات.نصف(50))    // 25
    
    إرجاع 0
}
```

**استيراد انتقائي متداخل / Nested Selective Import:**
```sad
من مكتبتي.أدوات.رياضية استورد ضاعف، نصف

دالة رقم رئيسية() {
    اطبع(ضاعف(7) + نصف(20))  // 14 + 10 = 24
    إرجاع 0
}
```

## 6. تصدير الدوال / Exporting Functions

### المثال السادس / Example 6

**ملف: `مكتبة_الحسابات.sad`**
```sad
// (AR) مكتبة عمليات حسابية متقدمة
// (EN) Advanced calculation library

// دالة خاصة - غير مُصدَّرة
دالة رقم _تحقق_صحة(رقم س) {
    إذا (س < 0) {
        رمي "القيمة يجب أن تكون موجبة"
    }
    إرجاع س
}

// دالة عامة - مُصدَّرة
صدّر دالة رقم جذر_مربع(رقم س) {
    _تحقق_صحة(س)
    إرجاع س ** 0.5
}

صدّر دالة رقم جذر_تكعيبي(رقم س) {
    إرجاع س ** (1.0 / 3.0)
}

صدّر دالة رقم مضروب(رقم ن) {
    إذا (ن <= 1) {
        إرجاع 1
    }
    إرجاع ن * مضروب(ن - 1)
}
```

**ملف: `main.sad`**
```sad
من مكتبة_الحسابات استورد جذر_مربع، مضروب

دالة رقم رئيسية() {
    اطبع("جذر 100 = " + جذر_مربع(100))  // 10
    اطبع("5! = " + مضروب(5))             // 120
    
    // ❌ هذا سيفشل - الدالة غير مُصدَّرة
    // اطبع(_تحقق_صحة(10))  // Error: _تحقق_صحة not exported
    
    إرجاع 0
}
```

## 7. تصدير الأصناف / Exporting Classes

### المثال السابع / Example 7

**ملف: `هندسة.sad`**
```sad
// (AR) مكتبة الأشكال الهندسية
// (EN) Geometric shapes library

صدّر صنف دائرة {
    خاص:
        رقم نصف_القطر
    
    عام:
        منشئ(رقم ن) {
            نصف_القطر = ن
        }
        
        دالة رقم احسب_المساحة() {
            إرجاع 3.14159 * نصف_القطر * نصف_القطر
        }
        
        دالة رقم احسب_المحيط() {
            إرجاع 2 * 3.14159 * نصف_القطر
        }
}

صدّر صنف مستطيل {
    خاص:
        رقم الطول
        رقم العرض
    
    عام:
        منشئ(رقم ط، رقم ع) {
            الطول = ط
            العرض = ع
        }
        
        دالة رقم احسب_المساحة() {
            إرجاع الطول * العرض
        }
        
        دالة رقم احسب_المحيط() {
            إرجاع 2 * (الطول + العرض)
        }
}
```

**ملف: `main.sad`**
```sad
من هندسة استورد دائرة، مستطيل

دالة رقم رئيسية() {
    // إنشاء دائرة
    دائرة د = جديد دائرة(5)
    اطبع("مساحة الدائرة = " + د.احسب_المساحة())
    اطبع("محيط الدائرة = " + د.احسب_المحيط())
    
    // إنشاء مستطيل
    مستطيل م = جديد مستطيل(10، 5)
    اطبع("مساحة المستطيل = " + م.احسب_المساحة())
    اطبع("محيط المستطيل = " + م.احسب_المحيط())
    
    إرجاع 0
}
```

#### **ملف: `رياضيات/أساسية.sad`**
```sad
// (AR) عمليات رياضية أساسية
// (EN) Basic mathematical operations

صدّر دالة رقم جذر(رقم س) {
    إذا (س < 0) {
        رمي "لا يمكن حساب جذر عدد سالب"
    }
    إرجاع س ** 0.5
}

صدّر دالة رقم قوة(رقم أساس، رقم أس) {
    إرجاع أساس ** أس
}

صدّر دالة رقم مطلق(رقم س) {
    إذا (س < 0) {
        إرجاع -س
    }
    إرجاع س
}

صدّر ثابت PI = 3.14159265359
صدّر ثابت E = 2.71828182846
صدّر ثابت PHI = 1.61803398875  // النسبة الذهبية
```

#### **ملف: `رياضيات/متقدمة.sad`**
```sad
// (AR) عمليات رياضية متقدمة
// (EN) Advanced mathematical operations

من .أساسية استورد E، قوة

صدّر دالة رقم لوغاريتم(رقم س، رقم أساس = E) {
    إذا (س <= 0) {
        رمي "اللوغاريتم معرف فقط للأعداد الموجبة"
    }
    // تنفيذ مبسط
    إرجاع ln(س) / ln(أساس)
}

صدّر دالة رقم جيب(رقم زاوية_راديان) {
    // تنفيذ متسلسلة تايلور
    رقم نتيجة = 0
    رقم حد = زاوية_راديان
    
    لكل (رقم ن في نطاق(0، 10)) {
        نتيجة = نتيجة + حد
        حد = حد * (-1) * زاوية_راديان * زاوية_راديان / ((2*ن+2) * (2*ن+3))
    }
    
    إرجاع نتيجة
}

صدّر دالة رقم جيب_تمام(رقم زاوية_راديان) {
    إرجاع جيب(1.5708 - زاوية_راديان)  // π/2 - x
}

صدّر دالة رقم ظل(رقم زاوية_راديان) {
    رقم جتا = جيب_تمام(زاوية_راديان)
    إذا (مطلق(جتا) < 0.0001) {
        رمي "الظل غير معرف عند π/2"
    }
    إرجاع جيب(زاوية_راديان) / جتا
}
```

#### **ملف: `رياضيات/إحصائيات.sad`**
```sad
// (AR) دوال إحصائية
// (EN) Statistical functions

من .أساسية استورد جذر، قوة

صدّر دالة رقم متوسط(قائمة أعداد) {
    إذا (طول(أعداد) == 0) {
        رمي "لا يمكن حساب متوسط قائمة فارغة"
    }
    
    رقم مجموع = 0
    لكل (رقم عدد في أعداد) {
        مجموع = مجموع + عدد
    }
    
    إرجاع مجموع / طول(أعداد)
}

صدّر دالة رقم وسيط(قائمة أعداد) {
    إذا (طول(أعداد) == 0) {
        رمي "لا يمكن حساب وسيط قائمة فارغة"
    }
    
    قائمة مرتبة = رتب(أعداد)
    رقم ن = طول(مرتبة)
    
    إذا (ن % 2 == 1) {
        إرجاع مرتبة[ن / 2]
    } وإلا {
        إرجاع (مرتبة[ن/2 - 1] + مرتبة[ن/2]) / 2
    }
}

صدّر دالة رقم انحراف_معياري(قائمة أعداد) {
    رقم م = متوسط(أعداد)
    رقم مجموع_مربعات = 0
    
    لكل (رقم عدد في أعداد) {
        رقم فرق = عدد - م
        مجموع_مربعات = مجموع_مربعات + قوة(فرق، 2)
    }
    
    إرجاع جذر(مجموع_مربعات / طول(أعداد))
}

صدّر دالة رقم تباين(قائمة أعداد) {
    رقم انحراف = انحراف_معياري(أعداد)
    إرجاع قوة(انحراف، 2)
}
```

#### **ملف: `main.sad`**
```sad
// (AR) البرنامج الرئيسي - يستخدم مكتبة الرياضيات الكاملة
// (EN) Main program - uses complete math library

استورد رياضيات كـ م

دالة رقم رئيسية() {
    اطبع("=== اختبار المكتبة الرياضية ===\n")
    
    // 1. العمليات الأساسية
    اطبع("-- العمليات الأساسية --")
    اطبع("جذر 144 = " + م.جذر(144))
    اطبع("2 أس 10 = " + م.قوة(2، 10))
    اطبع("مطلق -42 = " + م.مطلق(-42))
    اطبع("π = " + م.PI)
    اطبع()
    
    // 2. العمليات المتقدمة
    اطبع("-- العمليات المتقدمة --")
    اطبع("log₁₀(100) = " + م.لوغاريتم(100، 10))
    اطبع("sin(π/2) ≈ " + م.جيب(م.PI / 2))
    اطبع("cos(0) ≈ " + م.جيب_تمام(0))
    اطبع()
    
    // 3. الإحصائيات
    اطبع("-- الإحصائيات --")
    قائمة بيانات = [10، 20، 30، 40، 50]
    اطبع("البيانات: " + بيانات)
    اطبع("المتوسط = " + م.متوسط(بيانات))
    اطبع("الوسيط = " + م.وسيط(بيانات))
    اطبع("الانحراف المعياري = " + م.انحراف_معياري(بيانات))
    اطبع("التباين = " + م.تباين(بيانات))
    اطبع()
    
    // 4. مثال تطبيقي: حساب مساحة دائرة
    اطبع("-- مثال تطبيقي: حساب مساحة دائرة --")
    رقم نصف_القطر = 7.5
    رقم المساحة = م.PI * م.قوة(نصف_القطر، 2)
    رقم المحيط = 2 * م.PI * نصف_القطر
    
    اطبع("نصف القطر = " + نصف_القطر)
    اطبع("المساحة = " + المساحة)
    اطبع("المحيط = " + المحيط)
    
    إرجاع 0
}
```
