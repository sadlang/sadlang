```cpp
enum ValueType {
    INTEGER,    // الأرقام الصحيحة
    FLOAT,      // الأرقام العشرية
    STRING,     // النصوص
    BOOLEAN,    // القيم المنطقية
    ARRAY,      // المصفوفات
    MAP,        // القواميس/الخرائط
    OBJECT,     // الكائنات
    FUNCTION,   // الدوال
    NULL_TYPE   // القيم الفارغة
};

class Value {
public:
    ValueType type;
    union {
        int64_t intValue;
        double doubleValue;
        std::string* stringValue;
        bool boolValue;
        Array* arrayValue;
        Map* mapValue;
        Object* objectValue;
        Function* functionValue;
    };
    
    // عمليات متقدمة
    Value operator+(const Value& other);
    Value operator==(const Value& other);
    bool isTruthy() const;
    std::string toString() const;
};
```
