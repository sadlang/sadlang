# مكتبة الإدخال والإخراج

## القاعدة
مكتبة الإدخال والإخراج توفر دوال للتفاعل مع المستخدم عبر الطباعة والقراءة من لوحة المفاتيح والملفات.

---

## الشرح التفصيلي

### الدوال المضمنة للإدخال والإخراج

#### ١. دوال الطباعة

##### `اطبع()` - Print
طباعة نص بدون سطر جديد

**الصياغة:**
```s
اطبع(القيمة)
اطبع(القيمة١، القيمة٢، ...)
```

##### `اطبع_سطر()` - Print Line
طباعة نص مع سطر جديد

**الصياغة:**
```s
اطبع_سطر(القيمة)
اطبع_سطر(القيمة١، القيمة٢، ...)
```

##### `اطبع_خطأ()` - Print Error
طباعة رسالة خطأ إلى stderr

**الصياغة:**
```s
اطبع_خطأ(الرسالة)
```

#### ٢. دوال القراءة

##### `اقرأ()` - Read
قراءة حرف واحد من المستخدم

**الصياغة:**
```s
متغير حرف = اقرأ()
```

##### `اقرأ_سطر()` - Read Line
قراءة سطر كامل من المستخدم

**الصياغة:**
```s
متغير نص = اقرأ_سطر()
```

##### `اقرأ_عدد()` - Read Number
قراءة رقم مباشرة (تحويل تلقائي)

**الصياغة:**
```s
متغير رقم = اقرأ_عدد()
```

#### ٣. التنسيق والألوان

##### `نسق()` - Format
تنسيق نص مع متغيرات

**الصياغة:**
```s
متغير النص = نسق("مرحباً {0}، عمرك {1}"، الاسم، العمر)
```

##### `بلون()` - Colored
طباعة بلون محدد

**الصياغة:**
```s
بلون("أحمر"، "رسالة خطأ!")
بلون("أخضر"، "نجح!")
بلون("أزرق"، "معلومات")
```

الألوان المدعومة:
- `أحمر` / `red`
- `أخضر` / `green`
- `أزرق` / `blue`
- `أصفر` / `yellow`
- `أرجواني` / `purple`
- `سماوي` / `cyan`
- `أبيض` / `white`
- `أسود` / `black`

#### ٤. مسح الشاشة

##### `امسح_الشاشة()` - Clear Screen
مسح كل محتويات الشاشة

**الصياغة:**
```s
امسح_الشاشة()
```

### عمليات الملفات

#### فتح وإغلاق الملفات

##### `افتح_ملف()` - Open File
فتح ملف للقراءة أو الكتابة

**الصياغة:**
```s
متغير ملف = افتح_ملف("المسار"، الوضع)
```

**الأوضاع:**
- `"قراءة"` / `"r"`: القراءة فقط
- `"كتابة"` / `"w"`: الكتابة (يحذف المحتوى القديم)
- `"إضافة"` / `"a"`: الإضافة (يضيف للنهاية)
- `"قراءة_وكتابة"` / `"r+"`: القراءة والكتابة

##### `أغلق_ملف()` - Close File
إغلاق ملف مفتوح

**الصياغة:**
```s
أغلق_ملف(ملف)
```

#### القراءة من الملفات

##### `اقرأ_من_ملف()` - Read from File
قراءة محتوى الملف

**الصياغة:**
```s
متغير المحتوى = اقرأ_من_ملف(ملف)         # كل المحتوى
متغير سطر = اقرأ_سطر_من_ملف(ملف)          # سطر واحد
متغير أسطر = اقرأ_أسطر_من_ملف(ملف)        # قائمة بالأسطر
```

#### الكتابة للملفات

##### `اكتب_لملف()` - Write to File
كتابة نص للملف

**الصياغة:**
```s
اكتب_لملف(ملف، النص)
اكتب_سطر_لملف(ملف، النص)
```

```s
    وإلا
        اطبع_سطر("")
        اطبع_سطر("═══ قائمة المهام ═══")

        لكل مهمة في المهام افعل

                بلون("أخضر"، "✓ " + حول_لنص(رقم) + ". " + مهمة["النص"])
            وإلا
                اطبع_سطر("○ " + حول_لنص(رقم) + ". " + مهمة["النص"])
            نهاية
            رقم = رقم + 1
        نهاية
        
        اطبع_سطر("")
        متغير المكتملة = 0
        لكل مهمة في المهام افعل
            إذا مهمة["مكتملة"] إذن
                المكتملة = المكتملة + 1
            نهاية
        نهاية
        
        اطبع_سطر("المجموع: " + حول_لنص(طول(المهام)) + 
                   " | مكتملة: " + حول_لنص(المكتملة))
    نهاية
    
    اطبع("")
    اطبع("اضغط Enter للمتابعة...")
    اقرأ_سطر()
نهاية

# دالة لإتمام مهمة
دالة اتم_مهمة()
    اعرض_المهام()
    
    إذا طول(المهام) > 0 إذن
        اطبع("رقم المهمة المكتملة: ")
        متغير رقم = حول_لعدد_صحيح(اقرأ_سطر())

            بلون("أخضر"، "✓ أحسنت! تم إتمام المهمة")

            بلون("أحمر"، "✗ رقم غير صحيح!")
        نهاية
        
        اقرأ_سطر()
    نهاية
نهاية

# دالة لحذف مهمة
دالة احذف_مهمة()
    اعرض_المهام()
    
    إذا طول(المهام) > 0 إذن
        اطبع("رقم المهمة للحذف: ")
        متغير رقم = حول_لعدد_صحيح(اقرأ_سطر())
        
        إذا رقم > 0 و رقم <= طول(المهام) إذن
            احذف_بالموضع(المهام، رقم - 1)
            بلون("أخضر"، "✓ تم حذف المهمة")
        وإلا
            بلون("أحمر"، "✗ رقم غير صحيح!")
        نهاية
        
        اقرأ_سطر()
    نهاية
نهاية

# دالة لحفظ المهام في ملف

    متغير ملف = افتح_ملف("مهامي.txt"، "كتابة")
    
    لكل مهمة في المهام افعل
        متغير السطر = مهمة["النص"] + "|" + حول_لنص(مهمة["مكتملة"])
        اكتب_سطر_لملف(ملف، السطر)
    نهاية
    
    أغلق_ملف(ملف)
    بلون("أخضر"، "✓ تم حفظ المهام في مهامي.txt")
    اقرأ_سطر()
نهاية

# دالة لتحميل المهام من ملف
دالة حمل_المهام()
    إذا يوجد_ملف("مهامي.txt") إذن
        متغير ملف = افتح_ملف("مهامي.txt"، "قراءة")
        متغير الأسطر = اقرأ_أسطر_من_ملف(ملف)
        أغلق_ملف(ملف)
        
        المهام = []
        لكل سطر في الأسطر افعل
            متغير الأجزاء = اقسم(سطر، "|")
            إذا طول(الأجزاء) == 2 إذن
                متغير مهمة = {
                    "النص": الأجزاء[0]،
                    "مكتملة": الأجزاء[1] == "صح"
                }
                اضف(المهام، مهمة)
            نهاية
        نهاية
        
        بلون("أخضر"، "✓ تم تحميل " + حول_لنص(طول(المهام)) + " مهمة")
    وإلا
        بلون("أصفر"، "⚠️  لم يتم العثور على ملف المهام")
    نهاية
    
    اقرأ_سطر()
نهاية

# البرنامج الرئيسي
متغير مستمر = صح

بينما مستمر افعل
    اعرض_القائمة()
    متغير الاختيار = اقرأ_سطر()
    
    إذا الاختيار == "1" إذن
        اضف_مهمة()
    وإلا_إذا الاختيار == "2" إذن
        اعرض_المهام()
    وإلا_إذا الاختيار == "3" إذن
        اتم_مهمة()
    وإلا_إذا الاختيار == "4" إذن
        احذف_مهمة()
    وإلا_إذا الاختيار == "5" إذن
        احفظ_المهام()
    وإلا_إذا الاختيار == "6" إذن
        حمل_المهام()
    وإلا_إذا الاختيار == "0" إذن
        بلون("سماوي"، "👋 إلى اللقاء!")
        مستمر = خطأ
    وإلا
        بلون("أحمر"، "✗ اختيار غير صحيح!")
        اقرأ_سطر()
    نهاية
نهاية
```

**الشرح:**
- استخدام `امسح_الشاشة()` لتنظيف العرض
- استخدام `بلون()` للرسائل الملونة
- حفظ وتحميل البيانات من ملفات
- واجهة تفاعلية متكاملة

## المثال الثاني: نظام تسجيل الدخول

```s
    إذا يوجد_ملف("مستخدمين.txt") إذن
        متغير ملف = افتح_ملف("مستخدمين.txt"، "قراءة")
        متغير الأسطر = اقرأ_أسطر_من_ملف(ملف)

        لكل سطر في الأسطر افعل
            متغير البيانات = اقسم(سطر، ":")
            إذا طول(البيانات) == 2 إذن
                إذا البيانات[0] == اسم_المستخدم و البيانات[1] == كلمة_المرور إذن
                    ارجع صح
                نهاية
            نهاية
        نهاية
    نهاية
    
    ارجع خطأ
نهاية

# دالة لإضافة مستخدم جديد
دالة اضف_مستخدم(اسم_المستخدم، كلمة_المرور)
    متغير ملف = افتح_ملف("مستخدمين.txt"، "إضافة")
    متغير السطر = اسم_المستخدم + ":" + كلمة_المرور
    اكتب_سطر_لملف(ملف، السطر)

    اطبع("👤 اسم المستخدم: ")
    متغير اسم_المستخدم = اقرأ_سطر()
    
    متغير كلمة_المرور = اقرأ_كلمة_مرور()
    
    اطبع("🔑 أعد كلمة المرور: ")

        بلون("أخضر"، "")

        بلون("أحمر"، "")

# دالة تسجيل الدخول
```

## المثال الثالث: مسجل أحداث (Logger)

```s
# نظام تسجيل الأحداث والأخطاء

# دالة لإنشاء اسم ملف السجل اليومي
دالة اسم_ملف_السجل()
    متغير الوقت = احصل_على_الوقت_الحالي()
    متغير التاريخ = نسق_التاريخ(الوقت، "سنة-شهر-يوم")
    ارجع "سجل_" + التاريخ + ".log"
نهاية

# دالة لكتابة سجل
دالة سجل(المستوى، الرسالة)
    # أنواع المستويات: معلومات، تحذير، خطأ، نجاح
    متغير الوقت = احصل_على_الوقت_الحالي()
    متغير الطابع_الزمني = نسق_التاريخ(الوقت، "ساعة:دقيقة:ثانية")
    
    # تنسيق السطر
    متغير السطر = "[" + الطابع_الزمني + "] [" + المستوى + "] " + الرسالة
    
    # الطباعة للشاشة بلون حسب المستوى
    إذا المستوى == "خطأ" إذن
        بلون("أحمر"، السطر)
    وإلا_إذا المستوى == "تحذير" إذن
        بلون("أصفر"، السطر)
    وإلا_إذا المستوى == "نجاح" إذن
        بلون("أخضر"، السطر)
    وإلا
        بلون("سماوي"، السطر)
    نهاية
    
    # الكتابة للملف
    متغير اسم_الملف = اسم_ملف_السجل()
    متغير ملف = افتح_ملف(اسم_الملف، "إضافة")
    اكتب_سطر_لملف(ملف، السطر)
    أغلق_ملف(ملف)
نهاية

# دوال مساعدة
دالة سجل_معلومات(الرسالة)
    سجل("معلومات"، الرسالة)
نهاية

دالة سجل_تحذير(الرسالة)
    سجل("تحذير"، الرسالة)
نهاية

دالة سجل_خطأ(الرسالة)
```

## ملخص الدوال المضمنة

### دوال الطباعة
- `اطبع(نص)` - طباعة بدون سطر جديد
- `اطبع_سطر(نص)` - طباعة مع سطر جديد
- `اطبع_خطأ(نص)` - طباعة خطأ لـ stderr
- `بلون(لون، نص)` - طباعة ملونة
- `نسق(قالب، ...)` - تنسيق نص

### دوال القراءة
- `اقرأ()` - قراءة حرف
- `اقرأ_سطر()` - قراءة سطر
- `اقرأ_عدد()` - قراءة رقم

### دوال الملفات
- `افتح_ملف(مسار، وضع)` - فتح ملف
- `أغلق_ملف(ملف)` - إغلاق ملف
- `اقرأ_من_ملف(ملف)` - قراءة كل المحتوى
- `اقرأ_سطر_من_ملف(ملف)` - قراءة سطر
- `اقرأ_أسطر_من_ملف(ملف)` - قراءة كقائمة
- `اكتب_لملف(ملف، نص)` - كتابة نص
- `اكتب_سطر_لملف(ملف، نص)` - كتابة سطر
- `يوجد_ملف(مسار)` - التحقق من وجود ملف

### دوال أخرى
- `امسح_الشاشة()` - مسح الشاشة
- `حول_لنص(قيمة)` - تحويل لنص
- `حول_لعدد(نص)` - تحويل لعدد

## أفضل الممارسات

1. **أغلق الملفات دائماً**: تجنب تسرب الموارد
2. **استخدم الألوان بحكمة**: للتمييز بين أنواع الرسائل
3. **تحقق من وجود الملفات**: قبل محاولة فتحها
4. **استخدم التنسيق**: لرسائل واضحة ومنظمة
5. **احفظ السجلات**: لتتبع الأحداث وتصحيح الأخطاء
