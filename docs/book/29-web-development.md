# الفصل التاسع والعشرون: تطوير الويب (Web Development)

## مقدمة نظرية

تطوير الويب يشمل بناء خوادم ومواقع ديناميكية. لغة ص تدعم بناء خوادم HTTP وواجهات REST API كاملة.

---

## 1. خادم REST API

```sad
من "شبكة" استورد http_خادم

متغير خادم = جديد http_خادم(3000)
متغير طلاب = []

# GET — قراءة الكل
خادم.عالج("GET"، "/api/طلاب"، دالة(طلب، رد)
   رد.json({"بيانات": طلاب، "العدد": طول(طلاب)})
نهاية)

# POST — إضافة
خادم.عالج("POST"، "/api/طلاب"، دالة(طلب، رد)
   متغير بيانات = طلب.json()
   بيانات["id"] = طول(طلاب) + 1
   طلاب.أضف(بيانات)
   رد.الحالة(201)
   رد.json({"رسالة": "تمت الإضافة"، "طالب": بيانات})
نهاية)

# GET — طالب بالمعرف
خادم.عالج("GET"، "/api/طلاب/:id"، دالة(طلب، رد)
   متغير معرف = طلب.المعاملات["id"]
   لكل ط في طلاب
      إذا (ط["id"] == معرف)
         رد.json(ط)
         ارجع لاشيء
      نهاية
   نهاية
   رد.الحالة(404)
   رد.json({"خطأ": "الطالب غير موجود"})
نهاية)

اطبع_سطر("API تعمل على http://localhost:3000")
# خادم.ابدأ()
```

## 2. وسيط (Middleware)

```sad
# وسيط تسجيل الطلبات
دالة وسيط_تسجيل(طلب، رد، التالي)
   اطبع_سطر("[" + طلب.الطريقة + "] " + طلب.المسار)
   التالي()
نهاية

# وسيط CORS
دالة وسيط_cors(طلب، رد، التالي)
   رد.ترويسة("Access-Control-Allow-Origin"، "*")
   رد.ترويسة("Access-Control-Allow-Methods"، "GET,POST,PUT,DELETE")
   التالي()
نهاية

خادم.وسيط(وسيط_تسجيل)
خادم.وسيط(وسيط_cors)
```

## 3. خدمة ملفات ثابتة

```sad
من "شبكة" استورد http_خادم

متغير خادم = جديد http_خادم(8080)

# خدمة HTML
خادم.عالج("GET"، "/"، دالة(طلب، رد)
   رد.html("
      <html dir='rtl' lang='ar'>
      <head><title>موقع لغة ص</title></head>
      <body style='font-family: Cairo; text-align: center; padding: 50px;'>
         <h1>🌟 مرحباً بك في موقع لغة ص</h1>
         <p>أول لغة برمجة عربية حديثة</p>
         <a href='/معلومات'>معلومات عن اللغة</a>
      </body>
      </html>
   ")
نهاية)

خادم.عالج("GET"، "/معلومات"، دالة(طلب، رد)
   رد.json({
      "اسم": "لغة ص"،
      "إصدار": "1.0.0"،
      "ميزات": ["عربية", "كائنية", "متزامنة", "LLVM"]
   })
نهاية)

# خادم.ابدأ()
```

## 4. قوالب HTML ديناميكية

```sad
دالة قالب_طالب(طالب)
   ارجع "<div class='card'>" +
          "<h2>" + طالب["اسم"] + "</h2>" +
          "<p>الدرجة: " + طالب["درجة"] + "</p>" +
          "<p>التقدير: " + طالب["تقدير"] + "</p>" +
          "</div>"
نهاية

دالة صفحة_طلاب(طلاب)
   متغير محتوى = "<html dir='rtl'><body>"
   محتوى = محتوى + "<h1>سجل الطلاب</h1>"
   لكل ط في طلاب
      محتوى = محتوى + قالب_طالب(ط)
   نهاية
   محتوى = محتوى + "</body></html>"
   ارجع محتوى
نهاية
```

## خلاصة

| المفهوم | الوصف |
|---------|-------|
| REST API | `GET/POST/PUT/DELETE` مع مسارات ومعاملات |
| وسيط | دالة تُعالج الطلب قبل الوصول للمعالج |
| قوالب | HTML ديناميكي من بيانات |
| ملفات ثابتة | خدمة HTML/CSS/JS من مجلد |

## تمارين

1. ابنِ واجهة REST كاملة لنظام مدوّنة (تدوينات + تعليقات).
2. أضف مصادقة (authentication) بتوكن بسيط.
3. ابنِ صفحة بحث ديناميكية ترسل طلبات AJAX.
4. أضف تحقق من البيانات (validation) في واجهة POST.
