# الفصل الخامس والأربعون: الذكاء الاصطناعي وتعلم الآلة (AI & Machine Learning)

## مقدمة نظرية

**الذكاء الاصطناعي** (AI) هو محاكاة الذكاء البشري بالبرمجيات. **تعلم الآلة** (ML) فرع منه حيث البرنامج يتعلم من البيانات بدلاً من أن يُبرمج يدوياً.

| المفهوم | الوصف | مثال |
|---------|-------|------|
| **تعلم مُراقب** | تعلم من أمثلة مُعلَّمة | تصنيف بريد (عادي/مزعج) |
| **تعلم غير مُراقب** | اكتشاف أنماط بدون تعليم | تجميع عملاء متشابهين |
| **شبكة عصبية** | محاكاة الخلايا العصبية | التعرف على الصور |
| **التعلم العميق** | شبكات عصبية عميقة | ChatGPT، DALL-E |

## 1. الانحدار الخطي (Linear Regression)

أبسط خوارزمية تعلم — تجد خطاً يناسب البيانات:

```sad
صنف انحدار_خطي
   خاص متغير _ميل = 0        # الوزن w
   خاص متغير _تقاطع = 0     # الانحياز b
   # y = w*x + b

   ## يتدرب على بيانات (مصفوفة [س، ص])
   ## معدل_التعلم: حجم الخطوة (0.001-0.1)
   ## دورات: عدد التكرارات
   دالة تدرب(بيانات، معدل_التعلم، دورات)
      متغير ن = طول(بيانات)

      لكل دورة في نطاق(0، دورات)
         متغير مجموع_خطأ_ميل = 0
         متغير مجموع_خطأ_تقاطع = 0

         لكل نقطة في بيانات
            متغير س = نقطة[0]
            متغير ص = نقطة[1]
            متغير تنبؤ = _ميل * س + _تقاطع
            متغير خطأ = تنبؤ - ص

            مجموع_خطأ_ميل = مجموع_خطأ_ميل + خطأ * س
            مجموع_خطأ_تقاطع = مجموع_خطأ_تقاطع + خطأ
         نهاية

         # تحديث الأوزان (Gradient Descent)
         _ميل = _ميل - معدل_التعلم * (مجموع_خطأ_ميل / ن)
         _تقاطع = _تقاطع - معدل_التعلم * (مجموع_خطأ_تقاطع / ن)

         إذا (دورة % 100 == 0)
            متغير تكلفة = _احسب_تكلفة(بيانات)
            اطبع_سطر("دورة " + دورة + " | تكلفة: " + تكلفة)
         نهاية
      نهاية
   نهاية

   دالة تنبأ(س)
      ارجع _ميل * س + _تقاطع
   نهاية

   خاص دالة _احسب_تكلفة(بيانات)
      متغير مجموع = 0
      لكل نقطة في بيانات
         متغير خطأ = (_ميل * نقطة[0] + _تقاطع) - نقطة[1]
         مجموع = مجموع + خطأ * خطأ
      نهاية
      ارجع مجموع / (2 * طول(بيانات))
   نهاية
نهاية

# تجربة: توقع سعر منزل من مساحته
متغير بيانات = [
   [50، 150000]،    # 50م² → 150,000
   [80، 240000]،    # 80م² → 240,000
   [100، 300000]،   # 100م² → 300,000
   [120، 360000]،   # 120م² → 360,000
   [150، 450000]    # 150م² → 450,000
]

متغير نموذج = جديد انحدار_خطي()
نموذج.تدرب(بيانات، 0.0000001، 1000)

اطبع_سطر("توقع سعر 90م²: " + نموذج.تنبأ(90))
اطبع_سطر("توقع سعر 200م²: " + نموذج.تنبأ(200))
```

## 2. مصنف KNN (أقرب K جيران)

```sad
صنف مصنف_كيه_إن_إن
   خاص متغير _بيانات = []
   خاص متغير _ك = 3

   باني(ك)
      _ك = ك
   نهاية

   ## يُضيف عينة تدريب
   ## الميزات: مصفوفة أرقام، التصنيف: نص
   دالة أضف(ميزات، تصنيف)
      متغير عينة = خريطة()
      عينة["ميزات"] = ميزات
      عينة["تصنيف"] = تصنيف
      _بيانات[طول(_بيانات)] = عينة
   نهاية

   ## يُصنف نقطة جديدة بالنظر لأقرب K جيران
   دالة صنّف(ميزات)
      # حساب المسافات لكل عينة
      متغير مسافات = []
      لكل عينة في _بيانات
         متغير م = _مسافة(ميزات، عينة["ميزات"])
         متغير عنصر = خريطة()
         عنصر["مسافة"] = م
         عنصر["تصنيف"] = عينة["تصنيف"]
         مسافات[طول(مسافات)] = عنصر
      نهاية

      # ترتيب حسب المسافة (فقاعي مبسط)
      لكل ي في نطاق(0، طول(مسافات))
         لكل ج في نطاق(0، طول(مسافات) - 1)
            إذا (مسافات[ج]["مسافة"] > مسافات[ج + 1]["مسافة"])
               متغير مؤقت = مسافات[ج]
               مسافات[ج] = مسافات[ج + 1]
               مسافات[ج + 1] = مؤقت
            نهاية
         نهاية
      نهاية

      # التصويت بين أقرب K
      متغير أصوات = خريطة()
      لكل ي في نطاق(0، _ك)
         متغير ت = مسافات[ي]["تصنيف"]
         إذا (أصوات[ت] == لاشيء)
            أصوات[ت] = 0
         نهاية
         أصوات[ت] = أصوات[ت] + 1
      نهاية

      # أعلى تصويت
      متغير أفضل = ""
      متغير أعلى = 0
      لكل مفتاح في مفاتيح(أصوات)
         إذا (أصوات[مفتاح] > أعلى)
            أعلى = أصوات[مفتاح]
            أفضل = مفتاح
         نهاية
      نهاية
      ارجع أفضل
   نهاية

   خاص دالة _مسافة(أ، ب)
      متغير مجموع = 0
      لكل ي في نطاق(0، طول(أ))
         مجموع = مجموع + (أ[ي] - ب[ي]) ** 2
      نهاية
      ارجع جذر(مجموع)
   نهاية
نهاية

# تجربة: تصنيف الزهور (طول البتلة، عرضها)
متغير مصنف = جديد مصنف_كيه_إن_إن(3)

# بيانات تدريب: [طول_بتلة، عرض_بتلة]
مصنف.أضف([1.4، 0.2]، "سوسن_شائك")
مصنف.أضف([1.3، 0.3]، "سوسن_شائك")
مصنف.أضف([4.5، 1.5]، "سوسن_ملون")
مصنف.أضف([4.7، 1.4]، "سوسن_ملون")
مصنف.أضف([5.1، 1.8]، "سوسن_عذراء")
مصنف.أضف([5.9، 2.1]، "سوسن_عذراء")

# تصنيف زهرة جديدة
اطبع_سطر(مصنف.صنّف([1.5، 0.3]))  # → سوسن_شائك
اطبع_سطر(مصنف.صنّف([4.8، 1.6]))  # → سوسن_ملون
```

## 3. شبكة عصبية بسيطة (Perceptron)

```sad
## بيرسبترون — أبسط شبكة عصبية
## يتعلم تصنيف نقاط إلى فئتين
صنف بيرسبترون
   خاص متغير _أوزان = []
   خاص متغير _انحياز = 0

   باني(عدد_المدخلات)
      لكل ي في نطاق(0، عدد_المدخلات)
         _أوزان[ي] = 0.0
      نهاية
   نهاية

   خاص دالة _تنشيط(مجموع)
      إذا (مجموع >= 0)
         ارجع 1
      وإلا
         ارجع 0
      نهاية
   نهاية

   دالة تنبأ(مدخلات)
      متغير مجموع = _انحياز
      لكل ي في نطاق(0، طول(_أوزان))
         مجموع = مجموع + _أوزان[ي] * مدخلات[ي]
      نهاية
      ارجع _تنشيط(مجموع)
   نهاية

   دالة تدرب(بيانات، تصنيفات، معدل، دورات)
      لكل _ في نطاق(0، دورات)
         لكل ي في نطاق(0، طول(بيانات))
            متغير تنبؤ = تنبأ(بيانات[ي])
            متغير خطأ = تصنيفات[ي] - تنبؤ

            لكل ج في نطاق(0، طول(_أوزان))
               _أوزان[ج] = _أوزان[ج] + معدل * خطأ * بيانات[ي][ج]
            نهاية
            _انحياز = _انحياز + معدل * خطأ
         نهاية
      نهاية
   نهاية
نهاية

# تعلم بوابة AND
متغير بوابة = جديد بيرسبترون(2)
متغير بيانات = [[0،0]، [0،1]، [1،0]، [1،1]]
متغير نتائج = [0، 0، 0، 1]

بوابة.تدرب(بيانات، نتائج، 0.1، 100)

اطبع_سطر("0 AND 0 = " + بوابة.تنبأ([0، 0]))  # 0
اطبع_سطر("0 AND 1 = " + بوابة.تنبأ([0، 1]))  # 0
اطبع_سطر("1 AND 0 = " + بوابة.تنبأ([1، 0]))  # 0
اطبع_سطر("1 AND 1 = " + بوابة.تنبأ([1، 1]))  # 1
```

## 4. نظام توصيات بسيط

```sad
صنف محرك_توصيات
   خاص متغير _تقييمات = خريطة()

   ## يسجل تقييم مستخدم لعنصر (1-5)
   دالة قيّم(مستخدم، عنصر، تقييم)
      إذا (_تقييمات[مستخدم] == لاشيء)
         _تقييمات[مستخدم] = خريطة()
      نهاية
      _تقييمات[مستخدم][عنصر] = تقييم
   نهاية

   ## يجد مستخدماً مشابهاً ويوصي بما أعجبه
   دالة أوصِ(مستخدم)
      متغير أقرب = ""
      متغير أعلى_تشابه = -1

      لكل آخر في مفاتيح(_تقييمات)
         إذا (آخر == مستخدم)
            استمر
         نهاية
         متغير تشابه = _احسب_تشابه(مستخدم، آخر)
         إذا (تشابه > أعلى_تشابه)
            أعلى_تشابه = تشابه
            أقرب = آخر
         نهاية
      نهاية

      # أوصِ بما أعجب الأقرب ولم يشاهده المستخدم
      متغير توصيات = []
      لكل عنصر في مفاتيح(_تقييمات[أقرب])
         إذا (_تقييمات[مستخدم][عنصر] == لاشيء و
              _تقييمات[أقرب][عنصر] >= 4)
            توصيات[طول(توصيات)] = عنصر
         نهاية
      نهاية
      ارجع توصيات
   نهاية

   خاص دالة _احسب_تشابه(م1، م2)
      # تشابه بسيط: عدد العناصر المشتركة ذات التقييم المتقارب
      متغير نقاط = 0
      لكل عنصر في مفاتيح(_تقييمات[م1])
         إذا (_تقييمات[م2][عنصر] != لاشيء)
            متغير فرق = _تقييمات[م1][عنصر] - _تقييمات[م2][عنصر]
            إذا (فرق < 0)
               فرق = -فرق
            نهاية
            نقاط = نقاط + (5 - فرق)
         نهاية
      نهاية
      ارجع نقاط
   نهاية
نهاية

# تجربة
متغير محرك = جديد محرك_توصيات()

محرك.قيّم("أحمد"، "فيلم_أ"، 5)
محرك.قيّم("أحمد"، "فيلم_ب"، 4)
محرك.قيّم("أحمد"، "فيلم_ج"، 2)

محرك.قيّم("سارة"، "فيلم_أ"، 5)
محرك.قيّم("سارة"، "فيلم_ب"، 5)
محرك.قيّم("سارة"، "فيلم_د"، 5)
محرك.قيّم("سارة"، "فيلم_هـ"، 4)

متغير توصيات = محرك.أوصِ("أحمد")
اطبع_سطر("توصيات لأحمد: ")
لكل ت في توصيات
   اطبع_سطر("  - " + ت)
نهاية
# → فيلم_د، فيلم_هـ (ما أعجب سارة ولم يشاهده أحمد)
```

## خلاصة

| الخوارزمية | النوع | الاستخدام |
|-----------|-------|-----------|
| انحدار خطي | تعلم مُراقب | توقع أرقام (سعر، درجة) |
| KNN | تعلم مُراقب | تصنيف (فئات) |
| بيرسبترون | شبكة عصبية | تصنيف ثنائي |
| توصيات | تصفية تعاونية | اقتراح محتوى |

## تمارين

1. عدّل الانحدار الخطي ليدعم مُدخلات متعددة (مساحة + عدد غرف → سعر).
2. جرّب KNN مع K مختلفة (1، 3، 5، 7) ولاحظ تأثيرها على الدقة.
3. درّب بيرسبترون على بوابة OR و XOR. أيهما ينجح ولماذا؟
4. حسّن نظام التوصيات ليستخدم تشابه جيب التمام (Cosine Similarity).
