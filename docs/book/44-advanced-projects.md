## مشروع 1: محرك قوالب (Template Engine)

يحوّل قوالب نصية إلى مخرجات باستبدال المتغيرات:

```sad
صنف محرك_قوالب
   خاص متغير _متغيرات = خريطة()

   دالة عيّن(اسم، قيمة)
      _متغيرات[اسم] = قيمة
      ارجع هذا
   نهاية

   ## يُعالج قالباً ويستبدل {{اسم}} بالقيم
   ## مثال: "مرحباً {{الاسم}}" → "مرحباً أحمد"
   دالة عالج(قالب)
      متغير نتيجة = قالب

      لكل مفتاح في مفاتيح(_متغيرات)
         متغير علامة = "{{" + مفتاح + "}}"
         نتيجة = استبدل(نتيجة، علامة، نص(_متغيرات[مفتاح]))
      نهاية

      ارجع نتيجة
   نهاية

   ## يعالج حلقات {{#لكل عنصر في قائمة}}...{{/لكل}}
   دالة عالج_حلقات(قالب)
      متغير نتيجة = قالب
      # بحث عن أنماط التكرار
      # التنفيذ المبسط يستبدل القائمة بتكرار المحتوى
      لكل مفتاح في مفاتيح(_متغيرات)
         متغير القيمة = _متغيرات[مفتاح]
         متغير بداية = "{{#لكل " + مفتاح + "}}"
         متغير نهاية_علامة = "{{/لكل}}"

         إذا (يحتوي(نتيجة، بداية))
            متغير بداية_موقع = موقع(نتيجة، بداية)
            متغير نهاية_موقع = موقع(نتيجة، نهاية_علامة)
            متغير القالب_الداخلي = قطع(نتيجة،
               بداية_موقع + طول(بداية)، نهاية_موقع)

            متغير مُولَّد = ""
            لكل عنصر في القيمة
               مُولَّد = مُولَّد + استبدل(القالب_الداخلي، "{{.}}"، نص(عنصر))
            نهاية

            نتيجة = قطع(نتيجة، 0، بداية_موقع) +
                     مُولَّد +
                     قطع(نتيجة، نهاية_موقع + طول(نهاية_علامة)، طول(نتيجة))
         نهاية
      نهاية
      ارجع نتيجة
   نهاية
نهاية

# الاستخدام
متغير محرك = جديد محرك_قوالب()
محرك.عيّن("الاسم"، "أحمد")
     .عيّن("المدينة"، "الرياض")
     .عيّن("السنة"، 2026)

متغير قالب = "مرحباً {{الاسم}}! أنت من {{المدينة}}. السنة: {{السنة}}"
اطبع_سطر(محرك.عالج(قالب))
# → مرحباً أحمد! أنت من الرياض. السنة: 2026
```

## مشروع 2: مفسر تعبيرات رياضية

```sad
صنف مفسر_رياضي
   خاص متغير _النص
   خاص متغير _الموقع

   ## يحسب تعبيراً رياضياً من نص
   ## يدعم: +، -، *، /، ()، أعداد عشرية
   ## مثال: "3 + 4 * 2" → 11
   دالة احسب(تعبير)
      _النص = تعبير
      _الموقع = 0
      متغير نتيجة = _تعبير()
      ارجع نتيجة
   نهاية

   # تعبير: حد ((+ | -) حد)*
   خاص دالة _تعبير()
      متغير نتيجة = _حد()
      بينما (_الموقع < طول(_النص))
         _تجاوز_فراغات()
         إذا (_الموقع >= طول(_النص))
            توقف
         نهاية
         متغير عامل = _النص[_الموقع]
         إذا (عامل == "+" أو عامل == "-")
            _الموقع = _الموقع + 1
            متغير يمين = _حد()
            إذا (عامل == "+")
               نتيجة = نتيجة + يمين
            وإلا
               نتيجة = نتيجة - يمين
            نهاية
         وإلا
            توقف
         نهاية
      نهاية
      ارجع نتيجة
   نهاية

   # حد: عامل ((* | /) عامل)*
   خاص دالة _حد()
      متغير نتيجة = _عامل()
      بينما (_الموقع < طول(_النص))
         _تجاوز_فراغات()
         إذا (_الموقع >= طول(_النص))
            توقف
         نهاية
         متغير عامل = _النص[_الموقع]
         إذا (عامل == "*" أو عامل == "/")
            _الموقع = _الموقع + 1
            متغير يمين = _عامل()
            إذا (عامل == "*")
               نتيجة = نتيجة * يمين
            وإلا
               إذا (يمين == 0)
                  ارمي "خطأ: قسمة على صفر"
               نهاية
               نتيجة = نتيجة / يمين
            نهاية
         وإلا
            توقف
         نهاية
      نهاية
      ارجع نتيجة
   نهاية

   # عامل: رقم | '(' تعبير ')'
   خاص دالة _عامل()
      _تجاوز_فراغات()
      إذا (_النص[_الموقع] == "(")
         _الموقع = _الموقع + 1
         متغير نتيجة = _تعبير()
         _تجاوز_فراغات()
         إذا (_النص[_الموقع] == ")")
            _الموقع = _الموقع + 1
         وإلا
            ارمي "خطأ: قوس إغلاق مفقود"
         نهاية
         ارجع نتيجة
      وإلا
         ارجع _رقم()
      نهاية
   نهاية

   خاص دالة _رقم()
      _تجاوز_فراغات()
      متغير بداية = _الموقع
      بينما (_الموقع < طول(_النص) و
             (رقم(_النص[_الموقع]) أو _النص[_الموقع] == "."))
         _الموقع = _الموقع + 1
      نهاية
      ارجع عدد(قطع(_النص، بداية، _الموقع))
   نهاية

   خاص دالة _تجاوز_فراغات()
      بينما (_الموقع < طول(_النص) و _النص[_الموقع] == " ")
         _الموقع = _الموقع + 1
      نهاية
   نهاية
نهاية

# الاستخدام
متغير مفسر = جديد مفسر_رياضي()
اطبع_سطر(مفسر.احسب("3 + 4 * 2"))        # 11
اطبع_سطر(مفسر.احسب("(3 + 4) * 2"))      # 14
اطبع_سطر(مفسر.احسب("10 / 3 + 1"))       # 4.333...
```

## مشروع 3: نظام أحداث (Event System)

```sad
صنف ناقل_أحداث
   خاص متغير _مستمعون = خريطة()

   ## يسجل مستمعاً لحدث معين
   دالة استمع(حدث، دالة_معالجة)
      إذا (_مستمعون[حدث] == لاشيء)
         _مستمعون[حدث] = []
      نهاية
      متغير القائمة = _مستمعون[حدث]
      القائمة[طول(القائمة)] = دالة_معالجة
   نهاية

   ## يُطلق حدثاً ويُنبّه كل المستمعين
   دالة أطلق(حدث، بيانات)
      إذا (_مستمعون[حدث] == لاشيء)
         ارجع
      نهاية
      لكل معالج في _مستمعون[حدث]
         معالج(بيانات)
      نهاية
   نهاية

   ## يزيل كل مستمعي حدث معين
   دالة أزل_كل(حدث)
      _مستمعون[حدث] = لاشيء
   نهاية
نهاية

# تطبيق: نظام تنبيهات متجر
متغير ناقل = جديد ناقل_أحداث()

# تسجيل المستمعين
ناقل.استمع("طلب_جديد"، لامدا(بيانات)
   اطبع_سطر("📦 طلب جديد: " + بيانات["المنتج"])
نهاية)

ناقل.استمع("طلب_جديد"، لامدا(بيانات)
   اطبع_سطر("📧 إرسال تأكيد لـ: " + بيانات["العميل"])
نهاية)

ناقل.استمع("طلب_ملغي"، لامدا(بيانات)
   اطبع_سطر("❌ إلغاء طلب: " + بيانات["رقم_الطلب"])
نهاية)

# إطلاق الأحداث
متغير طلب = خريطة()
طلب["المنتج"] = "هاتف"
طلب["العميل"] = "محمد"
طلب["رقم_الطلب"] = 1001

ناقل.أطلق("طلب_جديد"، طلب)
# → 📦 طلب جديد: هاتف
# → 📧 إرسال تأكيد لـ: محمد
```

## مشروع 4: آلة حالات (State Machine)

```sad
صنف آلة_حالات
   خاص متغير _الحالة
   خاص متغير _انتقالات = خريطة()
   خاص متغير _عند_الدخول = خريطة()

   باني(حالة_ابتدائية)
      _الحالة = حالة_ابتدائية
   نهاية

   دالة أضف_انتقال(من، حدث، إلى)
      متغير مفتاح = من + ":" + حدث
      _انتقالات[مفتاح] = إلى
      ارجع هذا
   نهاية

   دالة عند_الدخول(حالة، دالة_معالجة)
      _عند_الدخول[حالة] = دالة_معالجة
      ارجع هذا
   نهاية

   دالة أرسل(حدث)
      متغير مفتاح = _الحالة + ":" + حدث
      متغير الجديدة = _انتقالات[مفتاح]
      إذا (الجديدة == لاشيء)
         ارمي "انتقال غير صالح: " + _الحالة + " + " + حدث
      نهاية
      _الحالة = الجديدة
      إذا (_عند_الدخول[_الحالة] != لاشيء)
         _عند_الدخول[_الحالة]()
      نهاية
   نهاية

   دالة الحالة()
      ارجع _الحالة
   نهاية
نهاية

# تطبيق: دورة حياة طلب
متغير طلب = جديد آلة_حالات("جديد")

طلب.أضف_انتقال("جديد"، "دفع"، "مدفوع")
     .أضف_انتقال("مدفوع"، "شحن"، "مشحون")
     .أضف_انتقال("مشحون"، "توصيل"، "مُكتمل")
     .أضف_انتقال("جديد"، "إلغاء"، "ملغي")
     .أضف_انتقال("مدفوع"، "إلغاء"، "ملغي")

طلب.عند_الدخول("مدفوع"، لامدا()
   اطبع_سطر("💰 تم الدفع — جاري تجهيز الطلب")
نهاية)
طلب.عند_الدخول("مشحون"، لامدا()
   اطبع_سطر("🚚 تم الشحن")
نهاية)

اطبع_سطر("الحالة: " + طلب.الحالة())   # جديد
طلب.أرسل("دفع")                         # 💰 تم الدفع
طلب.أرسل("شحن")                         # 🚚 تم الشحن
اطبع_سطر("الحالة: " + طلب.الحالة())   # مشحون
```

## خلاصة

| المشروع | المهارات المُستخدمة |
|---------|---------------------|
| محرك قوالب | نصوص، استبدال، Builder Pattern |
| مفسر رياضي | خوارزميات، Recursive Descent |
| نظام أحداث | Observer Pattern، دوال عليا |
| آلة حالات | State Machine، سلسلة الاستدعاءات |

## تمارين

1. أضف دعم الشروط (`{{#إذا شرط}}...{{/إذا}}`) لمحرك القوالب.
2. أضف دعم المتغيرات (`x = 5`) والدوال (`sin`, `cos`) للمفسر الرياضي.
3. ابنِ نظام أحداث يدعم الأولوية (المستمعون ذوو الأولوية العالية يُنفذون أولاً).
4. ابنِ آلة حالات لنظام مصعد (طوابق 1-10، أعلى، أسفل، توقف).
