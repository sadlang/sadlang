# 7. البرمجة الكائنية (OOP)

## تعريف صنف

```sad
صنف اسم_الصنف
    # الحقول (الخصائص)
    نص الاسم
    رقم العمر
    
    # الباني (المنشئ)
    باني(نص اسم، رقم عمر)
        الاسم = اسم
        العمر = عمر
    نهاية
    
    # الطرق (الدوال)
    دالة اعرض()
        اطبع_سطر(الاسم + " - " + العمر)
    نهاية
نهاية
```

## إنشاء كائن
```sad
متغير شخص = جديد اسم_الصنف("أحمد"، 25)
شخص.اعرض()
```

## الوراثة
```sad
صنف ابن يرث أب
    باني(معاملات)
        أساس(معاملات_الأب)    # أو أصل(...)
    نهاية
نهاية
```

## الأعضاء الساكنة (Static)
```sad
صنف عداد
    ساكن رقم العدد
    
    ساكن دالة رقم الإجمالي()
        ارجع العدد
    نهاية
نهاية

# الوصول: عداد.الإجمالي()
```

## الكلمة المفتاحية `هذا`
```sad
صنف مثال
    باني(الاسم)
        هذا.اسم = الاسم    # تفريق بين المعامل والحقل
    نهاية
نهاية
```

## تعدد الأشكال (Polymorphism)
```sad
# عند تعريف دالة بنفس الاسم في الصنف الأبن، تحل محل دالة الأب
```

### مثال 1 — مبتدئ: صنف بسيط
```sad
صنف حيوان
    نص الاسم
    
    باني(نص اسم)
        الاسم = اسم
    نهاية
    
    دالة تحدث()
        ارجع الاسم + " يتحدث"
    نهاية
نهاية

متغير ق = جديد حيوان("قط")
متغير ك = جديد حيوان("كلب")
اطبع_سطر(ق.تحدث())
اطبع_سطر(ك.تحدث())
```
**المخرجات:**
```
قط يتحدث
كلب يتحدث
```

### مثال 2 — مبتدئ: صنف مع عدة حقول وطرق
```sad
صنف كتاب
    نص العنوان
    نص المؤلف
    رقم السنة
    رقم متاح

    باني(نص عنوان، نص مؤلف، رقم سنة)
        العنوان = عنوان
        المؤلف = مؤلف
        السنة = سنة
        متاح = 1
    نهاية

    دالة اعرض()
        اطبع_سطر("  كتاب: " + العنوان + " - " + المؤلف + " (" + السنة + ")")
    نهاية

    دالة استعر()
        متاح = 0
        اطبع_سطر("  تمت استعارة: " + العنوان)
    نهاية

    دالة أرجع()
        متاح = 1
        اطبع_سطر("  تم إرجاع: " + العنوان)
    نهاية
نهاية

متغير ك1 = جديد كتاب("البؤساء"، "فيكتور هوغو"، 1862)
متغير ك2 = جديد كتاب("مئة عام من العزلة"، "ماركيز"، 1967)
ك1.اعرض()
ك2.اعرض()
ك1.استعر()
ك1.أرجع()
```
**المخرجات:**
```
  كتاب: البؤساء - فيكتور هوغو (1862)
  كتاب: مئة عام من العزلة - ماركيز (1967)
  تمت استعارة: البؤساء
  تم إرجاع: البؤساء
```

صنف عضو

### مثال 4 — متوسط: الوراثة
```sad
صنف شكل
    باني(اللون)
        هذا.اللون = اللون
    نهاية
    
    دالة وصف()
        ارجع "شكل بلون " + هذا.اللون
    نهاية
نهاية

صنف مستطيل يرث شكل
    باني(اللون، العرض، الارتفاع)
        أصل(اللون)
        هذا.العرض = العرض
        هذا.الارتفاع = الارتفاع
    نهاية
    
    دالة المساحة()
        ارجع هذا.العرض * هذا.الارتفاع
    نهاية
    
    دالة وصف()
        ارجع "مستطيل " + هذا.العرض + "x" + هذا.الارتفاع + " بلون " + هذا.اللون
    نهاية
نهاية

صنف مربع يرث مستطيل
    باني(اللون، الضلع)
        أصل(اللون، الضلع، الضلع)
    نهاية
    
    دالة وصف()
        ارجع "مربع ضلعه " + هذا.العرض + " بلون " + هذا.اللون
    نهاية
نهاية

متغير ش = جديد شكل("أحمر")
متغير م = جديد مستطيل("أزرق"، 5، 3)
متغير مر = جديد مربع("أخضر"، 4)

اطبع_سطر(ش.وصف())
اطبع_سطر(م.وصف())
اطبع_سطر("مساحة المستطيل = " + م.المساحة())
اطبع_سطر(مر.وصف())
اطبع_سطر("مساحة المربع = " + مر.المساحة())
```
**المخرجات:**
```
شكل بلون أحمر
مستطيل 5x3 بلون أزرق
مساحة المستطيل = 15
مربع ضلعه 4 بلون أخضر
مساحة المربع = 16
```

### مثال 5 — متوسط: وراثة متعددة المستويات
```sad
صنف حساب_بنكي
    نص صاحب_الحساب
    رقم الرصيد
    ساكن رقم عدد_الحسابات

    باني(نص صاحب، رقم رصيد_مبدئي)
        صاحب_الحساب = صاحب
        الرصيد = رصيد_مبدئي
        عدد_الحسابات = عدد_الحسابات + 1
    نهاية

    دالة ايداع(رقم مبلغ)
        الرصيد = الرصيد + مبلغ
        اطبع_سطر("  + إيداع " + مبلغ + " → الرصيد: " + الرصيد)
    نهاية

    دالة رقم سحب(رقم مبلغ)
        إذا (مبلغ > الرصيد)
            اطبع_سطر("  ! رصيد غير كافٍ")
            ارجع 0
        نهاية
        الرصيد = الرصيد - مبلغ
        اطبع_سطر("  - سحب " + مبلغ + " → الرصيد: " + الرصيد)
        ارجع 1
    نهاية

    دالة اعرض()
        اطبع_سطر("  " + صاحب_الحساب + " | الرصيد: " + الرصيد)
    نهاية

    ساكن دالة رقم اجمالي()
        ارجع عدد_الحسابات
    نهاية
نهاية

صنف حساب_توفير يرث حساب_بنكي
    رقم نسبة_الربح

    باني(نص صاحب، رقم رصيد، رقم نسبة)
        أساس(صاحب، رصيد)
        نسبة_الربح = نسبة
    نهاية

    دالة احسب_ربح()
        رقم ربح = الرصيد * نسبة_الربح / 100
        الرصيد = الرصيد + ربح
        اطبع_سطر("  $ ربح: " + ربح + " → الرصيد: " + الرصيد)
    نهاية

    دالة اعرض()
        اطبع_سطر("  [توفير] " + صاحب_الحساب + " | الرصيد: " + الرصيد + " | ربح: " + نسبة_الربح + "%")
    نهاية
نهاية

متغير ح1 = جديد حساب_بنكي("علي"، 5000)
متغير ح2 = جديد حساب_توفير("سلمى"، 10000، 5)

ح1.ايداع(2000)
ح1.سحب(1000)
ح2.احسب_ربح()

اطبع_سطر("")
ح1.اعرض()
ح2.اعرض()
اطبع_سطر("إجمالي الحسابات: " + حساب_بنكي.اجمالي())
```
**المخرجات:**
```
  + إيداع 2000 → الرصيد: 7000
  - سحب 1000 → الرصيد: 6000
  $ ربح: 500 → الرصيد: 10500

  علي | الرصيد: 6000
  [توفير] سلمى | الرصيد: 10500 | ربح: 5%
إجمالي الحسابات: 2
```

### مثال 6 — متقدم: استخدام `هذا`
```sad
صنف نقطة
    باني(س، ص)
        هذا.س = س
        هذا.ص = ص
    نهاية
    
    دالة بعد_عن(نقطة_أخرى)
        متغير فس = هذا.س - نقطة_أخرى.س
        متغير فص = هذا.ص - نقطة_أخرى.ص
        ارجع جذر_تربيعي(فس * فس + فص * فص)
    نهاية
    
    دالة انقل(دس، دص)
        هذا.س = هذا.س + دس
        هذا.ص = هذا.ص + دص
    نهاية
    
    دالة لنص()
        ارجع "(" + هذا.س + "، " + هذا.ص + ")"
    نهاية
نهاية

متغير ن1 = جديد نقطة(0، 0)
متغير ن2 = جديد نقطة(3، 4)
اطبع_سطر("ن1 = " + ن1.لنص())
اطبع_سطر("ن2 = " + ن2.لنص())
اطبع_سطر("البعد = " + ن1.بعد_عن(ن2))

ن1.انقل(1، 2)
اطبع_سطر("ن1 بعد النقل = " + ن1.لنص())
```
**المخرجات:**
```
ن1 = (0، 0)
ن2 = (3، 4)
البعد = 5
ن1 بعد النقل = (1، 2)
```

### مثال 7 — متقدم: نظام مدرسة كامل
```sad
صنف مادة
    نص الاسم
    رقم الدرجة_القصوى
    ساكن رقم عدد_المواد

    باني(نص اسم، رقم قصوى)
        الاسم = اسم
        الدرجة_القصوى = قصوى
        عدد_المواد = عدد_المواد + 1
    نهاية

    ساكن دالة رقم كم_مادة()
        ارجع عدد_المواد
    نهاية
نهاية

صنف طالب
    نص الاسم
    رقم المعرف
    رقم درجة_رياضيات
    رقم درجة_علوم
    رقم درجة_لغة
    ساكن رقم عدد_الطلاب

    باني(نص اسم، رقم معرف)
        الاسم = اسم
        المعرف = معرف
        درجة_رياضيات = 0
        درجة_علوم = 0
        درجة_لغة = 0
        عدد_الطلاب = عدد_الطلاب + 1
    نهاية

    دالة سجل_درجات(رقم رياضيات، رقم علوم، رقم لغة)
        درجة_رياضيات = رياضيات
        درجة_علوم = علوم
        درجة_لغة = لغة
    نهاية

    دالة رقم المعدل()
        ارجع (درجة_رياضيات + درجة_علوم + درجة_لغة) / 3
    نهاية

    دالة تقدير()
        رقم م = (درجة_رياضيات + درجة_علوم + درجة_لغة) / 3
        إذا (م >= 90)
            ارجع "ممتاز"
        والا_اذا (م >= 80)
            ارجع "جيد جداً"
        والا_اذا (م >= 70)
            ارجع "جيد"
        والا_اذا (م >= 60)
            ارجع "مقبول"
        وإلا
            ارجع "راسب"
        نهاية
    نهاية

    دالة اعرض()
        اطبع_سطر("  " + الاسم + " | رياضيات:" + درجة_رياضيات + " علوم:" + درجة_علوم + " لغة:" + درجة_لغة + " | " + تقدير())
    نهاية
نهاية

دالة رئيسية()
    اطبع_سطر("══════════════════════════════════")
    اطبع_سطر("       نظام إدارة المدرسة")
    اطبع_سطر("══════════════════════════════════")
    
    متغير ط1 = جديد طالب("أحمد"، 1001)
    متغير ط2 = جديد طالب("فاطمة"، 1002)
    متغير ط3 = جديد طالب("خالد"، 1003)
    
    ط1.سجل_درجات(95، 88، 92)
    ط2.سجل_درجات(78، 85، 90)
    ط3.سجل_درجات(60، 55، 65)
    
    ط1.اعرض()
    ط2.اعرض()
    ط3.اعرض()
    
    اطبع_سطر("══════════════════════════════════")
    اطبع_سطر("  عدد الطلاب: " + طالب.عدد_الطلاب)
    اطبع_سطر("══════════════════════════════════")
نهاية
```

### مثال 8 — متقدم جداً: نظام كتائن مع وراثة ثلاثية
```sad
# الصنف الأساسي
صنف حساب
    رقم رقم_ح
    رقم رصيد
    ساكن رقم عداد

    باني(رقم رقم_حساب، رقم رصيد_مبدئي)
        رقم_ح = رقم_حساب
        رصيد = رصيد_مبدئي
        عداد = عداد + 1
    نهاية

    دالة رقم الرصيد_الحالي()
        ارجع رصيد
    نهاية

    دالة رقم ايداع(رقم مبلغ)
        إذا (مبلغ <= 0)
            ارجع 0
        نهاية
        رصيد = رصيد + مبلغ
        ارجع 1
    نهاية

    دالة رقم سحب(رقم مبلغ)
        إذا (مبلغ > رصيد)
            ارجع 0
        نهاية
        رصيد = رصيد - مبلغ
        ارجع 1
    نهاية

    دالة عرض()
        اطبع_سطر("  حساب #" + رقم_ح + " | رصيد: " + رصيد)
    نهاية

    ساكن دالة رقم الإجمالي()
        ارجع عداد
    نهاية
نهاية

# المستوى الثاني: حساب توفير
صنف حساب_توفير يرث حساب
    رقم نسبة_ربح
    رقم ارباح

    باني(رقم رقم_حساب، رقم رصيد_مبدئي، رقم نسبة)
        أساس(رقم_حساب، رصيد_مبدئي)
        نسبة_ربح = نسبة
        ارباح = 0
    نهاية

    دالة رقم احسب_ربح()
        رقم ربح = رصيد * نسبة_ربح / 100
        ارباح = ارباح + ربح
        رصيد = رصيد + ربح
        ارجع ربح
    نهاية
نهاية

# المستوى الثالث: حساب استثمار
صنف حساب_استثمار يرث حساب_توفير
    رقم مخاطرة
    رقم عدد_استثمارات

    باني(رقم رقم_حساب، رقم رصيد_مبدئي، رقم نسبة، رقم مستوى_مخاطرة)
        أساس(رقم_حساب، رصيد_مبدئي، نسبة)
        مخاطرة = مستوى_مخاطرة
        عدد_استثمارات = 0
    نهاية

    دالة رقم استثمر(رقم مبلغ)
        إذا (مبلغ > رصيد)
            ارجع 0
        نهاية
        رقم عائد = مبلغ * نسبة_ربح * مخاطرة / 1000
        رصيد = رصيد + عائد
        عدد_استثمارات = عدد_استثمارات + 1
        ارجع عائد
    نهاية
نهاية

دالة رئيسية()
    اطبع_سطر("═══════════════════════════════════")
    اطبع_سطر("    نظام بنكي ثلاثي المستويات")
    اطبع_سطر("═══════════════════════════════════")
    
    متغير ح1 = جديد حساب(1001، 5000)
    متغير ح2 = جديد حساب_توفير(1002، 10000، 5)
    متغير ح3 = جديد حساب_استثمار(1003، 20000، 8، 3)
    
    # عمليات على الحساب العادي
    اطبع_سطر("")
    اطبع_سطر("[حساب عادي]")
    ح1.ايداع(2000)
    ح1.سحب(1000)
    ح1.عرض()
    
    # عمليات على حساب التوفير
    اطبع_سطر("")
    اطبع_سطر("[حساب توفير]")
    متغير ربح = ح2.احسب_ربح()
    اطبع_سطر("  الربح السنوي: " + ربح)
    ح2.عرض()
    
    # عمليات على حساب الاستثمار
    اطبع_سطر("")
    اطبع_سطر("[حساب استثمار]")
    متغير عائد = ح3.استثمر(5000)
    اطبع_سطر("  عائد الاستثمار: " + عائد)
    ح3.عرض()
    
    اطبع_سطر("")
    اطبع_سطر("═══════════════════════════════════")
    اطبع_سطر("  إجمالي الحسابات: " + حساب.الإجمالي())
    اطبع_سطر("═══════════════════════════════════")
نهاية
```
