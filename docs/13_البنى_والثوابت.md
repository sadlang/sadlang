# 13. البنى والثوابت

## البنى (Structs)

البنية هي نوع بيانات بسيط يجمع حقول مع قيم افتراضية:

```sad
بنية اسم_البنية
    نوع الحقل1 = قيمة_افتراضية
    نوع الحقل2 = قيمة_افتراضية
نهاية
```

### إنشاء نسخة من البنية
```sad
متغير ن = اسم_البنية(قيمة1، قيمة2)
```

### الوصول للحقول
```sad
اطبع_سطر(ن.الحقل1)
ن.الحقل1 = قيمة_جديدة
```

> **الفرق بين البنية والصنف**: البنية أبسط — لا تدعم الطرق أو الوراثة أو الباني. هي مناسبة لتجميع البيانات فقط.

## الثوابت

```sad
ثابت الاسم = قيمة
```

محاولة تعديل ثابت تلقي خطأ وقت التشغيل.

## 🔸 الأمثلة (8 أمثلة متدرجة)

### مثال 1 — مبتدئ: بنية نقطة بسيطة
```sad
بنية نقطة
    رقم س = 0
    رقم ص = 0
نهاية

متغير ن = نقطة(3، 5)
اطبع_سطر("س = " + ن.س)
اطبع_سطر("ص = " + ن.ص)
```
**المخرجات:**
```
س = 3
ص = 5
```

### مثال 2 — مبتدئ: بنية شخص
```sad
بنية شخص
    نص الاسم = "مجهول"
    رقم العمر = 0
نهاية

متغير أحمد = شخص("أحمد"، 30)
اطبع_سطر("الاسم: " + أحمد.الاسم)
اطبع_سطر("العمر: " + أحمد.العمر)

# باستخدام القيم الافتراضية
متغير مجهول = شخص("مجهول"، 0)
اطبع_سطر("الاسم: " + مجهول.الاسم)
```
**المخرجات:**
```
الاسم: أحمد
العمر: 30
الاسم: مجهول
```

### مثال 3 — مبتدئ: ثوابت أساسية
```sad
ثابت باي = 3.14
ثابت الاسم = "لغة ص"
ثابت نشط = صحيح
ثابت الاصدار = 1

اطبع_سطر("باي = " + باي)
اطبع_سطر("اللغة: " + الاسم)
اطبع_سطر("نشط: " + نشط)
اطبع_سطر("الإصدار: " + الاصدار)
```
**المخرجات:**
```
باي = 3.14
اللغة: لغة ص
نشط: صحيح
الإصدار: 1
```

### مثال 4 — متوسط: حماية الثوابت
```sad
ثابت باي = 3.14159

# لا يمكن تعديل الثابت
حاول
    باي = 3.15
    اطبع_سطر("فشل!")
امسك (خ)
    اطبع_سطر("نجح: " + خ)
نهاية

# لا يمكن استخدام += مع الثابت
حاول
    باي += 1
    اطبع_سطر("فشل!")
امسك (خ)
    اطبع_سطر("نجح: الثابت محمي من +=")
نهاية

# المتغيرات العادية يمكن تعديلها
متغير س = 10
س = 20
اطبع_سطر("المتغير بعد التعديل: " + س)
```
**المخرجات:**
```
نجح: لا يمكن تعديل الثابت 'باي'
نجح: الثابت محمي من +=
المتغير بعد التعديل: 20
```

### مثال 5 — متوسط: بنى متعددة
```sad
بنية لون
    رقم أحمر = 0
    رقم أخضر = 0
    رقم أزرق = 0
نهاية

بنية نقطة
    رقم س = 0
    رقم ص = 0
نهاية

متغير أحمر = لون(255، 0، 0)
متغير أخضر = لون(0، 255، 0)
متغير أزرق = لون(0، 0، 255)

اطبع_سطر("أحمر: (" + أحمر.أحمر + "، " + أحمر.أخضر + "، " + أحمر.أزرق + ")")
اطبع_سطر("أخضر: (" + أخضر.أحمر + "، " + أخضر.أخضر + "، " + أخضر.أزرق + ")")
اطبع_سطر("أزرق: (" + أزرق.أحمر + "، " + أزرق.أخضر + "، " + أزرق.أزرق + ")")

متغير مركز = نقطة(100، 200)
اطبع_سطر("المركز: (" + مركز.س + "، " + مركز.ص + ")")
```
**المخرجات:**
```
أحمر: (255، 0، 0)
أخضر: (0، 255، 0)
أزرق: (0، 0، 255)
المركز: (100، 200)
```

### مثال 6 — متوسط: بنية مع دوال مساعدة
```sad
بنية مستطيل
    رقم العرض = 0
    رقم الارتفاع = 0
نهاية

# دوال مساعدة تعمل مع البنية
دالة مساحة(م)
    ارجع م.العرض * م.الارتفاع
نهاية

دالة محيط(م)
    ارجع 2 * (م.العرض + م.الارتفاع)
نهاية

دالة اعرض_مستطيل(م)
    اطبع_سطر("مستطيل " + م.العرض + "×" + م.الارتفاع)
    اطبع_سطر("  المساحة: " + مساحة(م))
    اطبع_سطر("  المحيط: " + محيط(م))
نهاية

متغير م1 = مستطيل(10، 5)
متغير م2 = مستطيل(8، 8)

اعرض_مستطيل(م1)
اعرض_مستطيل(م2)
```
**المخرجات:**
```
مستطيل 10×5
  المساحة: 50
  المحيط: 30
مستطيل 8×8
  المساحة: 64
  المحيط: 32
```

### مثال 7 — متقدم: بنى في مصفوفات
```sad
بنية طالب
    نص الاسم = ""
    رقم الدرجة = 0
نهاية

متغير طلاب = [
    طالب("أحمد"، 95)،
    طالب("فاطمة"، 88)،
    طالب("خالد"، 72)،
    طالب("نورة"، 91)
]

اطبع_سطر("قائمة الطلاب:")
متغير مجموع = 0
لكل ط في طلاب
    متغير تقدير = ط.الدرجة >= 90 ? "ممتاز" : ط.الدرجة >= 80 ? "جيد جداً" : "جيد"
    اطبع_سطر("  " + ط.الاسم + ": " + ط.الدرجة + " - " + تقدير)
    مجموع = مجموع + ط.الدرجة
نهاية

اطبع_سطر("المتوسط: " + (مجموع / طول(طلاب)))
```

### مثال 8 — متقدم جداً: نظام إعدادات بالثوابت والبنى
```sad
# ثوابت النظام
ثابت اسم_النظام = "نظام إدارة المخزون"
ثابت الإصدار = 2
ثابت الحد_الأقصى = 1000

# بنى البيانات
بنية منتج
    نص الاسم = ""
    رقم السعر = 0
    رقم الكمية = 0
نهاية

# دوال مساعدة
دالة القيمة_الإجمالية(م)
    ارجع م.السعر * م.الكمية
نهاية

دالة تحقق_مخزون(م)
    إذا (م.الكمية <= 0)
        ارجع "نفذ"
    والا_اذا (م.الكمية < 10)
        ارجع "قليل"
    وإلا
        ارجع "متوفر"
    نهاية
نهاية

# البيانات
متغير مخزون = [
    منتج("حاسوب"، 5000، 15)،
    منتج("هاتف"، 3000، 50)،
    منتج("سماعات"، 200، 3)،
    منتج("لوحة مفاتيح"، 150، 0)،
    منتج("شاشة"، 2000، 8)
]

دالة رئيسية()
    اطبع_سطر("═══════════════════════════════════════")
    اطبع_سطر("  " + اسم_النظام + " v" + الإصدار)
    اطبع_سطر("═══════════════════════════════════════")
    
    متغير إجمالي = 0
    متغير نفذ = 0
    
    لكل م في مخزون
        متغير حالة = تحقق_مخزون(م)
        متغير قيمة = القيمة_الإجمالية(م)
        إجمالي = إجمالي + قيمة
        
        إذا (حالة == "نفذ")
            نفذ = نفذ + 1
        نهاية
        
        اطبع_سطر("  " + م.الاسم + " | " + م.السعر + " ر.س | كمية: " + م.الكمية + " | " + حالة + " | قيمة: " + قيمة)
    نهاية
    
    اطبع_سطر("═══════════════════════════════════════")
    اطبع_سطر("  إجمالي قيمة المخزون: " + إجمالي + " ر.س")
    اطبع_سطر("  عدد المنتجات: " + طول(مخزون))
    اطبع_سطر("  منتجات نفذت: " + نفذ)
    اطبع_سطر("  الحد الأقصى المسموح: " + الحد_الأقصى)
    اطبع_سطر("═══════════════════════════════════════")
نهاية
```
