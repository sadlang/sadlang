### (AR) نظرة عامة
نظام البرمجة كائنية التوجه في لغة "ص" يجمع بين أفضل ميزات Python و C++ و JavaScript:
- فئات Classes مع تغليف كامل (Encapsulation)
- وراثة Inheritance (أحادية ومتعددة)
- تعدد الأشكال Polymorphism (runtime & compile-time)

- Operator overloading
- إدارة ذاكرة تلقائية مع RAII

### (EN) Overview
Sad's OOP system combines the best features of Python, C++, and JavaScript with full encapsulation, single/multiple inheritance, polymorphism, interfaces, properties, operator overloading, and automatic memory management with RAII.

## 2️⃣ الخصائص والطرق / Properties & Methods

### 2.1 مستويات الوصول / Access Levels

```s
فئة حساب_بنكي
    # عام (Public) - الافتراضي
    نص اسم_صاحب_الحساب
    
    # خاص (Private) - داخل الفئة فقط
    خاص عشري _رصيد = 0.0
    خاص نص _رقم_سري
    
    # محمي (Protected) - الفئة والوارثات
    محمي نص _رقم_حساب
    
    # طريقة عامة للوصول للرصيد
    دالة احصل_على_رصيد() = عشري
        إرجاع _رصيد
    نهاية
    
    # طريقة عامة للإيداع
    دالة أودع(عشري مبلغ)
        إذا مبلغ > 0
            _رصيد += مبلغ
        نهاية
    نهاية
    
    # طريقة خاصة للتحقق
    خاص دالة تحقق_من_الرقم_السري(نص رقم) = منطقي
        إرجاع رقم == _رقم_سري
    نهاية
نهاية
```

### 2.2 الخصائص Properties (Getters/Setters)

```s
فئة دائرة
    خاص عشري _نصف_قطر = 0.0
    
    # خاصية نصف القطر مع getter/setter
    @خاصية
    دالة نصف_قطر() = عشري
        إرجاع _نصف_قطر
    نهاية
    
    @نصف_قطر.اضبط
    دالة نصف_قطر(عشري قيمة)
        إذا قيمة < 0
            ارمِ "نصف القطر يجب أن يكون موجباً!"
        نهاية
        _نصف_قطر = قيمة
    نهاية
    
    # خاصية محسوبة (read-only)
    @خاصية
    دالة مساحة() = عشري
        إرجاع 3.14159 * _نصف_قطر * _نصف_قطر
    نهاية
    
    @خاصية
    دالة محيط() = عشري
        إرجاع 2 * 3.14159 * _نصف_قطر
    نهاية
نهاية

# استخدام
دائرة د = دائرة()
د.نصف_قطر = 5.0        # يستدعي setter
اطبع(د.نصف_قطر)        # يستدعي getter: 5.0
اطبع(د.مساحة)          # 78.54 (محسوبة تلقائياً)
# د.مساحة = 100  # خطأ: read-only!
```

### 2.3 الطرق الثابتة Static Methods

```s
فئة رياضيات
    # ثابت من نوع الفئة
    ثابت عشري باي = 3.14159265359
    
    # طريقة ثابتة
    @ثابت
    دالة قيمة_مطلقة(رقم س) = رقم
        إرجاع س إذا س >= 0 وإلا -س
    نهاية
    
    @ثابت
    دالة أكبر(رقم أ، رقم ب) = رقم
        إرجاع أ إذا أ > ب وإلا ب
    نهاية
نهاية

# استخدام بدون إنشاء كائن
رقم مطلق = رياضيات.قيمة_مطلقة(-10)  # 10
رقم أقصى = رياضيات.أكبر(5، 8)        # 8
اطبع(رياضيات.باي)                   # 3.14159265359
```

### 2.4 طرق الفئة Class Methods

```s
فئة تاريخ
    رقم سنة
    رقم شهر
    رقم يوم
    
    # طريقة فئة لإنشاء كائن من نص
    @فئة
    دالة من_نص(فئة<تاريخ> ف، نص نص_تاريخ) = تاريخ
        مصفوفة<نص> أجزاء = نص_تاريخ.قسم("-")
        إرجاع ف(
            سنة: رقم(أجزاء[0])،
            شهر: رقم(أجزاء[1])،
            يوم: رقم(أجزاء[2])
        )
    نهاية
    
    # طريقة فئة للحصول على اليوم الحالي
    @فئة
    دالة اليوم(فئة<تاريخ> ف) = تاريخ
        # استخدام دالة من المكتبة القياسية
        (رقم س، رقم ش، رقم ي) = احصل_على_التاريخ_الحالي()
        إرجاع ف(سنة: س، شهر: ش، يوم: ي)
    نهاية
نهاية

# استخدام
تاريخ ت1 = تاريخ.من_نص("2025-11-04")
تاريخ ت2 = تاريخ.اليوم()
```

## 3️⃣ البناء والإتلاف / Constructors & Destructors

### 3.1 البناء Constructor

```s
فئة سيارة
    نص نوع
    رقم سنة_صنع
    عشري سرعة = 0.0
    
    # بناء افتراضي (ضمني)
    # بناء(نص النوع، رقم السنة)
    #     نوع = النوع
    #     سنة_صنع = السنة
    # نهاية
    
    # يمكن تعريف بناء مخصص
    بناء(نص النوع، رقم السنة، عشري السرعة_القصوى = 200.0)
        نوع = النوع
        سنة_صنع = السنة
        اطبع(ن"تم إنشاء سيارة {نوع} موديل {سنة_صنع}")
    نهاية
نهاية

# استخدام
سيارة س1 = سيارة("تويوتا"، 2025)
سيارة س2 = سيارة(النوع: "هوندا"، السنة: 2024)
```

### 3.2 بناءات متعددة Overloading

```s
فئة نقطة
    عشري س
    عشري ص
    
    # بناء افتراضي
    بناء()
        س = 0.0
        ص = 0.0
    نهاية
    
    # بناء بمعاملين
    بناء(عشري السينية، عشري الصادية)
        س = السينية
        ص = الصادية
    نهاية
    
    # بناء نسخ
    بناء(نقطة أخرى)
        س = أخرى.س
        ص = أخرى.ص
    نهاية
نهاية

# استخدام
نقطة ن1 = نقطة()              # (0، 0)
نقطة ن2 = نقطة(3.5، 7.2)     # (3.5، 7.2)
نقطة ن3 = نقطة(ن2)           # نسخ من ن2
```

### 3.3 الإتلاف Destructor

```s
فئة ملف
    خاص نص _اسم_ملف
    خاص أي _مؤشر_ملف
    
    بناء(نص اسم_ملف)
        _اسم_ملف = اسم_ملف
        _مؤشر_ملف = افتح_ملف(اسم_ملف)
        اطبع(ن"تم فتح الملف: {_اسم_ملف}")
    نهاية
    
    # إتلاف - يُستدعى تلقائياً عند حذف الكائن
    إتلاف()
        إذا _مؤشر_ملف != لاشيء
            أغلق_ملف(_مؤشر_ملف)
            اطبع(ن"تم إغلاق الملف: {_اسم_ملف}")
        نهاية
    نهاية
نهاية

# RAII - Resource Acquisition Is Initialization
{
    ملف م = ملف("data.txt")
    # استخدام الملف...
}  # يتم استدعاء الإتلاف تلقائياً هنا
```

## 4️⃣ الوراثة / Inheritance

### 4.1 وراثة أحادية Single Inheritance

```s
# فئة أساسية
فئة حيوان
    نص اسم
    رقم عمر
    
    بناء(نص الاسم، رقم العمر)
        اسم = الاسم
        عمر = العمر
    نهاية
    
    دالة أصدر_صوت()
        اطبع("صوت حيوان")
    نهاية
    
    دالة تحرك()
        اطبع(ن"{اسم} يتحرك")
    نهاية
نهاية

# فئة مشتقة
فئة كلب : حيوان
    نص فصيلة
    
    بناء(نص الاسم، رقم العمر، نص الفصيلة)
        # استدعاء بناء الفئة الأساسية
        أساس(الاسم، العمر)
        فصيلة = الفصيلة
    نهاية
    
    # تجاوز الطريقة Override
    دالة أصدر_صوت()
        اطبع("وف وف!")
    نهاية
    
    # طريقة جديدة
    دالة اجلب_الكرة()
        اطبع(ن"{اسم} يجلب الكرة")
    نهاية
نهاية

# استخدام
كلب ك = كلب("ريكس"، 3، "جولدن ريتريفر")
ك.أصدر_صوت()      # "وف وف!" (مُتجاوزة)
ك.تحرك()          # "ريكس يتحرك" (موروثة)
ك.اجلب_الكرة()    # "ريكس يجلب الكرة" (جديدة)
```

### 4.2 وراثة متعددة Multiple Inheritance

## 5️⃣ التعدد الشكلي / Polymorphism

### 5.1 التعدد الشكلي في وقت التنفيذ Runtime Polymorphism

```s
# استخدام الفئات السابقة
فئة حيوان
    دالة أصدر_صوت()
        اطبع("صوت حيوان")
    نهاية
نهاية

فئة قطة : حيوان
    دالة أصدر_صوت()
        اطبع("مياو!")
    نهاية
نهاية

فئة بقرة : حيوان
    دالة أصدر_صوت()
        اطبع("موو!")
    نهاية
نهاية

# دالة polymorphic
دالة دع_الحيوان_يصدر_صوتاً(حيوان ح)
    ح.أصدر_صوت()  # يستدعي الطريقة المناسبة حسب النوع الفعلي
نهاية

# استخدام
قطة ق = قطة()
كلب ك = كلب()
بقرة ب = بقرة()

دع_الحيوان_يصدر_صوتاً(ق)  # "مياو!"
دع_الحيوان_يصدر_صوتاً(ك)  # "وف وف!"
دع_الحيوان_يصدر_صوتاً(ب)  # "موو!"

# مصفوفة من أنواع مختلفة
مصفوفة<حيوان> حيوانات = [ق، ك، ب]
لكل حيوان في حيوانات
    حيوان.أصدر_صوت()
نهاية
```

### 5.2 الطرق المجردة Abstract Methods

```s
# فئة مجردة
فئة مجردة شكل_هندسي
    # طريقة مجردة (يجب تنفيذها في الفئات المشتقة)
    مجرد دالة احسب_مساحة() = عشري
    مجرد دالة احسب_محيط() = عشري
    
    # طريقة عادية (مُنفذة)
    دالة اطبع_معلومات()
        اطبع(ن"المساحة: {احسب_مساحة()}")
        اطبع(ن"المحيط: {احسب_محيط()}")
    نهاية
نهاية

# فئة ملموسة
فئة مستطيل : شكل_هندسي
    عشري عرض
    عشري ارتفاع
    
    بناء(عشري العرض، عشري الارتفاع)
        عرض = العرض
        ارتفاع = الارتفاع
    نهاية
    
    # تنفيذ الطرق المجردة
    دالة احسب_مساحة() = عشري
        إرجاع عرض * ارتفاع
    نهاية
    
    دالة احسب_محيط() = عشري
        إرجاع 2 * (عرض + ارتفاع)
    نهاية
نهاية

# لا يمكن إنشاء كائن من فئة مجردة
# شكل_هندسي ش = شكل_هندسي()  # خطأ!

# يمكن إنشاء كائن من فئة ملموسة
مستطيل م = مستطيل(5.0، 3.0)
م.اطبع_معلومات()
# المساحة: 15.0
# المحيط: 16.0
```

## 6️⃣ الواجهات / Interfaces

### 6.1 تعريف واجهة Interface

```s
# واجهة (Interface)
واجهة قابل_للرسم
    مجرد دالة ارسم()
    مجرد دالة احذف()
نهاية

واجهة قابل_للتحريك
    مجرد دالة حرك(عشري دلتا_س، عشري دلتا_ص)
نهاية

# فئة تُنفذ الواجهات
فئة دائرة : قابل_للرسم، قابل_للتحريك
    عشري س، ص
    عشري نصف_قطر
    
    بناء(عشري السينية، عشري الصادية، عشري النصف_قطر)
        س = السينية
        ص = الصادية
        نصف_قطر = النصف_قطر
    نهاية
    
    # تنفيذ طرق قابل_للرسم
    دالة ارسم()
        اطبع(ن"رسم دائرة عند ({س}، {ص}) بنصف قطر {نصف_قطر}")
    نهاية
    
    دالة احذف()
        اطبع("حذف الدائرة")
    نهاية
    
    # تنفيذ طرق قابل_للتحريك
    دالة حرك(عشري دلتا_س، عشري دلتا_ص)
        س += دلتا_س
        ص += دلتا_ص
    نهاية
نهاية

# استخدام
دائرة د = دائرة(10.0، 20.0، 5.0)
د.ارسم()           # "رسم دائرة..."
د.حرك(5.0، -3.0)
د.ارسم()           # موقع جديد
```

### 6.2 واجهات متعددة Multiple Interfaces

```s
واجهة قابل_للتسلسل
    مجرد دالة إلى_نص() = نص
    مجرد دالة من_نص(نص بيانات)
نهاية

واجهة قابل_للمقارنة
    مجرد دالة قارن(قابل_للمقارنة آخر) = رقم
نهاية

فئة طالب : قابل_للتسلسل، قابل_للمقارنة
    نص اسم
    رقم درجة
    
    بناء(نص الاسم، رقم الدرجة)
        اسم = الاسم
        درجة = الدرجة
    نهاية
    
    # تنفيذ قابل_للتسلسل
    دالة إلى_نص() = نص
        إرجاع ن"{اسم}:{درجة}"
    نهاية
    
    دالة من_نص(نص بيانات)
        مصفوفة<نص> أجزاء = بيانات.قسم(":")
        اسم = أجزاء[0]
        درجة = رقم(أجزاء[1])
    نهاية
    
    # تنفيذ قابل_للمقارنة
    دالة قارن(قابل_للمقارنة آخر) = رقم
        طالب ط = (طالب)آخر
        إرجاع درجة - ط.درجة
    نهاية
نهاية

# استخدام
طالب ط1 = طالب("أحمد"، 95)
نص بيانات = ط1.إلى_نص()  # "أحمد:95"

طالب ط2 = طالب("فاطمة"، 88)
رقم فرق = ط1.قارن(ط2)     # 7
```

## 7️⃣ تحميل العوامل / Operator Overloading

### 7.1 العوامل الحسابية Arithmetic Operators

```s
فئة عدد_مركب
    عشري حقيقي
    عشري تخيلي
    
    بناء(عشري ح، عشري ت)
        حقيقي = ح
        تخيلي = ت
    نهاية
    
    # تحميل عامل +
    عامل +(عدد_مركب آخر) = عدد_مركب
        إرجاع عدد_مركب(
            حقيقي + آخر.حقيقي،
            تخيلي + آخر.تخيلي
        )
    نهاية
    
    # تحميل عامل -
    عامل -(عدد_مركب آخر) = عدد_مركب
        إرجاع عدد_مركب(
            حقيقي - آخر.حقيقي،
            تخيلي - آخر.تخيلي
        )
    نهاية
    
    # تحميل عامل *
    عامل *(عدد_مركب آخر) = عدد_مركب
        # (a + bi)(c + di) = (ac - bd) + (ad + bc)i
        عشري ح = حقيقي * آخر.حقيقي - تخيلي * آخر.تخيلي
        عشري ت = حقيقي * آخر.تخيلي + تخيلي * آخر.حقيقي
        إرجاع عدد_مركب(ح، ت)
    نهاية
    
    # تحميل unary -
    عامل -() = عدد_مركب
        إرجاع عدد_مركب(-حقيقي، -تخيلي)
    نهاية
    
    # تحويل إلى نص
    دالة إلى_نص() = نص
        إرجاع ن"{حقيقي} + {تخيلي}i"
    نهاية
نهاية

# استخدام
عدد_مركب ع1 = عدد_مركب(3.0، 4.0)  # 3 + 4i
عدد_مركب ع2 = عدد_مركب(1.0، 2.0)  # 1 + 2i

عدد_مركب مجموع = ع1 + ع2            # 4 + 6i
عدد_مركب فرق = ع1 - ع2               # 2 + 2i
عدد_مركب ضرب = ع1 * ع2               # -5 + 10i
عدد_مركب نفي = -ع1                   # -3 - 4i
```

### 7.2 عوامل المقارنة Comparison Operators

```s
فئة نقطة
    عشري س، ص
    
    بناء(عشري السينية، عشري الصادية)
        س = السينية
        ص = الصادية
    نهاية
    
    # تحميل عامل ==
    عامل ==(نقطة أخرى) = منطقي
        إرجاع س == أخرى.س && ص == أخرى.ص
    نهاية
    
    # تحميل عامل !=
    عامل !=(نقطة أخرى) = منطقي
        إرجاع !(هذا == أخرى)
    نهاية
    
    # تحميل عامل < (المقارنة بالبعد عن الأصل)
    عامل <(نقطة أخرى) = منطقي
        عشري بعد1 = جذر_تربيعي(س * س + ص * ص)
        عشري بعد2 = جذر_تربيعي(أخرى.س * أخرى.س + أخرى.ص * أخرى.ص)
        إرجاع بعد1 < بعد2
    نهاية
نهاية

# استخدام
نقطة ن1 = نقطة(3.0، 4.0)
نقطة ن2 = نقطة(3.0، 4.0)
نقطة ن3 = نقطة(5.0، 12.0)

اطبع(ن1 == ن2)  # صحيح
اطبع(ن1 == ن3)  # خطأ
اطبع(ن1 < ن3)   # صحيح (البعد: 5 < 13)
```

### 7.3 عامل الفهرسة Indexing Operator

```s
فئة مصفوفة_ديناميكية<T>
    خاص مصفوفة<T> _بيانات = []
    
    # تحميل عامل []
    عامل [](رقم فهرس) = T
        إذا فهرس < 0 أو فهرس >= _بيانات.طول()
            ارمِ "فهرس خارج النطاق!"
        نهاية
        إرجاع _بيانات[فهرس]
    نهاية
    
    # تحميل عامل []= (setter)
    عامل []=(رقم فهرس، T قيمة)
        إذا فهرس < 0 أو فهرس >= _بيانات.طول()
            ارمِ "فهرس خارج النطاق!"
        نهاية
        _بيانات[فهرس] = قيمة
    نهاية
    
    دالة أضف(T عنصر)
        _بيانات.أضف(عنصر)
    نهاية
نهاية

# استخدام
مصفوفة_ديناميكية<رقم> م = مصفوفة_ديناميكية<رقم>()
م.أضف(10)
م.أضف(20)

اطبع(م[0])   # 10 (getter)
م[1] = 25     # (setter)
اطبع(م[1])   # 25
```

## 9️⃣ قواعد BNF / BNF Grammar

```bnf
<class_def> ::= <class_modifiers> "فئة" [<type_params>] <identifier> 
                [<inheritance>] <class_body> "نهاية"

<class_modifiers> ::= ["مجردة"] | ["ثابتة"]

<type_params> ::= "<" <identifier> ("،" <identifier>)* ">"

<inheritance> ::= ":" <base_list>

<base_list> ::= <identifier> ("،" <identifier>)*

<class_body> ::= <class_member>*

<class_member> ::= <field_def>
                 | <method_def>
                 | <constructor_def>
                 | <destructor_def>
                 | <operator_def>
                 | <property_def>

<field_def> ::= [<access_modifier>] [<field_modifiers>] <type> <identifier> ["=" <expression>]

<access_modifier> ::= "عام" | "public"
                    | "خاص" | "private"
                    | "محمي" | "protected"

<field_modifiers> ::= "ثابت" | "static"

<method_def> ::= [<access_modifier>] [<method_modifiers>] <function_def>

<method_modifiers> ::= ["مجرد" | "abstract"]
                     | ["افتراضي" | "virtual"]
                     | ["تجاوز" | "override"]
                     | ["ثابت" | "static"]
                     | ["@خاصية"]

<constructor_def> ::= [<access_modifier>] "بناء" "(" [<param_list>] ")" <function_body> "نهاية"

<destructor_def> ::= [<access_modifier>] "إتلاف" "(" ")" <function_body> "نهاية"

<operator_def> ::= "عامل" <operator> "(" [<param_list>] ")" "=" <type> <function_body> "نهاية"

<operator> ::= "+" | "-" | "*" | "/" | "%" | "=="  | "!=" | "<" | ">" | "<=" | ">=" 
             | "[]" | "[]=" | "()"

<property_def> ::= "@خاصية" <method_def>

<interface_def> ::= "واجهة" <identifier> <interface_body> "نهاية"

<interface_body> ::= <interface_member>*

<interface_member> ::= "مجرد" <function_signature>

<object_creation> ::= <identifier> ["<" <type_list> ">"] "(" [<arg_list>] ")"
```

## 🔟 تمثيل AST / AST Representation

```cpp
// include/parser/ast/class_ast.h

#pragma once
#include "ast_node.h"
#include "function_ast.h"
#include <vector>

namespace Sad {
namespace Parser {
namespace AST {

/**
 * @brief (AR) مستويات الوصول
 * @brief (EN) Access levels
 */
enum class AccessLevel {
    PUBLIC,      // عام
    PRIVATE,     // خاص
    PROTECTED    // محمي
};

/**
 * @brief (AR) حقل الفئة
 * @brief (EN) Class field
 */
struct Field {
    std::string name;
    std::shared_ptr<Type> type;
    AccessLevel access;
    bool isStatic;
    std::optional<std::shared_ptr<Expression>> initializer;
};

/**
 * @brief (AR) طريقة الفئة
 * @brief (EN) Class method
 */
struct Method {
    std::string name;
    std::shared_ptr<FunctionDef> function;
    AccessLevel access;
    bool isStatic;
    bool isAbstract;
    bool isVirtual;
    bool isOverride;
    std::vector<std::string> decorators;
};

/**
 * @brief (AR) تعريف الفئة
 * @brief (EN) Class definition
 */
class ClassDef : public Statement {
public:
    std::string name;
    std::vector<std::string> typeParameters;
    std::vector<std::string> baseClasses;
    std::vector<Field> fields;
    std::vector<Method> methods;
    std::optional<std::shared_ptr<FunctionDef>> constructor;
    std::optional<std::shared_ptr<FunctionDef>> destructor;
    bool isAbstract;
    bool isInterface;
    
    ClassDef(const std::string& name, bool isAbstract = false)
        : name(name), isAbstract(isAbstract), isInterface(false) {}
    
    NodeType getType() const override { return NodeType::CLASS_DEF; }
};

/**
 * @brief (AR) تعريف عامل
 * @brief (EN) Operator definition
 */
class OperatorDef : public Statement {
public:
    std::string operatorSymbol;
    std::vector<Parameter> parameters;
    std::shared_ptr<Type> returnType;
    std::vector<std::shared_ptr<Statement>> body;
    
    NodeType getType() const override { return NodeType::OPERATOR_DEF; }
};

/**
 * @brief (AR) وصول لعضو
 * @brief (EN) Member access
 */
class MemberAccess : public Expression {
public:
    std::shared_ptr<Expression> object;
    std::string memberName;
    
    MemberAccess(std::shared_ptr<Expression> obj, const std::string& member)
        : object(obj), memberName(member) {}
    
    NodeType getType() const override { return NodeType::MEMBER_ACCESS; }
};

} // namespace AST
} // namespace Parser
} // namespace Sad
```
