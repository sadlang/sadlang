## 2️⃣ المتغيرات والأنواع

### 2.1 الأرقام / Numbers

```s
# أرقام صحيحة
رقم عمر = 25
رقم سالب = -100
رقم ثنائي = 0b1010      # 10
رقم ست_عشري = 0xFF     # 255

# أرقام عشرية
عشري درجة = 95.5
عشري باي = 3.14159
عشري علمي = 3.0e8      # 300,000,000

# عمليات حسابية
رقم مجموع = 5 + 3        # 8
رقم فرق = 10 - 4         # 6
رقم ضرب = 6 * 7          # 42
رقم قسمة = 15 / 3        # 5
رقم باقي = 17 % 5        # 2
رقم أس = 2 ** 10         # 1024
```

### 2.2 النصوص / Strings

```s
# تعريف نصوص
نص اسم = "أحمد محمد"
نص تحية = 'مرحباً'
نص متعدد = "
    سطر أول
    سطر ثاني
    سطر ثالث
"

# دمج نصوص
نص كامل = اسم + " - " + "مبرمج"

# تنسيق f-string
رقم عمر = 25
نص رسالة = ن"الاسم: {اسم}، العمر: {عمر}"

# عمليات النصوص
رقم طول = طول(اسم)
نص جزء = اسم[0:4]
منطقي يحتوي = "أحمد" في اسم
```

### 2.3 القيم المنطقية / Booleans

```s
منطقي نشط = صحيح
منطقي مكتمل = خطأ

# عمليات منطقية
منطقي و = صحيح && خطأ      # خطأ
منطقي أو = صحيح || خطأ     # صحيح
منطقي نفي = !صحيح          # خطأ

# مقارنات
منطقي يساوي = 5 == 5       # صحيح
منطقي أكبر = 10 > 5        # صحيح
منطقي أصغر_أو_يساوي = 3 <= 5  # صحيح
```

## 3️⃣ البنى التحكمية

### 3.1 شرطية if/else

```s
رقم درجة = 85

إذا درجة >= 90
    اطبع("ممتاز")
وإلا_إذا درجة >= 80
    اطبع("جيد جداً")
وإلا_إذا درجة >= 70
    اطبع("جيد")
وإلا_إذا درجة >= 60
    اطبع("مقبول")
وإلا
    اطبع("راسب")
نهاية
```

### 3.2 حلقات Loops

```s
# حلقة بينما / while
رقم عداد = 0
بينما عداد < 5
    اطبع(عداد)
    عداد += 1
نهاية

# حلقة لكل / for-each
مصفوفة<نص> أسماء = ["أحمد"، "فاطمة"، "خالد"]
لكل اسم في أسماء
    اطبع(ن"مرحباً {اسم}")
نهاية

# حلقة مع نطاق / range
لكل ع في نطاق(1، 11)
    اطبع(ن"العدد: {ع}")
نهاية

# حلقة بخطوة / step
لكل ع في نطاق(0، 20، 2)  # أرقام زوجية
    اطبع(ع)
نهاية
```

### 3.3 التحكم في الحلقات

```s
# توقف / break
لكل ع في نطاق(1، 100)
    إذا ع == 10

    نهاية
    اطبع(ع)
نهاية

# تخطى / continue
لكل ع في نطاق(1، 11)
    إذا ع % 2 == 0
        تخطى  # تخطى الأرقام الزوجية
    نهاية
    اطبع(ع)  # فقط الأرقام الفردية

```

### 3.4 التعبيرات الشرطية / Ternary


## 4️⃣ الدوال

### 4.1 دوال بسيطة / Simple Functions

```s
# دالة بدون معاملات
دالة قل_مرحبا()
    اطبع("مرحباً!")
نهاية

# دالة بمعامل
دالة مربع(رقم س) = رقم
    إرجاع س * س
نهاية

# دالة سطر واحد
دالة ضعف(رقم س) = رقم: س * 2

# قيم إرجاع متعددة
دالة قسم(رقم أ، رقم ب) = (رقم، رقم)
    إرجاع (أ / ب، أ % ب)
نهاية

(خارج، باقي) = قسم(17، 5)
```

### 4.2 معاملات متقدمة / Advanced Parameters

```s
# معاملات اختيارية
دالة حي(نص اسم، نص تحية = "مرحباً")
    اطبع(ن"{تحية} {اسم}!")
نهاية

حي("أحمد")              # "مرحباً أحمد!"
حي("أحمد"، "أهلاً")     # "أهلاً أحمد!"

# معاملات varargs
دالة مجموع(*مصفوفة<رقم> أرقام) = رقم
    رقم نتيجة = 0
    لكل ر في أرقام
        نتيجة += ر
    نهاية
    إرجاع نتيجة
نهاية

اطبع(مجموع(1، 2، 3، 4، 5))  # 15

# معاملات kwargs
دالة أنشئ_مستخدم(**خريطة بيانات)
    اطبع(بيانات["اسم"])
    اطبع(بيانات["عمر"])
نهاية

أنشئ_مستخدم(اسم: "أحمد"، عمر: 25)
```

### 4.3 Lambda

```s
# lambda بسيطة
أي ضعف = lambda (رقم س) = س * 2
اطبع(ضعف(5))  # 10

# lambda مع map
مصفوفة<رقم> أرقام = [1، 2، 3، 4، 5]
مصفوفة<رقم> مربعات = أرقام.خرطة(lambda (رقم س) = س * س)

# lambda مع filter
مصفوفة<رقم> زوجية = أرقام.رشح(lambda (رقم س) = س % 2 == 0)

# lambda مع sort
مصفوفة<نص> أسماء = ["أحمد"، "فاطمة"، "علي"]
أسماء.رتب(مفتاح: lambda (نص اسم) = طول(اسم))
```

### 4.4 Closures

```s
# إغلاق بسيط
دالة منشئ_عداد() = دالة<رقم>
    رقم عداد = 0
    
    دالة زد()
        عداد += 1
        إرجاع عداد
    نهاية
    
    إرجاع زد
نهاية

أي عداد = منشئ_عداد()
اطبع(عداد())  # 1
اطبع(عداد())  # 2
اطبع(عداد())  # 3
```

## 5️⃣ المصفوفات والخرائط

### 5.1 المصفوفات / Arrays

```s
# تعريف
مصفوفة<رقم> أرقام = [1، 2، 3، 4، 5]
مصفوفة<نص> أسماء = ["أحمد"، "فاطمة"، "خالد"]

# الوصول
رقم أول = أرقام[0]
رقم آخر = أرقام[-1]

# التقطيع
مصفوفة<رقم> جزء = أرقام[1:4]  # [2، 3، 4]

# عمليات
أرقام.أضف(6)
أرقام.أدرج(0، 0)
أرقام.احذف(2)

# تكرار
لكل رقم في أرقام
    اطبع(رقم)
نهاية

# comprehension
مصفوفة<رقم> مربعات = [س * س لكل س في نطاق(1، 11)]
مصفوفة<رقم> زوجية = [س لكل س في أرقام إذا س % 2 == 0]
```

### 5.2 الخرائط / Dictionaries

```s
# تعريف
خريطة<نص، رقم> أعمار = {
    "أحمد": 25،
    "فاطمة": 30،
    "خالد": 28
}

# الوصول
رقم عمر_أحمد = أعمار["أحمد"]
رقم عمر_افتراضي = أعمار.احصل("علي"، 0)

# إضافة/تحديث
أعمار["سارة"] = 22
أعمار["أحمد"] = 26

# حذف
أعمار.احذف("خالد")

# تكرار
لكل (اسم، عمر) في أعمار.عناصر()
    اطبع(ن"{اسم}: {عمر}")
نهاية

# comprehension
خريطة<نص، رقم> أطوال = {اسم: طول(اسم) لكل اسم في أعمار.مفاتيح()}
```

## 6️⃣ الفئات والكائنات

### 6.1 فئة بسيطة / Simple Class

```s
فئة شخص
    نص اسم
    رقم عمر
    
    بناء(نص الاسم، رقم العمر)
        اسم = الاسم
        عمر = العمر
    نهاية
    
    دالة عرف_عن_نفسك()
        اطبع(ن"أنا {اسم}، عمري {عمر} سنة")
    نهاية
نهاية

# استخدام
شخص ش = شخص("أحمد"، 25)
ش.عرف_عن_نفسك()
```

### 6.2 وراثة / Inheritance

```s
فئة موظف : شخص
    نص وظيفة
    عشري راتب
    
    بناء(نص الاسم، رقم العمر، نص الوظيفة، عشري الراتب)
        أساس(الاسم، العمر)
        وظيفة = الوظيفة
        راتب = الراتب
    نهاية
    
    دالة عرف_عن_نفسك()
        أساس.عرف_عن_نفسك()
        اطبع(ن"أعمل كـ {وظيفة}")
    نهاية
نهاية

موظف م = موظف("فاطمة"، 30، "مهندسة"، 15000.0)
م.عرف_عن_نفسك()
```

### 6.3 خصائص Properties

```s
فئة حساب_بنكي
    خاص عشري _رصيد
    
    بناء()
        _رصيد = 0.0
    نهاية
    
    @خاصية
    دالة رصيد() = عشري
        إرجاع _رصيد
    نهاية
    
    دالة إيداع(عشري مبلغ)
        إذا مبلغ > 0
            _رصيد += مبلغ
        نهاية
    نهاية
    
    دالة سحب(عشري مبلغ) = منطقي
        إذا مبلغ > 0 && مبلغ <= _رصيد
            _رصيد -= مبلغ
            إرجاع صحيح
        نهاية
        إرجاع خطأ
    نهاية
نهاية

حساب_بنكي ح = حساب_بنكي()
ح.إيداع(1000.0)
اطبع(ح.رصيد)  # 1000.0
```

### 6.4 فئات عامة Generic Classes

```s
فئة<T> مكدس
    خاص مصفوفة<T> _عناصر = []
    
    دالة ادفع(T عنصر)
        _عناصر.أضف(عنصر)
    نهاية
    
    دالة اسحب() = T
        إذا _عناصر.فارغة()
            ارمِ "المكدس فارغ!"
        نهاية
        T عنصر = _عناصر[-1]
        _عناصر.أزل_آخر()
        إرجاع عنصر
    نهاية
    
    دالة فارغ() = منطقي
        إرجاع _عناصر.طول() == 0
    نهاية
نهاية

# استخدام
مكدس<رقم> أرقام = مكدس<رقم>()
أرقام.ادفع(10)
أرقام.ادفع(20)
اطبع(أرقام.اسحب())  # 20
```

## 7️⃣ أمثلة متقدمة

### 7.1 Decorator

```s
# تعريف decorator
دالة سجل(دالة<أي> دالة) = دالة<أي>

        اطبع(ن"انتهى {دالة.اسم}")
        إرجاع نتيجة
    نهاية
    إرجاع مغلفة
نهاية

# استخدام decorator
@سجل
دالة اجمع(رقم أ، رقم ب) = رقم
    إرجاع أ + ب
نهاية

اجمع(5، 3)
# استدعاء اجمع
# انتهى اجمع
```

### 7.2 Context Manager

```s
فئة مؤقت
    نص رسالة
    رقم وقت_بداية
    
    بناء(نص الرسالة)
        رسالة = الرسالة
    نهاية
    
    دالة __دخول__()
        وقت_بداية = احصل_على_الوقت()
        اطبع(ن"بدء: {رسالة}")
        إرجاع هذا
    نهاية
    
    دالة __خروج__(أي استثناء)
        رقم مدة = احصل_على_الوقت() - وقت_بداية
        اطبع(ن"انتهى في {مدة} ثانية")
    نهاية
نهاية

# استخدام
مع مؤقت("عملية معقدة") كـ ت
    # عمل...
    لكل ع في نطاق(1000000)
        رقم س = ع * 2
    نهاية
نهاية
```

### 7.3 Generator

```s

# استخدام
```

### 8.2 نظام إدارة طلاب

```s
فئة طالب
    نص اسم
    رقم رقم_تعريف
    مصفوفة<عشري> درجات
    
    بناء(نص الاسم، رقم الرقم)
        اسم = الاسم
        رقم_تعريف = الرقم
        درجات = []
    نهاية
    
    دالة أضف_درجة(عشري درجة)
        إذا درجة >= 0 && درجة <= 100
            درجات.أضف(درجة)
        وإلا
            ارمِ "درجة غير صحيحة!"
        نهاية
    نهاية
    
    @خاصية
    دالة متوسط() = عشري
        إذا درجات.طول() == 0
            إرجاع 0.0
        نهاية
        عشري مجموع = درجات.اطوِ(lambda (أ، ب) = أ + ب، 0.0)
        إرجاع مجموع / درجات.طول()
    نهاية
    
    دالة عرض_المعلومات()
        اطبع(ن"الطالب: {اسم} (ID: {رقم_تعريف})")
        اطبع(ن"المتوسط: {متوسط:.2f}")
        اطبع(ن"الدرجات: {درجات}")
    نهاية
نهاية

فئة نظام_طلاب
    خاص خريطة<رقم، طالب> _طلاب = {}
    
    دالة أضف_طالب(طالب ط)
        _طلاب[ط.رقم_تعريف] = ط
        اطبع(ن"✓ تمت إضافة الطالب: {ط.اسم}")
    نهاية
    
    دالة احصل_على_طالب(رقم رقم_تعريف) = طالب
        إذا رقم_تعريف في _طلاب
            إرجاع _طلاب[رقم_تعريف]
        نهاية
        إرجاع لاشيء
    نهاية
    
    دالة عرض_جميع_الطلاب()
        اطبع("\n=== قائمة الطلاب ===")
        لكل (رقم، طالب) في _طلاب.عناصر()
            طالب.عرض_المعلومات()
            اطبع("---")
        نهاية
    نهاية
    
    دالة أفضل_طالب() = طالب
        إذا _طلاب.طول() == 0
            إرجاع لاشيء
        نهاية
        
        طالب أفضل = لاشيء
        عشري أعلى_متوسط = 0.0
        
        لكل (رقم، طالب) في _طلاب.عناصر()
            إذا طالب.متوسط > أعلى_متوسط
                أفضل = طالب
                أعلى_متوسط = طالب.متوسط
            نهاية
        نهاية
        
        إرجاع أفضل
    نهاية
نهاية

# البرنامج الرئيسي
دالة main()
    نظام_طلاب نظام = نظام_طلاب()
    
    # إضافة طلاب
    طالب ط1 = طالب("أحمد محمد"، 101)
    ط1.أضف_درجة(95.0)
    ط1.أضف_درجة(88.0)
    ط1.أضف_درجة(92.0)
    نظام.أضف_طالب(ط1)
    
    طالب ط2 = طالب("فاطمة علي"، 102)
    ط2.أضف_درجة(98.0)
    ط2.أضف_درجة(96.0)
    ط2.أضف_درجة(100.0)
    نظام.أضف_طالب(ط2)
    
    طالب ط3 = طالب("خالد أحمد"، 103)
    ط3.أضف_درجة(85.0)
    ط3.أضف_درجة(90.0)
    ط3.أضف_درجة(87.0)
    نظام.أضف_طالب(ط3)
    
    # عرض جميع الطلاب
    نظام.عرض_جميع_الطلاب()
    
    # أفضل طالب
    طالب أفضل = نظام.أفضل_طالب()
    إذا أفضل != لاشيء
        اطبع("\n🏆 أفضل طالب:")
        أفضل.عرض_المعلومات()
    نهاية
نهاية

main()
```

### 8.3 لعبة تخمين الرقم
