## 1️⃣ Templates/Generics

### 1.1 Function Templates

```s
# دالة عامة بسيطة
دالة<T> أكبر(T أ، T ب) = T
    إرجاع أ إذا أ > ب وإلا ب
نهاية

# استخدام
رقم رقم_أكبر = أكبر<رقم>(5، 10)        # 10
نص نص_أكبر = أكبر<نص>("أحمد"، "محمد")  # "محمد"

# Type inference
رقم ر = أكبر(5، 10)  # يستنتج T = رقم تلقائياً
```

### 1.2 Class Templates

```s
# فئة عامة stack
فئة<T> مكدس
    خاص مصفوفة<T> _عناصر = []
    
    دالة ادفع(T عنصر)
        _عناصر.أضف(عنصر)
    نهاية
    
    دالة اسحب() = T
        إذا _عناصر.فارغة()
            ارمِ "المكدس فارغ!"
        نهاية
        T عنصر = _عناصر[-1]
        _عناصر.أزل_آخر()
        إرجاع عنصر
    نهاية
    
    دالة فارغ() = منطقي
        إرجاع _عناصر.طول() == 0
    نهاية
    
    دالة حجم() = رقم
        إرجاع _عناصر.طول()
    نهاية
نهاية

# استخدام
مكدس<رقم> أرقام = مكدس<رقم>()
أرقام.ادفع(10)
أرقام.ادفع(20)
رقم قمة = أرقام.اسحب()  # 20

مكدس<نص> نصوص = مكدس<نص>()
نصوص.ادفع("مرحباً")
```

## 2️⃣ RAII / Resource Management

### 2.1 RAII Pattern

```s
# إدارة ملف مع RAII
فئة مدير_ملف
    خاص نص _اسم_ملف
    خاص أي _مؤشر
    
    # البناء يحصل على المورد
    بناء(نص اسم_ملف، نص نمط)
        _اسم_ملف = اسم_ملف
        _مؤشر = افتح_ملف(_اسم_ملف، نمط)
        إذا _مؤشر == لاشيء
            ارمِ ن"فشل فتح الملف: {_اسم_ملف}"
        نهاية
        اطبع(ن"✓ فتح: {_اسم_ملف}")
    نهاية
    
    # الإتلاف يحرر المورد تلقائياً
    إتلاف()
        إذا _مؤشر != لاشيء
            أغلق_ملف(_مؤشر)
            اطبع(ن"✓ إغلاق: {_اسم_ملف}")
        نهاية
    نهاية
    
    دالة اكتب(نص بيانات)
        إذا _مؤشر == لاشيء
            ارمِ "الملف غير مفتوح!"
        نهاية
        اكتب_إلى_ملف(_مؤشر، بيانات)
    نهاية
نهاية

# استخدام - تلقائياً يُنظف
دالة معالج_بيانات()
    {
        مدير_ملف ملف = مدير_ملف("data.txt"، "كتابة")
        ملف.اكتب("بيانات مهمة")
        
        # حتى لو حدث استثناء، الملف سيُغلق
        إذا شرط_خطأ
            ارمِ "خطأ!"
        نهاية
    }  # ✓ يُستدعى الإتلاف هنا تلقائياً
نهاية
```

### 2.2 Lock Guard Pattern

```s
# قفل تلقائي للخيوط
فئة حارس_قفل
    خاص قفل _قفل
    
    بناء(قفل ق)
        _قفل = ق
        _قفل.اقفل()
    نهاية
    
    إتلاف()
        _قفل.افتح()
    نهاية
نهاية

# استخدام
قفل قفل_عام = قفل()
رقم عداد_مشترك = 0

دالة زد_العداد()
    {
        حارس_قفل حارس = حارس_قفل(قفل_عام)
        عداد_مشترك += 1
        # القفل يُفتح تلقائياً عند الخروج
    }
نهاية
```

## 3️⃣ Move Semantics

### 3.1 Move vs Copy

```s
فئة بيانات_كبيرة
    خاص مصفوفة<رقم> _بيانات
    
    # بناء نسخ (Copy Constructor)
    بناء(بيانات_كبيرة أخرى)
        اطبع("نسخ البيانات...")
        _بيانات = أخرى._بيانات.نسخ()  # نسخ عميق
    نهاية
    
    # بناء نقل (Move Constructor)
    بناء(بيانات_كبيرة&& أخرى)  # && = rvalue reference
        اطبع("نقل البيانات...")
        _بيانات = أخرى._بيانات  # نقل سريع
        أخرى._بيانات = لاشيء     # المصدر يصبح فارغ
    نهاية
نهاية

# استخدام
بيانات_كبيرة ب1 = بيانات_كبيرة()

# نسخ
بيانات_كبيرة ب2 = ب1          # Copy - "نسخ البيانات..."

# نقل
بيانات_كبيرة ب3 = نقل(ب1)     # Move - "نقل البيانات..."
# ب1 الآن فارغ، ب3 لديه البيانات
```

### 3.2 Perfect Forwarding

```s
# إرسال الوسائط بكفاءة
دالة<T> أنشئ_مُغلف(T&& كائن) = مُغلف<T>
    # forward يحافظ على نوع المرجع (copy/move)
    إرجاع مُغلف<T>(إرسال<T>(كائن))
نهاية

# استخدام
بيانات_كبيرة ب = بيانات_كبيرة()
مُغلف م1 = أنشئ_مُغلف(ب)         # Copy
مُغلف م2 = أنشئ_مُغلف(نقل(ب))    # Move
```

## 4️⃣ Operator Overloading

### 4.1 Arithmetic Operators

```s
فئة متجه
    عشري س، ص، ع
    
    بناء(عشري السينية، عشري الصادية، عشري العينية)
        س = السينية
        ص = الصادية
        ع = العينية
    نهاية
    
    # عامل +
    عامل +(متجه آخر) = متجه
        إرجاع متجه(س + آخر.س، ص + آخر.ص، ع + آخر.ع)
    نهاية
    
    # عامل - (ثنائي)
    عامل -(متجه آخر) = متجه
        إرجاع متجه(س - آخر.س، ص - آخر.ص، ع - آخر.ع)
    نهاية
    
    # عامل - (أحادي)
    عامل -() = متجه
        إرجاع متجه(-س، -ص، -ع)
    نهاية
    
    # عامل * (ضرب قياسي)
    عامل *(عشري معامل) = متجه
        إرجاع متجه(س * معامل، ص * معامل، ع * معامل)
    نهاية
    
    # عامل * (حاصل الضرب النقطي)
    عامل *(متجه آخر) = عشري
        إرجاع س * آخر.س + ص * آخر.ص + ع * آخر.ع
    نهاية
    
    # عامل +=
    عامل +=(متجه آخر) = متجه&
        س += آخر.س
        ص += آخر.ص
        ع += آخر.ع
        إرجاع هذا
    نهاية
    
    دالة طول() = عشري
        إرجاع جذر_تربيعي(س * س + ص * ص + ع * ع)
    نهاية
نهاية

# استخدام
متجه م1 = متجه(1.0، 2.0، 3.0)
متجه م2 = متجه(4.0، 5.0، 6.0)

متجه مجموع = م1 + م2              # (5، 7، 9)
متجه فرق = م1 - م2                # (-3، -3، -3)
متجه مضروب = م1 * 2.0             # (2، 4، 6)
عشري نقطي = م1 * م2               # 32.0
متجه معكوس = -م1                  # (-1، -2، -3)
```

### 4.2 Comparison Operators

```s
فئة تاريخ
    رقم سنة، شهر، يوم
    
    بناء(رقم س، رقم ش، رقم ي)
        سنة = س
        شهر = ش
        يوم = ي
    نهاية
    
    # عامل ==
    عامل ==(تاريخ آخر) = منطقي
        إرجاع سنة == آخر.سنة && شهر == آخر.شهر && يوم == آخر.يوم
    نهاية
    
    # عامل <
    عامل <(تاريخ آخر) = منطقي
        إذا سنة != آخر.سنة
            إرجاع سنة < آخر.سنة
        نهاية
        إذا شهر != آخر.شهر
            إرجاع شهر < آخر.شهر
        نهاية
        إرجاع يوم < آخر.يوم
    نهاية
    
    # بقية العوامل يمكن استنتاجها
    عامل >(تاريخ آخر) = منطقي
        إرجاع آخر < هذا
    نهاية
    
    عامل <=(تاريخ آخر) = منطقي
        إرجاع !(آخر < هذا)
    نهاية
نهاية
```

### 4.3 Stream Operators

```s
# عامل << للطباعة
فئة نقطة
    عشري س، ص
    
    # تحميل عامل <<
    عامل <<(مجرى خرج) = مجرى&
        خرج << ن"({س}، {ص})"
        إرجاع خرج
    نهاية
نهاية

# استخدام
نقطة ن = نقطة(3.5، 7.2)
اطبع << ن << سطر_جديد
# خرج: (3.5، 7.2)
```

## 5️⃣ Static Typing

### 5.1 Type Declarations

```s
# أنواع ثابتة صريحة
رقم عمر = 25
نص اسم = "أحمد"
منطقي نشط = صحيح

# خطأ: نوع غير متوافق
# عمر = "نص"  # Compile-time error!

# أنواع ديناميكية
أي متغير = 42
متغير = "الآن نص"  # OK
```

### 5.2 Type Inference

```s
# استنتاج النوع من القيمة
متغير عدد = 42              # يُستنتج: رقم
متغير رسالة = "مرحباً"       # يُستنتج: نص
متغير قائمة = [1، 2، 3]      # يُستنتج: مصفوفة<رقم>

# استنتاج من دالة
دالة احصل_عدد() = رقم
    إرجاع 42
نهاية

متغير ر = احصل_عدد()        # يُستنتج: رقم
```

### 5.3 Type Casting

```s
# Cast صريح
عشري ع = 3.14
رقم ر = رقم(ع)               # 3 (truncation)

# Cast آمن (مع فحص)
أي قيمة = "123"
إذا قيمة هو نص
    نص نص_قيمة = قيمة كـ نص
    رقم عدد = رقم(نص_قيمة)
نهاية

# Dynamic cast
فئة حيوان
    # ...
نهاية

فئة كلب : حيوان
    دالة انبح()
        اطبع("وف!")
    نهاية
نهاية

حيوان ح = كلب()
إذا ح هو كلب
    كلب ك = ح كـ كلب
    ك.انبح()
نهاية
```

## 6️⃣ References

### 6.1 Pass by Reference

```s
# تمرير بالقيمة (نسخ)
دالة ضعف_بالقيمة(رقم س)
    س *= 2
نهاية

رقم عدد = 10
ضعف_بالقيمة(عدد)
اطبع(عدد)  # 10 (لم يتغير)

# تمرير بالمرجع (بدون نسخ)
دالة ضعف_بالمرجع(مرجع رقم س)
    س *= 2
نهاية

ضعف_بالمرجع(مرجع عدد)
اطبع(عدد)  # 20 (تغير!)
```

### 6.2 Const References

```s
# مرجع const (للقراءة فقط، بدون نسخ)
دالة اطبع_كبير(مرجع ثابت مصفوفة<رقم> أرقام)
    # لا يمكن تعديل أرقام
    لكل ر في أرقام
        اطبع(ر)
    نهاية
    # أرقام.أضف(1)  # خطأ: const!
نهاية

# كفاءة: لا نسخ، وأمان: لا تعديل
مصفوفة<رقم> قائمة_كبيرة = [1، 2، 3، /* ... */]
اطبع_كبير(مرجع ثابت قائمة_كبيرة)
```

### 6.3 Return References

```s
فئة حاوية
    خاص مصفوفة<رقم> _بيانات
    
    # إرجاع مرجع للعنصر (للتعديل)
    دالة احصل(رقم فهرس) = مرجع رقم
        إرجاع _بيانات[فهرس]
    نهاية
    
    # إرجاع مرجع const (للقراءة فقط)
    دالة احصل(رقم فهرس) ثابت = مرجع ثابت رقم
        إرجاع _بيانات[فهرس]
    نهاية
نهاية

حاوية ح = حاوية()
ح.احصل(0) = 42  # تعديل مباشر
```

## 7️⃣ Smart Pointers

### 7.1 Unique Pointer

```s
# مؤشر فريد (ownership وحيد)
فئة<T> فريد
    خاص T* _مؤشر
    
    بناء(T* م)
        _مؤشر = م
    نهاية
    
    # لا يمكن نسخه
    بناء(فريد أخرى) = محذوف
    
    # يمكن نقله
    بناء(فريد&& أخرى)
        _مؤشر = أخرى._مؤشر
        أخرى._مؤشر = لاشيء
    نهاية
    
    إتلاف()
        إذا _مؤشر != لاشيء
            احذف _مؤشر
        نهاية
    نهاية
    
    دالة احصل() = T&
        إرجاع *_مؤشر
    نهاية
نهاية

# استخدام
{
    فريد<مصفوفة<رقم>> م = فريد(جديد مصفوفة<رقم>([1، 2، 3]))
    # م يمتلك البيانات
    
    # فريد<مصفوفة<رقم>> م2 = م  # خطأ: لا يمكن نسخه
    فريد<مصفوفة<رقم>> م2 = نقل(م)  # OK: نقل الملكية
}  # الذاكرة تُحرر تلقائياً
```

### 7.2 Shared Pointer

```s
# مؤشر مشترك (ownership متعدد)
فئة<T> مشترك
    خاص T* _مؤشر
    خاص رقم* _عداد_مراجع
    
    بناء(T* م)
        _مؤشر = م
        _عداد_مراجع = جديد رقم(1)
    نهاية
    
    # بناء نسخ (يزيد العداد)
    بناء(مشترك أخرى)
        _مؤشر = أخرى._مؤشر
        _عداد_مراجع = أخرى._عداد_مراجع
        *_عداد_مراجع += 1
    نهاية
    
    إتلاف()
        *_عداد_مراجع -= 1
        إذا *_عداد_مراجع == 0
            احذف _مؤشر
            احذف _عداد_مراجع
        نهاية
    نهاية
    
    دالة عدد_المراجع() = رقم
        إرجاع *_عداد_مراجع
    نهاية
نهاية

# استخدام
مشترك<نص> م1 = مشترك(جديد نص("بيانات مشتركة"))
اطبع(م1.عدد_المراجع())  # 1

{
    مشترك<نص> م2 = م1  # نسخ المرجع
    اطبع(م1.عدد_المراجع())  # 2
}  # م2 يُتلف، العداد يصبح 1

اطبع(م1.عدد_المراجع())  # 1
# م1 يُتلف، البيانات تُحذف
```

## 8️⃣ Const Correctness

### 8.1 Const Variables

```s
# متغير ثابت
ثابت رقم باي = 3.14159
# باي = 3  # خطأ: لا يمكن تعديل ثابت!

# مرجع ثابت
رقم قيمة = 100
ثابت رقم& مرجع_ثابت = قيمة
# مرجع_ثابت = 200  # خطأ!
قيمة = 200  # OK (المتغير الأصلي ليس const)
```

### 8.2 Const Methods

```s
فئة نقطة
    عشري س، ص
    
    # طريقة const (لا تعدل الحالة)
    دالة المسافة() ثابت = عشري
        إرجاع جذر_تربيعي(س * س + ص * ص)
    نهاية
    
    # طريقة عادية (تعدل الحالة)
    دالة حرك(عشري دلتا_س، عشري دلتا_ص)
        س += دلتا_س
        ص += دلتا_ص
    نهاية
نهاية

# كائن const
ثابت نقطة ن = نقطة(3.0، 4.0)
عشري م = ن.المسافة()  # OK (طريقة const)
# ن.حرك(1.0، 1.0)     # خطأ: لا يمكن استدعاء طريقة غير const
```

### 8.3 Const Parameters

```s
# معامل const (لا يمكن تعديله)
دالة اطبع_معلومات(ثابت مرجع نص اسم، ثابت رقم عمر)
    اطبع(ن"{اسم}، العمر: {عمر}")
    # اسم = "جديد"  # خطأ: const!
    # عمر += 1       # خطأ: const!
نهاية
```

| Template | `فئة<T>` أو `دالة<T>` | `فئة<T> صندوق` |
| RAII | `بناء()` + `إتلاف()` | Constructor acquires, destructor releases |
| Move | `نقل(كائن)` | `ب = نقل(أ)` |
| Reference | `مرجع نوع` | `دالة(مرجع رقم س)` |
| Const | `ثابت نوع` | `ثابت رقم باي = 3.14` |
| Unique Ptr | `فريد<T>` | `فريد<نص> م = ...` |
| Shared Ptr | `مشترك<T>` | `مشترك<نص> م = ...` |
| Operator | `عامل +` | `عامل +(نقطة أخرى)` |

## 🧪 اختبارات / Tests

```s
# tests/spec_tests/05_cpp_features_test.s

# اختبار 1: Template
فئة<T> صندوق
    T قيمة
    
    بناء(T ق)
        قيمة = ق
    نهاية
نهاية

صندوق<رقم> ص = صندوق<رقم>(42)
إذا ص.قيمة != 42
    ارمِ "فشل الاختبار: Template"
نهاية

# اختبار 2: RAII
رقم عداد_إنشاء = 0
رقم عداد_إتلاف = 0

فئة مورد
    بناء()
        عداد_إنشاء += 1
    نهاية
    
    إتلاف()
        عداد_إتلاف += 1
    نهاية
نهاية

{
    مورد م = مورد()
}  # الإتلاف يُستدعى هنا

إذا عداد_إنشاء != 1 أو عداد_إتلاف != 1
    ارمِ "فشل الاختبار: RAII"
نهاية

# اختبار 3: Reference
دالة ضعف(مرجع رقم س)
    س *= 2
نهاية

رقم عدد = 5
ضعف(مرجع عدد)
إذا عدد != 10
    ارمِ "فشل الاختبار: Reference"
نهاية

اطبع("✅ جميع اختبارات C++ Features نجحت")
```
