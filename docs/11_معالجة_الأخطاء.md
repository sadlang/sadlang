# 11. معالجة الأخطاء

## حاول / امسك / أخيراً

```sad
حاول
    # كود قد يُلقي خطأ
امسك (خ)
    # معالجة الخطأ
أخيراً
    # كود يُنفَّذ دائماً (اختياري)
نهاية
```

## إلقاء خطأ

```sad
ارمي "رسالة الخطأ"
```

## استثناءات مخصصة

```sad
صنف خطأ_مخصص
    باني(الرسالة)
        هذا.رسالة = الرسالة
    نهاية
نهاية

ارمي جديد خطأ_مخصص("حدث خطأ!")
```

## التقاط حسب النوع

```sad
حاول
    ارمي جديد خطأ_شبكة("انقطع الاتصال")
امسك (خطأ_شبكة خ)
    # معالجة أخطاء الشبكة
امسك (خطأ_ملف خ)
    # معالجة أخطاء الملفات
امسك (خ)
    # معالجة أي خطأ آخر
نهاية
```

## 🔸 الأمثلة (8 أمثلة متدرجة)

### مثال 1 — مبتدئ: حاول/امسك بسيط
```sad
حاول
    متغير نتيجة = 10 / 0
    اطبع_سطر("النتيجة: " + نتيجة)
امسك (خ)
    اطبع_سطر("حدث خطأ: " + خ)
نهاية

اطبع_سطر("البرنامج استمر بعد الخطأ")
```
**المخرجات:**
```
حدث خطأ: القسمة على صفر
البرنامج استمر بعد الخطأ
```

### مثال 2 — مبتدئ: إلقاء خطأ يدوياً
```sad
دالة قسمة(أ، ب)
    إذا (ب == 0)
        ارمي "لا يمكن القسمة على صفر!"
    نهاية
    ارجع أ / ب
نهاية

حاول
    اطبع_سطر("10 / 2 = " + قسمة(10، 2))
    اطبع_سطر("10 / 0 = " + قسمة(10، 0))
امسك (خ)
    اطبع_سطر("خطأ: " + خ)
نهاية
```
**المخرجات:**
```
10 / 2 = 5
خطأ: لا يمكن القسمة على صفر!
```

### مثال 3 — مبتدئ: حاول/امسك/أخيراً
```sad
اطبع_سطر("بداية البرنامج")

حاول
    ارمي "خطأ تجريبي"
امسك (خ)
    اطبع_سطر("تم التقاط: " + خ)
أخيراً
    اطبع_سطر("كتلة أخيراً: تُنفذ دائماً")
نهاية

اطبع_سطر("نهاية البرنامج")
```
**المخرجات:**
```
بداية البرنامج
تم التقاط: خطأ تجريبي
كتلة أخيراً: تُنفذ دائماً
نهاية البرنامج
```

### مثال 4 — متوسط: حماية الثوابت
```sad
ثابت باي = 3.14

حاول
    باي = 3.15
    اطبع_سطر("فشل - تم تعديل الثابت!")
امسك (خ)
    اطبع_سطر("نجح: الثوابت محمية - " + خ)
نهاية

# حماية += أيضاً
ثابت ثا = 50
حاول
    ثا += 10
    اطبع_سطر("فشل - ثابت تم تعديله بـ +=!")
امسك (خ)
    اطبع_سطر("نجح: الثابت محمي من += أيضاً")
نهاية
```
**المخرجات:**
```
نجح: الثوابت محمية - لا يمكن تعديل الثابت 'باي'
نجح: الثابت محمي من += أيضاً
```

### مثال 7 — متقدم: معالجة أخطاء متعددة
```sad
صنف خطأ_إدخال
    باني(الرسالة)
        هذا.رسالة = الرسالة
    نهاية
نهاية

صنف خطأ_حساب
    باني(الرسالة)
        هذا.رسالة = الرسالة
    نهاية
نهاية

دالة تحقق_عمر(عمر)
    إذا (عمر < 0)
        ارمي جديد خطأ_إدخال("العمر لا يمكن أن يكون سالباً")
    نهاية
    إذا (عمر > 150)
        ارمي جديد خطأ_إدخال("العمر غير واقعي")
    نهاية
    ارجع عمر
نهاية

دالة تحقق_قسمة(أ، ب)
    إذا (ب == 0)
        ارمي جديد خطأ_حساب("القسمة على صفر")
    نهاية
    ارجع أ / ب
نهاية

# اختبار عدة سيناريوهات
متغير اختبارات = [[-5، "عمر"]، [200، "عمر"]، [25، "عمر"]، [10، "قسمة_صفر"]، [10، "قسمة_عادية"]]

لكل اختبار في اختبارات
    حاول
        إذا (اختبار[1] == "عمر")
            متغير ع = تحقق_عمر(اختبار[0])
            اطبع_سطر("عمر صالح: " + ع)
        والا_اذا (اختبار[1] == "قسمة_صفر")
            تحقق_قسمة(اختبار[0]، 0)
        وإلا
            اطبع_سطر("قسمة: " + تحقق_قسمة(اختبار[0]، 2))
        نهاية
    امسك (خطأ_إدخال خ)
        اطبع_سطر("خطأ إدخال: " + خ)
    امسك (خطأ_حساب خ)
        اطبع_سطر("خطأ حساب: " + خ)
    نهاية
نهاية
```
**المخرجات:**
```
خطأ إدخال: العمر لا يمكن أن يكون سالباً
خطأ إدخال: العمر غير واقعي
عمر صالح: 25
خطأ حساب: القسمة على صفر
قسمة: 5
```

### مثال 8 — متقدم جداً: نظام تحقق شامل
```sad
صنف خطأ_تحقق

        هذا.حقل = الحقل
        هذا.رسالة = الرسالة
    نهاية
نهاية

صنف خطأ_عمل
    باني(العملية، الرسالة)
        هذا.عملية = العملية
        هذا.رسالة = الرسالة
    نهاية
نهاية

# نظام إنشاء حساب بنكي مع تحقق شامل
دالة تحقق_اسم(اسم)
    إذا (طول(اسم) < 2)
        ارمي جديد خطأ_تحقق("الاسم"، "الاسم قصير جداً (أقل من حرفين)")
    نهاية
    ارجع صحيح
نهاية

دالة تحقق_رصيد(رصيد)
    إذا (رصيد < 0)
        ارمي جديد خطأ_تحقق("الرصيد"، "الرصيد لا يمكن أن يكون سالباً")
    نهاية
    إذا (رصيد < 100)
        ارمي جديد خطأ_عمل("فتح_حساب"، "الحد الأدنى للفتح 100")
    نهاية
    ارجع صحيح
نهاية

دالة أنشئ_حساب(اسم، رصيد)
    متغير أخطاء = []
    
    حاول
        تحقق_اسم(اسم)
    امسك (خطأ_تحقق خ)
        أضف(أخطاء، "تحقق - " + خ)
    نهاية
    
    حاول
        تحقق_رصيد(رصيد)
    امسك (خطأ_تحقق خ)
        أضف(أخطاء، "تحقق - " + خ)
    امسك (خطأ_عمل خ)
        أضف(أخطاء، "عمل - " + خ)
    نهاية
    
    إذا (طول(أخطاء) > 0)
        اطبع_سطر("  ✗ فشل إنشاء حساب لـ '" + اسم + "':")
        لكل خ في أخطاء
            اطبع_سطر("    - " + خ)
        نهاية
    وإلا
        اطبع_سطر("  ✓ تم إنشاء حساب لـ " + اسم + " برصيد " + رصيد)
    نهاية
نهاية

دالة رئيسية()
    اطبع_سطر("═══════════════════════════════════")
    اطبع_سطر("   نظام فتح الحسابات البنكية")
    اطبع_سطر("═══════════════════════════════════")
    
    أنشئ_حساب("أحمد"، 5000)
    أنشئ_حساب("م"، 50)
    أنشئ_حساب("فاطمة"، -100)
    أنشئ_حساب("خالد"، 1000)
    أنشئ_حساب("ن"، -50)
    
    اطبع_سطر("═══════════════════════════════════")
نهاية
```
