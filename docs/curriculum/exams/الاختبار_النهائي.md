## القسم الرابع: اكتب الكود

### السؤال 26
```sad
صنف مستطيل
    متغير عرض
    متغير ارتفاع
    
    باني(العرض، الارتفاع)
        هذا.عرض = العرض
        هذا.ارتفاع = الارتفاع
    نهاية
    
    دالة مساحة()
        ارجع هذا.عرض * هذا.ارتفاع
    نهاية
    
    دالة محيط()
        ارجع 2 * (هذا.عرض + هذا.ارتفاع)
    نهاية
نهاية

# اختبار
متغير م = جديد مستطيل(10، 5)
اطبع_سطر("المساحة: " + م.مساحة())   # 50
اطبع_سطر("المحيط: " + م.محيط())     # 30
```

### السؤال 27
```sad
صنف حساب_بنكي
    خاص متغير رصيد = 0
    
    دالة إيداع(مبلغ)
        إذا (مبلغ <= 0)
            ارمي "المبلغ يجب أن يكون موجباً"
        نهاية
        هذا.رصيد = هذا.رصيد + مبلغ
    نهاية
    
    دالة سحب(مبلغ)
        إذا (مبلغ <= 0)
            ارمي "المبلغ يجب أن يكون موجباً"
        نهاية
        إذا (مبلغ > هذا.رصيد)
            ارمي "رصيد غير كافٍ"
        نهاية
        هذا.رصيد = هذا.رصيد - مبلغ
    نهاية
    
    دالة الرصيد()
        ارجع هذا.رصيد
    نهاية
نهاية

# اختبار
متغير ح = جديد حساب_بنكي()
ح.إيداع(1000)
ح.سحب(300)
اطبع_سطر("الرصيد: " + ح.الرصيد())  # 700
```
