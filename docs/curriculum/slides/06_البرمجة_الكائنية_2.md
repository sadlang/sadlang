# 🏷️ السمات (Traits/Interfaces)

> السمة تعريف لسلوكيات بدون تفاصيل التنفيذ

```sad
سمة قابل_للطباعة
    دالة اطبع()
نهاية

سمة قابل_للمقارنة
    دالة قارن(آخر)
نهاية
```

### لماذا السمات؟
- 📋 تحديد عقد (Contract)
- 🔄 تعدد الأشكال (Polymorphism)
- 🧩 فصل الواجهة عن التنفيذ

# ✅ تنفيذ السمات: `نفّذ`

```sad
سمة قابل_للطباعة
    دالة اطبع()
نهاية

صنف منتج نفّذ قابل_للطباعة
    متغير اسم
    متغير سعر
    
    باني(الاسم، السعر)
        هذا.اسم = الاسم
        هذا.سعر = السعر
    نهاية
    
    # تنفيذ الدالة المطلوبة
    دالة اطبع()
        اطبع_سطر(هذا.اسم + " - " + هذا.سعر + " ر.س")
    نهاية
نهاية

متغير م = جديد منتج("لابتوب"، 3500)
م.اطبع()  # لابتوب - 3500 ر.س
```

# 🗺️ الخرائط (Maps)

> مجموعة من أزواج (مفتاح، قيمة)

```sad
# إنشاء خريطة
متغير طالب = خريطة()

# إضافة عناصر
طالب["اسم"] = "محمد"
طالب["عمر"] = 20
طالب["تخصص"] = "حاسوب"

# الوصول للقيم
اطبع_سطر(طالب["اسم"])  # محمد
```

---

# 🔄 عمليات الخريطة

```sad
متغير قاموس = خريطة()

# إضافة
قاموس["hello"] = "مرحباً"
قاموس["world"] = "عالم"

# التحقق من وجود مفتاح
إذا ("hello" في قاموس)
    اطبع_سطر("موجود!")
نهاية

# المرور على العناصر
لكل مفتاح في قاموس
    اطبع_سطر(مفتاح + " = " + قاموس[مفتاح])
نهاية

# الحذف
احذف(قاموس، "hello")
```

# 🎨 نمط المفرد (Singleton)

> يضمن وجود نسخة واحدة فقط من الصنف

```sad
صنف قاعدة_البيانات
    ساكن متغير المثيل = لاشيء
    
    خاص باني()
        # إعداد الاتصال
    نهاية
    
    ساكن دالة مثيل()
        إذا (قاعدة_البيانات.المثيل == لاشيء)
            قاعدة_البيانات.المثيل = جديد قاعدة_البيانات()
        نهاية
        ارجع قاعدة_البيانات.المثيل
    نهاية
نهاية

# الاستخدام
متغير قب1 = قاعدة_البيانات.مثيل()
متغير قب2 = قاعدة_البيانات.مثيل()
# قب1 و قب2 نفس الكائن!
```

# 🏭 نمط المصنع (Factory)

> يُنشئ كائنات بدون تحديد الصنف بشكل صريح

```sad
صنف مصنع_الأشكال
    ساكن دالة انشئ(نوع، بُعد)
        طابق (نوع)
            عندما "مربع":
                ارجع جديد مربع(بُعد)
            عندما "دائرة":
                ارجع جديد دائرة(بُعد)
            عندما "مثلث":
                ارجع جديد مثلث_متساوي(بُعد)
            افتراضي:
                ارمي "نوع غير معروف"
        نهاية
    نهاية
نهاية

# الاستخدام
متغير شكل1 = مصنع_الأشكال.انشئ("دائرة"، 5)
متغير شكل2 = مصنع_الأشكال.انشئ("مربع"، 10)
```

# 👀 نمط المراقب (Observer)

> يُبلّغ كائنات متعددة عند حدوث تغيير

```sad
سمة مراقب
    دالة حدّث(حدث)
نهاية

صنف قناة_أخبار
    متغير المراقبون = []
    
    دالة اشترك(مراقب)
        أضف(هذا.المراقبون، مراقب)
    نهاية
    
    دالة أبلغ(خبر)
        لكل م في هذا.المراقبون
            م.حدّث(خبر)
        نهاية
    نهاية
نهاية

صنف مشترك نفّذ مراقب
    متغير اسم
    
    دالة حدّث(خبر)
        اطبع_سطر(هذا.اسم + " استلم: " + خبر)
    نهاية
نهاية
```

# 🛒 مثال عملي: عربة تسوق

```sad
صنف منتج
    متغير اسم
    متغير سعر
    
    باني(ا، س)
        هذا.اسم = ا
        هذا.سعر = س
    نهاية
نهاية

صنف عربة_تسوق
    متغير منتجات = []
    
    دالة أضف(منتج)
        أضف(هذا.منتجات، منتج)
    نهاية
    
    دالة المجموع()
        متغير مجموع = 0
        لكل م في هذا.منتجات
            مجموع = مجموع + م.سعر
        نهاية
        ارجع مجموع
    نهاية
    
    دالة عرض()
        لكل م في هذا.منتجات
            اطبع_سطر("- " + م.اسم + ": " + م.سعر + " ر.س")
        نهاية
        اطبع_سطر("المجموع: " + هذا.المجموع() + " ر.س")
    نهاية
نهاية
```
