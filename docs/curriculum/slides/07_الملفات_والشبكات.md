# 📖 قراءة ملف

```sad
استورد "io"

# قراءة الملف كاملاً
متغير محتوى = قراءة_ملف("بيانات.txt")
اطبع_سطر(محتوى)

# قراءة سطراً بسطر
متغير ملف = افتح("بيانات.txt"، "قراءة")

بينما (ليس نهاية_الملف(ملف))
    متغير سطر = اقرأ_سطر(ملف)
    اطبع_سطر(سطر)
نهاية

اغلق(ملف)
```

# 📋 JSON — القراءة

```sad
استورد "json"

# قراءة JSON من ملف
متغير بيانات = قراءة_json("config.json")

اطبع_سطر("الاسم: " + بيانات["اسم"])
```

# 🌐 طلبات HTTP — POST

```sad
استورد "http"
استورد "json"

# إنشاء البيانات
متغير بيانات = خريطة()
بيانات["اسم"] = "سارة"
بيانات["بريد"] = "sara@example.com"

# إرسال POST
متغير استجابة = http_post(
    "https://api.example.com/users"،
    إلى_json(بيانات)،
    خريطة("Content-Type"، "application/json")
)

إذا (استجابة.كود == 201)
    اطبع_سطر("تم الإنشاء بنجاح! 🎉")
    متغير نتيجة = من_json(استجابة.محتوى)
    اطبع_سطر("ID: " + نتيجة["id"])
وإلا
    اطبع_سطر("فشل: " + استجابة.كود)
نهاية
```

# 🌤️ مثال: تطبيق طقس

```sad
استورد "http"
استورد "json"

دالة احصل_الطقس(مدينة)
    متغير رابط = "https://api.weather.com/v1/current?city=" + مدينة
    
    حاول
        متغير استجابة = http_get(رابط)
        
        إذا (استجابة.كود == 200)
            متغير بيانات = من_json(استجابة.محتوى)
            
            اطبع_سطر("🌡️ الطقس في " + مدينة)
            اطبع_سطر("الحرارة: " + بيانات["درجة"] + "°")
            اطبع_سطر("الحالة: " + بيانات["حالة"])
            اطبع_سطر("الرطوبة: " + بيانات["رطوبة"] + "%")
        وإلا
            اطبع_سطر("❌ فشل في جلب البيانات")
        نهاية
    امسك (خطأ)
        اطبع_سطر("❌ خطأ في الاتصال: " + خطأ)
    نهاية
نهاية

احصل_الطقس("الرياض")
```

# 📊 مثال: دليل هاتف مع حفظ

```sad
صنف دليل_هاتف
    متغير جهات_الاتصال = خريطة()
    متغير ملف_الحفظ = "جهات_اتصال.json"
    
    باني()
        هذا.حمّل()
    نهاية
    
    دالة أضف(اسم، رقم)
        هذا.جهات_الاتصال[اسم] = رقم
        هذا.احفظ()
    نهاية
    
    دالة ابحث(اسم)
        إذا (اسم في هذا.جهات_الاتصال)
            ارجع هذا.جهات_الاتصال[اسم]
        نهاية
        ارجع لاشيء
    نهاية
    
    دالة احفظ()
        كتابة_json(هذا.ملف_الحفظ، هذا.جهات_الاتصال)
    نهاية
    
    دالة حمّل()
        إذا (ملف_موجود(هذا.ملف_الحفظ))
            هذا.جهات_الاتصال = قراءة_json(هذا.ملف_الحفظ)
        نهاية
    نهاية
نهاية
```
