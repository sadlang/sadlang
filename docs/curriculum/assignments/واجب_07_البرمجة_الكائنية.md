# 📝 الواجب 7: البرمجة الكائنية

## المستوى: متقدم
## الأسبوع: 9-10

---

## 🎯 الأهداف

- تصميم الأصناف
- استخدام الوراثة
- تطبيق السمات

---

## التمارين

### تمرين 1: صنف الطالب (25 نقاط)

أنشئ صنف `طالب` مع:
- الخصائص: الاسم، الرقم الجامعي، الدرجات (مصفوفة)
- الدوال:
  - `أضف_درجة(درجة)`
  - `احسب_المعدل()`
  - `حالة_النجاح()` — ناجح إذا المعدل >= 60
  - `اطبع_معلومات()`

---

### تمرين 2: نظام الأشكال (30 نقاط)

أنشئ نظام أشكال هندسية:
- صنف أساسي `شكل` مع `اسم` و `مساحة()` و `محيط()`
- صنف `مستطيل` يرث من `شكل`
- صنف `مربع` يرث من `مستطيل`
- صنف `دائرة` يرث من `شكل`
- صنف `مثلث` يرث من `شكل`

---

### تمرين 3: حساب بنكي (25 نقاط)

أنشئ نظام حسابات بنكية:
- صنف `حساب` مع: إيداع، سحب، رصيد
- صنف `حساب_توفير` (عائد شهري 2%)
- صنف `حساب_جاري` (رسوم على كل عملية)

---

### تمرين 4: نظام إدارة مهام (20 نقاط)

أنشئ نظام لإدارة المهام:
- صنف `مهمة` مع: عنوان، وصف، حالة
- صنف `قائمة_مهام` لإدارة مجموعة مهام

---

## 📊 نظام التقييم

| تمرين 1 | 25 |
| تمرين 2 | 30 |
| تمرين 3 | 25 |

```sad
# صنف الطالب

صنف طالب
    متغير الاسم
    متغير الرقم_الجامعي
    متغير الدرجات = []
    
    باني(اسم، رقم)
        هذا.الاسم = اسم
        هذا.الرقم_الجامعي = رقم
    نهاية
    
    دالة أضف_درجة(درجة)
        إذا (درجة < 0 أو درجة > 100)
            ارمي "درجة غير صالحة"
        نهاية
        أضف(هذا.الدرجات، درجة)
    نهاية
    
    دالة احسب_المعدل()
        إذا (طول(هذا.الدرجات) == 0)
            ارجع 0
        نهاية
        
        متغير مجموع = 0
        لكل درجة في هذا.الدرجات
            مجموع = مجموع + درجة
        نهاية
        
        ارجع مجموع / طول(هذا.الدرجات)
    نهاية
    
    دالة حالة_النجاح()
        إذا (هذا.احسب_المعدل() >= 60)
            ارجع "ناجح ✅"
        نهاية
        ارجع "راسب ❌"
    نهاية
    
    دالة التقدير()
        متغير معدل = هذا.احسب_المعدل()
        إذا (معدل >= 90)
            ارجع "ممتاز"
        وإلا إذا (معدل >= 80)
            ارجع "جيد جداً"
        وإلا إذا (معدل >= 70)
            ارجع "جيد"
        وإلا إذا (معدل >= 60)
            ارجع "مقبول"
        نهاية
        ارجع "راسب"
    نهاية
    
    دالة اطبع_معلومات()
        اطبع_سطر("═══════════════════════")
        اطبع_سطر("   بطاقة طالب")
        اطبع_سطر("═══════════════════════")
        اطبع_سطر("الاسم: " + هذا.الاسم)
        اطبع_سطر("الرقم: " + هذا.الرقم_الجامعي)
        اطبع_سطر("عدد المواد: " + طول(هذا.الدرجات))
        اطبع_سطر("المعدل: " + هذا.احسب_المعدل())
        اطبع_سطر("التقدير: " + هذا.التقدير())
        اطبع_سطر("الحالة: " + هذا.حالة_النجاح())
        اطبع_سطر("═══════════════════════")
    نهاية
نهاية

# اختبار

ط1.أضف_درجة(85)
ط1.أضف_درجة(90)
ط1.أضف_درجة(78)
ط1.أضف_درجة(92)
ط1.اطبع_معلومات()
```

```sad
# نظام الحسابات البنكية

صنف حساب
    محمي متغير رصيد = 0
    متغير رقم_الحساب
    متغير صاحب_الحساب
    متغير تاريخ_العمليات = []
    
    باني(رقم، صاحب)
        هذا.رقم_الحساب = رقم
        هذا.صاحب_الحساب = صاحب
    نهاية
    
    دالة إيداع(مبلغ)
        إذا (مبلغ <= 0)
            ارمي "المبلغ يجب أن يكون موجباً"
        نهاية
        هذا.رصيد = هذا.رصيد + مبلغ
        أضف(هذا.تاريخ_العمليات، "إيداع: " + مبلغ)
        اطبع_سطر("✅ تم إيداع " + مبلغ + " ر.س")
    نهاية
    
    دالة سحب(مبلغ)
        إذا (مبلغ <= 0)
            ارمي "المبلغ يجب أن يكون موجباً"
        نهاية
        إذا (مبلغ > هذا.رصيد)
            ارمي "رصيد غير كافٍ"
        نهاية
        هذا.رصيد = هذا.رصيد - مبلغ
        أضف(هذا.تاريخ_العمليات، "سحب: " + مبلغ)
        اطبع_سطر("✅ تم سحب " + مبلغ + " ر.س")
    نهاية
    
    دالة الرصيد()
        ارجع هذا.رصيد
    نهاية
    
    دالة اطبع_كشف()
        اطبع_سطر("═══════════════════════")
        اطبع_سطر("   كشف حساب")
        اطبع_سطر("═══════════════════════")
        اطبع_سطر("رقم الحساب: " + هذا.رقم_الحساب)
        اطبع_سطر("صاحب الحساب: " + هذا.صاحب_الحساب)
        اطبع_سطر("───────────────────────")
        لكل عملية في هذا.تاريخ_العمليات
            اطبع_سطر("  " + عملية)
        نهاية
        اطبع_سطر("───────────────────────")
        اطبع_سطر("الرصيد: " + هذا.رصيد + " ر.س")
        اطبع_سطر("═══════════════════════")
    نهاية
نهاية

صنف حساب_توفير يرث حساب
    متغير نسبة_العائد = 0.02  # 2% شهرياً
    
    باني(رقم، صاحب)
        الأساس(رقم، صاحب)
    نهاية
    
    دالة احسب_العائد()
        ارجع هذا.رصيد * هذا.نسبة_العائد
    نهاية
    
    دالة أضف_عائد_شهري()
        متغير عائد = هذا.احسب_العائد()
        هذا.رصيد = هذا.رصيد + عائد
        أضف(هذا.تاريخ_العمليات، "عائد شهري: " + عائد)
        اطبع_سطر("✅ تم إضافة عائد " + عائد + " ر.س")
    نهاية
نهاية

صنف حساب_جاري يرث حساب
    متغير رسوم_العملية = 1  # ريال لكل عملية
    
    باني(رقم، صاحب)
        الأساس(رقم، صاحب)
    نهاية
    
    دالة سحب(مبلغ)
        متغير المبلغ_الكلي = مبلغ + هذا.رسوم_العملية
        إذا (المبلغ_الكلي > هذا.رصيد)
            ارمي "رصيد غير كافٍ (مع الرسوم)"
        نهاية
        هذا.رصيد = هذا.رصيد - المبلغ_الكلي
        أضف(هذا.تاريخ_العمليات، "سحب: " + مبلغ + " + رسوم: " + هذا.رسوم_العملية)
        اطبع_سطر("✅ تم سحب " + مبلغ + " ر.س (+ " + هذا.رسوم_العملية + " رسوم)")
    نهاية
نهاية

# اختبار
متغير حساب1 = جديد حساب_توفير("SA001"، "أحمد")
حساب1.إيداع(10000)
حساب1.أضف_عائد_شهري()
حساب1.اطبع_كشف()
```

```sad
# نظام إدارة المهام

صنف مهمة
    متغير عنوان
    متغير وصف
    متغير حالة = "جديدة"  # جديدة، قيد التنفيذ، مكتملة
    متغير أولوية = "عادية"  # عاجلة، عالية، عادية، منخفضة
    
    باني(العنوان، الوصف)
        هذا.عنوان = العنوان
        هذا.وصف = الوصف
    نهاية
    
    دالة ابدأ()
        هذا.حالة = "قيد التنفيذ"
    نهاية
    
    دالة أكمل()
        هذا.حالة = "مكتملة"
    نهاية
    
    دالة رمز_الحالة()
        طابق (هذا.حالة)
            عندما "جديدة":
                ارجع "⬜"
            عندما "قيد التنفيذ":
                ارجع "🔄"
            عندما "مكتملة":
                ارجع "✅"
            افتراضي:
                ارجع "❓"
        نهاية
    نهاية
    
    دالة اطبع()
        اطبع_سطر(هذا.رمز_الحالة() + " " + هذا.عنوان)
        اطبع_سطر("   " + هذا.وصف)
    نهاية
نهاية

صنف قائمة_مهام
    متغير المهام = []
    متغير اسم_القائمة
    
    باني(اسم)
        هذا.اسم_القائمة = اسم
    نهاية
    
    دالة أضف(مهمة)
        أضف(هذا.المهام، مهمة)
    نهاية
    
    دالة المهام_الجديدة()
        متغير نتيجة = []
        لكل م في هذا.المهام
            إذا (م.حالة == "جديدة")
                أضف(نتيجة، م)
            نهاية
        نهاية
        ارجع نتيجة
    نهاية
    
    دالة المهام_المكتملة()
        متغير نتيجة = []
        لكل م في هذا.المهام
            إذا (م.حالة == "مكتملة")
                أضف(نتيجة، م)
            نهاية
        نهاية
        ارجع نتيجة
    نهاية
    
    دالة نسبة_الإنجاز()
        إذا (طول(هذا.المهام) == 0)
            ارجع 0
        نهاية
        ارجع طول(هذا.المهام_المكتملة()) * 100 / طول(هذا.المهام)
    نهاية
    
    دالة اطبع()
        اطبع_سطر("═══════════════════════")
        اطبع_سطر("📋 " + هذا.اسم_القائمة)
        اطبع_سطر("═══════════════════════")
        لكل م في هذا.المهام
            م.اطبع()
        نهاية
        اطبع_سطر("───────────────────────")
        اطبع_سطر("الإنجاز: " + هذا.نسبة_الإنجاز() + "%")
        اطبع_سطر("═══════════════════════")
    نهاية
نهاية

# اختبار
متغير قائمتي = جديد قائمة_مهام("مهام اليوم")

متغير م1 = جديد مهمة("إنهاء التقرير"، "تقرير المشروع الفصلي")
متغير م2 = جديد مهمة("اجتماع الفريق"، "الساعة 3 مساءً")
متغير م3 = جديد مهمة("مراجعة الكود"، "مراجعة PR #123")

قائمتي.أضف(م1)
قائمتي.أضف(م2)
قائمتي.أضف(م3)

م1.أكمل()
م2.ابدأ()

قائمتي.اطبع()
```
