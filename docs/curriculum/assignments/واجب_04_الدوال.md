# 📝 الواجب 4: الدوال

## المستوى: متوسط
## الأسبوع: 5

---

## 🎯 الأهداف

- تعريف واستدعاء الدوال
- استخدام المعاملات والقيم المرجعة
- الدوال التكرارية

---

## التمارين

### تمرين 1: دوال رياضية (15 نقاط)

أنشئ الدوال التالية:
- `مربع(س)` — يُرجع مربع الرقم
- `تكعيب(س)` — يُرجع مكعب الرقم
- `قوة(أساس، أس)` — يُرجع الأساس مرفوعاً للأس
- `مطلق(س)` — يُرجع القيمة المطلقة

---

### تمرين 2: دالة المضروب (15 نقاط)

اكتب دالة `مضروب(ن)` تحسب مضروب العدد.
- `5! = 5 × 4 × 3 × 2 × 1 = 120`

---

### تمرين 3: فيبوناتشي (20 نقاط)

اكتب دالة `فيبوناتشي(ن)` تُرجع الرقم رقم ن في متتالية فيبوناتشي.
- 0, 1, 1, 2, 3, 5, 8, 13, 21, ...

---

### تمرين 4: التحقق من عدد أولي (15 نقاط)

اكتب دالة `أولي(ن)` تُرجع `صحيح` إذا كان العدد أولياً.

---

### تمرين 5: أكبر قاسم مشترك (15 نقاط)

اكتب دالة `قمم(أ، ب)` تحسب أكبر قاسم مشترك لعددين.

# ✅ الحلول

## حل تمرين 1

```sad
# دوال رياضية

دالة مربع(س)
    ارجع س * س
نهاية

دالة تكعيب(س)
    ارجع س * س * س
نهاية

دالة قوة(أساس، أس)
    متغير نتيجة = 1
    متغير ع = 0
    بينما (ع < أس)
        نتيجة = نتيجة * أساس
        ع = ع + 1
    نهاية
    ارجع نتيجة
نهاية

دالة مطلق(س)
    إذا (س < 0)
        ارجع -س
    نهاية
    ارجع س
نهاية

# اختبار
اطبع_سطر("مربع(5) = " + مربع(5))       # 25
اطبع_سطر("تكعيب(3) = " + تكعيب(3))     # 27
اطبع_سطر("قوة(2، 8) = " + قوة(2، 8))   # 256
اطبع_سطر("مطلق(-7) = " + مطلق(-7))     # 7
```

## حل تمرين 2

```sad
# دالة المضروب

# طريقة تكرارية
دالة مضروب(ن)
    إذا (ن < 0)
        ارجع -1  # خطأ: لا يوجد مضروب للسالب
    نهاية
    
    متغير نتيجة = 1
    متغير ع = 1
    
    بينما (ع <= ن)
        نتيجة = نتيجة * ع
        ع = ع + 1
    نهاية
    
    ارجع نتيجة
نهاية

# طريقة تكررية (recursive)
دالة مضروب_تكرري(ن)
    إذا (ن <= 1)
        ارجع 1
    نهاية
    ارجع ن * مضروب_تكرري(ن - 1)
نهاية

# اختبار
اطبع_سطر("5! = " + مضروب(5))          # 120
اطبع_سطر("10! = " + مضروب(10))        # 3628800
اطبع_سطر("0! = " + مضروب(0))          # 1
```

## حل تمرين 3

```sad
# فيبوناتشي

# طريقة تكررية
دالة فيبوناتشي(ن)
    إذا (ن <= 0)
        ارجع 0
    نهاية
    إذا (ن == 1)
        ارجع 1
    نهاية
    ارجع فيبوناتشي(ن - 1) + فيبوناتشي(ن - 2)
نهاية

# طريقة تكرارية (أسرع)
دالة فيبوناتشي_سريع(ن)
    إذا (ن <= 0)
        ارجع 0
    نهاية
    إذا (ن == 1)
        ارجع 1
    نهاية
    
    متغير أ = 0
    متغير ب = 1
    متغير ع = 2
    
    بينما (ع <= ن)
        متغير ج = أ + ب
        أ = ب
        ب = ج
        ع = ع + 1
    نهاية
    
    ارجع ب
نهاية

# طباعة أول 10 أرقام
اطبع_سطر("متتالية فيبوناتشي:")
متغير ع = 0
بينما (ع < 10)
    اطبع(فيبوناتشي_سريع(ع) + " ")
    ع = ع + 1
نهاية
اطبع_سطر("")
```

## حل تمرين 4

```sad
# التحقق من عدد أولي

دالة أولي(ن)
    إذا (ن <= 1)
        ارجع خطأ
    نهاية
    
    إذا (ن == 2)
        ارجع صحيح
    نهاية
    
    إذا (ن % 2 == 0)
        ارجع خطأ
    نهاية
    
    متغير قاسم = 3
    بينما (قاسم * قاسم <= ن)
        إذا (ن % قاسم == 0)
            ارجع خطأ
        نهاية
        قاسم = قاسم + 2
    نهاية
    
    ارجع صحيح
نهاية

# اختبار
اطبع_سطر("أولي(7) = " + أولي(7))   # صحيح
اطبع_سطر("أولي(10) = " + أولي(10)) # خطأ
اطبع_سطر("أولي(17) = " + أولي(17)) # صحيح
```

## حل تمرين 5

```sad
# أكبر قاسم مشترك (خوارزمية إقليدس)

دالة قمم(أ، ب)
    بينما (ب != 0)
        متغير باقي = أ % ب
        أ = ب
        ب = باقي
    نهاية
    ارجع أ
نهاية

# طريقة تكررية
دالة قمم_تكرري(أ، ب)
    إذا (ب == 0)
        ارجع أ
    نهاية
    ارجع قمم_تكرري(ب، أ % ب)
نهاية

# اختبار
اطبع_سطر("قمم(48، 18) = " + قمم(48، 18))  # 6
اطبع_سطر("قمم(56، 98) = " + قمم(56، 98))  # 14
```
