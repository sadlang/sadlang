# 📝 الواجب 6: مطابقة الأنماط ومعالجة الأخطاء

## المستوى: متوسط
## الأسبوع: 7-8

---

## 🎯 الأهداف

- استخدام `طابق`/`عندما`
- معالجة الأخطاء بـ `حاول`/`امسك`
- التحقق من صحة المدخلات

---

## التمارين

### تمرين 1: محول الأشهر (15 نقاط)

اكتب دالة `اسم_الشهر(رقم)` تُرجع اسم الشهر بالعربية.

---

### تمرين 2: حاسبة متقدمة (20 نقاط)

اكتب دالة `احسب(أ، عملية، ب)` تدعم:
- `+`, `-`, `*`, `/`, `%`, `**`
- تُرمي استثناء للعمليات غير المعروفة
- تُرمي استثناء للقسمة على صفر

---

### تمرين 3: محلل النصوص (25 نقاط)

اكتب دالة `صنّف_حرف(ح)` تُرجع نوع الحرف:
- حرف عربي
- حرف إنجليزي
- رقم
- مسافة
- رمز خاص

---

### تمرين 4: التحقق من المدخلات (20 نقاط)

اكتب دوال التحقق:
- `تحقق_عمر(عمر)` — 0-150
- `تحقق_بريد(بريد)` — يحتوي @
- `تحقق_هاتف(رقم)` — يبدأ بـ 0 وطوله 10 أرقام

---

### تمرين 5: تقليص الكسور (20 نقاط)

اكتب صنف `كسر` مع دالة `بسّط()` لتبسيط الكسر.

---

# محلل النصوص

دالة صنّف_حرف(ح)
    # الأرقام
    إذا (ح >= "0" و ح <= "9")
        ارجع "رقم"
    نهاية
    
    # الحروف الإنجليزية
    إذا ((ح >= "a" و ح <= "z") أو (ح >= "A" و ح <= "Z"))
        ارجع "حرف إنجليزي"
    نهاية
    
    # المسافة
    إذا (ح == " ")
        ارجع "مسافة"
    نهاية
    
    # الحروف العربية (نطاق Unicode)
    # هذا مبسط - في التطبيق الحقيقي نتحقق من نطاق Unicode
    متغير عربي = "ابتثجحخدذرزسشصضطظعغفقكلمنهوي"
    لكل حرف في عربي
        إذا (ح == حرف)
            ارجع "حرف عربي"
        نهاية
    نهاية
    
    ارجع "رمز خاص"
نهاية

دالة حلل_نص(نص)
    متغير عربي = 0
    متغير إنجليزي = 0
    متغير أرقام = 0
    متغير مسافات = 0
    متغير رموز = 0
    
    لكل ح في نص
        متغير نوع = صنّف_حرف(ح)
        طابق (نوع)
            عندما "حرف عربي":
                عربي = عربي + 1
            عندما "حرف إنجليزي":
                إنجليزي = إنجليزي + 1
            عندما "رقم":
                أرقام = أرقام + 1
            عندما "مسافة":
                مسافات = مسافات + 1
            افتراضي:
                رموز = رموز + 1
        نهاية
    نهاية
    
    اطبع_سطر("تحليل النص:")
    اطبع_سطر("حروف عربية: " + عربي)
    اطبع_سطر("حروف إنجليزية: " + إنجليزي)
    اطبع_سطر("أرقام: " + أرقام)
    اطبع_سطر("مسافات: " + مسافات)
    اطبع_سطر("رموز: " + رموز)
نهاية

# صنف الكسر

دالة قمم(أ، ب)
    بينما (ب != 0)
        متغير باقي = أ % ب
        أ = ب
        ب = باقي
    نهاية
    ارجع أ
نهاية

صنف كسر
    متغير بسط
    متغير مقام
    
    باني(ب، م)

            ارمي "خطأ: المقام لا يمكن أن يكون صفراً"

        هذا.بسط = ب
        هذا.مقام = م

        متغير ق = قمم(هذا.بسط، هذا.مقام)

        ارجع هذا
    نهاية

    نهاية
نهاية
