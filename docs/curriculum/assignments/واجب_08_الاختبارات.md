## 🎯 الأهداف

- كتابة اختبارات الوحدات
- استخدام التأكيدات

## التمارين

### تمرين 1: اختبار دوال رياضية (25 نقاط)

اكتب اختبارات للدوال التالية:
- `جمع(أ، ب)`
- `طرح(أ، ب)`
- `ضرب(أ، ب)`
- `قسمة(أ، ب)` — مع اختبار القسمة على صفر

يجب اختبار:
- القيم الموجبة
- القيم السالبة
- الصفر
- الأعداد العشرية

---

### تمرين 2: اختبار صنف الطالب (30 نقاط)

اكتب اختبارات لصنف `طالب`:
- إنشاء طالب
- إضافة درجات
- حساب المعدل (حالة فارغة، درجة واحدة، درجات متعددة)
- حالة النجاح
- التحقق من صحة الدرجات

---

### تمرين 3: اختبار دوال النصوص (25 نقاط)

اكتب اختبارات لدوال النصوص:
- `عكس_نص(نص)`
- `عدد_الكلمات(نص)`
- `يبدأ_بـ(نص، بادئة)`
- `ينتهي_بـ(نص، لاحقة)`

---

### تمرين 4: اختبار نظام المهام (20 نقاط)

اكتب اختبارات لنظام المهام:
- إنشاء مهمة
- تغيير الحالة
- حساب نسبة الإنجاز

# اختبار دوال النصوص

استورد "test"

# الدوال
دالة عكس_نص(نص)
    متغير نتيجة = ""
    متغير ف = طول(نص) - 1
    بينما (ف >= 0)
        نتيجة = نتيجة + نص[ف]
        ف = ف - 1
    نهاية
    ارجع نتيجة
نهاية

دالة عدد_الكلمات(نص)
    إذا (طول(نص) == 0)
        ارجع 0
    نهاية
    
    متغير عداد = 0
    متغير في_كلمة = خطأ
    
    لكل ح في نص
        إذا (ح == " ")
            إذا (في_كلمة)
                عداد = عداد + 1
                في_كلمة = خطأ
            نهاية
        وإلا
            في_كلمة = صحيح
        نهاية
    نهاية
    
    إذا (في_كلمة)
        عداد = عداد + 1
    نهاية
    
    ارجع عداد
نهاية

دالة يبدأ_بـ(نص، بادئة)
    إذا (طول(بادئة) > طول(نص))
        ارجع خطأ
    نهاية
    
    متغير ف = 0
    بينما (ف < طول(بادئة))
        إذا (نص[ف] != بادئة[ف])
            ارجع خطأ
        نهاية
        ف = ف + 1
    نهاية
    
    ارجع صحيح
نهاية

دالة ينتهي_بـ(نص، لاحقة)
    إذا (طول(لاحقة) > طول(نص))
        ارجع خطأ
    نهاية
    
    متغير بداية = طول(نص) - طول(لاحقة)
    متغير ف = 0
    
    بينما (ف < طول(لاحقة))
        إذا (نص[بداية + ف] != لاحقة[ف])
            ارجع خطأ
        نهاية
        ف = ف + 1
    نهاية
    
    ارجع صحيح
نهاية
