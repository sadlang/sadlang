# 🧱 التجريد (Abstraction)

## مقدمة

التجريد (Abstraction) هو عملية إخفاء التفاصيل المعقدة وإظهار الميزات الأساسية فقط. في البرمجة الكائنية، يتم ذلك عبر الأصناف المجردة والواجهات التي تُعرِّف "ماذا" يجب أن يفعل الكائن دون تحديد "كيف".

## الفرق بين التغليف والتجريد

- **التغليف**: إخفاء التنفيذ الداخلي (كيف)
- **التجريد**: إخفاء التعقيد وإظهار ما هو ضروري فقط (ماذا)

```s
# مثال: قيادة السيارة
# التجريد: تعرف أن الدواسة تسرع والمقود يوجه
# لا تحتاج لمعرفة كيف يعمل المحرك أو ناقل الحركة

صنف مجرد مركبة
    مجرد دالة اشتغل()
    مجرد دالة تحرك()
    مجرد دالة توقف()
نهاية
```

## الأصناف المجردة

الصنف المجرد هو صنف لا يمكن إنشاء كائنات منه مباشرة، ويحتوي على دوال مجردة يجب على الأصناف الوارثة تنفيذها:

```s
صنف مجرد شكل
    نص الاسم
    
    منشئ(نص اسم)
        هذا.الاسم = اسم
    نهاية
    
    # دوال مجردة - يجب تنفيذها في الأصناف الوارثة
    مجرد دالة المساحة()
    مجرد دالة المحيط()
    
    # دالة عادية - يمكن استخدامها كما هي
    دالة وصف()
        اطبع("هذا شكل: " + هذا.الاسم)
        اطبع("المساحة: " + هذا.المساحة())
        اطبع("المحيط: " + هذا.المحيط())
    نهاية
نهاية

صنف مستطيل يرث شكل
    رقم الطول
    رقم العرض
    
    منشئ(رقم طول، رقم عرض)
        أبي("مستطيل")
        هذا.الطول = طول
        هذا.العرض = عرض
    نهاية
    
    # تنفيذ الدوال المجردة
    دالة المساحة()
        إرجاع هذا.الطول * هذا.العرض
    نهاية
    
    دالة المحيط()
        إرجاع 2 * (هذا.الطول + هذا.العرض)
    نهاية
نهاية

صنف دائرة يرث شكل
    رقم نصف_القطر
    
    منشئ(رقم نصف_قطر)
        أبي("دائرة")
        هذا.نصف_القطر = نصف_قطر
    نهاية
    
    دالة المساحة()
        إرجاع 3.14159 * هذا.نصف_القطر ** 2
    نهاية
    
    دالة المحيط()
        إرجاع 2 * 3.14159 * هذا.نصف_القطر
    نهاية
نهاية

دالة رئيسية()
    # جديد شكل()  # خطأ! لا يمكن إنشاء كائن من صنف مجرد
    
    متغير م = جديد مستطيل(5، 3)
    متغير د = جديد دائرة(4)
    
    م.وصف()
    اطبع("---")
    د.وصف()
نهاية
```

## الواجهات (Interfaces)

الواجهة تُعرِّف عقداً يجب على الأصناف المُنفِّذة الالتزام به:

```s
واجهة قابل_للحفظ
    دالة احفظ()
    دالة حمل(نص مسار)
نهاية

واجهة قابل_للطباعة
    دالة اطبع()
نهاية

صنف مستند ينفذ قابل_للحفظ، قابل_للطباعة
    نص العنوان
    نص المحتوى
    
    منشئ(نص عنوان، نص محتوى)
        هذا.العنوان = عنوان
        هذا.المحتوى = محتوى
    نهاية
    
    دالة احفظ()
        # حفظ المستند في ملف
        اطبع("💾 تم حفظ المستند: " + هذا.العنوان)
    نهاية
    
    دالة حمل(نص مسار)
        # تحميل المستند من ملف
        اطبع("📂 تم تحميل المستند من: " + مسار)
    نهاية
    
    دالة اطبع()
        اطبع("🖨️ طباعة: " + هذا.العنوان)
        اطبع(هذا.المحتوى)
    نهاية
نهاية

صنف صورة ينفذ قابل_للحفظ
    نص الاسم
    رقم العرض
    رقم الارتفاع
    
    منشئ(نص اسم، رقم عرض، رقم ارتفاع)
        هذا.الاسم = اسم
        هذا.العرض = عرض
        هذا.الارتفاع = ارتفاع
    نهاية
    
    دالة احفظ()
        اطبع("💾 تم حفظ الصورة: " + هذا.الاسم)
    نهاية
    
    دالة حمل(نص مسار)
        اطبع("📂 تم تحميل الصورة من: " + مسار)
    نهاية
نهاية

دالة احفظ_الكل(قائمة عناصر)
    # هذه الدالة تعمل مع أي شيء قابل للحفظ
    لكل عنصر في عناصر
        عنصر.احفظ()
    نهاية
نهاية

دالة رئيسية()
    قائمة قابلة_للحفظ = [
        جديد مستند("تقرير"، "محتوى التقرير...")،
        جديد صورة("صورة.jpg"، 1920، 1080)،
        جديد مستند("ملاحظات"، "ملاحظات مهمة...")
    ]
    
    احفظ_الكل(قابلة_للحفظ)
نهاية
```

## طبقات التجريد

```s
# المستوى الأعلى من التجريد
واجهة مخزن_بيانات
    دالة احفظ(نص مفتاح، متغير قيمة)
    دالة احصل(نص مفتاح)
    دالة احذف(نص مفتاح)
    دالة يوجد(نص مفتاح)
نهاية

# تنفيذ للذاكرة
صنف مخزن_ذاكرة ينفذ مخزن_بيانات
    خاص قاموس _البيانات = {}
    
    دالة احفظ(نص مفتاح، متغير قيمة)
        هذا._البيانات[مفتاح] = قيمة
    نهاية
    
    دالة احصل(نص مفتاح)
        إرجاع هذا._البيانات.احصل(مفتاح، عدم)
    نهاية
    
    دالة احذف(نص مفتاح)
        إذا (مفتاح في هذا._البيانات)
            احذف هذا._البيانات[مفتاح]
            إرجاع صحيح
        نهاية
        إرجاع خطأ
    نهاية
    
    دالة يوجد(نص مفتاح)
        إرجاع مفتاح في هذا._البيانات
    نهاية
نهاية

# تنفيذ للملفات
صنف مخزن_ملفات ينفذ مخزن_بيانات
    خاص نص _المسار
    
    منشئ(نص مسار)
        هذا._المسار = مسار
    نهاية
    
    دالة احفظ(نص مفتاح، متغير قيمة)
        نص مسار_الملف = هذا._المسار + "/" + مفتاح + ".json"
        اكتب_ملف(مسار_الملف، إلى_json(قيمة))
    نهاية
    
    دالة احصل(نص مفتاح)
        نص مسار_الملف = هذا._المسار + "/" + مفتاح + ".json"
        إذا (ملف_موجود(مسار_الملف))
            إرجاع من_json(اقرأ_ملف(مسار_الملف))
        نهاية
        إرجاع عدم
    نهاية
    
    دالة احذف(نص مفتاح)
        نص مسار_الملف = هذا._المسار + "/" + مفتاح + ".json"
        إذا (ملف_موجود(مسار_الملف))
            احذف_ملف(مسار_الملف)
            إرجاع صحيح
        نهاية
        إرجاع خطأ
    نهاية
    
    دالة يوجد(نص مفتاح)
        نص مسار_الملف = هذا._المسار + "/" + مفتاح + ".json"
        إرجاع ملف_موجود(مسار_الملف)
    نهاية
نهاية

# تنفيذ لقاعدة البيانات
صنف مخزن_قاعدة_بيانات ينفذ مخزن_بيانات
    خاص متغير _الاتصال
    
    منشئ(نص سلسلة_الاتصال)
        هذا._الاتصال = اتصل_بقاعدة_البيانات(سلسلة_الاتصال)
    نهاية
    
    دالة احفظ(نص مفتاح، متغير قيمة)
        هذا._الاتصال.نفذ(
            "INSERT OR REPLACE INTO مخزن (مفتاح، قيمة) VALUES (?, ?)"،
            [مفتاح، إلى_json(قيمة)]
        )
    نهاية
    
    دالة احصل(نص مفتاح)
        متغير نتيجة = هذا._الاتصال.استعلام(
            "SELECT قيمة FROM مخزن WHERE مفتاح = ?"،
            [مفتاح]
        )
        إذا (نتيجة)
            إرجاع من_json(نتيجة.قيمة)
        نهاية
        إرجاع عدم
    نهاية
    
    دالة احذف(نص مفتاح)
        رقم متأثرة = هذا._الاتصال.نفذ(
            "DELETE FROM مخزن WHERE مفتاح = ?"،
            [مفتاح]
        )
        إرجاع متأثرة > 0
    نهاية
    
    دالة يوجد(نص مفتاح)
        إرجاع هذا.احصل(مفتاح) != عدم
    نهاية
نهاية

# الكود الذي يستخدم المخزن لا يهتم بالتنفيذ
صنف خدمة_المستخدمين
    خاص مخزن_بيانات _المخزن
    
    منشئ(مخزن_بيانات مخزن)
        هذا._المخزن = مخزن
    نهاية
    
    دالة احفظ_مستخدم(قاموس مستخدم)
        هذا._المخزن.احفظ("مستخدم:" + مستخدم.المعرف، مستخدم)
    نهاية
    
    دالة احصل_على_مستخدم(نص معرف)
        إرجاع هذا._المخزن.احصل("مستخدم:" + معرف)
    نهاية
نهاية

دالة رئيسية()
    # يمكن تبديل التنفيذ بسهولة
    متغير مخزن
    
    # للتطوير
    مخزن = جديد مخزن_ذاكرة()
    
    # للإنتاج
    # مخزن = جديد مخزن_قاعدة_بيانات("...")
    
    متغير خدمة = جديد خدمة_المستخدمين(مخزن)
    خدمة.احفظ_مستخدم({"المعرف": "1"، "الاسم": "أحمد"})
    
    اطبع(خدمة.احصل_على_مستخدم("1"))
نهاية
```

## أمثلة عملية

### نظام الدفع المجرد

```s
صنف مجرد معالج_دفع
    خاص نص _المعرف
    
    منشئ()
        هذا._المعرف = ولد_معرف_فريد()
    نهاية
    
    # دوال مجردة
    مجرد دالة تهيئة()
    مجرد دالة عالج_الدفع(رقم مبلغ، قاموس تفاصيل)
    مجرد دالة استرد(نص معرف_العملية، رقم مبلغ)
    
    # دوال مشتركة
    دالة سجل_العملية(نص نوع، رقم مبلغ)
        اطبع("[" + هذا._المعرف + "] " + نوع + ": " + مبلغ)
    نهاية
    
    دالة تحقق_من_المبلغ(رقم مبلغ)
        إذا (مبلغ <= 0)
            ارفع خطأ("المبلغ يجب أن يكون موجباً")
        نهاية
        إذا (مبلغ > 1000000)
            ارفع خطأ("المبلغ يتجاوز الحد الأقصى")
        نهاية
    نهاية
نهاية

صنف Stripe يرث معالج_دفع
    خاص نص _مفتاح_API
    
    منشئ(نص مفتاح)
        أبي()
        هذا._مفتاح_API = مفتاح
    نهاية
    
    دالة تهيئة()
        اطبع("🔧 تهيئة Stripe...")
        # الاتصال بـ Stripe API
    نهاية
    
    دالة عالج_الدفع(رقم مبلغ، قاموس تفاصيل)
        هذا.تحقق_من_المبلغ(مبلغ)
        
        اطبع("💳 معالجة الدفع عبر Stripe: " + مبلغ)
        # استدعاء Stripe API
        
        هذا.سجل_العملية("دفع Stripe"، مبلغ)
        إرجاع {"نجاح": صحيح، "مرجع": "stripe_" + رقم_عشوائي(10000، 99999)}
    نهاية
    
    دالة استرد(نص معرف_العملية، رقم مبلغ)
        اطبع("↩️ استرداد عبر Stripe: " + مبلغ)
        إرجاع صحيح
    نهاية
نهاية

صنف PayPal يرث معالج_دفع
    خاص نص _معرف_العميل
    خاص نص _السر
    
    منشئ(نص معرف، نص سر)
        أبي()
        هذا._معرف_العميل = معرف
        هذا._السر = سر
    نهاية
    
    دالة تهيئة()
        اطبع("🔧 تهيئة PayPal...")
        # الحصول على رمز الوصول
    نهاية
    
    دالة عالج_الدفع(رقم مبلغ، قاموس تفاصيل)
        هذا.تحقق_من_المبلغ(مبلغ)
        
        اطبع("🅿️ معالجة الدفع عبر PayPal: " + مبلغ)
        
        هذا.سجل_العملية("دفع PayPal"، مبلغ)
        إرجاع {"نجاح": صحيح، "مرجع": "paypal_" + رقم_عشوائي(10000، 99999)}
    نهاية
    
    دالة استرد(نص معرف_العملية، رقم مبلغ)
        اطبع("↩️ استرداد عبر PayPal: " + مبلغ)
        إرجاع صحيح
    نهاية
نهاية
```

### نظام الإشعارات المجرد

```s
واجهة مرسل
    دالة أرسل(نص المستلم، نص الموضوع، نص المحتوى)
نهاية

صنف مجرد مرسل_إشعارات ينفذ مرسل
    خاص قائمة _السجلات = []
    
    دالة أرسل(نص المستلم، نص الموضوع، نص المحتوى)
        حاول
            هذا._نفذ_الإرسال(المستلم، الموضوع، المحتوى)
            هذا._سجل(المستلم، الموضوع، صحيح)
            إرجاع صحيح
        التقط خطأ كـ خ
            هذا._سجل(المستلم، الموضوع، خطأ، خ.الرسالة)
            إرجاع خطأ
        نهاية
    نهاية
    
    # يجب تنفيذها في الأصناف الفرعية
    مجرد دالة _نفذ_الإرسال(نص المستلم، نص الموضوع، نص المحتوى)
    
    خاص دالة _سجل(نص المستلم، نص الموضوع، منطقي نجح، نص خطأ = "")
        هذا._السجلات.أضف({
            "المستلم": المستلم،
            "الموضوع": الموضوع،
            "نجح": نجح،
            "خطأ": خطأ،
            "الوقت": الوقت_الحالي()
        })
    نهاية
    
    دالة السجلات()
        إرجاع هذا._السجلات.نسخة()
    نهاية
نهاية

صنف مرسل_بريد يرث مرسل_إشعارات
    خاص نص _خادم_SMTP
    
    منشئ(نص خادم)
        هذا._خادم_SMTP = خادم
    نهاية
    
    دالة _نفذ_الإرسال(نص المستلم، نص الموضوع، نص المحتوى)
        اطبع("📧 إرسال بريد إلى: " + المستلم)
        اطبع("   الموضوع: " + الموضوع)
        # تنفيذ الإرسال الفعلي
    نهاية
نهاية

صنف مرسل_SMS يرث مرسل_إشعارات

    منشئ(نص مزود)
        هذا._مزود = مزود
    نهاية
    
    دالة _نفذ_الإرسال(نص المستلم، نص الموضوع، نص المحتوى)

        # تنفيذ الإرسال الفعلي
```

## فوائد التجريد

### 1. تقليل التعقيد

```s
# بدلاً من معرفة كل تفاصيل قاعدة البيانات
# نستخدم واجهة بسيطة
واجهة مستودع_مستخدمين
    دالة احفظ(مستخدم م)
    دالة ابحث(نص معرف)
    دالة احذف(نص معرف)
    دالة الكل()
نهاية
```

### 2. قابلية التبديل

```s
# يمكن تبديل التنفيذ بسهولة
دالة رئيسية()
    متغير مستودع
    
    إذا (البيئة() == "اختبار")
        مستودع = جديد مستودع_ذاكرة()
    وإلا
        مستودع = جديد مستودع_MySQL()
    نهاية
    
    متغير تطبيق = جديد التطبيق(مستودع)
نهاية
```

### 3. قابلية الاختبار

```s
صنف مستودع_وهمي ينفذ مستودع_مستخدمين
    خاص قائمة _مستخدمين = []
    
    دالة احفظ(مستخدم م)
        هذا._مستخدمين.أضف(م)
    نهاية
    
    # ... باقي التنفيذ للاختبار
نهاية
```

## الخلاصة

التجريد في لغة ص:
- إخفاء التعقيد وإظهار الضروري فقط
- الأصناف المجردة تحدد القالب
- الواجهات تحدد العقد
- يسهل تبديل التنفيذ
- يحسن قابلية الاختبار
- يقلل الترابط بين الأجزاء
