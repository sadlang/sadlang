## مقدمة للاستخدامات المتقدمة

بعد أن تعلمنا الأساسيات وشاهدنا أمثلة عملية، حان الوقت لاستكشاف استخدامات أكثر تقدماً وأنماط برمجية احترافية للبرمجة غير المتزامنة في لغة ص. هذه الأنماط والتقنيات تساعدك على بناء تطبيقات أكثر قوة واحترافية.

## نمط المنسق (Coordinator Pattern)

نمط المنسق يُستخدم عندما تحتاج لتنسيق عدة عمليات غير متزامنة معقدة. المنسق هو دالة مركزية تدير تدفق العمل بين عدة دوال:

```
# دوال العمليات الفردية
غير_متزامن دالة التحقق_من_المخزون(نص منتج)
    اطبع("فحص المخزون لـ: " + منتج)
    إرجاع "متوفر"
نهاية

غير_متزامن دالة حساب_الشحن(نص عنوان، نص منتج)
    اطبع("حساب تكلفة الشحن إلى: " + عنوان)
    إرجاع 25
نهاية

غير_متزامن دالة حساب_الضرائب(عدد مبلغ، نص منطقة)
    اطبع("حساب الضرائب للمنطقة: " + منطقة)
    إرجاع مبلغ * 0.15
نهاية

غير_متزامن دالة التحقق_من_القسيمة(نص رمز_القسيمة)
    اطبع("التحقق من صلاحية القسيمة: " + رمز_القسيمة)
    
    إذا رمز_القسيمة == "SAVE20"
        إرجاع 20
    نهاية
    
    إرجاع 0
نهاية

# الدالة المنسقة الرئيسية
غير_متزامن دالة منسق_عملية_الشراء(نص منتج، عدد سعر_المنتج، نص عنوان، نص منطقة، نص قسيمة)
    اطبع("=== بدء تنسيق عملية الشراء ===")
    
    # المرحلة 1: التحققات الأولية
    اطبع("--- المرحلة 1: التحققات ---")
    نص حالة_المخزون = انتظر التحقق_من_المخزون(منتج)
    
    إذا حالة_المخزون != "متوفر"
        اطبع("المنتج غير متوفر")
        إرجاع "فشل: المنتج غير متوفر"
    نهاية
    اطبع("✓ المنتج متوفر")
    
    # المرحلة 2: الحسابات المالية
    اطبع("--- المرحلة 2: الحسابات ---")
    
    # حساب الشحن
    عدد تكلفة_الشحن = انتظر حساب_الشحن(عنوان، منتج)
    اطبع("تكلفة الشحن: " + تكلفة_الشحن + " ريال")
    
    # حساب المجموع الأولي
    عدد مجموع_أولي = سعر_المنتج + تكلفة_الشحن
    اطبع("المجموع قبل الضرائب: " + مجموع_أولي + " ريال")
    
    # حساب الضرائب
    عدد الضرائب = انتظر حساب_الضرائب(مجموع_أولي، منطقة)
    اطبع("الضرائب: " + الضرائب + " ريال")
    
    # المرحلة 3: الخصومات
    اطبع("--- المرحلة 3: الخصومات ---")
    عدد قيمة_الخصم = انتظر التحقق_من_القسيمة(قسيمة)
    
    إذا قيمة_الخصم > 0
        اطبع("✓ قسيمة صالحة! خصم: " + قيمة_الخصم + "%")
    نهاية
    
    # المرحلة 4: الحساب النهائي
    اطبع("--- المرحلة 4: المجموع النهائي ---")
    عدد مجموع_مع_ضرائب = مجموع_أولي + الضرائب
    عدد مبلغ_الخصم = مجموع_مع_ضرائب * (قيمة_الخصم / 100)
    عدد المجموع_النهائي = مجموع_مع_ضرائب - مبلغ_الخصم
    
    اطبع("سعر المنتج: " + سعر_المنتج + " ريال")
    اطبع("الشحن: " + تكلفة_الشحن + " ريال")
    اطبع("الضرائب: " + الضرائب + " ريال")
    اطبع("الخصم: " + مبلغ_الخصم + " ريال")
    اطبع("=========================")
    اطبع("المجموع النهائي: " + المجموع_النهائي + " ريال")
    
    اطبع("=== اكتمل التنسيق بنجاح ===")
    إرجاع "نجح"
نهاية

دالة رئيسية()
    اطبع_سطر("*** نظام منسق الشراء المتقدم ***")
    اطبع_سطر("")
    
    منسق_عملية_الشراء("لابتوب ديل"، 3500، "الرياض - حي النخيل"، "الوسطى"، "SAVE20")
نهاية
```

هذا النمط يجعل الكود أكثر تنظيماً ويسهل إضافة خطوات جديدة أو تعديل الخطوات الموجودة.

## نمط إعادة المحاولة (Retry Pattern)

في التطبيقات الحقيقية، قد تفشل بعض العمليات بسبب مشاكل مؤقتة في الشبكة أو الخادم. نمط إعادة المحاولة يسمح لك بإعادة محاولة العملية عدة مرات قبل الإعلان عن الفشل:

```
# دالة قد تفشل أحياناً
غير_متزامن دالة عملية_غير_مستقرة(نص بيانات)
    اطبع("محاولة معالجة: " + بيانات)
    # في الحقيقة هنا قد تفشل العملية أحياناً
    # للتبسيط، سنفترض النجاح دائماً في المحاولة الثالثة
    إرجاع "نجح"
نهاية

# دالة إعادة المحاولة الذكية
غير_متزامن دالة محاولة_مع_إعادة(نص بيانات، عدد أقصى_محاولات)
    اطبع("=== بدء العملية مع إمكانية إعادة المحاولة ===")
    اطبع("أقصى عدد محاولات: " + أقصى_محاولات)
    اطبع("")
    
    عدد محاولة_حالية = 1
    
    # المحاولة الأولى
    اطبع("المحاولة رقم: " + محاولة_حالية)
    نص نتيجة = انتظر عملية_غير_مستقرة(بيانات)
    
    إذا نتيجة == "نجح"
        اطبع("✓ نجحت العملية في المحاولة: " + محاولة_حالية)
        إرجاع "نجح"
    نهاية
    
    اطبع("✗ فشلت المحاولة " + محاولة_حالية)
    
    # المحاولة الثانية
    إذا محاولة_حالية < أقصى_محاولات
        محاولة_حالية = محاولة_حالية + 1
        اطبع("انتظار 2 ثانية قبل المحاولة التالية...")
        اطبع("المحاولة رقم: " + محاولة_حالية)
        نتيجة = انتظر عملية_غير_مستقرة(بيانات)
        
        إذا نتيجة == "نجح"
            اطبع("✓ نجحت العملية في المحاولة: " + محاولة_حالية)
            إرجاع "نجح"
        نهاية
        
        اطبع("✗ فشلت المحاولة " + محاولة_حالية)
    نهاية
    
    # المحاولة الثالثة
    إذا محاولة_حالية < أقصى_محاولات
        محاولة_حالية = محاولة_حالية + 1
        اطبع("انتظار 4 ثوانٍ قبل المحاولة التالية...")
        اطبع("المحاولة رقم: " + محاولة_حالية)
        نتيجة = انتظر عملية_غير_مستقرة(بيانات)
        
        إذا نتيجة == "نجح"
            اطبع("✓ نجحت العملية في المحاولة: " + محاولة_حالية)
            إرجاع "نجح"
        نهاية
        
        اطبع("✗ فشلت المحاولة " + محاولة_حالية)
    نهاية
    
    اطبع("")
    اطبع("=== فشلت جميع المحاولات ===")
    إرجاع "فشل بعد " + محاولة_حالية + " محاولات"
نهاية

دالة رئيسية()
    اطبع_سطر("*** نظام إعادة المحاولة الذكي ***")
    اطبع_سطر("")
    
    نص نتيجة = محاولة_مع_إعادة("بيانات مهمة"، 3)
    اطبع_سطر("")
    اطبع_سطر("النتيجة النهائية: " + نتيجة)
نهاية
```

هذا النمط مهم جداً للتعامل مع خدمات خارجية غير مستقرة مثل واجهات برمجة التطبيقات (APIs) أو قواعد البيانات البعيدة.

## نمط الذاكرة المؤقتة (Caching Pattern)

تحسين الأداء عن طريق حفظ نتائج العمليات المكلفة وإعادة استخدامها:

```
# دالة للبحث في قاعدة البيانات (عملية بطيئة)
غير_متزامن دالة البحث_في_قاعدة_البيانات(نص استعلام)
    اطبع("البحث في قاعدة البيانات: " + استعلام)
    اطبع("(هذه عملية بطيئة...)")
    # محاكاة وقت المعالجة
    إرجاع "نتيجة البحث عن: " + استعلام
نهاية

# نظام الذاكرة المؤقتة البسيط
# في الحقيقة سنستخدم قاموس لحفظ النتائج

# دالة مع ذاكرة مؤقتة
غير_متزامن دالة بحث_مع_ذاكرة_مؤقتة(نص استعلام)
    اطبع("--- طلب بحث: " + استعلام + " ---")
    
    # للتبسيط، سنحدد بعض الاستعلامات المحفوظة مسبقاً
    إذا استعلام == "منتجات"
        اطبع("✓ وُجد في الذاكرة المؤقتة!")
        إرجاع "نتيجة محفوظة: قائمة المنتجات"
    نهاية
    
    # إذا لم يكن محفوظاً، نبحث في قاعدة البيانات
    اطبع("لم يُعثر عليه في الذاكرة المؤقتة، البحث في قاعدة البيانات...")
    نص نتيجة = انتظر البحث_في_قاعدة_البيانات(استعلام)
    
    اطبع("✓ تم حفظ النتيجة في الذاكرة المؤقتة")
    إرجاع نتيجة
نهاية

# دالة لاختبار النظام
غير_متزامن دالة اختبار_نظام_الذاكرة_المؤقتة()
    اطبع("=== اختبار نظام الذاكرة المؤقتة ===")
    اطبع("")
    
    # البحث الأول - سيذهب لقاعدة البيانات
    اطبع("البحث الأول:")
    نص نتيجة1 = انتظر بحث_مع_ذاكرة_مؤقتة("عملاء")
    اطبع("النتيجة: " + نتيجة1)
    اطبع("")
    
    # البحث الثاني - نفس الاستعلام (محفوظ)
    اطبع("البحث الثاني (نفس الاستعلام):")
    نص نتيجة2 = انتظر بحث_مع_ذاكرة_مؤقتة("منتجات")
    اطبع("النتيجة: " + نتيجة2)
    اطبع("")
    
    # البحث الثالث - استعلام جديد
    اطبع("البحث الثالث (استعلام جديد):")
    نص نتيجة3 = انتظر بحث_مع_ذاكرة_مؤقتة("طلبات")
    اطبع("النتيجة: " + نتيجة3)
    اطبع("")
    
    اطبع("=== ملاحظة: البحث الثاني كان فورياً! ===")
    إرجاع "اكتمل"
نهاية

دالة رئيسية()
    اطبع_سطر("*** نظام الذاكرة المؤقتة للبحث ***")
    اطبع_سطر("")
    
    اختبار_نظام_الذاكرة_المؤقتة()
نهاية
```

نمط الذاكرة المؤقتة يحسن الأداء بشكل كبير خاصة للبيانات التي لا تتغير كثيراً.

## نمط المعالج التدريجي (Pipeline Pattern)

معالجة البيانات من خلال سلسلة من المراحل، كل مرحلة تعالج خرج المرحلة السابقة:

```
# المرحلة 1: جلب البيانات الخام
غير_متزامن دالة مرحلة_الجلب(نص مصدر)
    اطبع("[مرحلة 1] جلب البيانات من: " + مصدر)
    إرجاع "بيانات خام من " + مصدر
نهاية

# المرحلة 2: تنظيف البيانات
غير_متزامن دالة مرحلة_التنظيف(نص بيانات_خام)
    اطبع("[مرحلة 2] تنظيف البيانات...")
    إرجاع "بيانات منظفة: " + بيانات_خام
نهاية

# المرحلة 3: تحويل البيانات
غير_متزامن دالة مرحلة_التحويل(نص بيانات_منظفة)
    اطبع("[مرحلة 3] تحويل البيانات إلى الصيغة المطلوبة...")
    إرجاع "بيانات محولة: " + بيانات_منظفة
نهاية

# المرحلة 4: إثراء البيانات
غير_متزامن دالة مرحلة_الإثراء(نص بيانات_محولة)
    اطبع("[مرحلة 4] إضافة معلومات إضافية...")
    إرجاع "بيانات مثراة: " + بيانات_محولة
نهاية

# المرحلة 5: حفظ النتائج
غير_متزامن دالة مرحلة_الحفظ(نص بيانات_نهائية)
    اطبع("[مرحلة 5] حفظ النتائج في قاعدة البيانات...")
    إرجاع "تم الحفظ: " + بيانات_نهائية
نهاية

# المعالج التدريجي الرئيسي
غير_متزامن دالة معالج_بيانات_تدريجي(نص مصدر_البيانات)
    اطبع("=== بدء المعالج التدريجي ===")
    اطبع("المصدر: " + مصدر_البيانات)
    اطبع("")
    
    # تمرير البيانات عبر جميع المراحل
    نص بيانات = انتظر مرحلة_الجلب(مصدر_البيانات)
    اطبع("✓ مرحلة 1 مكتملة")
    اطبع("")
    
    بيانات = انتظر مرحلة_التنظيف(بيانات)
    اطبع("✓ مرحلة 2 مكتملة")
    اطبع("")
    
    بيانات = انتظر مرحلة_التحويل(بيانات)
    اطبع("✓ مرحلة 3 مكتملة")
    اطبع("")
    
    بيانات = انتظر مرحلة_الإثراء(بيانات)
    اطبع("✓ مرحلة 4 مكتملة")
    اطبع("")
    
    نص نتيجة_نهائية = انتظر مرحلة_الحفظ(بيانات)
    اطبع("✓ مرحلة 5 مكتملة")
    اطبع("")
    
    اطبع("=== اكتمل المعالج التدريجي ===")
    اطبع("النتيجة: " + نتيجة_نهائية)
    
    إرجاع "نجح"
نهاية

دالة رئيسية()
    اطبع_سطر("*** نظام المعالج التدريجي للبيانات ***")
    اطبع_سطر("")
    
    معالج_بيانات_تدريجي("ملف_بيانات_العملاء.csv")
نهاية
```

هذا النمط ممتاز لمعالجة البيانات الكبيرة والعمليات المعقدة التي تحتاج لعدة خطوات متتالية.

## نمط التجميع (Aggregation Pattern)

جمع نتائج من عدة مصادر مختلفة وتجميعها في نتيجة واحدة:

```
# دوال لجلب البيانات من مصادر مختلفة
غير_متزامن دالة جلب_من_قاعدة_البيانات_الأولى(نص استعلام)
    اطبع("جلب من قاعدة البيانات الأولى...")
    إرجاع "نتائج_قاعدة_1: 150 سجل"
نهاية

غير_متزامن دالة جلب_من_قاعدة_البيانات_الثانية(نص استعلام)
    اطبع("جلب من قاعدة البيانات الثانية...")
    إرجاع "نتائج_قاعدة_2: 200 سجل"
نهاية

غير_متزامن دالة جلب_من_واجهة_برمجية_خارجية(نص استعلام)
    اطبع("جلب من واجهة برمجية خارجية...")
    إرجاع "نتائج_API: 75 سجل"
نهاية

غير_متزامن دالة جلب_من_ملفات_محلية(نص مسار)
    اطبع("قراءة من الملفات المحلية...")
    إرجاع "نتائج_ملفات: 50 سجل"
نهاية

# دالة التجميع الرئيسية
غير_متزامن دالة مجمع_البيانات_الشامل(نص استعلام)
    اطبع("=== بدء عملية التجميع الشاملة ===")
    اطبع("الاستعلام: " + استعلام)
    اطبع("")
    
    اطبع("--- جمع البيانات من جميع المصادر ---")
    
    # جلب من المصدر 1
    نص نتائج1 = انتظر جلب_من_قاعدة_البيانات_الأولى(استعلام)
    اطبع("✓ " + نتائج1)
    
    # جلب من المصدر 2
    نص نتائج2 = انتظر جلب_من_قاعدة_البيانات_الثانية(استعلام)
    اطبع("✓ " + نتائج2)
    
    # جلب من المصدر 3
    نص نتائج3 = انتظر جلب_من_واجهة_برمجية_خارجية(استعلام)
    اطبع("✓ " + نتائج3)
    
    # جلب من المصدر 4
    نص نتائج4 = انتظر جلب_من_ملفات_محلية("/data")
    اطبع("✓ " + نتائج4)
    
    اطبع("")
    اطبع("--- تجميع النتائج ---")
    
    # حساب الإجماليات
    عدد مجموع_السجلات = 150 + 200 + 75 + 50
    
    اطبع("مجموع السجلات من جميع المصادر: " + مجموع_السجلات)
    اطبع("")
    اطبع("=== اكتمل التجميع ===")
    
    إرجاع "تم جمع " + مجموع_السجلات + " سجل من 4 مصادر"
نهاية

دالة رئيسية()
    اطبع_سطر("*** نظام تجميع البيانات من مصادر متعددة ***")
    اطبع_سطر("")
    
    نص نتيجة = مجمع_البيانات_الشامل("بحث عن العملاء")
    اطبع_سطر("")
    اطبع_سطر("النتيجة النهائية: " + نتيجة)
نهاية
```

نمط التجميع مفيد جداً عندما تحتاج لجمع معلومات من عدة مصادر مختلفة لبناء صورة شاملة.

## الخلاصة والنصائح النهائية

استخدام هذه الأنماط البرمجية المتقدمة يجعل كودك أكثر احترافية وسهولة في الصيانة. تذكر دائماً:

**أولاً:** اختر النمط المناسب للمشكلة. ليست كل مشكلة تحتاج نمطاً معقداً.

**ثانياً:** ابدأ بسيط ثم أضف التعقيد تدريجياً حسب الحاجة.

**ثالثاً:** اكتب كودك بطريقة تجعله سهل القراءة والفهم حتى بعد أشهر من كتابته.

**رابعاً:** اختبر كودك جيداً، خاصة الأجزاء غير المتزامنة.

**خامساً:** وثّق كودك بتعليقات واضحة، خاصة الأجزاء المعقدة.

البرمجة غير المتزامنة في لغة ص توفر لك أدوات قوية لبناء تطبيقات سريعة وفعالة. استخدمها بحكمة وستحصل على نتائج ممتازة.
