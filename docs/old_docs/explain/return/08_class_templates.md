# قوالب الأصناف
## دمج القوالب مع البرمجة الكائنية

## 🎓 مقدمة: ما هي قوالب الأصناف؟

**قوالب الأصناف (Class Templates)** تجمع قوة **القوالب** (البرمجة العامة) مع **البرمجة الكائنية**، لإنشاء أصناف تعمل مع **أي نوع من البيانات**.

### لماذا نحتاج قوالب الأصناف؟

تخيل أنك تريد إنشاء صندوق (Box) لتخزين قيمة واحدة. بدون القوالب:

```
# صندوق للأرقام
صنف صندوق_رقم
    رقم القيمة
نهاية

# صندوق للنصوص
صنف صندوق_نص
    نص القيمة
نهاية

# صندوق للأعداد العشرية
صنف صندوق_عشري
    عشري القيمة
نهاية
```

**مع قوالب الأصناف:**
```
# صنف عام واحد يعمل مع جميع الأنواع!
قالب<نوع ت>
صنف صندوق
    ت القيمة
نهاية

# الاستخدام
صندوق<رقم> صندوق_أرقام = جديد صندوق<رقم>()
صندوق<نص> صندوق_نصوص = جديد صندوق<نص>()
```

## 🏗️ الصيغة الأساسية

```
قالب<نوع معامل_النوع>
صنف اسم_الصنف
    # خصائص تستخدم معامل النوع
    معامل_النوع خاصية
    
    # دوال تستخدم معامل النوع
    دالة اسم_دالة(معامل_النوع معامل) ترجع معامل_النوع
        # ...
    نهاية
نهاية

# الاستخدام - تحديد النوع عند الإنشاء
اسم_الصنف<النوع_المحدد> كائن = جديد اسم_الصنف<النوع_المحدد>()
```

## 📦 مثال بسيط: صندوق عام

```
# تعريف قالب الصنف
قالب<نوع ت>
صنف صندوق
    خاصة ت المحتوى
    خاصة منطقي ممتلئ
    
    # المنشئ
    عامة دالة منشئ() فراغ
        هذا.ممتلئ = خطأ
    نهاية
    
    # وضع قيمة في الصندوق
    عامة دالة ضع(ت قيمة) فراغ
        هذا.المحتوى = قيمة
        هذا.ممتلئ = صحيح
        اطبع_سطر("✅ تم وضع القيمة في الصندوق")
    نهاية
    
    # أخذ القيمة من الصندوق
    عامة دالة خذ() ترجع ت
        اذا (!هذا.ممتلئ)
            اطبع_سطر("❌ الصندوق فارغ!")
            ارجع عدم
        نهاية
        
        هذا.ممتلئ = خطأ
        اطبع_سطر("✅ تم أخذ القيمة من الصندوق")
        ارجع هذا.المحتوى
    نهاية
    
    # التحقق من امتلاء الصندوق
    عامة دالة هل_ممتلئ() ترجع منطقي
        ارجع هذا.ممتلئ
    نهاية
    
    # نظرة خاطفة على المحتوى بدون أخذه
    عامة دالة انظر() ترجع ت
        اذا (!هذا.ممتلئ)
            اطبع_سطر("❌ الصندوق فارغ!")
            ارجع عدم
        نهاية
        
        ارجع هذا.المحتوى
    نهاية
نهاية

# ========== الاستخدام ==========

دالة رئيسية() رقم
    اطبع_سطر("═══ صندوق أرقام ═══")
    صندوق<رقم> صندوق_أرقام = جديد صندوق<رقم>()
    
    صندوق_أرقام.ضع(42)
    رقم قيمة = صندوق_أرقام.انظر()
    اطبع_سطر("القيمة: " + قيمة)
    
    رقم قيمة_مأخوذة = صندوق_أرقام.خذ()
    اطبع_سطر("أُخذت القيمة: " + قيمة_مأخوذة)
    
    اطبع_سطر("")
    اطبع_سطر("═══ صندوق نصوص ═══")
    صندوق<نص> صندوق_نصوص = جديد صندوق<نص>()
    
    صندوق_نصوص.ضع("مرحباً بك في لغة ص!")
    نص رسالة = صندوق_نصوص.خذ()
    اطبع_سطر("الرسالة: " + رسالة)
    
    ارجع 0
نهاية
```

## 📚 مثال: مكدس عام (Stack)

المكدس (Stack) هو بنية بيانات LIFO (آخر ما يدخل، أول ما يخرج):

```
قالب<نوع ت>
صنف مكدس
    خاصة مصفوفة العناصر
    خاصة رقم الحد_الأقصى
    
    # المنشئ
    عامة دالة منشئ(رقم حد_أقصى) فراغ
        هذا.العناصر = []
        هذا.الحد_الأقصى = حد_أقصى
        اطبع_سطر("📚 تم إنشاء مكدس بسعة " + حد_أقصى)
    نهاية
    
    # إضافة عنصر (Push)
    عامة دالة ادفع(ت عنصر) ترجع منطقي
        اذا (هذا.هل_ممتلئ())
            اطبع_سطر("❌ المكدس ممتلئ!")
            ارجع خطأ
        نهاية
        
        أضف(هذا.العناصر، عنصر)
        اطبع_سطر("➕ تم إضافة عنصر")
        ارجع صحيح
    نهاية
    
    # إزالة عنصر (Pop)
    عامة دالة اسحب() ترجع ت
        اذا (هذا.هل_فارغ())
            اطبع_سطر("❌ المكدس فارغ!")
            ارجع عدم
        نهاية
        
        رقم آخر_فهرس = طول(هذا.العناصر) - 1
        ت عنصر = هذا.العناصر[آخر_فهرس]
        احذف هذا.العناصر[آخر_فهرس]
        
        اطبع_سطر("➖ تم سحب عنصر")
        ارجع عنصر
    نهاية
    
    # النظر إلى العنصر العلوي (Peek)
    عامة دالة انظر_للأعلى() ترجع ت
        اذا (هذا.هل_فارغ())
            اطبع_سطر("❌ المكدس فارغ!")
            ارجع عدم
        نهاية
        
        رقم آخر_فهرس = طول(هذا.العناصر) - 1
        ارجع هذا.العناصر[آخر_فهرس]
    نهاية
    
    # التحقق من الفراغ
    عامة دالة هل_فارغ() ترجع منطقي
        ارجع طول(هذا.العناصر) == 0
    نهاية
    
    # التحقق من الامتلاء
    عامة دالة هل_ممتلئ() ترجع منطقي
        ارجع طول(هذا.العناصر) >= هذا.الحد_الأقصى
    نهاية
    
    # الحجم الحالي
    عامة دالة الحجم() ترجع رقم
        ارجع طول(هذا.العناصر)
    نهاية
    
    # طباعة المحتويات
    عامة دالة اطبع() فراغ
        اطبع_سطر("┌─────────────────┐")
        اطبع_سطر("│ محتويات المكدس │")
        اطبع_سطر("├─────────────────┤")
        
        اذا (هذا.هل_فارغ())
            اطبع_سطر("│ (فارغ)          │")
        نهاية_اذا_والا
            رقم عداد = طول(هذا.العناصر) - 1
            بينما (عداد >= 0)
                اطبع_سطر("│ " + عداد + ": " + هذا.العناصر[عداد])
                عداد = عداد - 1
            نهاية
        نهاية
        
        اطبع_سطر("└─────────────────┘")
        اطبع_سطر("الحجم: " + هذا.الحجم() + " / " + هذا.الحد_الأقصى)
    نهاية
نهاية

# ========== الاستخدام ==========

دالة رئيسية() رقم
    اطبع_سطر("═══ مكدس أرقام ═══")
    مكدس<رقم> مكدس_أرقام = جديد مكدس<رقم>(5)
    
    مكدس_أرقام.ادفع(10)
    مكدس_أرقام.ادفع(20)
    مكدس_أرقام.ادفع(30)
    
    اطبع_سطر("")
    مكدس_أرقام.اطبع()
    
    اطبع_سطر("")
    رقم علوي = مكدس_أرقام.انظر_للأعلى()
    اطبع_سطر("العنصر العلوي: " + علوي)
    
    اطبع_سطر("")
    رقم مسحوب = مكدس_أرقام.اسحب()
    اطبع_سطر("العنصر المسحوب: " + مسحوب)
    
    اطبع_سطر("")
    مكدس_أرقام.اطبع()
    
    اطبع_سطر("")
    اطبع_سطر("═══════════════════════")
    اطبع_سطر("")
    اطبع_سطر("═══ مكدس نصوص ═══")
    
    مكدس<نص> مكدس_نصوص = جديد مكدس<نص>(3)
    
    مكدس_نصوص.ادفع("أول")
    مكدس_نصوص.ادفع("ثاني")
    مكدس_نصوص.ادفع("ثالث")
    
    اطبع_سطر("")
    مكدس_نصوص.اطبع()
    
    ارجع 0
نهاية
```

## 🔗 مثال: قائمة مترابطة عامة (Linked List)

```
# عقدة في القائمة المترابطة
قالب<نوع ت>
صنف عقدة
    عامة ت البيانات
    عامة عقدة<ت> التالي
    
    عامة دالة منشئ(ت بيانات) فراغ
        هذا.البيانات = بيانات
        هذا.التالي = عدم
    نهاية
نهاية

# القائمة المترابطة
قالب<نوع ت>
صنف قائمة_مترابطة
    خاصة عقدة<ت> الرأس
    خاصة رقم الحجم
    
    عامة دالة منشئ() فراغ
        هذا.الرأس = عدم
        هذا.الحجم = 0
        اطبع_سطر("🔗 تم إنشاء قائمة مترابطة")
    نهاية
    
    # إضافة في البداية
    عامة دالة أضف_في_البداية(ت بيانات) فراغ
        عقدة<ت> عقدة_جديدة = جديد عقدة<ت>(بيانات)
        عقدة_جديدة.التالي = هذا.الرأس
        هذا.الرأس = عقدة_جديدة
        هذا.الحجم = هذا.الحجم + 1
        اطبع_سطر("➕ تمت الإضافة في البداية")
    نهاية
    
    # إضافة في النهاية
    عامة دالة أضف_في_النهاية(ت بيانات) فراغ
        عقدة<ت> عقدة_جديدة = جديد عقدة<ت>(بيانات)
        
        اذا (هذا.الرأس == عدم)
            هذا.الرأس = عقدة_جديدة
        نهاية_اذا_والا
            عقدة<ت> حالي = هذا.الرأس
            بينما (حالي.التالي != عدم)
                حالي = حالي.التالي
            نهاية
            حالي.التالي = عقدة_جديدة
        نهاية
        
        هذا.الحجم = هذا.الحجم + 1
        اطبع_سطر("➕ تمت الإضافة في النهاية")
    نهاية
    
    # البحث عن عنصر
    عامة دالة يوجد(ت بيانات) ترجع منطقي
        عقدة<ت> حالي = هذا.الرأس
        
        بينما (حالي != عدم)
            اذا (حالي.البيانات == بيانات)
                ارجع صحيح
            نهاية
            حالي = حالي.التالي
        نهاية
        
        ارجع خطأ
    نهاية
    
    # حذف عنصر
    عامة دالة احذف(ت بيانات) ترجع منطقي
        اذا (هذا.الرأس == عدم)
            ارجع خطأ
        نهاية
        
        # حذف من الرأس
        اذا (هذا.الرأس.البيانات == بيانات)
            هذا.الرأس = هذا.الرأس.التالي
            هذا.الحجم = هذا.الحجم - 1
            اطبع_سطر("➖ تم الحذف")
            ارجع صحيح
        نهاية
        
        # حذف من الوسط أو النهاية
        عقدة<ت> حالي = هذا.الرأس
        بينما (حالي.التالي != عدم)
            اذا (حالي.التالي.البيانات == بيانات)
                حالي.التالي = حالي.التالي.التالي
                هذا.الحجم = هذا.الحجم - 1
                اطبع_سطر("➖ تم الحذف")
                ارجع صحيح
            نهاية
            حالي = حالي.التالي
        نهاية
        
        ارجع خطأ
    نهاية
    
    # الحصول على الحجم
    عامة دالة احصل_الحجم() ترجع رقم
        ارجع هذا.الحجم
    نهاية
    
    # طباعة القائمة
    عامة دالة اطبع() فراغ
        اذا (هذا.الرأس == عدم)
            اطبع_سطر("القائمة فارغة")
            ارجع
        نهاية
        
        اطبع_سطر("┌─────────────────┐")
        اطبع_سطر("│ محتويات القائمة │")
        اطبع_سطر("├─────────────────┤")
        
        عقدة<ت> حالي = هذا.الرأس
        رقم عداد = 0
        
        بينما (حالي != عدم)
            اطبع_سطر("│ " + عداد + ": " + حالي.البيانات)
            حالي = حالي.التالي
            عداد = عداد + 1
        نهاية
        
        اطبع_سطر("└─────────────────┘")
        اطبع_سطر("الحجم: " + هذا.الحجم)
    نهاية
نهاية

# ========== الاستخدام ==========

دالة رئيسية() رقم
    اطبع_سطر("═══ قائمة أرقام ═══")
    قائمة_مترابطة<رقم> قائمة = جديد قائمة_مترابطة<رقم>()
    
    قائمة.أضف_في_النهاية(10)
    قائمة.أضف_في_النهاية(20)
    قائمة.أضف_في_البداية(5)
    قائمة.أضف_في_النهاية(30)
    
    اطبع_سطر("")
    قائمة.اطبع()
    
    اطبع_سطر("")
    منطقي موجود = قائمة.يوجد(20)
    اطبع_سطر("هل 20 موجود؟ " + موجود)
    
    اطبع_سطر("")
    قائمة.احذف(20)
    قائمة.اطبع()
    
    ارجع 0
نهاية
```

## 🎯 قوالب بمعاملي نوع: خريطة عامة (Map)

```
# زوج مفتاح-قيمة
قالب<نوع مفتاح، نوع قيمة>
صنف زوج
    عامة مفتاح المفتاح
    عامة قيمة القيمة
    
    عامة دالة منشئ(مفتاح م، قيمة ق) فراغ
        هذا.المفتاح = م
        هذا.القيمة = ق
    نهاية
نهاية

# خريطة بسيطة
قالب<نوع مفتاح، نوع قيمة>
صنف خريطة
    خاصة مصفوفة الأزواج
    
    عامة دالة منشئ() فراغ
        هذا.الأزواج = []
        اطبع_سطر("🗺️ تم إنشاء خريطة")
    نهاية
    
    # إضافة أو تحديث
    عامة دالة ضع(مفتاح م، قيمة ق) فراغ
        # البحث عن المفتاح
        لكل زوج في هذا.الأزواج
            اذا (زوج.المفتاح == م)
                زوج.القيمة = ق
                اطبع_سطر("✏️ تم تحديث القيمة")
                ارجع
            نهاية
        نهاية
        
        # إذا لم يوجد، أضف زوج جديد
        زوج<مفتاح، قيمة> زوج_جديد = جديد زوج<مفتاح، قيمة>(م، ق)
        أضف(هذا.الأزواج، زوج_جديد)
        اطبع_سطر("➕ تمت الإضافة")
    نهاية
    
    # الحصول على قيمة
    عامة دالة احصل(مفتاح م) ترجع قيمة
        لكل زوج في هذا.الأزواج
            اذا (زوج.المفتاح == م)
                ارجع زوج.القيمة
            نهاية
        نهاية
        
        اطبع_سطر("❌ المفتاح غير موجود")
        ارجع عدم
    نهاية
    
    # التحقق من وجود مفتاح
    عامة دالة يحتوي(مفتاح م) ترجع منطقي
        لكل زوج في هذا.الأزواج
            اذا (زوج.المفتاح == م)
                ارجع صحيح
            نهاية
        نهاية
        ارجع خطأ
    نهاية
    
    # حذف مفتاح
    عامة دالة احذف(مفتاح م) ترجع منطقي
        رقم فهرس = 0
        لكل زوج في هذا.الأزواج
            اذا (زوج.المفتاح == م)
                احذف هذا.الأزواج[فهرس]
                اطبع_سطر("➖ تم الحذف")
                ارجع صحيح
            نهاية
            فهرس = فهرس + 1
        نهاية
        ارجع خطأ
    نهاية
    
    # الحجم
    عامة دالة الحجم() ترجع رقم
        ارجع طول(هذا.الأزواج)
    نهاية
    
    # طباعة المحتويات
    عامة دالة اطبع() فراغ
        اطبع_سطر("╔═══════════════════════╗")
        اطبع_سطر("│ محتويات الخريطة      │")
        اطبع_سطر("╠═══════════════════════╣")
        
        اذا (طول(هذا.الأزواج) == 0)
            اطبع_سطر("│ (فارغة)              │")
        نهاية_اذا_والا
            لكل زوج في هذا.الأزواج
                اطبع_سطر("│ " + زوج.المفتاح + " => " + زوج.القيمة)
            نهاية
        نهاية
        
        اطبع_سطر("╚═══════════════════════╝")
        اطبع_سطر("الحجم: " + هذا.الحجم())
    نهاية
نهاية

# ========== الاستخدام ==========

دالة رئيسية() رقم
    اطبع_سطر("═══ خريطة: رقم → نص ═══")
    خريطة<رقم، نص> أسماء = جديد خريطة<رقم، نص>()
    
    أسماء.ضع(1، "أحمد")

    اطبع_سطر("الاسم رقم 2: " + اسم)
    
    اطبع_سطر("")
    أسماء.احذف(2)
    أسماء.اطبع()
    
    اطبع_سطر("")
    اطبع_سطر("═══════════════════════")
    اطبع_سطر("")
    اطبع_سطر("═══ خريطة: نص → عشري ═══")
    
    خريطة<نص، عشري> معدلات = جديد خريطة<نص، عشري>()
    
    معدلات.ضع("أحمد"، 4.75)
    معدلات.ضع("فاطمة"، 4.85)
    معدلات.ضع("علي"، 4.50)
    
    اطبع_سطر("")
    معدلات.اطبع()
    
    ارجع 0
نهاية
```

## 📚 الخلاصة

قوالب الأصناف توفر:
- 🔄 **قابلية إعادة الاستخدام القصوى** - صنف واحد، أنواع لا نهائية
- 🎯 **أمان النوع** - المترجم يتحقق من صحة الأنواع
- 📦 **بنيات بيانات عامة** - Stack, Queue, List, Map, etc.
- ⚡ **أداء عالي** - لا حاجة لـ casting أو boxing
