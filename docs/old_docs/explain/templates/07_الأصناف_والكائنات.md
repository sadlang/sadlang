# الأصناف والكائنات في لغة ص

## تعريف الصنف

الصنف هو قالب يُحدد بنية الكائنات وسلوكها. يحتوي على خصائص (البيانات) ودوال (السلوكيات).

### الصيغة الأساسية

```س
صنف اسم_الصنف
    خاص:
        # الخصائص الخاصة
    
    محمي:
        # الخصائص المحمية
    
    عام:
        # الخصائص والدوال العامة
نهاية
```

## مستويات الوصول

لغة ص تدعم ثلاثة مستويات للوصول:

### 1. عام (public)

الأعضاء العامة يمكن الوصول إليها من أي مكان:

```س
صنف شخص
    عام:
        نص الاسم        # يمكن الوصول من الخارج
        رقم العمر       # يمكن الوصول من الخارج
        
        دالة تحية()
            اطبع("مرحباً، أنا ")
            اطبع(الاسم)
        نهاية
نهاية

دالة رئيسية() رقم
    متغير ش = شخص()
    ش.الاسم = "أحمد"    # ✓ يعمل - الاسم عام
    ش.العمر = 25        # ✓ يعمل - العمر عام
    ش.تحية()           # ✓ يعمل - الدالة عامة
    ارجع 0
نهاية
```

### 2. خاص (private)

الأعضاء الخاصة يمكن الوصول إليها فقط من داخل الصنف نفسه:

```س
صنف حساب
    خاص:
        عشري الرصيد      # لا يمكن الوصول من الخارج
        نص كلمة_السر    # لا يمكن الوصول من الخارج
    
    عام:
        دالة منشئ(عشري رصيد_ابتدائي، نص كلمة)
            الرصيد = رصيد_ابتدائي
            كلمة_السر = كلمة
        نهاية
        
        دالة احصل_الرصيد() عشري
            ارجع الرصيد
        نهاية
        
        دالة إيداع(عشري مبلغ)
            الرصيد = الرصيد + مبلغ
        نهاية
نهاية

دالة رئيسية() رقم
    متغير ح = حساب(1000.0، "سري123")
    
    # ح.الرصيد = 1000000    # ✗ خطأ! الرصيد خاص
    # ح.كلمة_السر          # ✗ خطأ! كلمة_السر خاصة
    
    ح.إيداع(500.0)         # ✓ يعمل
    اطبع(ح.احصل_الرصيد())  # ✓ يعمل
    
    ارجع 0
نهاية
```

### 3. محمي (protected)

الأعضاء المحمية يمكن الوصول إليها من الصنف نفسه والأصناف الوارثة:

```س
صنف موظف
    محمي:
        نص الاسم
        عشري الراتب_الأساسي
    
    عام:
        دالة منشئ(نص اسم، عشري راتب)
            الاسم = اسم
            الراتب_الأساسي = راتب
        نهاية
نهاية

صنف مدير يرث موظف
    خاص:
        عشري علاوة_المدير
    
    عام:
        دالة منشئ(نص اسم، عشري راتب، عشري علاوة)
            أب.منشئ(اسم، راتب)
            علاوة_المدير = علاوة
        نهاية
        
        دالة احصل_الراتب_الكلي() عشري
            # يمكن الوصول للراتب_الأساسي لأنه محمي
            ارجع الراتب_الأساسي + علاوة_المدير
        نهاية
        
        دالة اطبع_المعلومات()
            # يمكن الوصول للاسم لأنه محمي
            اطبع("المدير: ")
            اطبع(الاسم)
            اطبع(" - الراتب الكلي: ")
            اطبع(احصل_الراتب_الكلي())
        نهاية
نهاية

دالة رئيسية() رقم
    متغير م = مدير("أحمد"، 5000.0، 2000.0)
    م.اطبع_المعلومات()
    
    # م.الاسم             # ✗ خطأ! الاسم محمي
    # م.الراتب_الأساسي   # ✗ خطأ! الراتب محمي
    
    ارجع 0
نهاية
```

## المُنشئ (Constructor)

المُنشئ هو دالة خاصة تُستدعى تلقائياً عند إنشاء كائن جديد. تُستخدم لتهيئة الخصائص.

### المُنشئ الافتراضي

```س
صنف كتاب
    عام:
        نص العنوان
        نص المؤلف
        
        # المُنشئ الافتراضي (بدون معاملات)
        دالة منشئ()
            العنوان = "غير معروف"
            المؤلف = "غير معروف"
        نهاية
نهاية

دالة رئيسية() رقم
    متغير ك = كتاب()
    اطبع(ك.العنوان)  # "غير معروف"
    ارجع 0
نهاية
```

### المُنشئ بمعاملات

```س
صنف كتاب
    عام:
        نص العنوان
        نص المؤلف
        رقم عدد_الصفحات
        
        # مُنشئ بمعاملات
        دالة منشئ(نص عنوان، نص مؤلف، رقم صفحات)
            العنوان = عنوان
            المؤلف = مؤلف
            عدد_الصفحات = صفحات
        نهاية
        
        دالة اطبع_المعلومات()
            اطبع("الكتاب: ")
            اطبع(العنوان)
            اطبع(" - المؤلف: ")
            اطبع(المؤلف)
            اطبع(" - عدد الصفحات: ")
            اطبع(عدد_الصفحات)
        نهاية
نهاية

دالة رئيسية() رقم
    متغير ك1 = كتاب("البؤساء"، "فيكتور هوغو"، 1500)
    متغير ك2 = كتاب("مئة عام من العزلة"، "غابرييل ماركيز"، 400)
    
    ك1.اطبع_المعلومات()
    ك2.اطبع_المعلومات()
    
    ارجع 0
نهاية
```

### تحميل المُنشئ الزائد (Constructor Overloading)

يمكن إنشاء أكثر من مُنشئ بمعاملات مختلفة:

```س
صنف نقطة
    عام:
        عشري س
        عشري ص
        
        # مُنشئ افتراضي
        دالة منشئ()
            س = 0.0
            ص = 0.0
        نهاية
        
        # مُنشئ بمعامل واحد (نقطة على المحور)
        دالة منشئ(عشري إحداثي)
            س = إحداثي
            ص = إحداثي
        نهاية
        
        # مُنشئ بمعاملين
        دالة منشئ(عشري إكس، عشري واي)
            س = إكس
            ص = واي
        نهاية
        
        دالة اطبع()
            اطبع("(")
            اطبع(س)
            اطبع("، ")
            اطبع(ص)
            اطبع(")")
        نهاية
نهاية

دالة رئيسية() رقم
    متغير ن1 = نقطة()         # (0، 0)
    متغير ن2 = نقطة(5.0)      # (5، 5)
    متغير ن3 = نقطة(3.0، 4.0) # (3، 4)
    
    ن1.اطبع()
    ن2.اطبع()
    ن3.اطبع()
    
    ارجع 0
نهاية
```

## المُدمّر (Destructor)

المُدمّر هو دالة تُستدعى تلقائياً عندما ينتهي عمر الكائن. تُستخدم لتحرير الموارد.

```س
صنف ملف
    خاص:
        نص اسم_الملف
        منطقي مفتوح
    
    عام:
        دالة منشئ(نص اسم)
            اسم_الملف = اسم
            مفتوح = صحيح
            اطبع("تم فتح الملف: ")
            اطبع(اسم_الملف)
        نهاية
        
        دالة مدمر()
            إذا(مفتوح)
                اطبع("تم إغلاق الملف: ")
                اطبع(اسم_الملف)
                مفتوح = خطأ
            نهاية
        نهاية
        
        دالة اكتب(نص محتوى)
            إذا(مفتوح)
                اطبع("كتابة في الملف: ")
                اطبع(محتوى)
            نهاية
        نهاية
نهاية

دالة رئيسية() رقم
    متغير م = ملف("بيانات.txt")
    م.اكتب("مرحباً بالعالم!")
    # المُدمّر يُستدعى تلقائياً عند انتهاء النطاق
    ارجع 0
نهاية
```

## دوال الوصول (Getters and Setters)

دوال الوصول تُوفر طريقة آمنة للقراءة والكتابة للخصائص الخاصة:

```س
صنف طالب
    خاص:
        نص الاسم
        رقم العمر
        عشري المعدل
    
    عام:
        دالة منشئ(نص اسم)
            الاسم = اسم
            العمر = 0
            المعدل = 0.0
        نهاية
        
        # Getters (دوال القراءة)
        دالة احصل_الاسم() نص
            ارجع الاسم
        نهاية
        
        دالة احصل_العمر() رقم
            ارجع العمر
        نهاية
        
        دالة احصل_المعدل() عشري
            ارجع المعدل
        نهاية
        
        # Setters (دوال الكتابة) مع تحقق
        دالة عيّن_العمر(رقم عمر)
            إذا(عمر >= 0 و عمر <= 120)
                العمر = عمر
            وإلا
                اطبع("خطأ: العمر يجب أن يكون بين 0 و 120")
            نهاية
        نهاية
        
        دالة عيّن_المعدل(عشري معدل)
            إذا(معدل >= 0.0 و معدل <= 4.0)
                المعدل = معدل
            وإلا
                اطبع("خطأ: المعدل يجب أن يكون بين 0 و 4")
            نهاية
        نهاية
        
        دالة اطبع_المعلومات()
            اطبع("الطالب: ")
            اطبع(الاسم)
            اطبع(" - العمر: ")
            اطبع(العمر)
            اطبع(" - المعدل: ")
            اطبع(المعدل)
        نهاية
نهاية

دالة رئيسية() رقم
    متغير ط = طالب("محمد")
    
    ط.عيّن_العمر(20)
    ط.عيّن_المعدل(3.5)
    ط.اطبع_المعلومات()
    
    # محاولة تعيين قيم غير صالحة
    ط.عيّن_العمر(-5)     # خطأ: العمر يجب أن يكون بين 0 و 120
    ط.عيّن_المعدل(5.0)   # خطأ: المعدل يجب أن يكون بين 0 و 4
    
    ارجع 0
نهاية
```

## الأعضاء الثابتة (Static Members)

الأعضاء الثابتة تنتمي للصنف نفسه وليس لكائنات محددة. تُشارك بين جميع الكائنات.

```س
صنف عداد
    خاص:
        ثابت رقم العدد_الكلي = 0  # متغير ثابت للصنف
        رقم المعرف
    
    عام:
        دالة منشئ()
            العدد_الكلي = العدد_الكلي + 1
            المعرف = العدد_الكلي
            اطبع("تم إنشاء كائن رقم: ")
            اطبع(المعرف)
        نهاية
        
        دالة ثابتة احصل_العدد_الكلي() رقم
            ارجع العدد_الكلي
        نهاية
        
        دالة احصل_المعرف() رقم
            ارجع المعرف
        نهاية
نهاية

دالة رئيسية() رقم
    اطبع("العدد الكلي قبل الإنشاء: ")
    اطبع(عداد::احصل_العدد_الكلي())  # 0
    
    متغير ع1 = عداد()  # تم إنشاء كائن رقم: 1
    متغير ع2 = عداد()  # تم إنشاء كائن رقم: 2
    متغير ع3 = عداد()  # تم إنشاء كائن رقم: 3
    
    اطبع("العدد الكلي بعد الإنشاء: ")
    اطبع(عداد::احصل_العدد_الكلي())  # 3
    
    ارجع 0
نهاية
```

## الكلمة المفتاحية "هذا" (this)

الكلمة المفتاحية `هذا` تُشير للكائن الحالي. مفيدة عندما يتطابق اسم المعامل مع اسم الخاصية:

```س
صنف مستطيل
    خاص:
        عشري الطول
        عشري العرض
    
    عام:
        دالة منشئ(عشري الطول، عشري العرض)
            # استخدام "هذا" للتمييز بين الخاصية والمعامل
            هذا.الطول = الطول
            هذا.العرض = العرض
        نهاية
        
        # إرجاع الكائن الحالي (للتسلسل)
        دالة عيّن_الطول(عشري الطول) مستطيل
            هذا.الطول = الطول
            ارجع هذا
        نهاية
        
        دالة عيّن_العرض(عشري العرض) مستطيل
            هذا.العرض = العرض
            ارجع هذا
        نهاية
        
        دالة المساحة() عشري
            ارجع هذا.الطول * هذا.العرض
        نهاية
        
        دالة اطبع()
            اطبع("مستطيل: ")
            اطبع(هذا.الطول)
            اطبع(" × ")
            اطبع(هذا.العرض)
            اطبع(" = ")
            اطبع(هذا.المساحة())
        نهاية
نهاية

دالة رئيسية() رقم
    متغير م = مستطيل(10.0، 5.0)
    م.اطبع()
    
    # استخدام التسلسل (Method Chaining)
    م.عيّن_الطول(20.0).عيّن_العرض(10.0)
    م.اطبع()
    
    ارجع 0
نهاية
```

## مثال شامل: نظام إدارة مكتبة

```س
صنف كتاب
    خاص:
        نص الرقم_التسلسلي
        نص العنوان
        نص المؤلف
        منطقي متاح
    
    عام:
        دالة منشئ(نص رقم، نص عنوان، نص مؤلف)
            الرقم_التسلسلي = رقم
            هذا.العنوان = عنوان
            هذا.المؤلف = مؤلف
            متاح = صحيح
        نهاية
        
        دالة احصل_الرقم() نص
            ارجع الرقم_التسلسلي
        نهاية
        
        دالة احصل_العنوان() نص
            ارجع العنوان
        نهاية
        
        دالة هل_متاح() منطقي
            ارجع متاح
        نهاية
        
        دالة استعر() منطقي
            إذا(متاح)
                متاح = خطأ
                ارجع صحيح
            نهاية
            ارجع خطأ
        نهاية
        
        دالة أرجع()
            متاح = صحيح
        نهاية
        
        دالة اطبع()
            اطبع("[")
            اطبع(الرقم_التسلسلي)
            اطبع("] ")
            اطبع(العنوان)
            اطبع(" - ")
            اطبع(المؤلف)
            إذا(متاح)
                اطبع(" (متاح)")
            وإلا
                اطبع(" (معار)")
            نهاية
        نهاية
نهاية

صنف عضو
    خاص:
        رقم رقم_العضوية
        نص الاسم
        كتاب[] الكتب_المستعارة
        رقم عدد_المستعارة
        ثابت رقم الحد_الأقصى = 5
    
    عام:
        دالة منشئ(رقم رقم، نص اسم)
            رقم_العضوية = رقم
            الاسم = اسم
            الكتب_المستعارة = كتاب[الحد_الأقصى]
            عدد_المستعارة = 0
        نهاية
        
        دالة احصل_الاسم() نص
            ارجع الاسم
        نهاية
        
        دالة استعر_كتاب(كتاب ك) منطقي
            إذا(عدد_المستعارة >= الحد_الأقصى)
                اطبع("تجاوزت الحد الأقصى للاستعارة!")
                ارجع خطأ
            نهاية
            
            إذا(!ك.هل_متاح())
                اطبع("الكتاب غير متاح حالياً!")
                ارجع خطأ
            نهاية
            
            ك.استعر()
            الكتب_المستعارة[عدد_المستعارة] = ك
            عدد_المستعارة = عدد_المستعارة + 1
            
            اطبع(الاسم)
            اطبع(" استعار: ")
            اطبع(ك.احصل_العنوان())
            
            ارجع صحيح
        نهاية
        
        دالة أرجع_كتاب(كتاب ك)
            متغير ع = 0
            طالما(ع < عدد_المستعارة)
                إذا(الكتب_المستعارة[ع].احصل_الرقم() == ك.احصل_الرقم())
                    ك.أرجع()
                    # نقل الكتب
                    متغير ج = ع
                    طالما(ج < عدد_المستعارة - 1)
                        الكتب_المستعارة[ج] = الكتب_المستعارة[ج + 1]
                        ج = ج + 1
                    نهاية
                    عدد_المستعارة = عدد_المستعارة - 1
                    اطبع(الاسم)
                    اطبع(" أرجع: ")
                    اطبع(ك.احصل_العنوان())
                    ارجع
                نهاية
                ع = ع + 1
            نهاية
            اطبع("لم يتم العثور على الكتاب في قائمة المستعارة!")
        نهاية
        
        دالة اطبع_المستعارة()
            اطبع("الكتب المستعارة لـ ")
            اطبع(الاسم)
            اطبع(":")
            
            إذا(عدد_المستعارة == 0)
                اطبع("  لا توجد كتب مستعارة")
                ارجع
            نهاية
            
            متغير ع = 0
            طالما(ع < عدد_المستعارة)
                اطبع("  - ")
                اطبع(الكتب_المستعارة[ع].احصل_العنوان())
                ع = ع + 1
            نهاية
        نهاية
نهاية

صنف مكتبة
    خاص:
        نص الاسم
        كتاب[] الكتب
        رقم عدد_الكتب
        رقم السعة
    
    عام:
        دالة منشئ(نص اسم، رقم سعة)
            الاسم = اسم
            السعة = سعة
            الكتب = كتاب[سعة]
            عدد_الكتب = 0
        نهاية
        
        دالة أضف_كتاب(كتاب ك) منطقي
            إذا(عدد_الكتب >= السعة)
                اطبع("المكتبة ممتلئة!")
                ارجع خطأ
            نهاية
            
            الكتب[عدد_الكتب] = ك
            عدد_الكتب = عدد_الكتب + 1
            اطبع("تمت إضافة: ")
            اطبع(ك.احصل_العنوان())
            ارجع صحيح
        نهاية
        
        دالة ابحث_بالعنوان(نص عنوان) كتاب
            متغير ع = 0
            طالما(ع < عدد_الكتب)
                إذا(الكتب[ع].احصل_العنوان() == عنوان)
                    ارجع الكتب[ع]
                نهاية
                ع = ع + 1
            نهاية
            ارجع فارغ
        نهاية
        
        دالة اطبع_الكتب()
            اطبع("=== مكتبة ")
            اطبع(الاسم)
            اطبع(" ===")
            
            متغير ع = 0
            طالما(ع < عدد_الكتب)
                الكتب[ع].اطبع()
                ع = ع + 1
            نهاية
        نهاية
        
        دالة عدد_المتاحة() رقم
            متغير عداد = 0
            متغير ع = 0
            طالما(ع < عدد_الكتب)
                إذا(الكتب[ع].هل_متاح())
                    عداد = عداد + 1
                نهاية
                ع = ع + 1
            نهاية
            ارجع عداد
        نهاية
نهاية

دالة رئيسية() رقم
    # إنشاء المكتبة
    متغير مكتبتنا = مكتبة("المكتبة المركزية"، 100)
    
    # إضافة كتب
    متغير ك1 = كتاب("001"، "البؤساء"، "فيكتور هوغو")
    متغير ك2 = كتاب("002"، "الأمير الصغير"، "أنطوان دو سانت")
    متغير ك3 = كتاب("003"، "ألف ليلة وليلة"، "مجهول")
    
    مكتبتنا.أضف_كتاب(ك1)
    مكتبتنا.أضف_كتاب(ك2)
    مكتبتنا.أضف_كتاب(ك3)
    
    # عرض الكتب
    مكتبتنا.اطبع_الكتب()
    
    # إنشاء عضو
    متغير عضو1 = عضو(1، "أحمد")
    
    # استعارة كتب
    عضو1.استعر_كتاب(ك1)
    عضو1.استعر_كتاب(ك2)
    
    # عرض الكتب بعد الاستعارة
    مكتبتنا.اطبع_الكتب()
    
    # عرض الكتب المستعارة للعضو
    عضو1.اطبع_المستعارة()
    
    # إرجاع كتاب
    عضو1.أرجع_كتاب(ك1)
    
    # عرض الحالة النهائية
    مكتبتنا.اطبع_الكتب()
    عضو1.اطبع_المستعارة()
    
    اطبع("عدد الكتب المتاحة: ")
    اطبع(مكتبتنا.عدد_المتاحة())
```

## ملخص

الأصناف والكائنات في لغة ص تُقدم لك:

- **مستويات الوصول**: عام، خاص، محمي
- **المُنشئات**: لتهيئة الكائنات
- **المُدمّرات**: لتحرير الموارد
- **دوال الوصول**: للقراءة والكتابة الآمنة
- **الأعضاء الثابتة**: للبيانات المشتركة
- **الكلمة "هذا"**: للإشارة للكائن الحالي

استخدم هذه المفاهيم لبناء برامج منظمة وقابلة للصيانة!
