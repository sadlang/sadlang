# أمثلة عملية على القوالب

## مقدمة

هذا الملف يحتوي على أمثلة عملية متنوعة لاستخدام القوالب في لغة ص. الأمثلة مُصنفة حسب المستوى والموضوع لتُسهّل التعلم والفهم.

## أمثلة للمبتدئين

### 1. دالة تبديل القيم

```س
قالب<نوع ت> دالة بدّل(ت أ، ت ب)
    متغير مؤقت = أ
    أ = ب
    ب = مؤقت
نهاية

دالة رئيسية() رقم
    # تبديل أرقام
    متغير س = 10
    متغير ص = 20
    اطبع("قبل: س=")
    اطبع(س)
    اطبع(" ص=")
    اطبع(ص)
    
    بدّل<رقم>(س، ص)
    
    اطبع("بعد: س=")
    اطبع(س)
    اطبع(" ص=")
    اطبع(ص)
    
    # تبديل نصوص
    متغير أ = "مرحباً"
    متغير ب = "عالم"
    بدّل<نص>(أ، ب)
    
    ارجع 0
نهاية
```

### 4. صندوق القيمة

```س
قالب<نوع ت> صنف صندوق
    خاص:
        ت القيمة
        منطقي يحتوي_قيمة
    
    عام:
        دالة منشئ()
            يحتوي_قيمة = خطأ
        نهاية
        
        دالة منشئ(ت قيمة_ابتدائية)
            القيمة = قيمة_ابتدائية
            يحتوي_قيمة = صحيح
        نهاية
        
        دالة ضع(ت قيمة)
            القيمة = قيمة
            يحتوي_قيمة = صحيح
        نهاية
        
        دالة احصل() ت
            ارجع القيمة
        نهاية
        
        دالة فارغ() منطقي
            ارجع !يحتوي_قيمة
        نهاية
        
        دالة افرغ()
            يحتوي_قيمة = خطأ
        نهاية
نهاية

دالة رئيسية() رقم
    متغير صندوق_رقم = صندوق<رقم>(42)
    متغير صندوق_نص = صندوق<نص>()
    
    اطبع("صندوق الرقم فارغ؟ ")
    اطبع(صندوق_رقم.فارغ())  # خطأ
    
    اطبع("قيمة صندوق الرقم: ")
    اطبع(صندوق_رقم.احصل())  # 42
    
    صندوق_نص.ضع("مرحباً")
    اطبع("صندوق النص: ")
    اطبع(صندوق_نص.احصل())
    
    ارجع 0

```

## أمثلة متوسطة المستوى

### 5. المصفوفة الديناميكية

```س
قالب<نوع ت> صنف مصفوفة_ديناميكية
    خاص:
        ت[] العناصر
        رقم الحجم
        رقم السعة
    
    عام:
        دالة منشئ(رقم سعة_ابتدائية)
            السعة = سعة_ابتدائية
            العناصر = ت[السعة]
            الحجم = 0
        نهاية
        
        دالة أضف(ت عنصر)
            إذا(الحجم >= السعة)
                # توسيع المصفوفة
                وسّع()
            نهاية
            العناصر[الحجم] = عنصر
            الحجم = الحجم + 1
        نهاية
        
        دالة احصل(رقم فهرس) ت
            إذا(فهرس < 0 أو فهرس >= الحجم)
                اطبع("خطأ: فهرس خارج الحدود!")
            نهاية
            ارجع العناصر[فهرس]
        نهاية
        
        دالة عيّن(رقم فهرس، ت قيمة)
            إذا(فهرس >= 0 و فهرس < الحجم)
                العناصر[فهرس] = قيمة
            نهاية
        نهاية
        
        دالة احصل_الحجم() رقم
            ارجع الحجم
        نهاية
        
        دالة فارغة() منطقي
            ارجع الحجم == 0
        نهاية
        
        خاص:
        دالة وسّع()
            متغير سعة_جديدة = السعة * 2
            متغير عناصر_جديدة = ت[سعة_جديدة]
            
            متغير ع = 0
            طالما(ع < الحجم)
                عناصر_جديدة[ع] = العناصر[ع]
                ع = ع + 1
            نهاية
            
            العناصر = عناصر_جديدة
            السعة = سعة_جديدة
        نهاية
نهاية

دالة رئيسية() رقم
    متغير أرقام = مصفوفة_ديناميكية<رقم>(2)
    
    أرقام.أضف(10)
    أرقام.أضف(20)
    أرقام.أضف(30)  # ستتوسع تلقائياً
    أرقام.أضف(40)
    
    اطبع("عناصر المصفوفة:")
    متغير ع = 0
    طالما(ع < أرقام.احصل_الحجم())
        اطبع(أرقام.احصل(ع))
        ع = ع + 1
    نهاية
    
    ارجع 0
نهاية
```

### 6. البحث والترتيب

```س
# البحث الخطي
قالب<نوع ت> دالة ابحث(ت[] مصفوفة، رقم حجم، ت قيمة) رقم
    متغير ع = 0
    طالما(ع < حجم)
        إذا(مصفوفة[ع] == قيمة)
            ارجع ع
        نهاية
        ع = ع + 1
    نهاية
    ارجع -1
نهاية

# الترتيب الفقاعي
قالب<نوع ت> دالة رتّب(ت[] مصفوفة، رقم حجم)
    متغير ع = 0
    طالما(ع < حجم - 1)
        متغير ج = 0
        طالما(ج < حجم - ع - 1)
            إذا(مصفوفة[ج] > مصفوفة[ج + 1])
                # تبديل
                متغير مؤقت = مصفوفة[ج]
                مصفوفة[ج] = مصفوفة[ج + 1]
                مصفوفة[ج + 1] = مؤقت
            نهاية
            ج = ج + 1
        نهاية
        ع = ع + 1
    نهاية
نهاية

# طباعة مصفوفة
قالب<نوع ت> دالة اطبع_مصفوفة(ت[] مصفوفة، رقم حجم)
    اطبع("[")
    متغير ع = 0
    طالما(ع < حجم)
        اطبع(مصفوفة[ع])
        إذا(ع < حجم - 1)
            اطبع("، ")
        نهاية
        ع = ع + 1
    نهاية
    اطبع("]")
نهاية

دالة رئيسية() رقم
    متغير أرقام = [64، 34، 25، 12، 22، 11، 90]
    
    اطبع("قبل الترتيب: ")
    اطبع_مصفوفة<رقم>(أرقام، 7)
    
    رتّب<رقم>(أرقام، 7)
    
    اطبع("بعد الترتيب: ")
    اطبع_مصفوفة<رقم>(أرقام، 7)
    
    متغير موقع = ابحث<رقم>(أرقام، 7، 34)
    اطبع("موقع الرقم 34: ")
    اطبع(موقع)
    
    ارجع 0
نهاية
```

## أمثلة متقدمة

### 8. المكدس

```س
قالب<نوع ت> صنف مكدس
    خاص:
        ت[] العناصر
        رقم القمة
        رقم السعة
    
    عام:
        دالة منشئ(رقم سعة)
            السعة = سعة
            العناصر = ت[سعة]
            القمة = -1
        نهاية
        
        دالة ادفع(ت عنصر) منطقي
            إذا(ممتلئ())
                اطبع("خطأ: المكدس ممتلئ!")
                ارجع خطأ
            نهاية
            
            القمة = القمة + 1
            العناصر[القمة] = عنصر
            ارجع صحيح
        نهاية
        
        دالة اسحب() ت
            إذا(فارغ())
                اطبع("خطأ: المكدس فارغ!")
            نهاية
            
            متغير قيمة = العناصر[القمة]
            القمة = القمة - 1
            ارجع قيمة
        نهاية
        
        دالة قمة() ت
            إذا(فارغ())
                اطبع("خطأ: المكدس فارغ!")
            نهاية
            ارجع العناصر[القمة]
        نهاية
        
        دالة فارغ() منطقي
            ارجع القمة < 0
        نهاية
        
        دالة ممتلئ() منطقي
            ارجع القمة >= السعة - 1
        نهاية
        
        دالة الحجم() رقم
            ارجع القمة + 1
        نهاية
نهاية

# استخدام المكدس لعكس نص
دالة اعكس_نص(نص النص) نص
    متغير م = مكدس<حرف>(100)
    
    # ادفع كل حرف
    متغير ع = 0
    طالما(ع < طول(النص))
        م.ادفع(النص[ع])
        ع = ع + 1
    نهاية
    
    # اسحب لإنشاء النص المعكوس
    متغير نتيجة = ""
    طالما(!م.فارغ())
        نتيجة = نتيجة + م.اسحب()
    نهاية
    
    ارجع نتيجة
نهاية

دالة رئيسية() رقم
    متغير م = مكدس<رقم>(5)
    
    م.ادفع(1)
    م.ادفع(2)
    م.ادفع(3)
    
    اطبع("القمة: ")
    اطبع(م.قمة())
    
    اطبع("السحب بالترتيب:")
    طالما(!م.فارغ())
        اطبع(م.اسحب())
    نهاية
    
    # عكس نص
    اطبع("عكس 'مرحبا': ")
    اطبع(اعكس_نص("مرحبا"))
    
    ارجع 0
نهاية
```

### 9. الطابور

```س
قالب<نوع ت> صنف طابور
    خاص:
        ت[] العناصر
        رقم الأمام
        رقم الخلف
        رقم السعة
        رقم العدد
    
    عام:
        دالة منشئ(رقم سعة)
            السعة = سعة
            العناصر = ت[سعة]
            الأمام = 0
            الخلف = -1
            العدد = 0
        نهاية
        
        دالة أدخل(ت عنصر) منطقي
            إذا(ممتلئ())
                اطبع("خطأ: الطابور ممتلئ!")
                ارجع خطأ
            نهاية
            
            الخلف = (الخلف + 1) % السعة
            العناصر[الخلف] = عنصر
            العدد = العدد + 1
            ارجع صحيح
        نهاية
        
        دالة أخرج() ت
            إذا(فارغ())
                اطبع("خطأ: الطابور فارغ!")
            نهاية
            
            متغير قيمة = العناصر[الأمام]
            الأمام = (الأمام + 1) % السعة
            العدد = العدد - 1
            ارجع قيمة
        نهاية
        
        دالة مقدمة() ت
            إذا(فارغ())
                اطبع("خطأ: الطابور فارغ!")
            نهاية
            ارجع العناصر[الأمام]
        نهاية
        
        دالة فارغ() منطقي
            ارجع العدد == 0
        نهاية
        
        دالة ممتلئ() منطقي
            ارجع العدد >= السعة
        نهاية
        
        دالة الحجم() رقم
            ارجع العدد
        نهاية
نهاية

دالة رئيسية() رقم
    # طابور للمهام
    متغير مهام = طابور<نص>(5)
    
    مهام.أدخل("مهمة 1")
    مهام.أدخل("مهمة 2")
    مهام.أدخل("مهمة 3")
    
    اطبع("معالجة المهام بالترتيب:")
    طالما(!مهام.فارغ())
        اطبع("معالجة: ")
        اطبع(مهام.أخرج())
    نهاية
    
    ارجع 0
نهاية
```

### 10. القائمة المتصلة

```س
قالب<نوع ت> صنف عقدة
    عام:
        ت البيانات
        عقدة<ت> التالي
        
        دالة منشئ(ت بيانات)
            البيانات = بيانات
            التالي = فارغ
        نهاية
نهاية

قالب<نوع ت> صنف قائمة_متصلة
    خاص:
        عقدة<ت> الرأس
        عقدة<ت> الذيل
        رقم العدد
    
    عام:
        دالة منشئ()
            الرأس = فارغ
            الذيل = فارغ
            العدد = 0
        نهاية
        
        دالة أضف_نهاية(ت قيمة)
            متغير جديدة = عقدة<ت>(قيمة)
            
            إذا(الرأس == فارغ)
                الرأس = جديدة
                الذيل = جديدة
            وإلا
                الذيل.التالي = جديدة
                الذيل = جديدة
            نهاية
            
            العدد = العدد + 1
        نهاية
        
        دالة أضف_بداية(ت قيمة)
            متغير جديدة = عقدة<ت>(قيمة)
            جديدة.التالي = الرأس
            الرأس = جديدة
            
            إذا(الذيل == فارغ)
                الذيل = جديدة
            نهاية
            
            العدد = العدد + 1
        نهاية
        
        دالة احصل(رقم فهرس) ت
            متغير حالي = الرأس
            متغير ع = 0
            
            طالما(حالي != فارغ و ع < فهرس)
                حالي = حالي.التالي
                ع = ع + 1
            نهاية
            
            إذا(حالي != فارغ)
                ارجع حالي.البيانات
            نهاية
            
            اطبع("خطأ: فهرس خارج الحدود!")
        نهاية
        
        دالة يحتوي(ت قيمة) منطقي
            متغير حالي = الرأس
            
            طالما(حالي != فارغ)
                إذا(حالي.البيانات == قيمة)
                    ارجع صحيح
                نهاية
                حالي = حالي.التالي
            نهاية
            
            ارجع خطأ
        نهاية
        
        دالة الحجم() رقم
            ارجع العدد
        نهاية
        
        دالة فارغة() منطقي
            ارجع العدد == 0
        نهاية
        
        دالة اطبع()
            اطبع("[")
            متغير حالي = الرأس
            
            طالما(حالي != فارغ)
                اطبع(حالي.البيانات)
                إذا(حالي.التالي != فارغ)
                    اطبع(" -> ")
                نهاية
                حالي = حالي.التالي
            نهاية
            
            اطبع("]")
        نهاية
نهاية

دالة رئيسية() رقم
    متغير قائمة = قائمة_متصلة<رقم>()
    
    قائمة.أضف_نهاية(10)
    قائمة.أضف_نهاية(20)
    قائمة.أضف_نهاية(30)
    قائمة.أضف_بداية(5)
    
    اطبع("القائمة: ")
    قائمة.اطبع()
    
    اطبع("العنصر في الفهرس 2: ")
    اطبع(قائمة.احصل(2))
    
    اطبع("هل تحتوي على 20؟ ")
    اطبع(قائمة.يحتوي(20))
    
    ارجع 0
نهاية
```

## تمارين

### تمرين 1
اكتب دالة قالب `عكس_مصفوفة` تعكس عناصر مصفوفة من أي نوع.

### تمرين 2
اكتب صنف قالب `مجموعة` يُخزن عناصر فريدة (بدون تكرار).

### تمرين 3
اكتب دالة قالب `فلتر` تُرشح عناصر مصفوفة بناءً على شرط.

## ملخص

هذه الأمثلة تُغطي:
- دوال القوالب الأساسية
- أصناف القوالب
- هياكل البيانات (مكدس، طابور، قائمة متصلة)
- خوارزميات البحث والترتيب
- القوالب مع معاملات متعددة

مارس هذه الأمثلة وطبّقها في مشاريعك!
