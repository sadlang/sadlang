# أنماط التصميم في لغة ص

## مقدمة

أنماط التصميم (Design Patterns) هي حلول مُجربة لمشاكل برمجية شائعة. تُساعد هذه الأنماط في كتابة كود منظم، قابل لإعادة الاستخدام، وسهل الصيانة. في هذا الدليل سنتعلم كيفية تطبيق أشهر أنماط التصميم في لغة ص.

## 1. نمط المفرد (Singleton)

### الشرح

نمط المفرد يضمن وجود نسخة واحدة فقط من الصنف في البرنامج كله. يُستخدم عندما نحتاج إلى نقطة وصول عالمية لمورد واحد مثل إعدادات التطبيق أو اتصال قاعدة البيانات.

### التنفيذ

```س
صنف إعدادات
    خاص:
        ثابت إعدادات النسخة = عدم  # النسخة الوحيدة
        نص اللغة
        نص الوضع
        رقم حجم_الخط
    
    عام:
        # منشئ خاص - لا يمكن إنشاء نسخ من الخارج
        دالة منشئ خاص()
            اللغة = "العربية"
            الوضع = "فاتح"
            حجم_الخط = 14
        نهاية
        
        # الحصول على النسخة الوحيدة
        دالة ثابتة احصل_النسخة() إعدادات
            إذا(النسخة == عدم)
                النسخة = إعدادات()
            نهاية
            ارجع النسخة
        نهاية
        
        # الدوال العامة
        دالة احصل_اللغة() نص
            ارجع اللغة
        نهاية
        
        دالة عيّن_اللغة(نص لغة)
            اللغة = لغة
        نهاية
        
        دالة احصل_الوضع() نص
            ارجع الوضع
        نهاية
        
        دالة عيّن_الوضع(نص وضع)
            الوضع = وضع
        نهاية
نهاية

دالة رئيسية() رقم
    # الحصول على نفس النسخة في كل مكان
    متغير إعدادات1 = إعدادات::احصل_النسخة()
    إعدادات1.عيّن_اللغة("English")
    
    متغير إعدادات2 = إعدادات::احصل_النسخة()
    اطبع(إعدادات2.احصل_اللغة())  # "English" - نفس النسخة!
    
    ارجع 0
نهاية
```

### مثال عملي: مدير السجلات

```س
صنف مدير_السجلات
    خاص:
        ثابت مدير_السجلات النسخة = عدم
        مصفوفة<نص> السجلات
        نص مستوى_التسجيل  # "معلومات" أو "تحذير" أو "خطأ"
    
    عام:
        دالة منشئ خاص()
            السجلات = مصفوفة<نص>()
            مستوى_التسجيل = "معلومات"
        نهاية
        
        دالة ثابتة احصل_النسخة() مدير_السجلات
            إذا(النسخة == عدم)
                النسخة = مدير_السجلات()
            نهاية
            ارجع النسخة
        نهاية
        
        دالة سجّل_معلومة(نص رسالة)
            متغير سجل = "[معلومة] " + رسالة
            السجلات.أضف(سجل)
            اطبع(سجل)
        نهاية
        
        دالة سجّل_تحذير(نص رسالة)
            متغير سجل = "[تحذير] " + رسالة
            السجلات.أضف(سجل)
            اطبع(سجل)
        نهاية
        
        دالة سجّل_خطأ(نص رسالة)
            متغير سجل = "[خطأ] " + رسالة
            السجلات.أضف(سجل)
            اطبع(سجل)
        نهاية
        
        دالة احصل_كل_السجلات() مصفوفة<نص>
            ارجع السجلات
        نهاية
نهاية

# الاستخدام في أي مكان في البرنامج
دالة معالجة_الطلب(رقم رقم_الطلب)
    مدير_السجلات::احصل_النسخة().سجّل_معلومة("بدء معالجة الطلب: " + رقم_الطلب)
    # معالجة...
    مدير_السجلات::احصل_النسخة().سجّل_معلومة("انتهاء معالجة الطلب: " + رقم_الطلب)
نهاية
```

## 2. نمط المصنع (Factory)

### الشرح

نمط المصنع يُوفر واجهة لإنشاء الكائنات دون تحديد الصنف الدقيق. يُفيد عندما يكون لدينا عدة أنواع من الكائنات ونريد إنشاءها بناءً على شرط معين.

### التنفيذ الأساسي

```س
# الصنف الأساسي
صنف مجرد زر
    عام:
        دالة مجردة ارسم()
        دالة مجردة عند_النقر()
نهاية

# الأنواع المختلفة
صنف زر_ويندوز يرث زر
    عام:
        دالة ارسم()
            اطبع("رسم زر بنمط ويندوز")
        نهاية
        
        دالة عند_النقر()
            اطبع("نقر على زر ويندوز")
        نهاية
نهاية

صنف زر_ماك يرث زر
    عام:
        دالة ارسم()
            اطبع("رسم زر بنمط ماك")
        نهاية
        
        دالة عند_النقر()
            اطبع("نقر على زر ماك")
        نهاية
نهاية

صنف زر_لينكس يرث زر
    عام:
        دالة ارسم()
            اطبع("رسم زر بنمط لينكس")
        نهاية
        
        دالة عند_النقر()
            اطبع("نقر على زر لينكس")
        نهاية
نهاية

# المصنع
صنف مصنع_الأزرار
    عام:
        دالة ثابتة أنشئ_زر(نص نظام_التشغيل) زر
            إذا(نظام_التشغيل == "ويندوز")
                ارجع زر_ويندوز()
            وإلا إذا(نظام_التشغيل == "ماك")
                ارجع زر_ماك()
            وإلا إذا(نظام_التشغيل == "لينكس")
                ارجع زر_لينكس()
            نهاية
            ارجع زر_ويندوز()  # افتراضي
        نهاية
نهاية

دالة رئيسية() رقم
    متغير نظام = "ويندوز"
    متغير زر = مصنع_الأزرار::أنشئ_زر(نظام)
    زر.ارسم()
    زر.عند_النقر()
    
    ارجع 0
نهاية
```

### مثال عملي: مصنع المستندات

```س
صنف مجرد مستند
    خاص:
        نص العنوان
        نص المحتوى
    
    عام:
        دالة عيّن_العنوان(نص عنوان)
            العنوان = عنوان
        نهاية
        
        دالة عيّن_المحتوى(نص محتوى)
            المحتوى = محتوى
        نهاية
        
        دالة احصل_العنوان() نص
            ارجع العنوان
        نهاية
        
        دالة احصل_المحتوى() نص
            ارجع المحتوى
        نهاية
        
        دالة مجردة احفظ(نص مسار)
        دالة مجردة اطبع()
نهاية

صنف مستند_PDF يرث مستند
    عام:
        دالة احفظ(نص مسار)
            اطبع("حفظ كـ PDF في: " + مسار)
        نهاية
        
        دالة اطبع()
            اطبع("طباعة مستند PDF")
            اطبع("العنوان: " + احصل_العنوان())
        نهاية
نهاية

صنف مستند_Word يرث مستند
    عام:
        دالة احفظ(نص مسار)
            اطبع("حفظ كـ Word في: " + مسار)
        نهاية
        
        دالة اطبع()
            اطبع("طباعة مستند Word")
            اطبع("العنوان: " + احصل_العنوان())
        نهاية
نهاية

صنف مستند_HTML يرث مستند
    عام:
        دالة احفظ(نص مسار)
            اطبع("حفظ كـ HTML في: " + مسار)
        نهاية
        
        دالة اطبع()
            اطبع("عرض في المتصفح")
            اطبع("<h1>" + احصل_العنوان() + "</h1>")
        نهاية
نهاية

صنف مصنع_المستندات
    عام:
        دالة ثابتة أنشئ_مستند(نص نوع) مستند
            إذا(نوع == "pdf")
                ارجع مستند_PDF()
            وإلا إذا(نوع == "word")
                ارجع مستند_Word()
            وإلا إذا(نوع == "html")
                ارجع مستند_HTML()
            نهاية
            ارجع مستند_PDF()  # افتراضي
        نهاية
نهاية

دالة رئيسية() رقم
    متغير مستند = مصنع_المستندات::أنشئ_مستند("html")
    مستند.عيّن_العنوان("تقرير المبيعات")
    مستند.عيّن_المحتوى("محتوى التقرير...")
    مستند.احفظ("report")
    مستند.اطبع()
    
    ارجع 0
نهاية
```

## 3. نمط المراقب (Observer)

### الشرح

نمط المراقب يُحدد علاقة واحد-إلى-كثير بين الكائنات، بحيث عندما يتغير حالة كائن واحد، يتم إبلاغ جميع الكائنات المعتمدة عليه تلقائياً.

### التنفيذ

```س
# واجهة المراقب
صنف مجرد مراقب
    عام:
        دالة مجردة حدّث(نص رسالة)
نهاية

# الموضوع (المُراقَب)
صنف موضوع
    خاص:
        مصفوفة<مراقب> المراقبون
        نص الحالة
    
    عام:
        دالة منشئ()
            المراقبون = مصفوفة<مراقب>()
        نهاية
        
        دالة أضف_مراقب(مراقب م)
            المراقبون.أضف(م)
        نهاية
        
        دالة أزل_مراقب(مراقب م)
            # إزالة المراقب من القائمة
        نهاية
        
        دالة أبلغ_الجميع()
            لكل م في المراقبون
                م.حدّث(الحالة)
            نهاية
        نهاية
        
        دالة عيّن_الحالة(نص حالة_جديدة)
            الحالة = حالة_جديدة
            أبلغ_الجميع()  # إبلاغ تلقائي عند التغيير
        نهاية
نهاية
```

### مثال عملي: نظام الإشعارات

```س
# المراقب الأساسي
صنف مجرد مشترك
    خاص:
        نص الاسم
    
    عام:
        دالة منشئ(نص اسم)
            الاسم = اسم
        نهاية
        
        دالة احصل_الاسم() نص
            ارجع الاسم
        نهاية
        
        دالة مجردة استقبل_إشعار(نص عنوان، نص محتوى)
نهاية

# أنواع مختلفة من المشتركين
صنف مشترك_بريد يرث مشترك
    خاص:
        نص البريد
    
    عام:
        دالة منشئ(نص اسم، نص بريد)
            أساس(اسم)
            البريد = بريد
        نهاية
        
        دالة استقبل_إشعار(نص عنوان، نص محتوى)
            اطبع("إرسال بريد إلى: " + البريد)
            اطبع("الموضوع: " + عنوان)
            اطبع("المحتوى: " + محتوى)
        نهاية
نهاية

صنف مشترك_SMS يرث مشترك
    خاص:
        نص رقم_الهاتف
    
    عام:
        دالة منشئ(نص اسم، نص رقم)
            أساس(اسم)
            رقم_الهاتف = رقم
        نهاية
        
        دالة استقبل_إشعار(نص عنوان، نص محتوى)
            اطبع("إرسال SMS إلى: " + رقم_الهاتف)
            اطبع("الرسالة: " + عنوان + " - " + محتوى)
        نهاية
نهاية

صنف مشترك_تطبيق يرث مشترك
    عام:
        دالة منشئ(نص اسم)
            أساس(اسم)
        نهاية
        
        دالة استقبل_إشعار(نص عنوان، نص محتوى)
            اطبع("إشعار في التطبيق للمستخدم: " + احصل_الاسم())
            اطبع("العنوان: " + عنوان)
            اطبع("المحتوى: " + محتوى)
        نهاية
نهاية

# الناشر
صنف قناة_أخبار
    خاص:
        نص اسم_القناة
        مصفوفة<مشترك> المشتركون
    
    عام:
        دالة منشئ(نص اسم)
            اسم_القناة = اسم
            المشتركون = مصفوفة<مشترك>()
        نهاية
        
        دالة اشترك(مشترك م)
            المشتركون.أضف(م)
            اطبع(م.احصل_الاسم() + " اشترك في قناة " + اسم_القناة)
        نهاية
        
        دالة ألغِ_الاشتراك(مشترك م)
            # إزالة من القائمة
            اطبع(م.احصل_الاسم() + " ألغى اشتراكه")
        نهاية
        
        دالة انشر_خبر(نص عنوان، نص محتوى)
            اطبع("نشر خبر جديد: " + عنوان)
            اطبع("---")
            
            لكل مشترك في المشتركون
                مشترك.استقبل_إشعار(عنوان، محتوى)
                اطبع("---")
            نهاية
        نهاية
نهاية

دالة رئيسية() رقم
    متغير قناة = قناة_أخبار("الأخبار التقنية")
    
    متغير أحمد = مشترك_بريد("أحمد", "ahmad@email.com")
    متغير سارة = مشترك_SMS("سارة", "0501234567")
    متغير محمد = مشترك_تطبيق("محمد")
    
    قناة.اشترك(أحمد)
    قناة.اشترك(سارة)
    قناة.اشترك(محمد)
    
    قناة.انشر_خبر("إصدار جديد من لغة ص!", "تم إصدار النسخة 2.0 من لغة البرمجة ص")
    
    ارجع 0
نهاية
```

## 4. نمط الإستراتيجية (Strategy)

### الشرح

نمط الإستراتيجية يُتيح تعريف عائلة من الخوارزميات، وتغليف كل منها، وجعلها قابلة للتبديل. يُتيح للخوارزمية أن تختلف بشكل مستقل عن العملاء الذين يستخدمونها.

### التنفيذ

```س
# واجهة الإستراتيجية
صنف مجرد إستراتيجية_الدفع
    عام:
        دالة مجردة ادفع(عشري مبلغ) منطقي
        دالة مجردة احصل_الاسم() نص
نهاية

# الإستراتيجيات المختلفة
صنف دفع_نقدي يرث إستراتيجية_الدفع
    عام:
        دالة ادفع(عشري مبلغ) منطقي
            اطبع("الدفع نقداً: " + مبلغ + " ريال")
            ارجع صحيح
        نهاية
        
        دالة احصل_الاسم() نص
            ارجع "نقدي"
        نهاية
نهاية

صنف دفع_بطاقة يرث إستراتيجية_الدفع
    خاص:
        نص رقم_البطاقة
        نص تاريخ_الانتهاء
    
    عام:
        دالة منشئ(نص رقم، نص تاريخ)
            رقم_البطاقة = رقم
            تاريخ_الانتهاء = تاريخ
        نهاية
        
        دالة ادفع(عشري مبلغ) منطقي
            اطبع("الدفع بالبطاقة: " + مبلغ + " ريال")
            اطبع("رقم البطاقة: ****" + رقم_البطاقة.آخر(4))
            # التحقق من البطاقة...
            ارجع صحيح
        نهاية
        
        دالة احصل_الاسم() نص
            ارجع "بطاقة"
        نهاية
نهاية

صنف دفع_محفظة يرث إستراتيجية_الدفع
    خاص:
        نص رقم_المحفظة
        عشري الرصيد
    
    عام:
        دالة منشئ(نص رقم، عشري رصيد)
            رقم_المحفظة = رقم
            الرصيد = رصيد
        نهاية
        
        دالة ادفع(عشري مبلغ) منطقي
            إذا(مبلغ > الرصيد)
                اطبع("رصيد المحفظة غير كافٍ")
                ارجع خطأ
            نهاية
            
            الرصيد = الرصيد - مبلغ
            اطبع("الدفع من المحفظة: " + مبلغ + " ريال")
            اطبع("الرصيد المتبقي: " + الرصيد)
            ارجع صحيح
        نهاية
        
        دالة احصل_الاسم() نص
            ارجع "محفظة إلكترونية"
        نهاية
نهاية

# السياق الذي يستخدم الإستراتيجية
صنف سلة_التسوق
    خاص:
        مصفوفة<نص> المنتجات
        عشري الإجمالي
        إستراتيجية_الدفع طريقة_الدفع
    
    عام:
        دالة منشئ()
            المنتجات = مصفوفة<نص>()
            الإجمالي = 0.0
        نهاية
        
        دالة أضف_منتج(نص منتج، عشري سعر)
            المنتجات.أضف(منتج)
            الإجمالي = الإجمالي + سعر
        نهاية
        
        دالة عيّن_طريقة_الدفع(إستراتيجية_الدفع طريقة)
            طريقة_الدفع = طريقة
        نهاية
        
        دالة أتمم_الشراء() منطقي
            إذا(طريقة_الدفع == عدم)
                اطبع("الرجاء اختيار طريقة الدفع")
                ارجع خطأ
            نهاية
            
            اطبع("=== إتمام الشراء ===")
            اطبع("عدد المنتجات: " + المنتجات.الطول())
            اطبع("الإجمالي: " + الإجمالي + " ريال")
            اطبع("طريقة الدفع: " + طريقة_الدفع.احصل_الاسم())
            
            ارجع طريقة_الدفع.ادفع(الإجمالي)
        نهاية
نهاية

دالة رئيسية() رقم
    متغير سلة = سلة_التسوق()
    سلة.أضف_منتج("هاتف ذكي", 2500.0)
    سلة.أضف_منتج("سماعات", 350.0)
    
    # الدفع ببطاقة
    متغير بطاقة = دفع_بطاقة("1234567890123456", "12/25")
    سلة.عيّن_طريقة_الدفع(بطاقة)
    سلة.أتمم_الشراء()
    
    اطبع("")
    
    # تغيير طريقة الدفع بسهولة
    متغير محفظة = دفع_محفظة("0501234567", 5000.0)
    سلة.عيّن_طريقة_الدفع(محفظة)
    سلة.أتمم_الشراء()
    
    ارجع 0
نهاية
```

## 5. نمط المُزخرف (Decorator)

### الشرح

نمط المُزخرف يُتيح إضافة سلوكيات جديدة للكائنات ديناميكياً عن طريق تغليفها في كائنات "مُزخرفة". يُوفر بديلاً مرناً للوراثة لتوسيع الوظائف.

### التنفيذ

```س
# المكون الأساسي
صنف مجرد قهوة
    عام:
        دالة مجردة احصل_الوصف() نص
        دالة مجردة احصل_السعر() عشري
نهاية

# القهوة الأساسية
صنف قهوة_عادية يرث قهوة
    عام:
        دالة احصل_الوصف() نص
            ارجع "قهوة عادية"
        نهاية
        
        دالة احصل_السعر() عشري
            ارجع 10.0
        نهاية
نهاية
```

## 6. نمط المُحول (Adapter)

### الشرح

نمط المُحول يُتيح للأصناف ذات الواجهات غير المتوافقة العمل معاً. يُحول واجهة صنف إلى واجهة أخرى يتوقعها العميل.

## ملخص أنماط التصميم

| النمط | الغرض | متى يُستخدم |
|-------|-------|-------------|
| المفرد | نسخة واحدة من الصنف | إعدادات، اتصال قاعدة البيانات |
| المصنع | إنشاء كائنات بدون تحديد الصنف | إنشاء كائنات حسب شرط |
| المراقب | إبلاغ الكائنات بالتغييرات | نظام الإشعارات، الأحداث |
| الإستراتيجية | تبديل الخوارزميات | طرق دفع، خوارزميات ترتيب |
| المُزخرف | إضافة سلوكيات جديدة | إضافة ميزات اختيارية |
| المُحول | توافق الواجهات | دمج أنظمة قديمة |

## نصائح للاختيار

1. **استخدم المفرد** عندما تحتاج نقطة وصول عالمية لمورد واحد
2. **استخدم المصنع** عندما يعتمد نوع الكائن على شرط وقت التشغيل
3. **استخدم المراقب** عندما تحتاج إبلاغ عدة كائنات بتغيير حالة
4. **استخدم الإستراتيجية** عندما تحتاج تبديل خوارزميات بسهولة
5. **استخدم المُزخرف** عندما تحتاج إضافة مسؤوليات بشكل مرن
6. **استخدم المُحول** عندما تحتاج دمج كود قديم مع واجهة جديدة
