# التغليف في لغة ص

## ما هو التغليف؟

التغليف (Encapsulation) هو أحد أركان البرمجة الكائنية الأساسية. يعني إخفاء التفاصيل الداخلية للكائن وتوفير واجهة محددة ومُتحكم بها للتعامل معه. التغليف يحمي البيانات من التعديل غير المقصود ويُسهّل صيانة الكود.

تخيل جهاز التلفاز - أنت تستخدم جهاز التحكم لتغيير القناة والصوت، لكنك لا تتعامل مباشرة مع الدوائر الإلكترونية الداخلية. هذا هو التغليف: واجهة بسيطة للمستخدم مع إخفاء التعقيد الداخلي.

## مبادئ التغليف

### 1. إخفاء البيانات (Data Hiding)

جعل الخصائص خاصة والوصول إليها عبر دوال:

```س
صنف حساب_مصرفي
    خاص:
        عشري الرصيد          # لا يمكن الوصول مباشرة
        نص رقم_الحساب       # لا يمكن الوصول مباشرة
        نص كلمة_السر        # لا يمكن الوصول مباشرة
    
    عام:
        دالة منشئ(نص رقم، نص كلمة، عشري رصيد_ابتدائي)
            رقم_الحساب = رقم
            كلمة_السر = كلمة
            
            # التحقق من صحة الرصيد الابتدائي
            إذا(رصيد_ابتدائي >= 0)
                الرصيد = رصيد_ابتدائي
            وإلا
                الرصيد = 0
                اطبع("تحذير: الرصيد الابتدائي لا يمكن أن يكون سالباً")
            نهاية
        نهاية
        
        # واجهة للوصول للرصيد (للقراءة فقط)
        دالة احصل_الرصيد() عشري
            ارجع الرصيد
        نهاية
        
        # واجهة للإيداع مع تحقق
        دالة إيداع(عشري مبلغ) منطقي
            إذا(مبلغ <= 0)
                اطبع("خطأ: مبلغ الإيداع يجب أن يكون موجباً")
                ارجع خطأ
            نهاية
            
            الرصيد = الرصيد + مبلغ
            اطبع("تم إيداع ")
            اطبع(مبلغ)
            اطبع(" ريال")
            ارجع صحيح
        نهاية
        
        # واجهة للسحب مع تحقق
        دالة سحب(عشري مبلغ، نص كلمة) منطقي
            # التحقق من كلمة السر
            إذا(كلمة != كلمة_السر)
                اطبع("خطأ: كلمة السر غير صحيحة")
                ارجع خطأ
            نهاية
            
            # التحقق من المبلغ
            إذا(مبلغ <= 0)
                اطبع("خطأ: مبلغ السحب يجب أن يكون موجباً")
                ارجع خطأ
            نهاية
            
            # التحقق من الرصيد
            إذا(مبلغ > الرصيد)
                اطبع("خطأ: الرصيد غير كافٍ")
                ارجع خطأ
            نهاية
            
            الرصيد = الرصيد - مبلغ
            اطبع("تم سحب ")
            اطبع(مبلغ)
            اطبع(" ريال")
            ارجع صحيح
        نهاية
نهاية

دالة رئيسية() رقم
    متغير حسابي = حساب_مصرفي("123456"، "سر123"، 1000.0)
    
    # لا يمكن الوصول للخصائص الخاصة مباشرة:
    # حسابي.الرصيد = 1000000        # خطأ!
    # حسابي.كلمة_السر              # خطأ!
    
    # نستخدم الواجهة المتاحة:
    اطبع("الرصيد الحالي: ")
    اطبع(حسابي.احصل_الرصيد())
    
    حسابي.إيداع(500.0)
    حسابي.سحب(200.0، "سر123")
    
    # محاولة سحب بكلمة سر خاطئة
    حسابي.سحب(100.0، "خطأ")  # خطأ: كلمة السر غير صحيحة
    
    اطبع("الرصيد النهائي: ")
    اطبع(حسابي.احصل_الرصيد())
    
    ارجع 0
نهاية
```

### 2. دوال الوصول (Getters and Setters)

توفير دوال للقراءة والكتابة مع تحقق:

```س
صنف موظف
    خاص:
        نص الاسم
        رقم العمر
        عشري الراتب
        نص المنصب
    
    عام:
        دالة منشئ(نص اسم)
            الاسم = اسم
            العمر = 0
            الراتب = 0.0
            المنصب = "غير محدد"
        نهاية
        
        # ===== Getters (دوال القراءة) =====
        
        دالة احصل_الاسم() نص
            ارجع الاسم
        نهاية
        
        دالة احصل_العمر() رقم
            ارجع العمر
        نهاية
        
        دالة احصل_الراتب() عشري
            ارجع الراتب
        نهاية
        
        دالة احصل_المنصب() نص
            ارجع المنصب
        نهاية
        
        # ===== Setters (دوال الكتابة مع تحقق) =====
        
        دالة عيّن_الاسم(نص اسم_جديد)
            # التحقق من أن الاسم ليس فارغاً
            إذا(طول(اسم_جديد) > 0)
                الاسم = اسم_جديد
            وإلا
                اطبع("خطأ: الاسم لا يمكن أن يكون فارغاً")
            نهاية
        نهاية
        
        دالة عيّن_العمر(رقم عمر_جديد)
            # التحقق من نطاق العمر المعقول
            إذا(عمر_جديد >= 18 و عمر_جديد <= 70)
                العمر = عمر_جديد
            وإلا
                اطبع("خطأ: العمر يجب أن يكون بين 18 و 70")
            نهاية
        نهاية
        
        دالة عيّن_الراتب(عشري راتب_جديد)
            # التحقق من أن الراتب موجب
            إذا(راتب_جديد >= 0)
                الراتب = راتب_جديد
            وإلا
                اطبع("خطأ: الراتب لا يمكن أن يكون سالباً")
            نهاية
        نهاية
        
        دالة عيّن_المنصب(نص منصب_جديد)
            # التحقق من المناصب المسموحة
            إذا(منصب_جديد == "مبتدئ" أو منصب_جديد == "متوسط" أو 
               منصب_جديد == "خبير" أو منصب_جديد == "مدير")
                المنصب = منصب_جديد
            وإلا
                اطبع("خطأ: منصب غير معروف. المناصب المتاحة: مبتدئ، متوسط، خبير، مدير")
            نهاية
        نهاية
        
        # دالة مساعدة لطباعة المعلومات
        دالة اطبع_المعلومات()
            اطبع("━━━━━━━━━━━━━━━━━━━━")
            اطبع("الاسم: ")
            اطبع(الاسم)
            اطبع("العمر: ")
            اطبع(العمر)
            اطبع("المنصب: ")
            اطبع(المنصب)
            اطبع("الراتب: ")
            اطبع(الراتب)
            اطبع(" ريال")
        نهاية
نهاية

دالة رئيسية() رقم
    متغير موظف1 = موظف("أحمد")
    
    # استخدام الـ setters مع تحقق
    موظف1.عيّن_العمر(25)      # ✓ يعمل
    موظف1.عيّن_الراتب(5000.0) # ✓ يعمل
    موظف1.عيّن_المنصب("متوسط") # ✓ يعمل
    
    موظف1.اطبع_المعلومات()
    
    # محاولات غير صالحة
    موظف1.عيّن_العمر(15)       # خطأ: العمر يجب أن يكون بين 18 و 70
    موظف1.عيّن_الراتب(-1000.0) # خطأ: الراتب لا يمكن أن يكون سالباً
    موظف1.عيّن_المنصب("رئيس")  # خطأ: منصب غير معروف
    
    ارجع 0
نهاية
```

### 3. الثبات (Immutability)

إنشاء كائنات لا يمكن تعديلها بعد الإنشاء:

```س
صنف نقطة_ثابتة
    خاص:
        عشري س
        عشري ص
    
    عام:
        # تُعيّن القيم في المُنشئ فقط
        دالة منشئ(عشري إكس، عشري واي)
            س = إكس
            ص = واي
        نهاية
        
        # دوال قراءة فقط (لا يوجد setters)
        دالة احصل_س() عشري
            ارجع س
        نهاية
        
        دالة احصل_ص() عشري
            ارجع ص
        نهاية
        
        # بدلاً من تعديل النقطة، نُنشئ نقطة جديدة
        دالة حرّك(عشري د_س، عشري د_ص) نقطة_ثابتة
            ارجع نقطة_ثابتة(س + د_س، ص + د_ص)
        نهاية
        
        دالة اطبع()
            اطبع("(")
            اطبع(س)
            اطبع("، ")
            اطبع(ص)
            اطبع(")")
        نهاية
نهاية

دالة رئيسية() رقم
    متغير ن1 = نقطة_ثابتة(3.0، 4.0)
    ن1.اطبع()  # (3، 4)
    
    # بدلاً من تعديل ن1، نحصل على نقطة جديدة
    متغير ن2 = ن1.حرّك(2.0، 1.0)
    
    ن1.اطبع()  # (3، 4) - لم تتغير!
    ن2.اطبع()  # (5، 5) - نقطة جديدة
    
    ارجع 0
نهاية
```

## أمثلة عملية شاملة

### نظام إدارة المخزون

```س
صنف منتج
    خاص:
        نص الكود
        نص الاسم
        عشري السعر
        رقم الكمية
        رقم الحد_الأدنى
    
    عام:
        دالة منشئ(نص كود، نص اسم، عشري سعر، رقم كمية)
            الكود = كود
            الاسم = اسم
            
            # تحقق من السعر
            إذا(سعر > 0)
                السعر = سعر
            وإلا
                السعر = 0.01
                اطبع("تحذير: تم تعيين سعر افتراضي")
            نهاية
            
            # تحقق من الكمية
            إذا(كمية >= 0)
                الكمية = كمية
            وإلا
                الكمية = 0
            نهاية
            
            الحد_الأدنى = 5  # الحد الأدنى الافتراضي
        نهاية
        
        # Getters
        دالة احصل_الكود() نص
            ارجع الكود
        نهاية
        
        دالة احصل_الاسم() نص
            ارجع الاسم
        نهاية
        
        دالة احصل_السعر() عشري
            ارجع السعر
        نهاية
        
        دالة احصل_الكمية() رقم
            ارجع الكمية
        نهاية
        
        # Setters مع تحقق
        دالة عيّن_السعر(عشري سعر_جديد) منطقي
            إذا(سعر_جديد <= 0)
                اطبع("خطأ: السعر يجب أن يكون موجباً")
                ارجع خطأ
            نهاية
            السعر = سعر_جديد
            ارجع صحيح
        نهاية
        
        دالة عيّن_الحد_الأدنى(رقم حد)
            إذا(حد >= 0)
                الحد_الأدنى = حد
            نهاية
        نهاية
        
        # عمليات المخزون
        دالة أضف_كمية(رقم كمية_جديدة) منطقي
            إذا(كمية_جديدة <= 0)
                اطبع("خطأ: الكمية يجب أن تكون موجبة")
                ارجع خطأ
            نهاية
            
            الكمية = الكمية + كمية_جديدة
            اطبع("تمت إضافة ")
            اطبع(كمية_جديدة)
            اطبع(" وحدة من ")
            اطبع(الاسم)
            ارجع صحيح
        نهاية
        
        دالة اسحب_كمية(رقم كمية_مطلوبة) منطقي
            إذا(كمية_مطلوبة <= 0)
                اطبع("خطأ: الكمية يجب أن تكون موجبة")
                ارجع خطأ
            نهاية
            
            إذا(كمية_مطلوبة > الكمية)
                اطبع("خطأ: الكمية المطلوبة غير متوفرة")
                اطبع("المتوفر: ")
                اطبع(الكمية)
                ارجع خطأ
            نهاية
            
            الكمية = الكمية - كمية_مطلوبة
            
            # تحذير إذا وصلنا للحد الأدنى
            إذا(الكمية <= الحد_الأدنى)
                اطبع("⚠️ تحذير: مخزون ")
                اطبع(الاسم)
                اطبع(" وصل للحد الأدنى!")
            نهاية
            
            ارجع صحيح
        نهاية
        
        دالة احسب_قيمة_المخزون() عشري
            ارجع السعر * الكمية
        نهاية
        
        دالة يحتاج_إعادة_طلب() منطقي
            ارجع الكمية <= الحد_الأدنى
        نهاية
        
        دالة اطبع_المعلومات()
            اطبع("┌────────────────────────────┐")
            اطبع("│ كود: ")
            اطبع(الكود)
            اطبع("│ المنتج: ")
            اطبع(الاسم)
            اطبع("│ السعر: ")
            اطبع(السعر)
            اطبع(" ريال")
            اطبع("│ الكمية: ")
            اطبع(الكمية)
            اطبع("│ قيمة المخزون: ")
            اطبع(احسب_قيمة_المخزون())
            اطبع(" ريال")
            إذا(يحتاج_إعادة_طلب())
                اطبع("│ ⚠️ يحتاج إعادة طلب!")
            نهاية
            اطبع("└────────────────────────────┘")
        نهاية
نهاية

صنف مخزن
    خاص:
        منتج[] المنتجات
        رقم عدد_المنتجات
        رقم السعة
    
    عام:
        دالة منشئ(رقم سعة)
            السعة = سعة
            المنتجات = منتج[سعة]
            عدد_المنتجات = 0
        نهاية
        
        دالة أضف_منتج(منتج م) منطقي
            إذا(عدد_المنتجات >= السعة)
                اطبع("خطأ: المخزن ممتلئ!")
                ارجع خطأ
            نهاية
            
            # التحقق من عدم تكرار الكود
            متغير ع = 0
            طالما(ع < عدد_المنتجات)
                إذا(المنتجات[ع].احصل_الكود() == م.احصل_الكود())
                    اطبع("خطأ: كود المنتج موجود مسبقاً!")
                    ارجع خطأ
                نهاية
                ع = ع + 1
            نهاية
            
            المنتجات[عدد_المنتجات] = م
            عدد_المنتجات = عدد_المنتجات + 1
            اطبع("تمت إضافة المنتج: ")
            اطبع(م.احصل_الاسم())
            ارجع صحيح
        نهاية
        
        دالة ابحث_بالكود(نص كود) منتج
            متغير ع = 0
            طالما(ع < عدد_المنتجات)
                إذا(المنتجات[ع].احصل_الكود() == كود)
                    ارجع المنتجات[ع]
                نهاية
                ع = ع + 1
            نهاية
            ارجع فارغ
        نهاية
        
        دالة قائمة_المنتجات_الناقصة()
            اطبع("=== المنتجات التي تحتاج إعادة طلب ===")
            متغير وُجد = خطأ
            
            متغير ع = 0
            طالما(ع < عدد_المنتجات)
                إذا(المنتجات[ع].يحتاج_إعادة_طلب())
                    اطبع("- ")
                    اطبع(المنتجات[ع].احصل_الاسم())
                    اطبع(" (الكمية: ")
                    اطبع(المنتجات[ع].احصل_الكمية())
                    اطبع(")")
                    وُجد = صحيح
                نهاية
                ع = ع + 1
            نهاية
            
            إذا(!وُجد)
                اطبع("جميع المنتجات متوفرة بكميات كافية")
            نهاية
        نهاية
        
        دالة إجمالي_قيمة_المخزون() عشري
            متغير إجمالي = 0.0
            متغير ع = 0
            طالما(ع < عدد_المنتجات)
                إجمالي = إجمالي + المنتجات[ع].احسب_قيمة_المخزون()
                ع = ع + 1
            نهاية
            ارجع إجمالي
        نهاية
        
        دالة اطبع_تقرير()
            اطبع("╔════════════════════════════════════╗")
            اطبع("║       تقرير المخزون               ║")
            اطبع("╠════════════════════════════════════╣")
            
            متغير ع = 0
            طالما(ع < عدد_المنتجات)
                المنتجات[ع].اطبع_المعلومات()
                ع = ع + 1
            نهاية
            
            اطبع("╠════════════════════════════════════╣")
            اطبع("║ إجمالي عدد المنتجات: ")
            اطبع(عدد_المنتجات)
            اطبع("║ إجمالي قيمة المخزون: ")
            اطبع(إجمالي_قيمة_المخزون())
            اطبع(" ريال")
            اطبع("╚════════════════════════════════════╝")
        نهاية
نهاية

    متغير م1 = منتج("P001"، "قلم"، 5.0، 100)
```

### نظام إدارة الطلاب

```س
صنف درجة
    خاص:
        نص المادة
        عشري الدرجة
        عشري الحد_الأقصى
    
    عام:
        دالة منشئ(نص مادة، عشري درجة، عشري حد_أقصى)
            المادة = مادة
            الحد_الأقصى = حد_أقصى
            
            # التحقق من صحة الدرجة
            إذا(درجة >= 0 و درجة <= حد_أقصى)
                الدرجة = درجة
            وإلا إذا(درجة < 0)
                الدرجة = 0
            وإلا
                الدرجة = حد_أقصى
            نهاية
        نهاية
        
        دالة احصل_المادة() نص
            ارجع المادة
        نهاية
        
        دالة احصل_الدرجة() عشري
            ارجع الدرجة
        نهاية
        
        دالة احصل_النسبة() عشري
            ارجع (الدرجة / الحد_الأقصى) * 100
        نهاية
        
        دالة احصل_التقدير() نص
            متغير نسبة = احصل_النسبة()
            
            إذا(نسبة >= 90)
                ارجع "ممتاز"
            وإلا إذا(نسبة >= 80)
                ارجع "جيد جداً"
            وإلا إذا(نسبة >= 70)
                ارجع "جيد"
            وإلا إذا(نسبة >= 60)
                ارجع "مقبول"
            وإلا
                ارجع "راسب"
            نهاية
        نهاية
        
        دالة ناجح() منطقي
            ارجع احصل_النسبة() >= 60
        نهاية
نهاية

صنف طالب
```

## فوائد التغليف

1. **حماية البيانات**: منع التعديل غير المقصود
2. **التحقق من الصحة**: التأكد من صحة القيم قبل التخزين
3. **المرونة**: تغيير التنفيذ الداخلي بدون التأثير على الكود الخارجي
4. **سهولة الصيانة**: الكود منظم ومركزي
5. **تقليل الأخطاء**: الواجهة المحددة تقلل سوء الاستخدام

## ملخص

التغليف في لغة ص يُقدم لك:

- **مستويات الوصول**: خاص، محمي، عام
- **دوال الوصول**: Getters و Setters
- **التحقق من البيانات**: ضمان صحة القيم
- **إخفاء التعقيد**: واجهة بسيطة للمستخدم

استخدم التغليف لبناء كود آمن ومنظم وقابل للصيانة!
