# معالجة الأخطاء في لغة ص

## مقدمة

معالجة الأخطاء جزء أساسي من كتابة برامج قوية وموثوقة. في هذا الدليل سنتعلم كيفية التعامل مع الأخطاء والاستثناءات في لغة ص بطرق مختلفة ومتقدمة.

## 1. أنواع الأخطاء

### أخطاء وقت الترجمة

هذه الأخطاء تُكتشف أثناء ترجمة البرنامج ولا تسمح بتشغيله.

```س

# أخطاء نوع
رقم س = "نص"  # لا يمكن تعيين نص لمتغير رقمي

# أخطاء تعريف
اطبع(س)  # المتغير س غير معرف
```

### أخطاء وقت التشغيل

هذه الأخطاء تحدث أثناء تنفيذ البرنامج.

```س
# القسمة على صفر
متغير نتيجة = 10 / 0

# الوصول لفهرس غير موجود
متغير قائمة = [1, 2, 3]
اطبع(قائمة[10])

# استدعاء دالة على قيمة عدم
متغير كائن = عدم
كائن.دالة()
```

### أخطاء منطقية

هذه الأخطاء لا تُوقف البرنامج لكنها تُنتج نتائج خاطئة.

```س
# خطأ منطقي: حساب المتوسط بشكل خاطئ
دالة متوسط_خاطئ(مصفوفة<رقم> أرقام) عشري
    متغير مجموع = 0
    لكل رقم في أرقام
        مجموع = مجموع + رقم
    نهاية
    ارجع مجموع / 10  # خطأ! يجب القسمة على عدد العناصر
نهاية

# الصحيح
دالة متوسط_صحيح(مصفوفة<رقم> أرقام) عشري
    متغير مجموع = 0
    لكل رقم في أرقام
        مجموع = مجموع + رقم
    نهاية
    ارجع مجموع / أرقام.الطول()
نهاية
```

## 2. نمط النتيجة (Result Pattern)

طريقة آمنة للتعامل مع العمليات التي قد تفشل.

### تعريف صنف النتيجة

```س
# نتيجة عامة
قالب<نوع ت>
صنف نتيجة
    خاص:
        منطقي _نجاح
        ت _قيمة
        نص _خطأ
    
    عام:
        # إنشاء نتيجة ناجحة
        دالة ثابتة نجاح(ت قيمة) نتيجة<ت>
            متغير ن = نتيجة<ت>()
            ن._نجاح = صحيح
            ن._قيمة = قيمة
            ن._خطأ = ""
            ارجع ن
        نهاية
        
        # إنشاء نتيجة فاشلة
        دالة ثابتة فشل(نص رسالة) نتيجة<ت>
            متغير ن = نتيجة<ت>()
            ن._نجاح = خطأ
            ن._خطأ = رسالة
            ارجع ن
        نهاية
        
        # التحقق من النجاح
        دالة هل_نجح() منطقي
            ارجع _نجاح
        نهاية
        
        # الحصول على القيمة
        دالة القيمة() ت
            ارجع _قيمة
        نهاية
        
        # الحصول على رسالة الخطأ
        دالة الخطأ() نص
            ارجع _خطأ
        نهاية
نهاية
```

### استخدام نمط النتيجة

```س
صنف حساب_بنكي
    خاص:
        نص رقم_الحساب
        عشري الرصيد
    
    عام:
        دالة منشئ(نص رقم، عشري رصيد_ابتدائي)
            رقم_الحساب = رقم
            الرصيد = رصيد_ابتدائي
        نهاية
        
        دالة سحب(عشري مبلغ) نتيجة<عشري>
            # التحقق من صحة المبلغ
            إذا(مبلغ <= 0)
                ارجع نتيجة<عشري>::فشل("المبلغ يجب أن يكون موجباً")
            نهاية
            
            # التحقق من الرصيد
            إذا(مبلغ > الرصيد)
                ارجع نتيجة<عشري>::فشل("الرصيد غير كافٍ. المتاح: " + الرصيد)
            نهاية
            
            # تنفيذ السحب
            الرصيد = الرصيد - مبلغ
            ارجع نتيجة<عشري>::نجاح(الرصيد)
        نهاية
        
        دالة إيداع(عشري مبلغ) نتيجة<عشري>
            إذا(مبلغ <= 0)
                ارجع نتيجة<عشري>::فشل("المبلغ يجب أن يكون موجباً")
            نهاية
            
            إذا(مبلغ > 50000)
                ارجع نتيجة<عشري>::فشل("لا يمكن إيداع أكثر من 50,000 في عملية واحدة")
            نهاية
            
            الرصيد = الرصيد + مبلغ
            ارجع نتيجة<عشري>::نجاح(الرصيد)
        نهاية
        
        دالة تحويل(حساب_بنكي حساب_آخر، عشري مبلغ) نتيجة<نص>
            # محاولة السحب
            متغير نتيجة_سحب = سحب(مبلغ)
            إذا(!نتيجة_سحب.هل_نجح())
                ارجع نتيجة<نص>::فشل("فشل السحب: " + نتيجة_سحب.الخطأ())
            نهاية
            
            # محاولة الإيداع
            متغير نتيجة_إيداع = حساب_آخر.إيداع(مبلغ)
            إذا(!نتيجة_إيداع.هل_نجح())
                # إرجاع المبلغ للحساب الأصلي
                إيداع(مبلغ)
                ارجع نتيجة<نص>::فشل("فشل الإيداع: " + نتيجة_إيداع.الخطأ())
            نهاية
            
            ارجع نتيجة<نص>::نجاح("تم التحويل بنجاح")
        نهاية
نهاية

دالة رئيسية() رقم
    متغير حسابي = حساب_بنكي("123", 1000.0)
    
    # محاولة سحب صحيحة
    متغير نتيجة1 = حسابي.سحب(500.0)
    إذا(نتيجة1.هل_نجح())
        اطبع("تم السحب. الرصيد الجديد: " + نتيجة1.القيمة())
    وإلا
        اطبع("خطأ: " + نتيجة1.الخطأ())
    نهاية
    
    # محاولة سحب فاشلة
    متغير نتيجة2 = حسابي.سحب(1000.0)  # أكثر من الرصيد
    إذا(نتيجة2.هل_نجح())
        اطبع("تم السحب. الرصيد الجديد: " + نتيجة2.القيمة())
    وإلا
        اطبع("خطأ: " + نتيجة2.الخطأ())  # "الرصيد غير كافٍ"
    نهاية
    
    ارجع 0
نهاية
```

## 3. نمط الاختياري (Optional Pattern)

للتعامل مع القيم التي قد لا تكون موجودة.

```س
قالب<نوع ت>
صنف اختياري
    خاص:
        منطقي _موجود
        ت _قيمة
    
    عام:
        # إنشاء قيمة موجودة
        دالة ثابتة بعض(ت قيمة) اختياري<ت>
            متغير خ = اختياري<ت>()
            خ._موجود = صحيح
            خ._قيمة = قيمة
            ارجع خ
        نهاية
        
        # إنشاء قيمة غير موجودة
        دالة ثابتة لا_شيء() اختياري<ت>
            متغير خ = اختياري<ت>()
            خ._موجود = خطأ
            ارجع خ
        نهاية
        
        # التحقق من الوجود
        دالة موجود() منطقي
            ارجع _موجود
        نهاية
        
        # الحصول على القيمة
        دالة القيمة() ت
            ارجع _قيمة
        نهاية
        
        # الحصول على القيمة أو البديل
        دالة أو(ت بديل) ت
            إذا(_موجود)
                ارجع _قيمة
            نهاية
            ارجع بديل
        نهاية
نهاية
```

### استخدام الاختياري

```س
صنف قاموس
    خاص:
        مصفوفة<نص> المفاتيح
        مصفوفة<نص> القيم
    
    عام:
        دالة منشئ()
            المفاتيح = مصفوفة<نص>()
            القيم = مصفوفة<نص>()
        نهاية
        
        دالة أضف(نص مفتاح، نص قيمة)
            المفاتيح.أضف(مفتاح)
            القيم.أضف(قيمة)
        نهاية
        
        دالة ابحث(نص مفتاح) اختياري<نص>
            من أجل(رقم ع = 0؛ ع < المفاتيح.الطول()؛ ع++)
                إذا(المفاتيح[ع] == مفتاح)
                    ارجع اختياري<نص>::بعض(القيم[ع])
                نهاية
            نهاية
            ارجع اختياري<نص>::لا_شيء()
        نهاية
نهاية

دالة رئيسية() رقم
    متغير قاموس_عربي = قاموس()
    قاموس_عربي.أضف("hello", "مرحباً")
    قاموس_عربي.أضف("world", "عالم")
    قاموس_عربي.أضف("book", "كتاب")
    
    # البحث عن كلمة موجودة
    متغير نتيجة1 = قاموس_عربي.ابحث("hello")
    إذا(نتيجة1.موجود())
        اطبع("الترجمة: " + نتيجة1.القيمة())
    وإلا
        اطبع("الكلمة غير موجودة")
    نهاية
    
    # البحث عن كلمة غير موجودة مع قيمة افتراضية
    متغير ترجمة = قاموس_عربي.ابحث("cat").أو("غير معروف")
    اطبع("ترجمة cat: " + ترجمة)  # "غير معروف"
    
    ارجع 0
نهاية
```

## 4. التحقق من المدخلات

### التحقق الأساسي

```س
صنف مدقق_المدخلات
    عام:
        # التحقق من البريد الإلكتروني
        دالة ثابتة بريد_صالح(نص بريد) نتيجة<نص>
            إذا(بريد.الطول() == 0)
                ارجع نتيجة<نص>::فشل("البريد الإلكتروني مطلوب")
            نهاية
            
            إذا(!بريد.يحتوي("@"))
                ارجع نتيجة<نص>::فشل("البريد يجب أن يحتوي على @")
            نهاية
            
            إذا(!بريد.يحتوي("."))
                ارجع نتيجة<نص>::فشل("البريد يجب أن يحتوي على نقطة")
            نهاية
            
            ارجع نتيجة<نص>::نجاح(بريد)
        نهاية
        
        # التحقق من كلمة المرور
        دالة ثابتة كلمة_مرور_صالحة(نص كلمة) نتيجة<نص>
            إذا(كلمة.الطول() < 8)
                ارجع نتيجة<نص>::فشل("كلمة المرور يجب أن تكون 8 أحرف على الأقل")
            نهاية
            
            متغير بها_رقم = خطأ
            متغير بها_حرف = خطأ
            
            لكل حرف في كلمة
                إذا(حرف.هل_رقم())
                    بها_رقم = صحيح
                نهاية
                إذا(حرف.هل_حرف())
                    بها_حرف = صحيح
                نهاية
            نهاية
            
            إذا(!بها_رقم)
                ارجع نتيجة<نص>::فشل("كلمة المرور يجب أن تحتوي على رقم واحد على الأقل")
            نهاية
            
            إذا(!بها_حرف)
                ارجع نتيجة<نص>::فشل("كلمة المرور يجب أن تحتوي على حرف واحد على الأقل")
            نهاية
            
            ارجع نتيجة<نص>::نجاح(كلمة)
        نهاية
        
        # التحقق من رقم الهاتف السعودي
        دالة ثابتة هاتف_صالح(نص هاتف) نتيجة<نص>
            إذا(هاتف.الطول() != 10)
                ارجع نتيجة<نص>::فشل("رقم الهاتف يجب أن يكون 10 أرقام")
            نهاية
            
            إذا(!هاتف.يبدأ_بـ("05"))
                ارجع نتيجة<نص>::فشل("رقم الهاتف يجب أن يبدأ بـ 05")
            نهاية
            
            لكل حرف في هاتف
                إذا(!حرف.هل_رقم())
                    ارجع نتيجة<نص>::فشل("رقم الهاتف يجب أن يحتوي على أرقام فقط")
                نهاية
            نهاية
            
            ارجع نتيجة<نص>::نجاح(هاتف)
        نهاية
        
        # التحقق من العمر
        دالة ثابتة عمر_صالح(رقم عمر) نتيجة<رقم>
            إذا(عمر < 0)
                ارجع نتيجة<رقم>::فشل("العمر لا يمكن أن يكون سالباً")
            نهاية
            
            إذا(عمر > 150)
                ارجع نتيجة<رقم>::فشل("العمر غير منطقي")
            نهاية
            
            ارجع نتيجة<رقم>::نجاح(عمر)
        نهاية
نهاية
```

### مثال تطبيقي: نموذج تسجيل

```س
صنف نموذج_تسجيل
    خاص:
        نص الاسم
        نص البريد
        نص الهاتف
        نص كلمة_المرور

        نهاية
        
        دالة عيّن_الاسم(نص اسم)
            إذا(اسم.الطول() < 2)
                الأخطاء.أضف("الاسم يجب أن يكون حرفين على الأقل")
            وإلا
                الاسم = اسم
            نهاية
        نهاية
        
        دالة عيّن_البريد(نص بريد)
            متغير نتيجة = مدقق_المدخلات::بريد_صالح(بريد)
            إذا(نتيجة.هل_نجح())
                البريد = بريد
            وإلا
                الأخطاء.أضف(نتيجة.الخطأ())
            نهاية
        نهاية
        
        دالة عيّن_الهاتف(نص هاتف)
            متغير نتيجة = مدقق_المدخلات::هاتف_صالح(هاتف)
            إذا(نتيجة.هل_نجح())
                الهاتف = هاتف
            وإلا
                الأخطاء.أضف(نتيجة.الخطأ())
            نهاية
        نهاية
        
        دالة عيّن_كلمة_المرور(نص كلمة)
            متغير نتيجة = مدقق_المدخلات::كلمة_مرور_صالحة(كلمة)
            إذا(نتيجة.هل_نجح())
                كلمة_المرور = كلمة
            وإلا
                الأخطاء.أضف(نتيجة.الخطأ())
            نهاية
        نهاية
        
        دالة عيّن_العمر(رقم عمر)
            متغير نتيجة = مدقق_المدخلات::عمر_صالح(عمر)
            إذا(نتيجة.هل_نجح())
                العمر = عمر
            وإلا
                الأخطاء.أضف(نتيجة.الخطأ())
            نهاية
        نهاية
        
        دالة هل_صالح() منطقي
            ارجع الأخطاء.الطول() == 0
        نهاية
        
        دالة احصل_الأخطاء() مصفوفة<نص>
            ارجع الأخطاء
        نهاية
        
        دالة أرسل() نتيجة<نص>
            إذا(!هل_صالح())
                متغير رسالة = "فشل التسجيل بسبب الأخطاء التالية:\n"
                لكل خطأ في الأخطاء
                    رسالة = رسالة + "- " + خطأ + "\n"
                نهاية
                ارجع نتيجة<نص>::فشل(رسالة)
            نهاية
            
            # تنفيذ التسجيل
            ارجع نتيجة<نص>::نجاح("تم التسجيل بنجاح!")
        نهاية
نهاية

دالة رئيسية() رقم
    متغير نموذج = نموذج_تسجيل()
    
    نموذج.عيّن_الاسم("أحمد محمد")
    نموذج.عيّن_البريد("ahmad@email.com")
    نموذج.عيّن_الهاتف("0501234567")
    نموذج.عيّن_كلمة_المرور("Pass123!")
    نموذج.عيّن_العمر(25)
    
    متغير نتيجة = نموذج.أرسل()
    إذا(نتيجة.هل_نجح())
        اطبع(نتيجة.القيمة())
    وإلا
        اطبع(نتيجة.الخطأ())
    نهاية
    
    ارجع 0
نهاية
```

## 5. سلسلة المسؤولية (Chain of Responsibility)

للتعامل مع أنواع مختلفة من الأخطاء.

```س
صنف مجرد معالج_أخطاء
    محمي:
        معالج_أخطاء المعالج_التالي
    
    عام:
        دالة عيّن_التالي(معالج_أخطاء تالي) معالج_أخطاء
            المعالج_التالي = تالي
            ارجع تالي
        نهاية
        
        دالة مجردة معالجة(نص نوع_الخطأ، نص رسالة)
نهاية

صنف معالج_أخطاء_الشبكة يرث معالج_أخطاء
    عام:
        دالة معالجة(نص نوع_الخطأ، نص رسالة)
            إذا(نوع_الخطأ == "شبكة")
                اطبع("[معالج الشبكة] محاولة إعادة الاتصال...")
                اطبع("  الخطأ: " + رسالة)
                # محاولة إعادة الاتصال
            وإلا إذا(المعالج_التالي != عدم)
                المعالج_التالي.معالجة(نوع_الخطأ، رسالة)
            نهاية
        نهاية
نهاية

صنف معالج_أخطاء_قاعدة_البيانات يرث معالج_أخطاء
    عام:
        دالة معالجة(نص نوع_الخطأ، نص رسالة)
            إذا(نوع_الخطأ == "قاعدة_بيانات")
                اطبع("[معالج قاعدة البيانات] محاولة إعادة الاتصال...")
                اطبع("  الخطأ: " + رسالة)
                # محاولة إعادة الاتصال
            وإلا إذا(المعالج_التالي != عدم)
                المعالج_التالي.معالجة(نوع_الخطأ، رسالة)
            نهاية
        نهاية
نهاية

صنف معالج_أخطاء_ملفات يرث معالج_أخطاء
    عام:
        دالة معالجة(نص نوع_الخطأ، نص رسالة)
            إذا(نوع_الخطأ == "ملف")
                اطبع("[معالج الملفات] محاولة الوصول للملف...")
                اطبع("  الخطأ: " + رسالة)
            وإلا إذا(المعالج_التالي != عدم)
                المعالج_التالي.معالجة(نوع_الخطأ، رسالة)
            نهاية
        نهاية
نهاية

صنف معالج_افتراضي يرث معالج_أخطاء
    عام:
        دالة معالجة(نص نوع_الخطأ، نص رسالة)
            اطبع("[معالج افتراضي] خطأ غير معروف: " + نوع_الخطأ)
            اطبع("  الرسالة: " + رسالة)
        نهاية
نهاية

دالة رئيسية() رقم
    # بناء سلسلة المعالجات
    متغير معالج_شبكة = معالج_أخطاء_الشبكة()
    متغير معالج_قاعدة = معالج_أخطاء_قاعدة_البيانات()
    متغير معالج_ملفات = معالج_أخطاء_ملفات()
    متغير معالج_افتراضي = معالج_افتراضي()
    
    معالج_شبكة.عيّن_التالي(معالج_قاعدة)
        .عيّن_التالي(معالج_ملفات)
        .عيّن_التالي(معالج_افتراضي)
    
    # اختبار المعالجات
    معالج_شبكة.معالجة("شبكة", "انقطع الاتصال بالخادم")
```

## ملخص

### قواعد معالجة الأخطاء

1. **تحقق دائماً من المدخلات** قبل استخدامها
2. **استخدم نمط النتيجة** للعمليات التي قد تفشل
3. **سجّل الأخطاء** للتصحيح والمراقبة
4. **قدّم رسائل خطأ واضحة** للمستخدم
5. **لا تتجاهل الأخطاء** - عالجها أو مررها للأعلى
6. **اختبر حالات الخطأ** في الكود

### متى تستخدم كل نمط

| النمط | الاستخدام |
|-------|-----------|
| النتيجة | عمليات قد تفشل مع سبب |
| الاختياري | قيم قد لا تكون موجودة |
| التحقق | فحص المدخلات |
| سلسلة المسؤولية | أنواع مختلفة من الأخطاء |
| التسجيل | تتبع وتصحيح الأخطاء |
