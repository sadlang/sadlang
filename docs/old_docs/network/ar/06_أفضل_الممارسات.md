## 📖 مقدمة

الفرق بين كود "يعمل" وكود "احترافي" هو اتباع أفضل الممارسات. في هذا الدليل، سنستكشف الممارسات المثبتة التي تستخدمها الشركات الكبرى لبناء تطبيقات شبكية موثوقة وآمنة وقابلة للصيانة.

---

## 🎯 المبادئ الأساسية

### 1. دائماً توقع الأسوأ

**المبدأ:** الشبكات غير موثوقة بطبيعتها. افترض دائماً أن كل عملية قد تفشل.

**❌ سيء:**
```sad
عميل = شبكة.اتصل_tcp("server.com", 8080)
بيانات = عميل.استقبل(1024)
عميل.ارسل(بيانات)
```

**✅ جيد:**
```sad
محاولة {
    عميل = شبكة.اتصل_tcp("server.com", 8080)
    عميل.عين_مهلة_الاستقبال(5000)
    عميل.عين_مهلة_الإرسال(5000)
    
    محاولة {
        بيانات = عميل.استقبل(1024)
        
        إذا بيانات.طول() > 0 {
            عميل.ارسل_الكل(بيانات)
        }
    } نهائياً {
        عميل.اغلق()
    }
} اصطياد (خطأ_اتصال خطأ) {
    اطبع("فشل الاتصال: " + خطأ.رسالة())
    // خطة بديلة...
} اصطياد (خطأ_مهلة خطأ) {
    اطبع("انتهت مهلة العملية")
    // خطة بديلة...
}
```

### 2. أغلق الموارد دائماً

**المبدأ:** المقابس هي موارد نظام محدودة. يجب إغلاقها دائماً، حتى عند حدوث أخطاء.

**❌ سيء:**
```sad
دالة معالجة_طلب() {
    عميل = شبكة.اتصل_tcp("server.com", 8080)
    بيانات = عميل.استقبل(1024)
    
    إذا بيانات.طول() == 0 {
        ارجع  // ⚠️ لم نغلق العميل!
    }
    
    عميل.اغلق()
}
```

**✅ جيد:**
```sad
دالة معالجة_طلب() {
    عميل = باطل
    
    محاولة {
        عميل = شبكة.اتصل_tcp("server.com", 8080)
        بيانات = عميل.استقبل(1024)
        
        إذا بيانات.طول() == 0 {
            ارجع
        }
        
        // معالجة البيانات...
        
    } نهائياً {
        // سيُنفذ دائماً، حتى عند return أو exception
        إذا عميل != باطل {
            عميل.اغلق()
        }
    }
}
```

**🎯 أفضل - استخدام مدير سياق:**
```sad
دالة معالجة_طلب() {
    مع (عميل = شبكة.اتصل_tcp("server.com", 8080)) {
        // العميل سيُغلق تلقائياً عند الخروج من الكتلة
        بيانات = عميل.استقبل(1024)
        // معالجة البيانات...
    }
}
```

### 3. استخدم مهلات زمنية معقولة

**المبدأ:** بدون مهلات، تطبيقك قد "يتجمد" إلى الأبد.

**❌ سيء:**
```sad
// لا مهلة - قد ننتظر للأبد!
بيانات = عميل.استقبل(1024)
```

**✅ جيد:**
```sad
// مهلة 30 ثانية للعمليات العادية
عميل.عين_مهلة_الاستقبال(30000)
بيانات = عميل.استقبل(1024)
```

**🎯 أفضل - مهلات حسب نوع العملية:**
```sad
صنف مهلات {
    ثابت اتصال = 10000      // 10 ثوانٍ للاتصال
    ثابت قراءة_سريعة = 5000   // 5 ثوانٍ للعمليات السريعة
    ثابت قراءة_عادية = 30000  // 30 ثانية للعمليات العادية
    ثابت رفع_ملف = 300000     // 5 دقائق لرفع ملفات كبيرة
}

// حسب السياق
عميل.عين_مهلة_الاستقبال(مهلات.قراءة_عادية)
```

### 4. استخدم `ارسل_الكل()` و `استقبل_الكل()`

**المبدأ:** `ارسل()` و `استقبل()` العاديان قد لا يرسلون/يستقبلون كل البيانات في مرة واحدة.

**❌ سيء:**
```sad
// قد لا يرسل كل البيانات!
عميل.ارسل(بيانات_كبيرة)
```

**✅ جيد:**
```sad
// يضمن إرسال كل البيانات
عميل.ارسل_الكل(بيانات_كبيرة)
```

**📝 ملاحظة مهمة:**
```sad
// استقبال حجم محدد بالضبط
بيانات = عميل.استقبل_الكل(1024)  // سيستقبل 1024 بايت أو يرمي خطأ

// استقبال "حتى" حجم معين
بيانات = عميل.استقبل(1024)  // قد يستقبل أقل من 1024
```

### 5. لا تثق بالمدخلات أبداً

**المبدأ:** أي بيانات من الشبكة قد تكون ضارة. تحقق وفلتر دائماً.

**❌ سيء:**
```sad
// خطر أمني! لا تحقق من الحجم
حجم_بايتات = عميل.استقبل(8)
حجم = نص.الى_رقم(نص.من_بايتات(حجم_بايتات))

// ماذا لو كان حجم = 2GB؟ سيتعطل البرنامج!
بيانات = عميل.استقبل_الكل(حجم)
```

**✅ جيد:**
```sad
// تحديد حد أقصى معقول
الحد_الأقصى = 10 * 1024 * 1024  // 10MB

حجم_بايتات = عميل.استقبل(8)
حجم = نص.الى_رقم(نص.من_بايتات(حجم_بايتات))

// التحقق من الحجم
إذا حجم < 0 أو حجم > الحد_الأقصى {
    ارمِ خطأ("حجم غير صالح: " + حجم)
}

بيانات = عميل.استقبل_الكل(حجم)
```

**🎯 أفضل - فلترة شاملة:**
```sad
دالة استقبل_بيانات_آمن(عميل، حد_أقصى=10485760) {
    محاولة {
        // استقبال الحجم
        حجم_بايتات = عميل.استقبل_الكل(4)
        
        إذا حجم_بايتات.طول() != 4 {
            ارمِ خطأ("بيانات تالفة")
        }
        
        حجم = فك_رقم_صحيح(حجم_بايتات)
        
        // التحقق من الحدود
        إذا حجم < 0 {
            ارمِ خطأ("حجم سالب: " + حجم)
        }
        
        إذا حجم > حد_أقصى {
            ارمِ خطأ("حجم كبير جداً: " + حجم)
        }
        
        إذا حجم == 0 {
            ارجع []  // بيانات فارغة صالحة
        }
        
        // استقبال البيانات
        بيانات = عميل.استقبل_الكل(حجم)
        
        // التحقق من الاستقبال الكامل
        إذا بيانات.طول() != حجم {
            ارمِ خطأ("استقبال غير كامل")
        }
        
        ارجع بيانات
        
    } اصطياد (خطأ) {
        اطبع("خطأ في استقبال البيانات: " + خطأ.رسالة())
        ارمِ خطأ
    }
}
```

### 6. استخدم بروتوكولات واضحة ومحددة

**المبدأ:** حدد بنية الرسائل بوضوح لتجنب الغموض وسوء الفهم.

**❌ سيء - بروتوكول غامض:**
```sad
// ما هيكل هذه البيانات؟ لا أحد يعلم!
عميل.ارسل("بيانات مهمة")
```

**✅ جيد - بروتوكول واضح:**
```sad
// بروتوكول محدد: [نوع: 1 بايت] [حجم: 4 بايتات] [بيانات]

دالة ارسل_رسالة(عميل، نوع، بيانات) {
    حزمة = []
    
    // 1. نوع الرسالة (1 بايت)
    حزمة.أضف(نوع)
    
    // 2. حجم البيانات (4 بايتات)
    حجم = بيانات.طول()
    حزمة.أضف_كل(رمز_رقم_صحيح(حجم، 4))
    
    // 3. البيانات نفسها
    حزمة.أضف_كل(بيانات)
    
    // إرسال الحزمة كاملة
    عميل.ارسل_الكل(حزمة)
}

دالة استقبل_رسالة(عميل) {
    // 1. استقبال النوع
    نوع = عميل.استقبل_الكل(1)[0]
    
    // 2. استقبال الحجم
    حجم_بايتات = عميل.استقبل_الكل(4)
    حجم = فك_رقم_صحيح(حجم_بايتات)
    
    // 3. استقبال البيانات
    بيانات = عميل.استقبل_الكل(حجم)
    
    ارجع {نوع: نوع، بيانات: بيانات}
}
```

**🎯 أفضل - استخدام JSON للبنية المعقدة:**
```sad
استخدم json

// للرسائل المعقدة، JSON أوضح
دالة ارسل_رسالة_json(عميل، رسالة) {
    // تحويل للـ JSON
    نص_json = json.من_كائن(رسالة)
    بيانات = نص.الى_بايتات(نص_json)
    
    // إرسال الحجم ثم البيانات
    عميل.ارسل_الكل(رمز_رقم_صحيح(بيانات.طول()، 4))
    عميل.ارسل_الكل(بيانات)
}

// مثال استخدام
رسالة = {
    نوع: "دردشة"،
    مرسل: "أحمد"،
    محتوى: "مرحباً!"،
    وقت: وقت.الآن()
}

ارسل_رسالة_json(عميل، رسالة)
```

### 7. سجل كل شيء (مع الحذر)

**المبدأ:** التسجيل الجيد يسهل تتبع المشاكل، لكن احذر من تسجيل معلومات حساسة.

**❌ سيء:**
```sad
// لا تسجيل - من المستحيل تتبع المشاكل
عميل.ارسل(بيانات)
```

**❌ سيء أيضاً:**
```sad
// تسجيل معلومات حساسة!
اطبع("إرسال كلمة المرور: " + كلمة_المرور)
```

**✅ جيد:**
```sad
صنف مسجل {
    مستوى: نص
    
    دالة معلومات(رسالة، سياق={}) {
        طابع_زمني = وقت.الآن_بتنسيق("%Y-%m-%d %H:%M:%S")
        اطبع("[" + طابع_زمني + "] [معلومات] " + رسالة)
        
        إذا سياق.طول() > 0 {
            // لا تسجل كلمات المرور أو توكنات!
            سياق_آمن = فلتر_بيانات_حساسة(سياق)
            اطبع("  السياق: " + نص.من_كائن(سياق_آمن))
        }
    }
    
    دالة خطأ(رسالة، استثناء=باطل) {
        طابع_زمني = وقت.الآن_بتنسيق("%Y-%m-%d %H:%M:%S")
        اطبع("[" + طابع_زمني + "] [خطأ] " + رسالة)
        
        إذا استثناء != باطل {
            اطبع("  النوع: " + استثناء.نوع())
            اطبع("  الرسالة: " + استثناء.رسالة())
            اطبع("  الكود: " + استثناء.كود())
        }
    }
}

دالة فلتر_بيانات_حساسة(بيانات) {
    // حذف الحقول الحساسة
    مفاتيح_حساسة = ["كلمة_المرور", "توكن", "مفتاح_api", "سر"]
    
    بيانات_آمنة = {}
    لكل (مفتاح، قيمة) في بيانات {
        إذا مفتاح في مفاتيح_حساسة {
            بيانات_آمنة[مفتاح] = "***مخفي***"
        } وإلا {
            بيانات_آمنة[مفتاح] = قيمة
        }
    }
    
    ارجع بيانات_آمنة
}

// الاستخدام
مسجل = مسجل()

محاولة {
    عميل = شبكة.اتصل_tcp("server.com", 8080)
    مسجل.معلومات("تم الاتصال بنجاح", {
        خادم: "server.com:8080"،
        بروتوكول: "TCP"
    })
    
} اصطياد (خطأ) {
    مسجل.خطأ("فشل الاتصال", خطأ)
}
```

### 8. استخدم المخازن المؤقتة المناسبة

**المبدأ:** حجم المخزن المؤقت يؤثر على الأداء والذاكرة.

**❌ سيء:**
```sad
// مخزن صغير جداً - استدعاءات كثيرة
بينما صحيح {
    دفعة = عميل.استقبل(16)  // 16 بايت فقط!
    // معالجة...
}
```

**❌ سيء أيضاً:**
```sad
// مخزن كبير جداً - هدر للذاكرة
بينما صحيح {
    دفعة = عميل.استقبل(100 * 1024 * 1024)  // 100MB!
    // معالجة...
}
```

**✅ جيد:**
```sad
// أحجام معقولة حسب الاستخدام
صنف أحجام_مخزن {
    ثابت صغير = 1024         // 1KB - للرسائل القصيرة
    ثابت متوسط = 8192        // 8KB - للرسائل العادية
    ثابت كبير = 65536        // 64KB - لنقل الملفات
    ثابت ضخم = 1048576       // 1MB - للملفات الكبيرة جداً
}

// للرسائل النصية
رسالة = عميل.استقبل(أحجام_مخزن.متوسط)

// لنقل ملف
بينما صحيح {
    دفعة = عميل.استقبل(أحجام_مخزن.كبير)
    إذا دفعة.طول() == 0 { اكسر }
    ملف.اكتب(دفعة)
}
```

### 9. تعامل مع الإنهاء بأناقة

**المبدأ:** أغلق الاتصالات بشكل لطيف قبل إنهاء البرنامج.

**❌ سيء:**
```sad
// إغلاق فجائي
عميل.اغلق()
```

**✅ جيد:**
```sad
// إغلاق لطيف - أخبر الطرف الآخر أولاً
دالة اغلاق_لطيف(عميل) {
    محاولة {
        // 1. أرسل رسالة وداع
        عميل.ارسل_الكل("BYE")
        
        // 2. أوقف الإرسال (لكن يمكن الاستقبال)
        عميل.اوقف_الإرسال()
        
        // 3. انتظر رد الطرف الآخر (مع مهلة)
        عميل.عين_مهلة_الاستقبال(1000)
        رد = عميل.استقبل(16)
        
        // 4. الآن أغلق تماماً
        عميل.اغلق()
        
        اطبع("✅ تم الإغلاق بأناقة")
        
    } اصطياد (خطأ) {
        // لا مشكلة - أغلق على أي حال
        محاولة {
            عميل.اغلق()
        } اصطياد {}
    }
}
```

### 10. راقب الأداء والموارد

**المبدأ:** تتبع استهلاك الموارد لاكتشاف المشاكل مبكراً.

**✅ جيد:**
```sad
صنف مراقب_أداء {
    عدد_طلبات: رقم
    عدد_أخطاء: رقم
    وقت_بداية: رقم
    بايتات_مرسلة: رقم
    بايتات_مستقبلة: رقم
    
    دالة __init__() {
        هذا.عدد_طلبات = 0
        هذا.عدد_أخطاء = 0
        هذا.وقت_بداية = وقت.الآن()
        هذا.بايتات_مرسلة = 0
        هذا.بايتات_مستقبلة = 0
    }
    
    دالة سجل_طلب(نجح، بايتات_مرسلة، بايتات_مستقبلة) {
        هذا.عدد_طلبات = هذا.عدد_طلبات + 1
        
        إذا !نجح {
            هذا.عدد_أخطاء = هذا.عدد_أخطاء + 1
        }
        
        هذا.بايتات_مرسلة = هذا.بايتات_مرسلة + بايتات_مرسلة
        هذا.بايتات_مستقبلة = هذا.بايتات_مستقبلة + بايتات_مستقبلة
    }
    
    دالة احصائيات() {
        مدة = (وقت.الآن() - هذا.وقت_بداية) / 1000.0
        
        اطبع("📊 إحصائيات الأداء:")
        اطبع("  المدة: " + مدة.مقرب(2) + " ثانية")
        اطبع("  الطلبات: " + هذا.عدد_طلبات)
        اطبع("  الأخطاء: " + هذا.عدد_أخطاء + 
              " (" + ((هذا.عدد_أخطاء * 100) / هذا.عدد_طلبات).مقرب(2) + "%)")
        اطبع("  البيانات المرسلة: " + حجم_بشري(هذا.بايتات_مرسلة))
        اطبع("  البيانات المستقبلة: " + حجم_بشري(هذا.بايتات_مستقبلة))
        اطبع("  السرعة: " + (هذا.عدد_طلبات / مدة).مقرب(2) + " طلب/ثانية")
    }
}

// الاستخدام
مراقب = مراقب_أداء()

لكل طلب في الطلبات {
    محاولة {
        عميل = شبكة.اتصل_tcp("server.com", 8080)
        عميل.ارسل_الكل(طلب)
        رد = عميل.استقبل(4096)
        
        مراقب.سجل_طلب(صحيح، طلب.طول()، رد.طول())
        
    } اصطياد (خطأ) {
        مراقب.سجل_طلب(خطأ، 0، 0)
    }
}

مراقب.احصائيات()
```

## 🎓 خلاصة

هذه الممارسات ليست مجرد "اقتراحات" - إنها دروس مستفادة من سنوات من الخبرة وآلاف المشاكل المحلولة. اتبعها لبناء تطبيقات:

✅ **موثوقة** - تعمل حتى في أسوأ الظروف  
✅ **آمنة** - محمية من الهجمات والاستغلال  
✅ **قابلة للصيانة** - يسهل فهمها وتعديلها  
✅ **فعالة** - تستخدم الموارد بحكمة  
✅ **احترافية** - جاهزة للإنتاج
