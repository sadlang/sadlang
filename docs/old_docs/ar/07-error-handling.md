# ⚠️ معالجة الأخطاء

## مقدمة

الأخطاء جزء طبيعي من البرمجة. معالجة الأخطاء بشكل صحيح تجعل برامجك:
- **أكثر موثوقية**: لا تنهار عند حدوث مشكلة
- **أسهل في التصحيح**: رسائل واضحة تساعد على الفهم
- **أفضل تجربة للمستخدم**: رسائل مفهومة بدلاً من انهيار البرنامج

---

## 🔍 أنواع الأخطاء

### 1. أخطاء الصياغة (Syntax Errors)
تحدث عند كتابة كود غير صحيح:

```س
# ❌ خطأ: نسيان كلمة نهاية
دالة رئيسية()
    اطبع("مرحبا")
# نهاية  ← منسية!

# ❌ خطأ: أقواس غير متطابقة
اطبع("مرحبا"  # ← قوس ناقص
```

### 2. أخطاء وقت التشغيل (Runtime Errors)
تحدث أثناء تنفيذ البرنامج:

```س
# القسمة على صفر
رقم نتيجة = 10 / 0  # ❌ خطأ!

# الوصول لفهرس غير موجود
مصفوفة أرقام = [1، 2، 3]
اطبع(أرقام[10])  # ❌ خطأ!

# استخدام متغير فارغ
نص اسم = فارغ
اطبع(طول(اسم))  # ❌ خطأ!
```

### 3. أخطاء منطقية (Logic Errors)
البرنامج يعمل لكن النتيجة خاطئة:

```س
# ❌ خطأ منطقي: استخدام < بدلاً من >
إذا (الدرجة < 60)  # كان يجب أن يكون >=
    اطبع("ناجح")
نهاية
```

## 🛡️ معالجة الاستثناءات

### حاول/اصطاد (try/catch)

```س
دالة رئيسية()
    حاول
        # الكود الذي قد يسبب خطأ
        رقم نتيجة = 10 / 0
        اطبع(نتيجة)
    اصطاد (خطأ)
        # التعامل مع الخطأ
        اطبع("❌ حدث خطأ: ")
        اطبع(خطأ.رسالة)
    نهاية
    
    اطبع("\n✅ البرنامج استمر في العمل!")
    ارجع 0
نهاية
```

### اصطياد أنواع محددة من الأخطاء

```س
دالة رئيسية()
    حاول
        مصفوفة أرقام = [1، 2، 3]
        اطبع(أرقام[10])
    اصطاد (خطأ_فهرس خ)
        اطبع("❌ خطأ في الفهرس: الموقع غير موجود\n")
    اصطاد (خطأ_نوع خ)
        اطبع("❌ خطأ في النوع\n")
    اصطاد (خطأ خ)
        # أي خطأ آخر
        اطبع("❌ خطأ غير متوقع: " + خ.رسالة + "\n")
    نهاية
    
    ارجع 0
نهاية
```

### كتلة أخيراً (finally)

تُنفَّذ دائماً، سواء حدث خطأ أم لا:

```س
دالة اقرأ_ملف(نص مسار)
    متغير ملف = فارغ
    
    حاول
        ملف = افتح(مسار، "قراءة")
        نص محتوى = ملف.اقرأ_الكل()
        اطبع(محتوى)
    اصطاد (خطأ_ملف خ)
        اطبع("❌ لم يتم العثور على الملف: " + مسار + "\n")
    أخيراً
        # يُنفَّذ دائماً - مهم لتنظيف الموارد
        إذا (ملف != فارغ)
            ملف.أغلق()
            اطبع("📁 تم إغلاق الملف\n")
        نهاية
    نهاية
نهاية
```

## 🚨 إنشاء استثناءات مخصصة

### تعريف استثناء مخصص

```س
# تعريف استثناء مخصص
استثناء خطأ_تحقق يرث خطأ
    نص الحقل
    
    دالة أنشئ(نص رسالة، نص حقل)
        فوق.أنشئ(رسالة)
        هذا.الحقل = حقل
    نهاية
نهاية

استثناء خطأ_رصيد_غير_كافٍ يرث خطأ
    رقم الرصيد_الحالي
    رقم المبلغ_المطلوب
    
    دالة أنشئ(رقم رصيد، رقم مبلغ)
        فوق.أنشئ("الرصيد غير كافٍ")
        هذا.الرصيد_الحالي = رصيد
        هذا.المبلغ_المطلوب = مبلغ
    نهاية
نهاية
```

### رمي استثناء

```س
دالة سجل_مستخدم(نص الاسم، نص البريد، رقم العمر)
    # التحقق من الاسم
    إذا (طول(الاسم) < 2)
        ارمِ خطأ_تحقق.أنشئ("الاسم قصير جداً"، "الاسم")
    نهاية
    
    # التحقق من البريد
    إذا (ليس يحتوي(البريد، "@"))
        ارمِ خطأ_تحقق.أنشئ("البريد غير صالح"، "البريد")
    نهاية
    
    # التحقق من العمر
    إذا (العمر < 13 أو العمر > 120)
        ارمِ خطأ_تحقق.أنشئ("العمر غير صالح"، "العمر")
    نهاية
    
    اطبع("✅ تم تسجيل المستخدم: " + الاسم + "\n")
نهاية

دالة رئيسية()
    حاول
        سجل_مستخدم("أ"، "test@example.com"، 25)
    اصطاد (خطأ_تحقق خ)
        اطبع("❌ خطأ في الحقل '" + خ.الحقل + "': " + خ.رسالة + "\n")
    نهاية
    
    حاول
        سجل_مستخدم("أحمد"، "invalid-email"، 25)
    اصطاد (خطأ_تحقق خ)
        اطبع("❌ خطأ في الحقل '" + خ.الحقل + "': " + خ.رسالة + "\n")
    نهاية
    
    حاول
        سجل_مستخدم("أحمد"، "ahmed@example.com"، 25)
    اصطاد (خطأ_تحقق خ)
        اطبع("❌ خطأ في الحقل '" + خ.الحقل + "': " + خ.رسالة + "\n")
    نهاية
    
    ارجع 0
نهاية
```

## 📋 أنماط معالجة الأخطاء

### 1. نمط النتيجة (Result Pattern)

```س
# تعريف نوع النتيجة
فئة نتيجة<ن>
    ن القيمة
    نص الخطأ
    منطقي ناجح
    
    دالة نجاح(ن قيمة) -> نتيجة<ن>
        نتيجة ن = نتيجة.أنشئ()
        ن.القيمة = قيمة
        ن.ناجح = صحيح
        ارجع ن
    نهاية
    
    دالة فشل(نص خطأ) -> نتيجة<ن>
        نتيجة ن = نتيجة.أنشئ()
        ن.الخطأ = خطأ
        ن.ناجح = خطأ
        ارجع ن
    نهاية
نهاية

# استخدام نمط النتيجة
دالة قسمة_آمنة(رقم أ، رقم ب) -> نتيجة<رقم>
    إذا (ب == 0)
        ارجع نتيجة.فشل("لا يمكن القسمة على صفر")
    نهاية
    ارجع نتيجة.نجاح(أ / ب)
نهاية

دالة رئيسية()
    نتيجة ن1 = قسمة_آمنة(10، 2)
    إذا (ن1.ناجح)
        اطبع("النتيجة: ")
        اطبع(ن1.القيمة)
    وإلا
        اطبع("خطأ: " + ن1.الخطأ)
    نهاية
    
    ارجع 0
نهاية
```

### 2. نمط القيمة الاختيارية (Optional)

```س
# دالة قد ترجع قيمة أو فارغ
دالة ابحث_عن_مستخدم(نص الاسم) -> مستخدم؟
    # ... البحث في قاعدة البيانات
    إذا (ليس_موجود)
        ارجع فارغ
    نهاية
    ارجع المستخدم
نهاية

دالة رئيسية()
    مستخدم؟ م = ابحث_عن_مستخدم("أحمد")
    
    # التحقق الآمن من القيمة
    إذا (م != فارغ)
        اطبع("مرحباً " + م.الاسم)
    وإلا
        اطبع("المستخدم غير موجود")
    نهاية
    
    # أو باستخدام المعامل الآمن
    اطبع(م?.الاسم ?? "غير معروف")
    
    ارجع 0
نهاية
```

### 3. نمط التأكيد (Assertions)

```س
دالة احسب_الجذر_التربيعي(عشري س) -> عشري
    # التأكيد: س يجب أن يكون غير سالب
    تأكد(س >= 0، "لا يمكن حساب الجذر التربيعي لرقم سالب")
    
    ارجع جذر(س)
نهاية

دالة رئيسية()
    اطبع(احسب_الجذر_التربيعي(16))  # ✅ يعمل: 4
    اطبع(احسب_الجذر_التربيعي(-4)) # ❌ يفشل مع رسالة واضحة
    
    ارجع 0
نهاية
```

## 🎯 أمثلة عملية

### مثال 1: قراءة ملف إعدادات

```س
دالة اقرأ_الإعدادات(نص مسار_الملف) -> قاموس
    قاموس إعدادات = {}
    
    حاول
        متغير ملف = افتح(مسار_الملف، "قراءة")
        
        حاول
            لكل (سطر في ملف.اقرأ_أسطر())
                # تجاهل الأسطر الفارغة والتعليقات
                إذا (طول(قص(سطر)) == 0 أو يبدأ_بـ(سطر، "#"))
                    استمر
                نهاية
                
                # تقسيم السطر: مفتاح=قيمة
                مصفوفة أجزاء = قسّم(سطر، "=")
                إذا (طول(أجزاء) != 2)
                    ارمِ خطأ.أنشئ("صيغة غير صالحة: " + سطر)
                نهاية
                
                نص مفتاح = قص(أجزاء[0])
                نص قيمة = قص(أجزاء[1])
                إعدادات[مفتاح] = قيمة
            نهاية
        أخيراً
            ملف.أغلق()
        نهاية
        
    اصطاد (خطأ_ملف خ)
        اطبع("⚠️ ملف الإعدادات غير موجود، جاري استخدام القيم الافتراضية\n")
        إعدادات = {
            "اللغة": "ar"،
            "المنفذ": "8080"،
            "وضع_التصحيح": "خطأ"
        }
    اصطاد (خطأ خ)
        اطبع("❌ خطأ في قراءة الإعدادات: " + خ.رسالة + "\n")
        ارمِ خ  # إعادة رمي الخطأ
    نهاية
    
    ارجع إعدادات
نهاية
```

### مثال 2: التحقق من صحة النموذج

```س
فئة محقق_النموذج
    مصفوفة الأخطاء
    
    دالة أنشئ()
        هذا.الأخطاء = []
    نهاية
    
    دالة تحقق_من_مطلوب(قيمة، نص اسم_الحقل) -> هذا
        إذا (قيمة == فارغ أو قيمة == "")
            أضف(هذا.الأخطاء، اسم_الحقل + " مطلوب")
        نهاية
        ارجع هذا  # للسلسلة
    نهاية
    
    دالة تحقق_من_البريد(نص البريد) -> هذا
        إذا (ليس يحتوي(البريد، "@") أو ليس يحتوي(البريد، "."))
            أضف(هذا.الأخطاء، "البريد الإلكتروني غير صالح")
        نهاية
        ارجع هذا
    نهاية
    
    دالة تحقق_من_الطول(نص النص، رقم الحد_الأدنى، رقم الحد_الأقصى، نص اسم_الحقل) -> هذا
        رقم الطول = طول(النص)
        إذا (الطول < الحد_الأدنى)
            أضف(هذا.الأخطاء، اسم_الحقل + " قصير جداً (الحد الأدنى: " + لنص(الحد_الأدنى) + ")")
        وإلا_إذا (الطول > الحد_الأقصى)
            أضف(هذا.الأخطاء، اسم_الحقل + " طويل جداً (الحد الأقصى: " + لنص(الحد_الأقصى) + ")")
        نهاية
        ارجع هذا
    نهاية
    
    دالة تحقق_من_الرقم_في_نطاق(رقم القيمة، رقم الحد_الأدنى، رقم الحد_الأقصى، نص اسم_الحقل) -> هذا
        إذا (القيمة < الحد_الأدنى أو القيمة > الحد_الأقصى)
            أضف(هذا.الأخطاء، اسم_الحقل + " يجب أن يكون بين " + لنص(الحد_الأدنى) + " و " + لنص(الحد_الأقصى))
        نهاية
        ارجع هذا
    نهاية
    
    دالة صالح() -> منطقي
        ارجع طول(هذا.الأخطاء) == 0
    نهاية
    
    دالة اعرض_الأخطاء()
        إذا (هذا.صالح())
            اطبع("✅ النموذج صالح\n")
        وإلا
            اطبع("❌ أخطاء التحقق:\n")
            لكل (خطأ في هذا.الأخطاء)
                اطبع("   • " + خطأ + "\n")
            نهاية
        نهاية
    نهاية
نهاية

دالة رئيسية()
    # بيانات النموذج
    نص الاسم = "أ"
    نص البريد = "invalid"
    رقم العمر = 10
    
    # التحقق
    محقق_النموذج محقق = محقق_النموذج.أنشئ()
    محقق
        .تحقق_من_مطلوب(الاسم، "الاسم")
        .تحقق_من_الطول(الاسم، 2، 50، "الاسم")
        .تحقق_من_مطلوب(البريد، "البريد")
        .تحقق_من_البريد(البريد)
        .تحقق_من_الرقم_في_نطاق(العمر، 13، 120، "العمر")
    
    محقق.اعرض_الأخطاء()
    
    إذا (محقق.صالح())
        اطبع("جاري إنشاء الحساب...\n")
    نهاية
    
    ارجع 0
نهاية
```

### مثال 3: معالجة أخطاء الشبكة

```س
دالة أرسل_طلب_http(نص الرابط) -> نص
    رقم محاولات = 0
    رقم الحد_الأقصى = 3
    
    بينما (محاولات < الحد_الأقصى)
        محاولات = محاولات + 1
        
        حاول
            اطبع("📡 المحاولة " + لنص(محاولات) + "...\n")
            
            متغير اتصال = http.اتصل(الرابط)
            نص استجابة = اتصال.اقرأ()
            
            إذا (اتصال.رمز_الحالة >= 400)
                ارمِ خطأ_http.أنشئ(اتصال.رمز_الحالة)
            نهاية
            
            اطبع("✅ نجح الطلب!\n")
            ارجع استجابة
            
        اصطاد (خطأ_انتهاء_المهلة خ)
            اطبع("⏰ انتهت المهلة، جاري إعادة المحاولة...\n")
            انتظر(1000)  # انتظر ثانية
            
        اصطاد (خطأ_اتصال خ)
            اطبع("🔌 فشل الاتصال: " + خ.رسالة + "\n")
            إذا (محاولات >= الحد_الأقصى)
                ارمِ خ
            نهاية
            انتظر(2000)  # انتظر ثانيتين
            
        اصطاد (خطأ_http خ)
            طابق خ.رمز_الحالة
                حالة 404:
                    ارمِ خطأ.أنشئ("الصفحة غير موجودة")
                حالة 500:
                    اطبع("⚠️ خطأ في الخادم، جاري إعادة المحاولة...\n")
                حالة _:
                    ارمِ خ
            نهاية
        نهاية
    نهاية
    
    ارمِ خطأ.أنشئ("فشلت جميع المحاولات")
نهاية
```

## 💡 أفضل الممارسات

### 1. لا تبتلع الأخطاء

```س
# ❌ سيء: إخفاء الخطأ
حاول
    شيء_خطير()
اصطاد (خطأ خ)
    # لا شيء!
نهاية

# ✅ جيد: على الأقل سجّل الخطأ
حاول
    شيء_خطير()
اصطاد (خطأ خ)
    سجّل.خطأ("فشل شيء_خطير: " + خ.رسالة)
نهاية
```

### 2. استخدم رسائل خطأ واضحة

```س
# ❌ سيء
ارمِ خطأ.أنشئ("خطأ")

# ✅ جيد
ارمِ خطأ.أنشئ("فشل حفظ الملف: المسار غير موجود: " + المسار)
```

### 3. نظّف الموارد دائماً

```س
# ✅ استخدم أخيراً للتنظيف
حاول
    ملف = افتح("بيانات.txt"، "قراءة")
    # ... عمليات على الملف
أخيراً
    إذا (ملف != فارغ)
        ملف.أغلق()
    نهاية
نهاية
```

### 4. تجنب الإفراط في حاول/اصطاد

```س
# ❌ سيء: حاول/اصطاد في كل مكان
حاول
    س = 1
اصطاد نهاية
حاول
    ص = 2
اصطاد نهاية

# ✅ جيد: تجميع العمليات المرتبطة
حاول
    س = 1
    ص = 2
    ز = س + ص
اصطاد (خطأ خ)
    # معالجة الخطأ
نهاية
```

## ✅ تمارين

### تمرين 1: قسمة آمنة
اكتب دالة للقسمة تتعامل مع القسمة على صفر.

### تمرين 2: قراءة قائمة من ملف
اكتب دالة تقرأ قائمة أرقام من ملف مع معالجة الأخطاء.

### تمرين 3: التحقق من كلمة المرور
اكتب دالة للتحقق من قوة كلمة المرور مع إرجاع أخطاء محددة.
