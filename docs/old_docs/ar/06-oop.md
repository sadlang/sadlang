# 🏗️ البرمجة الكائنية (OOP)

## مقدمة

البرمجة الكائنية (Object-Oriented Programming) هي نمط برمجي يعتمد على تنظيم الكود حول **الكائنات** بدلاً من الدوال والمنطق فقط.

### المفاهيم الأساسية

| المفهوم | الشرح | مثال |
|---------|-------|------|
| **الفئة (Class)** | قالب لإنشاء كائنات | فئة `سيارة` |
| **الكائن (Object)** | نسخة من الفئة | سيارتي، سيارتك |
| **الخصائص (Properties)** | بيانات الكائن | اللون، السرعة |
| **الدوال (Methods)** | سلوكيات الكائن | تسريع، فرملة |

## 📦 الفئات والكائنات

### تعريف فئة بسيطة

```س
# تعريف فئة السيارة
فئة سيارة
    # الخصائص
    نص اللون
    نص الماركة
    رقم السنة
    رقم السرعة_الحالية
    
    # دالة المُنشئ (Constructor)
    دالة أنشئ(نص لون، نص ماركة، رقم سنة)
        هذا.اللون = لون
        هذا.الماركة = ماركة
        هذا.السنة = سنة
        هذا.السرعة_الحالية = 0
    نهاية
    
    # دالة التسريع
    دالة سرّع(رقم مقدار)
        هذا.السرعة_الحالية = هذا.السرعة_الحالية + مقدار
        اطبع("السرعة الآن: ")
        اطبع(هذا.السرعة_الحالية)
        اطبع(" كم/س\n")
    نهاية
    
    # دالة الفرملة
    دالة افرمل(رقم مقدار)
        هذا.السرعة_الحالية = هذا.السرعة_الحالية - مقدار
        إذا (هذا.السرعة_الحالية < 0)
            هذا.السرعة_الحالية = 0
        نهاية
    نهاية
    
    # دالة عرض المعلومات
    دالة اعرض()
        اطبع("=== معلومات السيارة ===\n")
        اطبع("الماركة: " + هذا.الماركة + "\n")
        اطبع("اللون: " + هذا.اللون + "\n")
        اطبع("السنة: ")
        اطبع(هذا.السنة)
        اطبع("\nالسرعة: ")
        اطبع(هذا.السرعة_الحالية)
        اطبع(" كم/س\n")
    نهاية
نهاية
```

### استخدام الفئة

```س
دالة رئيسية()
    # إنشاء كائن من فئة السيارة
    سيارة سيارتي = سيارة.أنشئ("أحمر"، "تويوتا"، 2023)
    
    # عرض المعلومات
    سيارتي.اعرض()
    
    # استخدام الدوال
    سيارتي.سرّع(50)   # السرعة: 50
    سيارتي.سرّع(30)   # السرعة: 80
    سيارتي.افرمل(20)  # السرعة: 60
    
    # الوصول المباشر للخصائص
    اطبع("لون سيارتي: " + سيارتي.اللون)
    
    ارجع 0
نهاية
```

## 🔒 التغليف (Encapsulation)

التغليف يحمي البيانات الداخلية من الوصول المباشر:

```س
فئة حساب_بنكي
    # خصائص خاصة (لا يمكن الوصول إليها مباشرة من الخارج)
    خاص رقم _الرصيد
    خاص نص _رقم_الحساب
    نص اسم_صاحب_الحساب
    
    دالة أنشئ(نص الاسم، نص رقم_الحساب، رقم الرصيد_الابتدائي)
        هذا.اسم_صاحب_الحساب = الاسم
        هذا._رقم_الحساب = رقم_الحساب
        هذا._الرصيد = الرصيد_الابتدائي
    نهاية
    
    # دالة الإيداع
    دالة أودع(رقم المبلغ)
        إذا (المبلغ > 0)
            هذا._الرصيد = هذا._الرصيد + المبلغ
            اطبع("تم إيداع ")
            اطبع(المبلغ)
            اطبع(" ريال\n")
            ارجع صحيح
        نهاية
        اطبع("❌ المبلغ يجب أن يكون موجباً\n")
        ارجع خطأ
    نهاية
    
    # دالة السحب
    دالة اسحب(رقم المبلغ) -> منطقي
        إذا (المبلغ <= 0)
            اطبع("❌ المبلغ يجب أن يكون موجباً\n")
            ارجع خطأ
        نهاية
        
        إذا (المبلغ > هذا._الرصيد)
            اطبع("❌ الرصيد غير كافٍ\n")
            ارجع خطأ
        نهاية
        
        هذا._الرصيد = هذا._الرصيد - المبلغ
        اطبع("تم سحب ")
        اطبع(المبلغ)
        اطبع(" ريال\n")
        ارجع صحيح
    نهاية
    
    # دالة الحصول على الرصيد (قراءة فقط)
    دالة الرصيد() -> رقم
        ارجع هذا._الرصيد
    نهاية
    
    # دالة عرض معلومات الحساب
    دالة اعرض_الملخص()
        اطبع("=== ملخص الحساب ===\n")
        اطبع("صاحب الحساب: " + هذا.اسم_صاحب_الحساب + "\n")
        اطبع("رقم الحساب: ***" + هذا._رقم_الحساب[-4:] + "\n")
        اطبع("الرصيد: ")
        اطبع(هذا._الرصيد)
        اطبع(" ريال\n")
    نهاية
نهاية

دالة رئيسية()
    حساب_بنكي حسابي = حساب_بنكي.أنشئ("أحمد محمد"، "1234567890"، 1000)
    
    حسابي.اعرض_الملخص()
    
    حسابي.أودع(500)        # تم إيداع 500 ريال
    حسابي.اسحب(200)        # تم سحب 200 ريال
    حسابي.اسحب(2000)       # الرصيد غير كافٍ
    
    اطبع("\nالرصيد الحالي: ")
    اطبع(حسابي.الرصيد())
    
    ارجع 0
نهاية
```

## 👪 الوراثة (Inheritance)

الوراثة تسمح لفئة بأن ترث خصائص ودوال فئة أخرى:

```س
# الفئة الأساسية
فئة حيوان
    نص الاسم
    رقم العمر
    
    دالة أنشئ(نص اسم، رقم عمر)
        هذا.الاسم = اسم
        هذا.العمر = عمر
    نهاية
    
    دالة أصدر_صوت()
        اطبع("...")
    نهاية
    
    دالة تحرك()
        اطبع(هذا.الاسم + " يتحرك\n")
    نهاية
    
    دالة اعرض()
        اطبع("الاسم: " + هذا.الاسم + "\n")
        اطبع("العمر: ")
        اطبع(هذا.العمر)
        اطبع(" سنوات\n")
    نهاية
نهاية

# فئة مشتقة: القط
فئة قط يرث حيوان
    نص اللون
    
    دالة أنشئ(نص اسم، رقم عمر، نص لون)
        فوق.أنشئ(اسم، عمر)  # استدعاء منشئ الفئة الأب
        هذا.اللون = لون
    نهاية
    
    # إعادة تعريف دالة الصوت
    دالة أصدر_صوت()
        اطبع(هذا.الاسم + ": مياو! 🐱\n")
    نهاية
    
    # دالة خاصة بالقط
    دالة تسلق()
        اطبع(هذا.الاسم + " يتسلق الشجرة 🌳\n")
    نهاية
نهاية

# فئة مشتقة: الكلب
فئة كلب يرث حيوان
    نص السلالة
    
    دالة أنشئ(نص اسم، رقم عمر، نص سلالة)
        فوق.أنشئ(اسم، عمر)
        هذا.السلالة = سلالة
    نهاية
    
    دالة أصدر_صوت()
        اطبع(هذا.الاسم + ": هاو هاو! 🐕\n")
    نهاية
    
    دالة احضر_الكرة()
        اطبع(هذا.الاسم + " يحضر الكرة ⚽\n")
    نهاية
نهاية

دالة رئيسية()
    قط بسبس = قط.أنشئ("بسبس"، 3، "أبيض")
    كلب ريكس = كلب.أنشئ("ريكس"، 5، "جيرمن شيبرد")
    
    # استخدام الدوال الموروثة
    بسبس.اعرض()
    بسبس.تحرك()
    بسبس.أصدر_صوت()
    بسبس.تسلق()
    
    اطبع("\n---\n\n")
    
    ريكس.اعرض()
    ريكس.تحرك()
    ريكس.أصدر_صوت()
    ريكس.احضر_الكرة()
    
    ارجع 0
نهاية
```

## 🎭 تعدد الأشكال (Polymorphism)

تعدد الأشكال يسمح لكائنات مختلفة بالاستجابة لنفس الرسالة بطرق مختلفة:

```س
فئة شكل
    دالة احسب_المساحة() -> عشري
        ارجع 0.0
    نهاية
    
    دالة اعرض()
        اطبع("شكل غير محدد")
    نهاية
نهاية

فئة مستطيل يرث شكل
    عشري الطول
    عشري العرض
    
    دالة أنشئ(عشري طول، عشري عرض)
        هذا.الطول = طول
        هذا.العرض = عرض
    نهاية
    
    دالة احسب_المساحة() -> عشري
        ارجع هذا.الطول * هذا.العرض
    نهاية
    
    دالة اعرض()
        اطبع("مستطيل ")
        اطبع(هذا.الطول)
        اطبع(" × ")
        اطبع(هذا.العرض)
    نهاية
نهاية

فئة دائرة يرث شكل
    عشري نصف_القطر
    
    دالة أنشئ(عشري نصف_قطر)
        هذا.نصف_القطر = نصف_قطر
    نهاية
    
    دالة احسب_المساحة() -> عشري
        # π ≈ 3.14159
        ارجع 3.14159 * هذا.نصف_القطر * هذا.نصف_القطر
    نهاية
    
    دالة اعرض()
        اطبع("دائرة نصف قطرها ")
        اطبع(هذا.نصف_القطر)
    نهاية
نهاية

فئة مثلث يرث شكل
    عشري القاعدة
    عشري الارتفاع
    
    دالة أنشئ(عشري قاعدة، عشري ارتفاع)
        هذا.القاعدة = قاعدة
        هذا.الارتفاع = ارتفاع
    نهاية
    
    دالة احسب_المساحة() -> عشري
        ارجع 0.5 * هذا.القاعدة * هذا.الارتفاع
    نهاية
    
    دالة اعرض()
        اطبع("مثلث قاعدته ")
        اطبع(هذا.القاعدة)
        اطبع(" وارتفاعه ")
        اطبع(هذا.الارتفاع)
    نهاية
نهاية

# دالة تعمل مع أي شكل (تعدد الأشكال)
دالة اطبع_معلومات_الشكل(شكل ش)
    ش.اعرض()
    اطبع(" - المساحة: ")
    اطبع(ش.احسب_المساحة())
    اطبع("\n")
نهاية

دالة رئيسية()
    # إنشاء أشكال مختلفة
    مصفوفة أشكال = [
        مستطيل.أنشئ(5.0، 3.0)،
        دائرة.أنشئ(4.0)،
        مثلث.أنشئ(6.0، 4.0)،
        مستطيل.أنشئ(10.0، 2.0)
    ]
    
    اطبع("=== الأشكال ومساحاتها ===\n")
    
    # تعدد الأشكال: نفس الدالة تعمل مع أنواع مختلفة
    لكل (شكل في أشكال)
        اطبع_معلومات_الشكل(شكل)
    نهاية
    
    # حساب المساحة الإجمالية
    عشري المجموع = 0.0
    لكل (شكل في أشكال)
        المجموع = المجموع + شكل.احسب_المساحة()
    نهاية
    
    اطبع("\nالمساحة الإجمالية: ")
    اطبع(المجموع)
    
    ارجع 0
نهاية
```

## 🔧 الواجهات (Interfaces)

الواجهة تحدد العقد الذي يجب أن تلتزم به الفئات:

```س
# تعريف واجهة
واجهة قابل_للطباعة
    دالة اطبع_نفسك()
نهاية

واجهة قابل_للمقارنة
    دالة قارن(آخر) -> رقم  # -1، 0، 1
نهاية

# فئة تطبق واجهات متعددة
فئة منتج يطبق قابل_للطباعة، قابل_للمقارنة
    نص الاسم
    عشري السعر
    
    دالة أنشئ(نص اسم، عشري سعر)
        هذا.الاسم = اسم
        هذا.السعر = سعر
    نهاية
    
    # تطبيق واجهة قابل_للطباعة
    دالة اطبع_نفسك()
        اطبع("المنتج: " + هذا.الاسم)
        اطبع(" - السعر: ")
        اطبع(هذا.السعر)
        اطبع(" ريال\n")
    نهاية
    
    # تطبيق واجهة قابل_للمقارنة
    دالة قارن(منتج آخر) -> رقم
        إذا (هذا.السعر < آخر.السعر)
            ارجع -1
        وإلا_إذا (هذا.السعر > آخر.السعر)
            ارجع 1
        وإلا
            ارجع 0
        نهاية
    نهاية
نهاية

دالة رئيسية()
    مصفوفة منتجات = [
        منتج.أنشئ("لابتوب"، 3500.0)،
        منتج.أنشئ("هاتف"، 2000.0)،
        منتج.أنشئ("سماعات"، 500.0)
    ]
    
    # طباعة جميع المنتجات (باستخدام الواجهة)
    لكل (م في منتجات)
        م.اطبع_نفسك()
    نهاية
    
    ارجع 0
نهاية
```

## 🎯 مثال شامل: نظام إدارة المكتبة

```س
# فئة الكتاب
فئة كتاب
    نص العنوان
    نص المؤلف
    نص الرقم_التسلسلي
    منطقي متاح
    
    دالة أنشئ(نص عنوان، نص مؤلف، نص رقم)
        هذا.العنوان = عنوان
        هذا.المؤلف = مؤلف
        هذا.الرقم_التسلسلي = رقم
        هذا.متاح = صحيح
    نهاية
    
    دالة اعرض()
        اطبع("📚 " + هذا.العنوان + "\n")
        اطبع("   المؤلف: " + هذا.المؤلف + "\n")
        اطبع("   الرقم: " + هذا.الرقم_التسلسلي + "\n")
        إذا (هذا.متاح)
            اطبع("   الحالة: متاح ✅\n")
        وإلا
            اطبع("   الحالة: مُعار ❌\n")
        نهاية
    نهاية
نهاية

# فئة العضو
فئة عضو
    نص الاسم
    نص رقم_العضوية
    مصفوفة الكتب_المستعارة
    
    دالة أنشئ(نص اسم، نص رقم)
        هذا.الاسم = اسم
        هذا.رقم_العضوية = رقم
        هذا.الكتب_المستعارة = []
    نهاية
    
    دالة استعر(كتاب ك) -> منطقي
        إذا (ليس ك.متاح)
            اطبع("❌ الكتاب غير متاح للإعارة\n")
            ارجع خطأ
        نهاية
        
        إذا (طول(هذا.الكتب_المستعارة) >= 3)
            اطبع("❌ لا يمكن استعارة أكثر من 3 كتب\n")
            ارجع خطأ
        نهاية
        
        ك.متاح = خطأ
        أضف(هذا.الكتب_المستعارة، ك)
        اطبع("✅ تم استعارة: " + ك.العنوان + "\n")
        ارجع صحيح
    نهاية
    
    دالة أرجع(كتاب ك) -> منطقي
        لكل (كتاب_مستعار، فهرس في هذا.الكتب_المستعارة)
            إذا (كتاب_مستعار.الرقم_التسلسلي == ك.الرقم_التسلسلي)
                ك.متاح = صحيح
                احذف(هذا.الكتب_المستعارة، فهرس)
                اطبع("✅ تم إرجاع: " + ك.العنوان + "\n")
                ارجع صحيح
            نهاية
        نهاية
        
        اطبع("❌ هذا الكتاب غير مستعار من هذا العضو\n")
        ارجع خطأ
    نهاية
    
    دالة اعرض_المستعارات()
        اطبع("📋 الكتب المستعارة لـ " + هذا.الاسم + ":\n")
        إذا (طول(هذا.الكتب_المستعارة) == 0)
            اطبع("   لا توجد كتب مستعارة\n")
        وإلا
            لكل (ك في هذا.الكتب_المستعارة)
                اطبع("   - " + ك.العنوان + "\n")
            نهاية
        نهاية
    نهاية
نهاية

# فئة المكتبة
فئة مكتبة
    نص الاسم
    مصفوفة الكتب
    مصفوفة الأعضاء
    
    دالة أنشئ(نص اسم)
        هذا.الاسم = اسم
        هذا.الكتب = []
        هذا.الأعضاء = []
    نهاية
    
    دالة أضف_كتاب(كتاب ك)
        أضف(هذا.الكتب، ك)
        اطبع("📚 تمت إضافة: " + ك.العنوان + "\n")
    نهاية
    
    دالة سجل_عضو(عضو ع)
        أضف(هذا.الأعضاء، ع)
        اطبع("👤 تم تسجيل العضو: " + ع.الاسم + "\n")
    نهاية
    
    دالة ابحث_عن_كتاب(نص العنوان) -> كتاب
        لكل (ك في هذا.الكتب)
            إذا (يحتوي(ك.العنوان، العنوان))
                ارجع ك
            نهاية
        نهاية
        ارجع فارغ
    نهاية
    
    دالة اعرض_الكتب_المتاحة()
        اطبع("=== الكتب المتاحة في " + هذا.الاسم + " ===\n")
        لكل (ك في هذا.الكتب)
            إذا (ك.متاح)
                ك.اعرض()
            نهاية
        نهاية
    نهاية
    
    دالة إحصائيات()
        رقم متاح = 0
        رقم مُعار = 0
        
        لكل (ك في هذا.الكتب)
            إذا (ك.متاح)
                متاح = متاح + 1
            وإلا
                مُعار = مُعار + 1
            نهاية
        نهاية
        
        اطبع("=== إحصائيات المكتبة ===\n")
        اطبع("إجمالي الكتب: ")
        اطبع(طول(هذا.الكتب))
        اطبع("\nالكتب المتاحة: ")
        اطبع(متاح)
        اطبع("\nالكتب المُعارة: ")
        اطبع(مُعار)
        اطبع("\nعدد الأعضاء: ")
        اطبع(طول(هذا.الأعضاء))
        اطبع("\n")
    نهاية
نهاية

دالة رئيسية()
    # إنشاء المكتبة
    مكتبة مكتبتنا = مكتبة.أنشئ("مكتبة المعرفة")
    
    # إضافة كتب
    مكتبتنا.أضف_كتاب(كتاب.أنشئ("مقدمة في البرمجة"، "أحمد علي"، "001"))
    مكتبتنا.أضف_كتاب(كتاب.أنشئ("تعلم لغة ص"، "محمد سعيد"، "002"))
    مكتبتنا.أضف_كتاب(كتاب.أنشئ("الخوارزميات"، "سارة خالد"، "003"))
    
    # تسجيل أعضاء
    عضو أحمد = عضو.أنشئ("أحمد محمد"، "M001")
    مكتبتنا.سجل_عضو(أحمد)
    
    اطبع("\n")
    
    # عرض الكتب المتاحة
    مكتبتنا.اعرض_الكتب_المتاحة()
    
    # استعارة كتاب
    اطبع("\n--- عملية الاستعارة ---\n")
    كتاب كتاب_برمجة = مكتبتنا.ابحث_عن_كتاب("برمجة")
    إذا (كتاب_برمجة != فارغ)
        أحمد.استعر(كتاب_برمجة)
    نهاية
    
    # عرض الكتب المستعارة
    اطبع("\n")
    أحمد.اعرض_المستعارات()
    
    # الإحصائيات
    اطبع("\n")
    مكتبتنا.إحصائيات()
    
    ارجع 0
نهاية
```
