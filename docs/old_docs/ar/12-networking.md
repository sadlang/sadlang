# 🌐 برمجة الشبكات

## مقدمة

لغة ص توفر دعماً قوياً لبرمجة الشبكات، مما يمكّنك من:
- إنشاء خوادم وعملاء TCP/UDP
- التعامل مع WebSocket
- إرسال طلبات HTTP
- بناء تطبيقات الويب

---

## 📡 بروتوكول TCP

### خادم TCP بسيط

```س
استورد شبكة من "stdlib/network"

دالة رئيسية()
    اطبع("🚀 بدء خادم TCP...\n")
    
    # إنشاء مقبس الخادم
    متغير خادم = شبكة.مقبس_tcp()
    
    # السماح بإعادة استخدام العنوان
    خادم.عين_اعادة_استخدام_عنوان(صحيح)
    
    # الربط بالمنفذ 8080
    حاول
        خادم.اربط(8080)
        اطبع("✅ تم ربط الخادم بالمنفذ 8080\n")
    اصطاد (خطأ)
        اطبع("❌ خطأ في الربط: " + خطأ.رسالة + "\n")
        ارجع 1
    نهاية
    
    # الاستماع للاتصالات
    خادم.استمع(10)
    اطبع("👂 في انتظار الاتصالات...\n")
    
    # حلقة قبول الاتصالات
    بينما (صحيح)
        # قبول عميل جديد
        متغير عميل = خادم.اقبل()
        اطبع("✨ اتصل عميل جديد!\n")
        
        # إرسال رسالة ترحيب
        عميل.ارسل("مرحباً بك في الخادم!\n")
        
        # قراءة البيانات من العميل
        بينما (صحيح)
            نص بيانات = عميل.استقبل(1024)
            
            إذا (طول(بيانات) == 0)
                اطبع("👋 انقطع اتصال العميل\n")
                اكسر
            نهاية
            
            اطبع("📨 استقبل: " + بيانات)
            
            # إرجاع الرسالة (Echo)
            عميل.ارسل("الرد: " + بيانات)
        نهاية
        
        عميل.اغلق()
    نهاية
    
    ارجع 0
نهاية
```

## 📮 بروتوكول UDP

### خادم UDP

```س
استورد شبكة من "stdlib/network"

دالة رئيسية()
    اطبع("🚀 بدء خادم UDP...\n")
    
    متغير خادم = شبكة.مقبس_udp()
    خادم.اربط(9000)
    
    اطبع("👂 في انتظار الرسائل على المنفذ 9000...\n")
    
    بينما (صحيح)
        # استقبال رسالة مع عنوان المرسل
        (بيانات، عنوان) = خادم.استقبل_من(1024)
        
        اطبع("📨 استقبل من " + عنوان.لنص() + ": " + بيانات)
        
        # إرسال رد
        خادم.ارسل_إلى("تم استلام رسالتك!\n"، عنوان)
    نهاية
    
    ارجع 0
نهاية
```

## 🔗 WebSocket

### خادم WebSocket

```س
استورد ويب_سوكت من "stdlib/network/websocket"

دالة رئيسية()
    اطبع("🚀 بدء خادم WebSocket...\n")
    
    متغير خادم = ويب_سوكت.خادم(8765)
    
    # معالج الاتصال الجديد
    خادم.عند_اتصال = دالة(عميل)
        اطبع("✨ عميل جديد اتصل\n")
        عميل.ارسل("مرحباً بك!")
    نهاية
    
    # معالج الرسائل
    خادم.عند_رسالة = دالة(عميل، رسالة)
        اطبع("📨 استقبل: " + رسالة + "\n")
        
        # بث الرسالة لجميع العملاء
        لكل (ع في خادم.العملاء)
            ع.ارسل("بث: " + رسالة)
        نهاية
    نهاية
    
    # معالج قطع الاتصال
    خادم.عند_قطع_اتصال = دالة(عميل)
        اطبع("👋 عميل غادر\n")
    نهاية
    
    خادم.ابدأ()
    اطبع("✅ الخادم يعمل على ws://localhost:8765\n")
    
    # إبقاء الخادم يعمل
    انتظر_إلى_ما_لا_نهاية()
    
    ارجع 0
نهاية
```

## 🌍 طلبات HTTP

### إرسال طلب GET

```س
استورد http من "stdlib/network/http"

دالة رئيسية()
    # طلب GET بسيط
    حاول
        متغير استجابة = http.احصل("https://api.example.com/users")
        
        اطبع("رمز الحالة: ")
        اطبع(استجابة.رمز_الحالة)
        اطبع("\n")
        
        اطبع("المحتوى:\n")
        اطبع(استجابة.نص)
        
    اصطاد (خطأ_http خ)
        اطبع("❌ خطأ HTTP: " + خ.رسالة + "\n")
    اصطاد (خطأ_اتصال خ)
        اطبع("❌ خطأ في الاتصال\n")
    نهاية
    
    ارجع 0
نهاية
```

### إرسال طلب POST

```س
استورد http من "stdlib/network/http"

دالة رئيسية()
    # بيانات للإرسال
    قاموس بيانات = {
        "الاسم": "أحمد"،
        "البريد": "ahmed@example.com"،
        "العمر": 25
    }
    
    # إرسال طلب POST
    حاول
        متغير استجابة = http.انشر(
            "https://api.example.com/users"،
            جسم: بيانات،
            ترويسات: {
                "Content-Type": "application/json"،
                "Authorization": "Bearer token123"
            }
        )
        
        إذا (استجابة.رمز_الحالة == 201)
            اطبع("✅ تم إنشاء المستخدم بنجاح!\n")
            قاموس نتيجة = استجابة.json()
            اطبع("المعرف الجديد: " + نتيجة["id"] + "\n")
        وإلا
            اطبع("❌ فشل: " + استجابة.نص + "\n")
        نهاية
        
    اصطاد (خطأ خ)
        اطبع("❌ خطأ: " + خ.رسالة + "\n")
    نهاية
    
    ارجع 0
نهاية
```

### عميل HTTP متقدم

```س
استورد http من "stdlib/network/http"

فئة عميل_واجهة_برمجية
    نص الرابط_الأساسي
    نص التوكن
    
    دالة أنشئ(نص رابط، نص توكن)
        هذا.الرابط_الأساسي = رابط
        هذا.التوكن = توكن
    نهاية
    
    # ترويسات مشتركة
    دالة _ترويسات() -> قاموس
        ارجع {
            "Authorization": "Bearer " + هذا.التوكن،
            "Content-Type": "application/json"،
            "Accept": "application/json"
        }
    نهاية
    
    دالة احصل(نص مسار) -> قاموس
        متغير استجابة = http.احصل(
            هذا.الرابط_الأساسي + مسار،
            ترويسات: هذا._ترويسات()
        )
        هذا._تحقق_من_الاستجابة(استجابة)
        ارجع استجابة.json()
    نهاية
    
    دالة انشر(نص مسار، قاموس بيانات) -> قاموس
        متغير استجابة = http.انشر(
            هذا.الرابط_الأساسي + مسار،
            جسم: بيانات،
            ترويسات: هذا._ترويسات()
        )
        هذا._تحقق_من_الاستجابة(استجابة)
        ارجع استجابة.json()
    نهاية
    
    دالة _تحقق_من_الاستجابة(استجابة)
        إذا (استجابة.رمز_الحالة >= 400)
            ارمِ خطأ_واجهة_برمجية.أنشئ(
                استجابة.رمز_الحالة،
                استجابة.نص
            )
        نهاية
    نهاية
نهاية

دالة رئيسية()
    عميل_واجهة_برمجية عميل = عميل_واجهة_برمجية.أنشئ(
        "https://api.example.com"،
        "my-secret-token"
    )
    
    حاول
        # جلب قائمة المستخدمين
        قاموس مستخدمين = عميل.احصل("/users")
        اطبع("عدد المستخدمين: ")
        اطبع(طول(مستخدمين["data"]))
        
        # إنشاء مستخدم جديد
        قاموس مستخدم_جديد = عميل.انشر("/users"، {
            "الاسم": "محمد"،
            "البريد": "mohamed@example.com"
        })
        اطبع("\nتم إنشاء المستخدم: " + مستخدم_جديد["id"])
        
    اصطاد (خطأ_واجهة_برمجية خ)
        اطبع("❌ خطأ API: " + خ.رسالة + "\n")
    نهاية
    
    ارجع 0
نهاية
```

## 🏠 خادم HTTP بسيط

```س
استورد http من "stdlib/network/http"

دالة رئيسية()
    اطبع("🚀 بدء خادم HTTP...\n")
    
    متغير خادم = http.خادم(8080)
    
    # معالج الصفحة الرئيسية
    خادم.عالج("/"، دالة(طلب، استجابة)
        استجابة.ترويسة("Content-Type"، "text/html; charset=utf-8")
        استجابة.اكتب("""
            <!DOCTYPE html>
            <html dir="rtl" lang="ar">
            <head>
                <title>مرحباً</title>
                <style>
                    body { font-family: Arial; text-align: center; padding: 50px; }
                </style>
            </head>
            <body>
                <h1>🎉 مرحباً بك في خادم لغة ص!</h1>
                <p>الخادم يعمل بنجاح.</p>
            </body>
            </html>
        """)
    نهاية)
    
    # معالج API
    خادم.عالج("/api/معلومات"، دالة(طلب، استجابة)
        استجابة.ترويسة("Content-Type"، "application/json")
        استجابة.json({
            "الإصدار": "1.0.0"،
            "اللغة": "لغة ص"،
            "الحالة": "يعمل"
        })
    نهاية)
    
    # معالج المستخدمين
    خادم.عالج("/api/مستخدمين"، دالة(طلب، استجابة)
        طابق طلب.الطريقة
            حالة "GET":
                استجابة.json([
                    {"id": 1، "الاسم": "أحمد"}،
                    {"id": 2، "الاسم": "سارة"}
                ])
            حالة "POST":
                قاموس بيانات = طلب.json()
                اطبع("إنشاء مستخدم: " + بيانات["الاسم"] + "\n")
                استجابة.رمز_الحالة(201)
                استجابة.json({"نجاح": صحيح، "id": 3})
            حالة _:
                استجابة.رمز_الحالة(405)
                استجابة.json({"خطأ": "الطريقة غير مسموحة"})
        نهاية
    نهاية)
    
    # معالج الملفات الثابتة
    خادم.ملفات_ثابتة("/static"، "./public")
    
    خادم.ابدأ()
    اطبع("✅ الخادم يعمل على http://localhost:8080\n")
    
    انتظر_إلى_ما_لا_نهاية()
    ارجع 0
نهاية
```

## 🔄 البث المتعدد (Multicast)

### مرسل Multicast

```س
استورد شبكة من "stdlib/network"

دالة رئيسية()
    متغير مقبس = شبكة.مقبس_udp()
    
    # عنوان المجموعة
    نص عنوان_المجموعة = "239.255.255.250"
    رقم المنفذ = 5000
    
    # إرسال رسائل للمجموعة
    رقم عداد = 0
    بينما (صحيح)
        عداد = عداد + 1
        نص رسالة = "رسالة رقم " + لنص(عداد)
        
        مقبس.ارسل_إلى(رسالة، (عنوان_المجموعة، المنفذ))
        اطبع("📤 تم إرسال: " + رسالة + "\n")
        
        انتظر(1000)  # كل ثانية
    نهاية
    
    ارجع 0
نهاية
```

## 💡 أفضل الممارسات

### 1. إدارة الاتصالات

```س
# ✅ أغلق الاتصالات دائماً
متغير اتصال = شبكة.مقبس_tcp()
حاول
    اتصال.اتصل(العنوان، المنفذ)
    # ... العمليات
أخيراً
    اتصال.اغلق()
نهاية
```

### 2. المهلات الزمنية

```س
# ✅ حدد مهلة للعمليات
اتصال.عين_المهلة(5000)  # 5 ثواني

حاول
    اتصال.اتصل(العنوان، المنفذ)
اصطاد (خطأ_انتهاء_المهلة)
    اطبع("انتهت المهلة!")
نهاية
```

### 3. إعادة المحاولة

```س
# ✅ أعد المحاولة عند الفشل
دالة اتصل_مع_إعادة_المحاولة(نص العنوان، رقم المنفذ، رقم المحاولات) -> مقبس
    رقم محاولة = 0
    بينما (محاولة < المحاولات)
        حاول
            متغير اتصال = شبكة.مقبس_tcp()
            اتصال.اتصل(العنوان، المنفذ)
            ارجع اتصال
        اصطاد (خطأ)
            محاولة = محاولة + 1
            انتظر(1000 * محاولة)  # انتظار متزايد
        نهاية
    نهاية
    ارمِ خطأ.أنشئ("فشلت جميع المحاولات")
نهاية
```
