# 🚀 البداية السريعة مع لغة ص

## التثبيت

### Windows
```powershell
# 1. استنسخ المشروع
git clone https://github.com/user/s_language.git
cd s_language

# 2. ابنِ المشروع
cmake -S . -B build
cmake --build build --config Debug --target sad

# 3. تأكد من نجاح التثبيت
.\build\bin\Debug\sad.exe --version
```

## أول برنامج

أنشئ ملف `مرحبا.ص`:
```
اطبع("مرحباً بالعالم! 🌍")
```

شغّله:
```powershell
.\build\bin\Debug\sad.exe مرحبا.ص
```

## أساسيات اللغة

### المتغيرات
```
// تعريف متغير
متغير اسم = "أحمد"
متغير عمر = 25
متغير نشط = صحيح

// ثابت (لا يتغير)
ثابت باي = 3.14159
```

### الشروط
```
إذا (عمر >= 18) {
    اطبع("بالغ")
} وإلا {
    اطبع("قاصر")
}
```

### الحلقات
```
// حلقة تكرار
كرر (متغير ع = 0؛ ع < 10؛ ع++) {
    اطبع(ع)
}

// حلقة طالما
طالما (نشط) {
    // ...
}

// حلقة لكل
لكل عنصر في [1, 2, 3, 4, 5] {
    اطبع(عنصر)
}
```

### الدوال
```
دالة جمع(أ، ب) {
    أرجع أ + ب
}

متغير نتيجة = جمع(3, 7)
اطبع(نتيجة)  // 10
```

### الأصناف (البرمجة الكائنية)
```
صنف شخص {
    باني(اسم، عمر) {
        هذا.اسم = اسم
        هذا.عمر = عمر
    }

    دالة تحية() {
        اطبع("مرحباً، أنا " + هذا.اسم)
    }
}

متغير شخص1 = شخص("أحمد"، 25)
شخص1.تحية()
```

### المصفوفات
```
متغير أرقام = [1, 2, 3, 4, 5]
اطبع(أرقام.طول())      // 5
أرقام.أضف(6)
اطبع(أرقام[0])         // 1
```

### معالجة الأخطاء
```
حاول {
    // كود قد يفشل
    متغير نتيجة = 10 / 0
} أمسك (خطأ) {
    اطبع("حدث خطأ: " + خطأ)
}
```
