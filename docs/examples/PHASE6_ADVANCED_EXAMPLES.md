## <a name="switch-advanced"></a>1. أمثلة Switch/Case المتقدمة

### 1.1 آلة حاسبة بسيطة / Simple Calculator

```s
دالة احسب(رقم أ، نص العملية، رقم ب) -> رقم {
    رقم النتيجة = 0
    
    حالة (العملية)
        عندما "+": النتيجة = أ + ب
        عندما "-": النتيجة = أ - ب
        عندما "*": النتيجة = أ * ب
        عندما "/": {
            إذا (ب == 0) {
                اطبع("خطأ: القسمة على صفر")
                إرجاع 0
            }
            النتيجة = أ / ب
        }
        افتراضي: {
            اطبع("عملية غير معروفة:", العملية)
            إرجاع 0
        }
    نهاية
    
    إرجاع النتيجة
}

// الاستخدام / Usage
اطبع("5 + 3 =", احسب(5, "+", 3))      // 8
اطبع("10 - 4 =", احسب(10, "-", 4))    // 6
اطبع("6 * 7 =", احسب(6, "*", 7))      // 42
اطبع("15 / 3 =", احسب(15, "/", 3))    // 5
```

### 2.2 حساب الخصومات / Discount Calculator

```s
دالة احسب_السعر_النهائي(رقم السعر، رقم الكمية) -> عشري {
    // خصم حسب الكمية / Quantity-based discount
    عشري الخصم = الكمية >= 100 ? 0.30 :  // 30% للكميات الكبيرة
                   الكمية >= 50 ? 0.20 :   // 20% للكميات المتوسطة
                   الكمية >= 10 ? 0.10 :   // 10% للكميات الصغيرة
                   0.0                       // لا خصم
    
    عشري السعر_بعد_الخصم = السعر * (1.0 - الخصم)
    
    // خصم إضافي للمبالغ الكبيرة / Additional discount for large amounts
    عشري السعر_النهائي = السعر_بعد_الخصم > 1000 ? 
                           السعر_بعد_الخصم * 0.95 :  // خصم 5% إضافي
                           السعر_بعد_الخصم
    
    إرجاع السعر_النهائي
}

// الاستخدام / Usage
اطبع("كمية 5 => سعر:", احسب_السعر_النهائي(100, 5))
اطبع("كمية 15 => سعر:", احسب_السعر_النهائي(100, 15))
اطبع("كمية 60 => سعر:", احسب_السعر_النهائي(100, 60))
اطبع("كمية 120 => سعر:", احسب_السعر_النهائي(100, 120))
```

### 2.3 تنسيق الوقت / Time Formatting

```s
دالة نسق_الوقت(رقم الساعات، رقم الدقائق، منطقي نظام_24) -> نص {
    // تحويل إلى نظام 12 ساعة إذا لزم الأمر
    رقم ساعة_منسقة = نظام_24 ? الساعات :
                        الساعات == 0 ? 12 :
                        الساعات > 12 ? الساعات - 12 :
                        الساعات
    
    // تحديد AM/PM
    نص الفترة = نظام_24 ? "" :
                  الساعات < 12 ? " AM" : " PM"
    
    // إضافة صفر أمام الأرقام الأحادية
    نص ساعة_نص = ساعة_منسقة < 10 ? "0" + نص(ساعة_منسقة) : نص(ساعة_منسقة)
    نص دقيقة_نص = الدقائق < 10 ? "0" + نص(الدقائق) : نص(الدقائق)
    
    إرجاع ساعة_نص + ":" + دقيقة_نص + الفترة
}

// الاختبار / Testing
اطبع("نظام 24:", نسق_الوقت(14, 30, صحيح))   // 14:30
اطبع("نظام 12:", نسق_الوقت(14, 30, خطأ))     // 02:30 PM
اطبع("نظام 12:", نسق_الوقت(0, 15, خطأ))      // 12:15 AM
```

### 2.4 التحقق من قوة كلمة المرور / Password Strength

```s
دالة تحقق_من_قوة_كلمة_المرور(نص كلمة_المرور) -> نص {
    رقم الطول = طول(كلمة_المرور)
    
    // تقييم بسيط للقوة / Simple strength evaluation
    نص القوة = الطول < 6 ? "ضعيفة جداً" :
                 الطول < 8 ? "ضعيفة" :
                 الطول < 12 ? "متوسطة" :
                 الطول < 16 ? "قوية" :
                 "قوية جداً"
    
    // توصية / Recommendation
    نص التوصية = الطول < 8 ? " - يُنصح بزيادة الطول" :
                   الطول >= 12 ? " - ممتازة!" :
                   ""
    
    إرجاع القوة + التوصية
}

// الاختبار / Testing
اطبع("'abc' =>", تحقق_من_قوة_كلمة_المرور("abc"))
اطبع("'password' =>", تحقق_من_قوة_كلمة_المرور("password"))
اطبع("'MyStr0ngP@ss' =>", تحقق_من_قوة_كلمة_المرور("MyStr0ngP@ss"))
```

## <a name="combining-features"></a>3. الدمج بين الميزات

### 3.1 نظام تسعير متقدم / Advanced Pricing System

```s
دالة احسب_السعر(نص نوع_العميل، رقم المبلغ، منطقي عضو_مميز) -> عشري {
    // تحديد نسبة الخصم الأساسية
    عشري خصم_العميل = 0.0
    
    حالة (نوع_العميل)
        عندما "vip": خصم_العميل = 0.25
        عندما "premium": خصم_العميل = 0.15
        عندما "regular": خصم_العميل = 0.05
        افتراضي: خصم_العميل = 0.0
    نهاية
    
    // خصم إضافي للأعضاء المميزين
    عشري خصم_إضافي = عضو_مميز ? 0.10 : 0.0
    
    // خصم بناءً على المبلغ
    عشري خصم_المبلغ = المبلغ > 1000 ? 0.10 :
                        المبلغ > 500 ? 0.05 :
                        0.0
    
    // حساب الخصم الكلي (لا يتجاوز 50%)
    عشري الخصم_الكلي = خصم_العميل + خصم_إضافي + خصم_المبلغ
    الخصم_الكلي = الخصم_الكلي > 0.50 ? 0.50 : الخصم_الكلي
    
    عشري السعر_النهائي = المبلغ * (1.0 - الخصم_الكلي)
    
    اطبع("نوع:", نوع_العميل, "- مميز:", عضو_مميز ? "نعم" : "لا")
    اطبع("المبلغ الأصلي:", المبلغ)
    اطبع("الخصم:", (الخصم_الكلي * 100), "%")
    اطبع("السعر النهائي:", السعر_النهائي)
    اطبع("---")
    
    إرجاع السعر_النهائي
}

// الاختبار / Testing
احسب_السعر("regular", 300, خطأ)
احسب_السعر("premium", 600, صحيح)
احسب_السعر("vip", 1500, صحيح)
```

### 3.2 معالج الحالات / State Machine

```s
صنف معالج_الحالة {
    نص الحالة_الحالية
    
    منشئ() {
        هذا.الحالة_الحالية = "idle"
    }
    
    دالة عالج_حدث(نص الحدث) -> منطقي {
        نص الحالة_الجديدة = ""
        منطقي نجح = صحيح
        
        حالة (هذا.الحالة_الحالية)
            عندما "idle": {
                حالة (الحدث)
                    عندما "start": الحالة_الجديدة = "running"
                    افتراضي: {
                        نجح = خطأ
                        اطبع("حدث غير صالح في حالة idle:", الحدث)
                    }
                نهاية
            }
            
            عندما "running": {
                حالة (الحدث)
                    عندما "pause": الحالة_الجديدة = "paused"
                    عندما "stop": الحالة_الجديدة = "stopped"
                    عندما "error": الحالة_الجديدة = "error"
                    افتراضي: {
                        نجح = خطأ
                        اطبع("حدث غير صالح في حالة running:", الحدث)
                    }
                نهاية
            }
            
            عندما "paused": {
                حالة (الحدث)
                    عندما "resume": الحالة_الجديدة = "running"
                    عندما "stop": الحالة_الجديدة = "stopped"
                    افتراضي: {
                        نجح = خطأ
                        اطبع("حدث غير صالح في حالة paused:", الحدث)
                    }
                نهاية
            }
            
            افتراضي: {
                اطبع("حالة غير معروفة:", هذا.الحالة_الحالية)
                نجح = خطأ
            }
        نهاية
        
        // تحديث الحالة إذا نجحت
        إذا (نجح && الحالة_الجديدة != "") {
            اطبع("انتقال:", هذا.الحالة_الحالية, "->", الحالة_الجديدة)
            هذا.الحالة_الحالية = الحالة_الجديدة
        }
        
        إرجاع نجح
    }
}

// الاستخدام / Usage
معالج_الحالة آلة = جديد معالج_الحالة()
آلة.عالج_حدث("start")
آلة.عالج_حدث("pause")
آلة.عالج_حدث("resume")
آلة.عالج_حدث("stop")
```

### 4.1 نظام طلبات مطعم / Restaurant Order System

```s
صنف طلب_مطعم {
    نص الصنف
    رقم الكمية
    عشري السعر_الوحدة
    
    دالة احسب_المجموع() -> عشري {
        عشري المجموع_الأساسي = هذا.الكمية * هذا.السعر_الوحدة
        
        // خصم على الكميات الكبيرة
        عشري الخصم = هذا.الكمية >= 10 ? 0.15 :
                       هذا.الكمية >= 5 ? 0.10 :
                       هذا.الكمية >= 3 ? 0.05 :
                       0.0
        
        إرجاع المجموع_الأساسي * (1.0 - الخصم)
    }
    
    دالة احصل_على_وقت_التحضير() -> رقم {
        رقم الوقت = 0
        
        حالة (هذا.الصنف)
            عندما "burger": الوقت = 15
            عندما "pizza": الوقت = 25
            عندما "pasta": الوقت = 20
            عندما "salad": الوقت = 10
            عندما "soup": الوقت = 15
            افتراضي: الوقت = 10
        نهاية
        
        // وقت إضافي للكميات الكبيرة
        رقم وقت_إضافي = هذا.الكمية > 5 ? (هذا.الكمية - 5) * 2 : 0
        
        إرجاع الوقت + وقت_إضافي
    }
}

// مثال استخدام / Usage example
طلب_مطعم طلب1 = جديد طلب_مطعم()
طلب1.الصنف = "pizza"
طلب1.الكمية = 3
طلب1.السعر_الوحدة = 12.99

اطبع("الصنف:", طلب1.الصنف)
اطبع("الكمية:", طلب1.الكمية)
اطبع("المجموع:", طلب1.احسب_المجموع())
اطبع("وقت التحضير:", طلب1.احصل_على_وقت_التحضير(), "دقيقة")
```

### 4.2 محلل الطقس / Weather Analyzer

```s
دالة حلل_الطقس(رقم الحرارة، رقم الرطوبة، رقم سرعة_الرياح) -> نص {
    // تحديد حالة الحرارة
    نص حالة_الحرارة = الحرارة < 0 ? "متجمد" :
                        الحرارة < 10 ? "بارد جداً" :
                        الحرارة < 20 ? "بارد" :
                        الحرارة < 25 ? "معتدل" :
                        الحرارة < 30 ? "دافئ" :
                        الحرارة < 35 ? "حار" :
                        "حار جداً"
    
    // تحديد مستوى الراحة
    نص مستوى_الراحة = ""
    
    حالة (حالة_الحرارة)
        عندما "معتدل": {
            مستوى_الراحة = الرطوبة < 60 && سرعة_الرياح < 20 ?
                           "مريح جداً" : "مريح"
        }
        عندما "دافئ": {
            مستوى_الراحة = الرطوبة > 70 ?
                           "خانق" :
                           سرعة_الرياح > 30 ?
                           "عاصف" : "مقبول"
        }
        عندما "حار":
        عندما "حار جداً": {
            مستوى_الراحة = الرطوبة > 60 ?
                           "خانق وغير مريح" :
                           "حار ولكن محتمل"
        }
        افتراضي: {
            مستوى_الراحة = سرعة_الرياح > 40 ?
                           "عاصف جداً" : "عادي"
        }
    نهاية
    
    // توصيات
    نص التوصية = ""
    
    حالة (حالة_الحرارة)
        عندما "متجمد":
        عندما "بارد جداً":
        عندما "بارد":
            التوصية = "ارتدِ ملابس ثقيلة"
        عندما "معتدل":
        عندما "دافئ":
            التوصية = "طقس مثالي للخروج"
        عندما "حار":
        عندما "حار جداً":
            التوصية = "ابقَ في الظل واشرب الماء"
        افتراضي:
            التوصية = "كن حذراً"
    نهاية
    
    إرجاع "الحرارة: " + حالة_الحرارة + "\n" +
           "الراحة: " + مستوى_الراحة + "\n" +
           "التوصية: " + التوصية
}

// الاستخدام / Usage
اطبع("=== تحليل الطقس ===")
اطبع(حلل_الطقس(23, 50, 15))
اطبع("\n")
اطبع(حلل_الطقس(35, 75, 10))
```
