## 🔌 TCP

### خادم TCP بسيط

```sad
استخدم شبكة

خادم = شبكة.انشئ_خادم_tcp(8080)
اطبع("الخادم يعمل...")

بينما صحيح {
    عميل = خادم.اقبل()
    بيانات = عميل.استقبل(1024)
    عميل.ارسل_الكل(بيانات)  // echo
    عميل.اغلق()
}
```

### عميل TCP بسيط

```sad
استخدم شبكة

عميل = شبكة.اتصل_tcp("127.0.0.1", 8080)
عميل.ارسل_الكل("مرحباً!")
رد = عميل.استقبل(1024)
اطبع("الرد: " + نص.من_بايتات(رد))
عميل.اغلق()
```

### خادم بمهلات

```sad
خادم = شبكة.انشئ_خادم_tcp(8080)
عميل = خادم.اقبل()

عميل.عين_مهلة_الاستقبال(5000)  // 5 ثوانٍ
محاولة {
    بيانات = عميل.استقبل(1024)
} اصطياد (خطأ_شبكة.مهلة_انتهت) {
    اطبع("انتهت المهلة")
}
```

### خادم متعدد العملاء

```sad
استخدم شبكة
استخدم خيط

خادم = شبكة.انشئ_خادم_tcp(8080)

بينما صحيح {
    عميل = خادم.اقبل()
    خيط.ابدأ(عالج_عميل، [عميل])
}

دالة عالج_عميل(عميل) {
    بينما صحيح {
        بيانات = عميل.استقبل(4096)
        إذا بيانات.طول() == 0 { اكسر }
        عميل.ارسل_الكل(بيانات)
    }
    عميل.اغلق()
}
```

### إرسال واستقبال نصوص

```sad
// إرسال
عميل.ارسل_الكل(نص.الى_بايتات("مرحباً"))

// استقبال
بيانات = عميل.استقبل(1024)
نص = نص.من_بايتات(بيانات)
```

### تفعيل خيارات TCP

```sad
عميل = شبكة.اتصل_tcp("server.com", 8080)

عميل.عين_بدون_تأخير(صحيح)           // TCP_NODELAY
عميل.عين_حافظ_على_الحياة(صحيح)       // Keep-Alive
عميل.عين_مهلة_الإرسال(5000)          // مهلة إرسال
عميل.عين_مهلة_الاستقبال(10000)       // مهلة استقبال
عميل.عين_حجم_مخزن_الإرسال(256*1024)  // 256KB
```

## 📡 UDP

### خادم UDP

```sad
استخدم شبكة

مقبس = شبكة.انشئ_مقبس_udp()
مقبس.اربط(9090)
اطبع("خادم UDP يعمل...")

بينما صحيح {
    (بيانات، مرسل) = مقبس.استقبل_من(4096)
    اطبع("من " + مرسل.نصي() + ": " + بيانات)
    مقبس.ارسل_الى("ACK", مرسل)
}
```

### عميل UDP

```sad
استخدم شبكة

مقبس = شبكة.انشئ_مقبس_udp()
خادم = شبكة.انشئ_عنوان("127.0.0.1", 9090)

مقبس.ارسل_الى("رسالة", خادم)
(رد، _) = مقبس.استقبل_من(1024)
اطبع("الرد: " + رد)
```

### UDP المتصل

```sad
مقبس = شبكة.انشئ_مقبس_udp()
مقبس.اتصل("127.0.0.1", 9090)

// الآن يمكن استخدام ارسل() بدون عنوان
مقبس.ارسل("رسالة 1")
مقبس.ارسل("رسالة 2")
رد = مقبس.استقبل(1024)
```

### البث العام (Broadcast)

```sad
مقبس = شبكة.انشئ_مقبس_udp()
مقبس.فعل_البث()

عنوان_بث = شبكة.انشئ_عنوان("255.255.255.255", 9999)
مقبس.ارسل_الى("رسالة للجميع!", عنوان_بث)
```

### البث الجماعي (Multicast)

```sad
// المرسل
مقبس = شبكة.انشئ_مقبس_udp()
مجموعة = شبكة.انشئ_عنوان("239.0.0.1", 5000)
مقبس.ارسل_الى("رسالة multicast", مجموعة)

// المستقبِل
مقبس = شبكة.انشئ_مقبس_udp()
مقبس.اربط(5000)
مقبس.انضم_لمجموعة_multicast("239.0.0.1")

بينما صحيح {
    (بيانات، مرسل) = مقبس.استقبل_من(4096)
    اطبع(بيانات)
}
```

## 🔧 معالجة الأخطاء

### الأخطاء الشائعة

```sad
محاولة {
    عميل = شبكة.اتصل_tcp("server.com", 8080)
    
} اصطياد (خطأ_شبكة.اتصال_مرفوض) {
    اطبع("الخادم لا يعمل")
    
} اصطياد (خطأ_شبكة.مهلة_انتهت) {
    اطبع("انتهت المهلة")
    
} اصطياد (خطأ_شبكة.عنوان_غير_صالح) {
    اطبع("عنوان خاطئ")
    
} اصطياد (خطأ_شبكة خطأ) {
    اطبع("خطأ عام: " + خطأ.رسالة())
}
```

### إعادة محاولة تلقائية

```sad
دالة اتصل_مع_إعادة(عنوان، منفذ، محاولات=3) {
    لـ ع من 1 إلى محاولات {
        محاولة {
            ارجع شبكة.اتصل_tcp(عنوان، منفذ)
        } اصطياد {
            إذا ع < محاولات {
                اطبع("إعادة المحاولة...")
                نم(2000)
            }
        }
    }
    ارمِ خطأ("فشل الاتصال")
}
```

## 📦 البروتوكولات

### بروتوكول بسيط: حجم + بيانات

```sad
// الإرسال
دالة ارسل_رسالة(عميل، رسالة) {
    بيانات = نص.الى_بايتات(رسالة)
    حجم = بيانات.طول()
    
    // إرسال الحجم (4 بايتات)
    عميل.ارسل_الكل(رمز_رقم_صحيح(حجم، 4))
    // إرسال البيانات
    عميل.ارسل_الكل(بيانات)
}

// الاستقبال
دالة استقبل_رسالة(عميل) {
    // استقبال الحجم
    حجم_بايتات = عميل.استقبل_الكل(4)
    حجم = فك_رقم_صحيح(حجم_بايتات)
    
    // استقبال البيانات
    بيانات = عميل.استقبل_الكل(حجم)
    ارجع نص.من_بايتات(بيانات)
}
```

### بروتوكول JSON

```sad
استخدم json

// الإرسال
دالة ارسل_json(عميل، كائن) {
    نص_json = json.من_كائن(كائن)
    بيانات = نص.الى_بايتات(نص_json)
    
    عميل.ارسل_الكل(رمز_رقم_صحيح(بيانات.طول()، 4))
    عميل.ارسل_الكل(بيانات)
}

// الاستقبال
دالة استقبل_json(عميل) {
    حجم = فك_رقم_صحيح(عميل.استقبل_الكل(4))
    بيانات = عميل.استقبل_الكل(حجم)
    نص_json = نص.من_بايتات(بيانات)
    ارجع json.الى_كائن(نص_json)
}

// الاستخدام
ارسل_json(عميل، {اسم: "أحمد", عمر: 25})
كائن = استقبل_json(عميل)
```

## 📁 نقل ملفات

### إرسال ملف

```sad
استخدم ملف

دالة ارسل_ملف(عميل، مسار_ملف) {
    ملف_دخل = ملف.افتح(مسار_ملف، "rb")
    
    بينما صحيح {
        دفعة = ملف_دخل.اقرأ(65536)  // 64KB
        إذا دفعة.طول() == 0 { اكسر }
        
        عميل.ارسل_الكل(دفعة)
    }
    
    ملف_دخل.اغلق()
}
```

### استقبال ملف

```sad
دالة استقبل_ملف(عميل، مسار_ملف، حجم_كامل) {
    ملف_خرج = ملف.افتح(مسار_ملف، "wb")
    مستقبل = 0
    
    بينما مستقبل < حجم_كامل {
        باقي = حجم_كامل - مستقبل
        حجم_دفعة = اصغر(65536، باقي)
        
        دفعة = عميل.استقبل_الكل(حجم_دفعة)
        ملف_خرج.اكتب(دفعة)
        
        مستقبل = مستقبل + دفعة.طول()
        
        // عرض التقدم
        نسبة = (مستقبل * 100) / حجم_كامل
        اطبع(نسبة.صحيح() + "%", نهاية="\r")
    }
    
    ملف_خرج.اغلق()
}
```

## 🎯 أنماط شائعة

### خادم echo

```sad
خادم = شبكة.انشئ_خادم_tcp(8080)

بينما صحيح {
    عميل = خادم.اقبل()
    بيانات = عميل.استقبل(4096)
    عميل.ارسل_الكل(بيانات)
    عميل.اغلق()
}
```

### خادم بروتوكول طلب-رد

```sad
بينما صحيح {
    عميل = خادم.اقبل()
    
    بينما صحيح {
        طلب = استقبل_رسالة(عميل)
        إذا طلب.طول() == 0 { اكسر }
        
        رد = عالج_طلب(طلب)
        ارسل_رسالة(عميل، رد)
    }
    
    عميل.اغلق()
}
```

### عميل بإعادة اتصال

```sad
بينما صحيح {
    محاولة {
        عميل = شبكة.اتصل_tcp("server.com", 8080)
        
        // العمل العادي
        بينما صحيح {
            عميل.ارسل_الكل("ping")
            رد = عميل.استقبل(16)
            نم(1000)
        }
        
    } اصطياد (خطأ) {
        اطبع("انقطع الاتصال، إعادة الاتصال...")
        نم(5000)  // انتظر 5 ثوانٍ
    }
}
```

## 🛠️ دوال مساعدة

### تحويل رقم لبايتات

```sad
دالة رمز_رقم_صحيح(رقم: رقم، بايتات: رقم) -> قائمة<بايت> {
    نتيجة = []
    لـ ع من 0 إلى بايتات {
        نتيجة.أضف((رقم >> (ع * 8)) & 0xFF)
    }
    ارجع نتيجة
}
```

### تحويل بايتات لرقم

```sad
دالة فك_رقم_صحيح(بايتات: قائمة<بايت>) -> رقم {
    نتيجة = 0
    لـ ع من 0 إلى بايتات.طول() {
        نتيجة = نتيجة أو (بايتات[ع] << (ع * 8))
    }
    ارجع نتيجة
}
```

### حجم بصيغة بشرية

```sad
دالة حجم_بشري(بايتات: رقم) -> نص {
    إذا بايتات < 1024 {
        ارجع بايتات + " B"
    } وإلا إذا بايتات < 1024 * 1024 {
        ارجع (بايتات / 1024.0).مقرب(2) + " KB"
    } وإلا إذا بايتات < 1024 * 1024 * 1024 {
        ارجع (بايتات / 1024.0 / 1024.0).مقرب(2) + " MB"
    } وإلا {
        ارجع (بايتات / 1024.0 / 1024.0 / 1024.0).مقرب(2) + " GB"
    }
}
```
