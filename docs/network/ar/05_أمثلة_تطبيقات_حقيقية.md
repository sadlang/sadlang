## 📖 مقدمة

الآن بعد أن تعلمت الأساسيات، حان الوقت لبناء تطبيقات حقيقية! في هذا الدليل، سنبني ستة تطبيقات عملية كاملة من الصفر. كل تطبيق سيعلمك مفاهيم ومهارات جديدة تستطيع تطبيقها في مشاريعك الخاصة.

**ما ستتعلمه:**
- بناء أنظمة عميل-خادم كاملة
- التعامل مع البروتوكولات الشائعة
- إدارة الاتصالات المتعددة

## 💬 تطبيق 1: نظام دردشة متعدد المستخدمين

### الوصف
نظام دردشة كامل يسمح لعدة مستخدمين بالدردشة في نفس الوقت. الخادم يدير كل الاتصالات ويوزع الرسائل على الجميع.

### المتطلبات
- ✅ خادم يدعم عدة مستخدمين في نفس الوقت
- ✅ كل مستخدم له اسم فريد
- ✅ إرسال رسائل لكل المتصلين
- ✅ إشعارات انضمام/مغادرة
- ✅ قائمة المستخدمين المتصلين

### كود الخادم الكامل

```sad
استخدم شبكة
استخدم نص
استخدم خيط
استخدم وقت

صنف خادم_دردشة {
    مقبس_خادم: مقبس_tcp
    عملاء: قائمة<عميل_دردشة>
    قفل: قفل_خيط
    
    دالة __init__(منفذ) {
        هذا.مقبس_خادم = شبكة.انشئ_خادم_tcp(منفذ)
        هذا.عملاء = []
        هذا.قفل = قفل_خيط.جديد()
        
        إذا هذا.مقبس_خادم == باطل {
            ارمِ خطأ("فشل إنشاء الخادم على المنفذ " + منفذ)
        }
        
        // تمكين إعادة استخدام العنوان
        هذا.مقبس_خادم.عين_اعادة_استخدام_العنوان(صحيح)
    }
    
    دالة ابدأ() {
        اطبع("═══════════════════════════════════")
        اطبع("💬 خادم الدردشة - لغة ص")
        اطبع("═══════════════════════════════════")
        اطبع("✅ الخادم يعمل على المنفذ 8888")
        اطبع("⏳ في انتظار متصلين...")
        اطبع("")
        
        // حلقة قبول اتصالات جديدة
        بينما صحيح {
            محاولة {
                مقبس_عميل = هذا.مقبس_خادم.اقبل()
                
                // إنشاء خيط جديد لهذا العميل
                خيط.ابدأ(هذا.عالج_عميل، [مقبس_عميل])
                
            } اصطياد (خطأ) {
                اطبع("❌ خطأ في قبول اتصال: " + خطأ.رسالة())
            }
        }
    }
    
    دالة عالج_عميل(مقبس_عميل) {
        عميل = باطل
        
        محاولة {
            // 1. استقبال اسم المستخدم
            بيانات_اسم = مقبس_عميل.استقبل(256)
            اسم_مستخدم = نص.من_بايتات(بيانات_اسم).قص()
            
            // التحقق من أن الاسم غير مستخدم
            إذا هذا.اسم_مستخدم_موجود(اسم_مستخدم) {
                مقبس_عميل.ارسل_الكل("خطأ:الاسم_مستخدم")
                مقبس_عميل.اغلق()
                ارجع
            }
            
            // إرسال تأكيد
            مقبس_عميل.ارسل_الكل("موافق")
            
            // 2. إضافة العميل للقائمة
            عميل = {
                مقبس: مقبس_عميل،
                اسم: اسم_مستخدم،
                عنوان: مقبس_عميل.عنوان_البعيد()،
                وقت_الانضمام: وقت.الآن()
            }
            
            هذا.قفل.اقفل()
            هذا.عملاء.أضف(عميل)
            هذا.قفل.افتح()
            
            // 3. إعلان الانضمام
            اطبع("✅ انضم " + اسم_مستخدم + " من " + عميل.عنوان)
            هذا.بث("النظام: انضم " + اسم_مستخدم + " للدردشة! 👋")
            
            // إرسال قائمة المستخدمين للعميل الجديد
            قائمة = هذا.احصل_على_قائمة_مستخدمين()
            مقبس_عميل.ارسل_الكل("قائمة:" + قائمة)
            
            // 4. حلقة استقبال الرسائل
            بينما صحيح {
                بيانات = مقبس_عميل.استقبل(4096)
                
                إذا بيانات.طول() == 0 {
                    اكسر  // العميل قطع الاتصال
                }
                
                رسالة = نص.من_بايتات(بيانات).قص()
                
                // معالجة الأوامر الخاصة
                إذا رسالة.تبدأ_بـ("/") {
                    هذا.عالج_أمر(عميل، رسالة)
                } وإلا {
                    // رسالة عادية - بث للجميع
                    رسالة_كاملة = "[" + اسم_مستخدم + "]: " + رسالة
                    اطبع(رسالة_كاملة)
                    هذا.بث(رسالة_كاملة، استثناء=عميل)
                }
            }
            
        } اصطياد (خطأ) {
            اطبع("⚠️ خطأ مع عميل: " + خطأ.رسالة())
        } نهائياً {
            // تنظيف عند المغادرة
            إذا عميل != باطل {
                هذا.قفل.اقفل()
                هذا.عملاء.احذف(عميل)
                هذا.قفل.افتح()
                
                اطبع("👋 غادر " + عميل.اسم)
                هذا.بث("النظام: غادر " + عميل.اسم + " الدردشة. 👋")
                
                محاولة {
                    عميل.مقبس.اغلق()
                } اصطياد {}
            }
        }
    }
    
    دالة بث(رسالة، استثناء=باطل) {
        بيانات = نص.الى_بايتات(رسالة)
        
        هذا.قفل.اقفل()
        لكل عميل في هذا.عملاء {
            إذا عميل == استثناء {
                استمر  // لا ترسل للمرسل نفسه
            }
            
            محاولة {
                عميل.مقبس.ارسل_الكل(بيانات)
            } اصطياد (خطأ) {
                اطبع("⚠️ فشل إرسال لـ " + عميل.اسم)
            }
        }
        هذا.قفل.افتح()
    }
    
    دالة عالج_أمر(عميل، أمر) {
        إذا أمر == "/قائمة" {
            قائمة = هذا.احصل_على_قائمة_مستخدمين()
            عميل.مقبس.ارسل_الكل("👥 المتصلون: " + قائمة)
            
        } وإلا إذا أمر == "/وقت" {
            وقت_حالي = وقت.الآن_بتنسيق("%H:%M:%S")
            عميل.مقبس.ارسل_الكل("🕐 الوقت: " + وقت_حالي)
            
        } وإلا إذا أمر == "/مساعدة" {
            مساعدة = "📋 الأوامر المتاحة:\n"
            مساعدة = مساعدة + "  /قائمة - عرض المستخدمين المتصلين\n"
            مساعدة = مساعدة + "  /وقت - عرض الوقت الحالي\n"
            مساعدة = مساعدة + "  /خروج - مغادرة الدردشة"
            عميل.مقبس.ارسل_الكل(مساعدة)
            
        } وإلا {
            عميل.مقبس.ارسل_الكل("❌ أمر غير معروف. اكتب /مساعدة")
        }
    }
    
    دالة احصل_على_قائمة_مستخدمين() {
        هذا.قفل.اقفل()
        أسماء = []
        لكل عميل في هذا.عملاء {
            أسماء.أضف(عميل.اسم)
        }
        هذا.قفل.افتح()
        
        ارجع نص.اربط(أسماء، ", ")
    }
    
    دالة اسم_مستخدم_موجود(اسم) {
        هذا.قفل.اقفل()
        لكل عميل في هذا.عملاء {
            إذا عميل.اسم == اسم {
                هذا.قفل.افتح()
                ارجع صحيح
            }
        }
        هذا.قفل.افتح()
        ارجع خطأ
    }
}

// تشغيل الخادم
دالة رئيسية() {
    محاولة {
        خادم = خادم_دردشة(8888)
        خادم.ابدأ()
    } اصطياد (خطأ) {
        اطبع("❌ فشل تشغيل الخادم: " + خطأ.رسالة())
    }
}
```

### كود العميل الكامل

```sad
استخدم شبكة
استخدم نص
استخدم خيط
استخدم إدخال_إخراج

صنف عميل_دردشة {
    مقبس: مقبس_tcp
    اسم: نص
    متصل: منطقي
    
    دالة اتصل(عنوان_خادم، منفذ، اسم_مستخدم) {
        هذا.اسم = اسم_مستخدم
        هذا.متصل = خطأ
        
        اطبع("🔌 جاري الاتصال بالخادم...")
        
        محاولة {
            هذا.مقبس = شبكة.اتصل_tcp(عنوان_خادم، منفذ)
        } اصطياد (خطأ) {
            اطبع("❌ فشل الاتصال: " + خطأ.رسالة())
            ارجع خطأ
        }
        
        // إرسال اسم المستخدم
        هذا.مقبس.ارسل_الكل(نص.الى_بايتات(اسم_مستخدم))
        
        // استقبال تأكيد
        رد_بايتات = هذا.مقبس.استقبل(256)
        رد = نص.من_بايتات(رد_بايتات)
        
        إذا رد == "خطأ:الاسم_مستخدم" {
            اطبع("❌ الاسم '" + اسم_مستخدم + "' مستخدم بالفعل!")
            هذا.مقبس.اغلق()
            ارجع خطأ
        }
        
        هذا.متصل = صحيح
        اطبع("✅ تم الاتصال بنجاح!")
        اطبع("")
        
        ارجع صحيح
    }
    
    دالة ابدأ() {
        إذا !هذا.متصل {
            اطبع("❌ لست متصلاً!")
            ارجع
        }
        
        اطبع("═══════════════════════════════════")
        اطبع("💬 عميل الدردشة - لغة ص")
        اطبع("═══════════════════════════════════")
        اطبع("أهلاً " + هذا.اسم + "! 👋")
        اطبع("اكتب رسالتك واضغط Enter")
        اطبع("الأوامر: /قائمة /وقت /مساعدة /خروج")
        اطبع("═══════════════════════════════════")
        اطبع("")
        
        // بدء خيط الاستقبال
        خيط.ابدأ(هذا.خيط_استقبال)
        
        // حلقة الإرسال (الخيط الرئيسي)
        بينما هذا.متصل {
            محاولة {
                // قراءة من المستخدم
                رسالة = إدخال_إخراج.اقرأ_سطر()
                
                إذا رسالة.طول() == 0 {
                    استمر
                }
                
                // التحقق من أمر الخروج
                إذا رسالة == "/خروج" {
                    اطبع("👋 جاري قطع الاتصال...")
                    هذا.متصل = خطأ
                    اكسر
                }
                
                // إرسال الرسالة
                هذا.مقبس.ارسل_الكل(نص.الى_بايتات(رسالة))
                
            } اصطياد (خطأ) {
                اطبع("❌ خطأ في الإرسال: " + خطأ.رسالة())
                هذا.متصل = خطأ
                اكسر
            }
        }
        
        // إغلاق الاتصال
        محاولة {
            هذا.مقبس.اغلق()
        } اصطياد {}
        
        اطبع("✅ تم قطع الاتصال")
    }
    
    دالة خيط_استقبال() {
        بينما هذا.متصل {
            محاولة {
                بيانات = هذا.مقبس.استقبل(4096)
                
                إذا بيانات.طول() == 0 {
                    هذا.متصل = خطأ
                    اطبع("\n⚠️ الخادم قطع الاتصال")
                    اكسر
                }
                
                رسالة = نص.من_بايتات(بيانات)
                اطبع("\n" + رسالة)
                
            } اصطياد (خطأ) {
                إذا هذا.متصل {
                    اطبع("\n❌ خطأ في الاستقبال: " + خطأ.رسالة())
                    هذا.متصل = خطأ
                }
                اكسر
            }
        }
    }
}

// تشغيل العميل
دالة رئيسية() {
    اطبع("═══════════════════════════════════")
    اطبع("💬 عميل الدردشة - لغة ص")
    اطبع("═══════════════════════════════════")
    اطبع("")
    
    // طلب اسم المستخدم
    اطبع("ما اسمك؟ ", نهاية="")
    اسم = إدخال_إخراج.اقرأ_سطر().قص()
    
    إذا اسم.طول() == 0 {
        اطبع("❌ يجب إدخال اسم!")
        ارجع
    }
    
    اطبع("")
    
    // إنشاء العميل والاتصال
    عميل = عميل_دردشة()
    
    إذا عميل.اتصل("127.0.0.1", 8888, اسم) {
        عميل.ابدأ()
    }
}
```

## 📁 تطبيق 2: نظام نقل الملفات

### الوصف
نظام نقل ملفات قوي يدعم:
- ✅ إرسال ملفات بأي حجم
- ✅ شريط تقدم
- ✅ استئناف التحميل بعد الانقطاع
- ✅ ضغط الملفات تلقائياً
- ✅ التحقق من صحة الملف (MD5)

### كود الخادم

```sad
استخدم شبكة
استخدم ملف
استخدم نص
استخدم تجزئة
استخدم ضغط

صنف خادم_ملفات {
    مقبس: مقبس_tcp
    مجلد_التحميل: نص
    
    دالة __init__(منفذ، مجلد="uploads") {
        هذا.مقبس = شبكة.انشئ_خادم_tcp(منفذ)
        هذا.مجلد_التحميل = مجلد
        
        // إنشاء مجلد التحميل إذا لم يكن موجوداً
        إذا !ملف.موجود(مجلد) {
            ملف.انشئ_مجلد(مجلد)
        }
        
        اطبع("📁 خادم نقل الملفات")
        اطبع("📂 مجلد التحميل: " + مجلد)
        اطبع("✅ الخادم يعمل على المنفذ " + منفذ)
    }
    
    دالة ابدأ() {
        بينما صحيح {
            عميل = هذا.مقبس.اقبل()
            اطبع("\n✅ عميل جديد: " + عميل.عنوان_البعيد())
            
            محاولة {
                هذا.استقبل_ملف(عميل)
            } اصطياد (خطأ) {
                اطبع("❌ خطأ: " + خطأ.رسالة())
            } نهائياً {
                عميل.اغلق()
            }
        }
    }
    
    دالة استقبل_ملف(عميل) {
        // 1. استقبال معلومات الملف (JSON)
        معلومات_بايتات = هذا.استقبل_حزمة(عميل)
        معلومات = نص.من_json(نص.من_بايتات(معلومات_بايتات))
        
        اسم_ملف = معلومات["اسم"]
        حجم = معلومات["حجم"]
        مضغوط = معلومات["مضغوط"]
        تجزئة_md5 = معلومات["md5"]
        
        اطبع("📄 الملف: " + اسم_ملف)
        اطبع("📊 الحجم: " + حجم_بشري(حجم))
        اطبع("🗜️ مضغوط: " + (مضغوط ? "نعم" : "لا"))
        
        // 2. التحقق من الاستئناف
        مسار_كامل = هذا.مجلد_التحميل + "/" + اسم_ملف
        موقع_حالي = 0
        
        إذا ملف.موجود(مسار_كامل) {
            موقع_حالي = ملف.حجم(مسار_كامل)
            اطبع("🔄 استئناف من " + حجم_بشري(موقع_حالي))
        }
        
        // إرسال موقع الاستئناف للعميل
        عميل.ارسل_الكل(نص.الى_بايتات(موقع_حالي.نص()))
        
        // 3. استقبال البيانات
        ملف_خرج = ملف.افتح(مسار_كامل، موقع_حالي > 0 ? "ab" : "wb")
        
        مستقبل = موقع_حالي
        وقت_بداية = وقت.الآن()
        
        بينما مستقبل < حجم {
            // استقبال دفعة
            باقي = حجم - مستقبل
            حجم_دفعة = اصغر(65536، باقي)  // 64KB
            
            دفعة = عميل.استقبل_الكل(حجم_دفعة)
            ملف_خرج.اكتب(دفعة)
            
            مستقبل = مستقبل + دفعة.طول()
            
            // عرض التقدم
            نسبة = (مستقبل * 100) / حجم
            سرعة = (مستقبل - موقع_حالي) / ((وقت.الآن() - وقت_بداية) / 1000.0)
            
            اطبع("📥 " + نسبة.صحيح() + "% | " + 
                  حجم_بشري(مستقبل) + "/" + حجم_بشري(حجم) + 
                  " | " + حجم_بشري(سرعة) + "/s", نهاية="\r")
        }
        
        ملف_خرج.اغلق()
        اطبع("")  // سطر جديد
        
        // 4. فك الضغط إذا لزم الأمر
        إذا مضغوط {
            اطبع("🗜️ جاري فك الضغط...")
            مسار_مفكوك = مسار_كامل.استبدل(".gz", "")
            ضغط.فك(مسار_كامل، مسار_مفكوك)
            ملف.احذف(مسار_كامل)
            مسار_كامل = مسار_مفكوك
        }
        
        // 5. التحقق من التجزئة
        اطبع("🔐 التحقق من صحة الملف...")
        تجزئة_محسوبة = حساب_md5(مسار_كامل)
        
        إذا تجزئة_محسوبة == تجزئة_md5 {
            اطبع("✅ الملف صحيح!")
            عميل.ارسل_الكل("نجح")
        } وإلا {
            اطبع("❌ الملف تالف!")
            عميل.ارسل_الكل("فشل")
            ملف.احذف(مسار_كامل)
        }
    }
    
    دالة استقبل_حزمة(عميل) {
        // استقبال حجم الحزمة (4 بايتات)
        حجم_بايتات = عميل.استقبل_الكل(4)
        حجم = نص.الى_رقم(نص.من_بايتات(حجم_بايتات))
        
        // استقبال البيانات
        ارجع عميل.استقبل_الكل(حجم)
    }
}

// دوال مساعدة
دالة حجم_بشري(بايتات) {
    إذا بايتات < 1024 {
        ارجع بايتات + " B"
    } وإلا إذا بايتات < 1024 * 1024 {
        ارجع (بايتات / 1024.0).مقرب(2) + " KB"
    } وإلا إذا بايتات < 1024 * 1024 * 1024 {
        ارجع (بايتات / 1024.0 / 1024.0).مقرب(2) + " MB"
    } وإلا {
        ارجع (بايتات / 1024.0 / 1024.0 / 1024.0).مقرب(2) + " GB"
    }
}

دالة حساب_md5(مسار_ملف) {
    محزّم = تجزئة.md5()
    ملف_دخل = ملف.افتح(مسار_ملف، "rb")
    
    بينما صحيح {
        دفعة = ملف_دخل.اقرأ(65536)
        إذا دفعة.طول() == 0 {
            اكسر
        }
        محزّم.حدث(دفعة)
    }
    
    ملف_دخل.اغلق()
    ارجع محزّم.hex()
}

دالة اصغر(أ، ب) {
    ارجع أ < ب ? أ : ب
}

// تشغيل الخادم
دالة رئيسية() {
    خادم = خادم_ملفات(9000)
    خادم.ابدأ()
}
```
