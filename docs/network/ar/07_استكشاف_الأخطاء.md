## 📖 مقدمة

حتى أفضل المبرمجين يواجهون مشاكل شبكية. الفرق ليس في عدم مواجهة المشاكل، بل في القدرة على تشخيصها وحلها بسرعة. في هذا الدليل، سنستكشف المشاكل الأكثر شيوعاً في برمجة الشبكات وكيفية حلها خطوة بخطوة.

## 🚫 المشكلة 1: "فشل الاتصال - Connection Refused"

### الأعراض
```
❌ خطأ: فشل الاتصال بالخادم
   نوع الخطأ: ConnectionRefused
   الرسالة: الاتصال رُفض من الطرف الآخر
```

### الأسباب المحتملة

**1. الخادم لا يعمل**
```sad
// تحقق من تشغيل الخادم
اطبع("هل الخادم يعمل؟")
اطبع("شغّله أولاً: sad خادم.ص")
```

**2. رقم المنفذ خاطئ**
```sad
// تأكد من استخدام نفس المنفذ
// الخادم:
خادم = شبكة.انشئ_خادم_tcp(8080)  // المنفذ 8080

// العميل:
عميل = شبكة.اتصل_tcp("localhost", 8080)  // نفس المنفذ!
```

**3. عنوان IP خاطئ**
```sad
// للاتصال بنفس الجهاز، استخدم:
عميل = شبكة.اتصل_tcp("127.0.0.1", 8080)
// أو
عميل = شبكة.اتصل_tcp("localhost", 8080)

// لأجهزة أخرى على الشبكة:
عميل = شبكة.اتصل_tcp("192.168.1.100", 8080)
```

**4. جدار ناري يحجب الاتصال**
```bash
# على Windows، افتح PowerShell كمسؤول:
New-NetFirewallRule -DisplayName "لغة ص" -Direction Inbound -Program "C:\path\to\sad.exe" -Action Allow

# على Linux:
sudo ufw allow 8080/tcp
```

### الحل الشامل

```sad
دالة اتصل_مع_تشخيص(عنوان، منفذ) {
    اطبع("🔍 تشخيص الاتصال...")
    اطبع("   العنوان: " + عنوان)
    اطبع("   المنفذ: " + منفذ)
    اطبع("")
    
    // 1. التحقق من صلاحية العنوان
    إذا عنوان.طول() == 0 {
        اطبع("❌ العنوان فارغ!")
        ارجع باطل
    }
    
    // 2. التحقق من صلاحية المنفذ
    إذا منفذ < 1 أو منفذ > 65535 {
        اطبع("❌ المنفذ غير صالح! يجب أن يكون بين 1 و 65535")
        ارجع باطل
    }
    
    // 3. محاولة الاتصال
    اطبع("⏳ جاري الاتصال...")
    
    محاولة {
        عميل = شبكة.اتصل_tcp(عنوان، منفذ)
        اطبع("✅ تم الاتصال بنجاح!")
        ارجع عميل
        
    } اصطياد (خطأ_شبكة.اتصال_مرفوض) {
        اطبع("❌ الاتصال رُفض")
        اطبع("")
        اطبع("🔧 الحلول المقترحة:")
        اطبع("   1. تأكد من تشغيل الخادم")
        اطبع("   2. تحقق من رقم المنفذ الصحيح")
        اطبع("   3. تأكد من العنوان الصحيح")
        اطبع("   4. تحقق من جدار الحماية")
        
    } اصطياد (خطأ_شبكة.مهلة_انتهت) {
        اطبع("⏰ انتهت مهلة الاتصال")
        اطبع("")
        اطبع("🔧 الحلول المقترحة:")
        اطبع("   1. الخادم بطيء جداً")
        اطبع("   2. مشكلة في الشبكة")
        اطبع("   3. العنوان غير صحيح")
        
    } اصطياد (خطأ) {
        اطبع("❌ خطأ: " + خطأ.رسالة())
    }
    
    ارجع باطل
}
```

## ⏰ المشكلة 2: "البرنامج 'يتجمد' ولا يستجيب"

### الأعراض
- البرنامج يتوقف عند استدعاء `استقبل()` ولا يعود أبداً
- لا يمكن إيقاف البرنامج إلا بـ Ctrl+C

### السبب
عدم استخدام مهلات زمنية، والانتظار اللانهائي لبيانات قد لا تصل أبداً.

### الحل

```sad
// ❌ سيء - قد ننتظر للأبد
بيانات = عميل.استقبل(1024)

// ✅ جيد - مهلة 10 ثوانٍ
عميل.عين_مهلة_الاستقبال(10000)

محاولة {
    بيانات = عميل.استقبل(1024)
    اطبع("✅ استقبلت البيانات")
    
} اصطياد (خطأ_شبكة.مهلة_انتهت) {
    اطبع("⏰ لم تصل بيانات خلال 10 ثوانٍ")
    اطبع("الخادم ربما بطيء أو معطل")
}
```

**أداة تشخيص:**
```sad
دالة استقبل_مع_تتبع(عميل، حجم، مهلة=5000) {
    اطبع("⏳ في انتظار البيانات (مهلة: " + مهلة/1000 + "ث)...")
    
    عميل.عين_مهلة_الاستقبال(مهلة)
    وقت_بداية = وقت.الآن()
    
    محاولة {
        بيانات = عميل.استقبل(حجم)
        مدة = وقت.الآن() - وقت_بداية
        
        اطبع("✅ استقبلت " + بيانات.طول() + " بايت في " + مدة + "ms")
        ارجع بيانات
        
    } اصطياد (خطأ_شبكة.مهلة_انتهت) {
        اطبع("⏰ انتهت المهلة بعد " + مهلة + "ms")
        اطبع("💡 ربما:")
        اطبع("   - الخادم بطيء")
        اطبع("   - الشبكة بطيئة")
        اطبع("   - الخادم لا يرسل البيانات")
        ارمِ خطأ
    }
}
```

## 🔌 المشكلة 3: "العنوان قيد الاستخدام - Address Already in Use"

### الأعراض
```
❌ خطأ: فشل إنشاء الخادم
   نوع الخطأ: AddressInUse
   الرسالة: العنوان 0.0.0.0:8080 قيد الاستخدام بالفعل
```

### السبب
برنامج آخر (أو نسخة قديمة من برنامجك) يستخدم نفس المنفذ.

### الحل السريع

```sad
// الحل 1: تفعيل SO_REUSEADDR
خادم = شبكة.انشئ_خادم_tcp(8080)
خادم.عين_اعادة_استخدام_العنوان(صحيح)
```

### الحل الشامل

```sad
دالة انشئ_خادم_مرن(منفذ_مفضل، محاولات=10) {
    لـ ع من 0 إلى محاولات {
        منفذ = منفذ_مفضل + ع
        
        محاولة {
            خادم = شبكة.انشئ_خادم_tcp(منفذ)
            خادم.عين_اعادة_استخدام_العنوان(صحيح)
            
            اطبع("✅ الخادم يعمل على المنفذ " + منفذ)
            ارجع خادم
            
        } اصطياد (خطأ_شبكة.عنوان_مستخدم) {
            إذا ع == 0 {
                اطبع("⚠️ المنفذ " + منفذ + " مستخدم، أجرب البدائل...")
            }
            استمر
        }
    }
    
    اطبع("❌ فشل إيجاد منفذ متاح!")
    ارجع باطل
}

// الاستخدام
خادم = انشئ_خادم_مرن(8080)
```

### إيجاد البرنامج المستخدم للمنفذ

**على Windows:**
```bash
# PowerShell
netstat -ano | findstr :8080
# سترى PID (معرف العملية)

# إنهاء العملية
taskkill /PID <رقم_PID> /F
```

**على Linux:**
```bash
# إيجاد البرنامج
sudo lsof -i :8080

# إنهاء العملية
sudo kill <PID>
```

## 📦 المشكلة 4: "البيانات المستقبلة ناقصة أو مقطعة"

### الأعراض
- أرسلت 1000 بايت لكن `استقبل()` أعاد 500 بايت فقط
- الرسائل تصل مقطعة أو ممزوجة

### السبب
TCP يعمل كـ"تدفق بايتات" وليس "رسائل منفصلة". قد يجمع أو يقسم البيانات.

### الحل

```sad
// ❌ سيء - لا ضمان لاستقبال كل البيانات
بيانات = عميل.استقبل(1000)

// ✅ جيد - استخدم استقبل_الكل()
بيانات = عميل.استقبل_الكل(1000)  // سيستقبل 1000 بايت بالضبط
```

**بروتوكول محدد الطول:**
```sad
// الإرسال: أرسل الحجم أولاً
دالة ارسل_رسالة(عميل، رسالة) {
    بيانات = نص.الى_بايتات(رسالة)
    حجم = بيانات.طول()
    
    // 1. أرسل الحجم (4 بايتات)
    عميل.ارسل_الكل(رمز_رقم_صحيح(حجم، 4))
    
    // 2. أرسل البيانات
    عميل.ارسل_الكل(بيانات)
}

// الاستقبال: اقرأ الحجم أولاً
دالة استقبل_رسالة(عميل) {
    // 1. استقبل الحجم (4 بايتات)
    حجم_بايتات = عميل.استقبل_الكل(4)
    حجم = فك_رقم_صحيح(حجم_بايتات)
    
    // 2. استقبل البيانات (بالحجم المحدد)
    بيانات = عميل.استقبل_الكل(حجم)
    
    ارجع نص.من_بايتات(بيانات)
}
```

**بروتوكول محدد بفاصل:**
```sad
// الإرسال: أضف فاصل
دالة ارسل_سطر(عميل، سطر) {
    عميل.ارسل_الكل(سطر + "\n")
}

// الاستقبال: اقرأ حتى الفاصل
دالة استقبل_سطر(عميل) {
    مخزن = ""
    
    بينما صحيح {
        حرف = عميل.استقبل(1)  // حرف واحد
        
        إذا حرف == "\n" {
            اكسر  // وصلنا لنهاية السطر
        }
        
        مخزن = مخزن + حرف
    }
    
    ارجع مخزن
}
```

## 🐌 المشكلة 5: "النقل بطيء جداً"

### الأعراض
- نقل ملف 10MB يستغرق عدة دقائق
- السرعة أقل بكثير من سرعة الشبكة

### الأسباب والحلول

**1. خوارزمية Nagle مفعّلة**
```sad
// الحل: تعطيل Nagle
عميل = شبكة.اتصل_tcp("server.com", 8080)
عميل.عين_بدون_تأخير(صحيح)  // TCP_NODELAY
```

**2. مخازن مؤقتة صغيرة**
```sad
// ❌ سيء - مخزن صغير = استدعاءات كثيرة
بينما صحيح {
    دفعة = ملف.اقرأ(1024)  // 1KB فقط
    عميل.ارسل(دفعة)
}

// ✅ جيد - مخزن كبير = استدعاءات أقل
بينما صحيح {
    دفعة = ملف.اقرأ(65536)  // 64KB
    عميل.ارسل_الكل(دفعة)
}
```

**3. عدم استخدام SO_SNDBUF/SO_RCVBUF**
```sad
// زيادة أحجام المخازن
عميل = شبكة.اتصل_tcp("server.com", 8080)
عميل.عين_حجم_مخزن_الإرسال(256 * 1024)   // 256KB
عميل.عين_حجم_مخزن_الاستقبال(256 * 1024)  // 256KB
```

**أداة قياس السرعة:**
```sad
دالة قياس_سرعة_نقل(عميل، حجم_اختبار=10485760) {
    // 10MB اختبار
    اطبع("📊 قياس سرعة النقل...")
    اطبع("   حجم الاختبار: " + حجم_بشري(حجم_اختبار))
    
    // إنشاء بيانات عشوائية
    بيانات = انشئ_بيانات_عشوائية(حجم_اختبار)
    
    // قياس الإرسال
    وقت_بداية = وقت.الآن()
    عميل.ارسل_الكل(بيانات)
    مدة = (وقت.الآن() - وقت_بداية) / 1000.0
    
    // حساب السرعة
    سرعة_بايت_ثانية = حجم_اختبار / مدة
    سرعة_mb_ثانية = سرعة_بايت_ثانية / 1024 / 1024
    
    اطبع("✅ النتيجة:")
    اطبع("   المدة: " + مدة.مقرب(2) + " ثانية")
    اطبع("   السرعة: " + سرعة_mb_ثانية.مقرب(2) + " MB/s")
    
    إذا سرعة_mb_ثانية < 1 {
        اطبع("⚠️ السرعة بطيئة! الأسباب المحتملة:")
        اطبع("   - مخازن صغيرة")
        اطبع("   - Nagle's algorithm مفعلة")
        اطبع("   - الشبكة بطيئة")
    }
}
```

## 🔄 المشكلة 6: "حزم UDP تضيع"

### الأعراض
- ترسل 100 حزمة UDP، لكن 20 منها لا تصل
- الألعاب/البث يتقطع

### السبب الطبيعي
UDP لا يضمن التوصيل. هذا عادي ومتوقع!

### الحلول

**1. إعادة إرسال البيانات الحرجة**
```sad
دالة ارسل_مع_تأكيد(مقبس، بيانات، عنوان، محاولات=3) {
    لـ ع من 1 إلى محاولات {
        // إرسال
        مقبس.ارسل_الى(بيانات، عنوان)
        
        // انتظار تأكيد
        مقبس.عين_مهلة_الاستقبال(1000)
        
        محاولة {
            (تأكيد، مرسل) = مقبس.استقبل_من(16)
            
            إذا تأكيد == "ACK" {
                ارجع صحيح  // نجح!
            }
        } اصطياد (مهلة) {
            اطبع("⏰ المحاولة " + ع + " فشلت")
        }
    }
    
    اطبع("❌ فشل بعد " + محاولات + " محاولات")
    ارجع خطأ
}
```

**2. رقم تسلسل للكشف عن الفقد**
```sad
// الإرسال
رقم_تسلسل = 0

دالة ارسل_مع_رقم(مقبس، بيانات، عنوان) {
    // إضافة رقم تسلسل في البداية
    حزمة = رمز_رقم_صحيح(رقم_تسلسل، 4) + بيانات
    مقبس.ارسل_الى(حزمة، عنوان)
    
    رقم_تسلسل = رقم_تسلسل + 1
}

// الاستقبال
رقم_متوقع = 0

دالة استقبل_مع_كشف_فقد(مقبس) {
    (حزمة، مرسل) = مقبس.استقبل_من(4096)
    
    // استخراج رقم التسلسل
    رقم = فك_رقم_صحيح(حزمة[0:4])
    بيانات = حزمة[4:]
    
    // كشف الفقد
    إذا رقم != رقم_متوقع {
        فاقد = رقم - رقم_متوقع
        اطبع("⚠️ ضاعت " + فاقد + " حزم!")
    }
    
    رقم_متوقع = رقم + 1
    ارجع بيانات
}
```

**3. Forward Error Correction (للمتقدمين)**
أرسل بيانات زائدة تسمح باستعادة الحزم المفقودة رياضياً.

## 🎓 خلاصة: خطوات التشخيص العامة

عند مواجهة أي مشكلة شبكية:

**1. اقرأ رسالة الخطأ بعناية**
```sad
محاولة {
    // كودك
} اصطياد (خطأ) {
    اطبع("نوع الخطأ: " + خطأ.نوع())
    اطبع("الرسالة: " + خطأ.رسالة())
    اطبع("الكود: " + خطأ.كود())
}
```

**2. أضف تسجيلاً تفصيلياً**
```sad
اطبع("[DEBUG] جاري الاتصال بـ " + عنوان + ":" + منفذ)
محاولة {
    عميل = شبكة.اتصل_tcp(عنوان، منفذ)
    اطبع("[DEBUG] تم الاتصال بنجاح")
} اصطياد (خطأ) {
    اطبع("[DEBUG] فشل الاتصال: " + خطأ.رسالة())
}
```

**3. اختبر المكونات بشكل منفصل**
- هل الخادم يعمل؟
- هل العميل يتصل بالعنوان الصحيح؟
- هل البيانات ترسل/تستقبل؟

**4. استخدم أدوات الشبكة**
- `ping` - هل الخادم موجود؟
- `telnet` - هل المنفذ مفتوح؟
- `netstat` - ما المنافذ المستخدمة؟

**5. قلل التعقيد حتى يعمل**
ابدأ ببرنامج بسيط جداً، ثم أضف الميزات تدريجياً.
