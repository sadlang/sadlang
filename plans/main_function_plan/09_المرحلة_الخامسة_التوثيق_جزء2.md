### المهمة 5.5: أمثلة تطبيقية

#### مثال 1: تطبيق قائمة مهام

**الملف:** `examples/main_function/04_real_world/todo_app.s`

```sad
// ========================================
// تطبيق قائمة المهام
// To-Do List Application
// ========================================
// الاستخدام:
//   sad todo_app.s add "مهمة جديدة"     - إضافة مهمة
//   sad todo_app.s list                 - عرض المهام
//   sad todo_app.s done <رقم>           - تعليم مهمة كمكتملة
//   sad todo_app.s remove <رقم>         - حذف مهمة
// ========================================

// ============ الثوابت ============
ثابت نص ملف_المهام = "tasks.txt"

// ============ الهياكل ============
صنف مهمة
    رقم المعرف
    نص الوصف
    منطق مكتملة
    
    دالة نص إلى_نص()
        نص الحالة = إذا مكتملة "✓" غير_ذلك "☐"
        إرجاع الحالة + " " + المعرف + ". " + الوصف
    نهاية
نهاية

// ============ الدوال المساعدة ============

/**
 * قراءة جميع المهام من الملف
 */
دالة مهمة[] قراءة_المهام()
    إذا !ملف_موجود(ملف_المهام)
        إرجاع []
    نهاية
    
    مهمة[] المهام = []
    نص[] الأسطر = قراءة_أسطر(ملف_المهام)
    
    لكل سطر في الأسطر
        إذا طول(سطر) > 0
            // تحليل السطر: معرف|وصف|حالة
            نص[] الأجزاء = تقسيم(سطر، "|")
            
            مهمة م = جديد مهمة()
            م.المعرف = إلى_رقم(الأجزاء[0])
            م.الوصف = الأجزاء[1]
            م.مكتملة = الأجزاء[2] == "1"
            
            إضافة(المهام، م)
        نهاية
    نهاية
    
    إرجاع المهام
نهاية

/**
 * حفظ المهام في الملف
 */
دالة عدم حفظ_المهام(مهمة[] المهام)
    نص[] الأسطر = []
    
    لكل مهمة في المهام
        نص السطر = مهمة.المعرف + "|" + 
                     مهمة.الوصف + "|" + 
                     (إذا مهمة.مكتملة "1" غير_ذلك "0")
        إضافة(الأسطر، السطر)
    نهاية
    
    كتابة_أسطر(ملف_المهام، الأسطر)
نهاية

/**
 * إضافة مهمة جديدة
 */
دالة صحيح إضافة_مهمة(نص الوصف)
    مهمة[] المهام = قراءة_المهام()
    
    مهمة م = جديد مهمة()
    م.المعرف = طول(المهام) + 1
    م.الوصف = الوصف
    م.مكتملة = خطأ
    
    إضافة(المهام، م)
    حفظ_المهام(المهام)
    
    اطبع("✓ تمت إضافة المهمة #" + م.المعرف)
    إرجاع 0
نهاية

/**
 * عرض جميع المهام
 */
دالة صحيح عرض_المهام()
    مهمة[] المهام = قراءة_المهام()
    
    إذا طول(المهام) == 0
        اطبع("لا توجد مهام حالياً")
        إرجاع 0
    نهاية
    
    اطبع("========================================")
    اطبع("قائمة المهام")
    اطبع("========================================")
    
    لكل مهمة في المهام
        اطبع(مهمة.إلى_نص())
    نهاية
    
    اطبع("========================================")
    إرجاع 0
نهاية

/**
 * تعليم مهمة كمكتملة
 */
دالة صحيح تعليم_مكتملة(رقم المعرف)
    مهمة[] المهام = قراءة_المهام()
    منطق وُجدت = خطأ
    
    لكل مهمة في المهام
        إذا مهمة.المعرف == المعرف
            مهمة.مكتملة = صحيح
            وُجدت = صحيح
            اطبع("✓ تم تعليم المهمة #" + المعرف + " كمكتملة")
        نهاية
    نهاية
    
    إذا !وُجدت
        اطبع("❌ خطأ: المهمة #" + المعرف + " غير موجودة")
        إرجاع 1
    نهاية
    
    حفظ_المهام(المهام)
    إرجاع 0
نهاية

/**
 * حذف مهمة
 */
دالة صحيح حذف_مهمة(رقم المعرف)
    مهمة[] المهام = قراءة_المهام()
    مهمة[] المهام_الجديدة = []
    منطق وُجدت = خطأ
    
    لكل مهمة في المهام
        إذا مهمة.المعرف != المعرف
            إضافة(المهام_الجديدة، مهمة)
        غير_ذلك
            وُجدت = صحيح
        نهاية
    نهاية
    
    إذا !وُجدت
        اطبع("❌ خطأ: المهمة #" + المعرف + " غير موجودة")
        إرجاع 1
    نهاية
    
    حفظ_المهام(المهام_الجديدة)
    اطبع("✓ تم حذف المهمة #" + المعرف)
    إرجاع 0
نهاية

/**
 * عرض المساعدة
 */
دالة عدم عرض_المساعدة()
    اطبع("========================================")
    اطبع("تطبيق قائمة المهام")
    اطبع("========================================")
    اطبع("")
    اطبع("الاستخدام:")
    اطبع("  sad todo_app.s <أمر> [وسائط]")
    اطبع("")
    اطبع("الأوامر:")
    اطبع("  add <وصف>      إضافة مهمة جديدة")
    اطبع("  list            عرض جميع المهام")
    اطبع("  done <رقم>      تعليم مهمة كمكتملة")
    اطبع("  remove <رقم>    حذف مهمة")
    اطبع("  help            عرض هذه المساعدة")
    اطبع("")
    اطبع("أمثلة:")
    اطبع("  sad todo_app.s add \"شراء الحليب\"")
    اطبع("  sad todo_app.s list")
    اطبع("  sad todo_app.s done 1")
    اطبع("  sad todo_app.s remove 2")
    اطبع("========================================")
نهاية

// ============ الدالة الرئيسية ============

دالة صحيح رئيسية(نص[] الوسائط)
    // التحقق من الوسائط
    إذا طول(الوسائط) == 0
        عرض_المساعدة()
        إرجاع 1
    نهاية
    
    نص الأمر = الوسائط[0]
    
    // معالجة الأوامر
    إذا الأمر == "add"
        إذا طول(الوسائط) < 2
            اطبع("❌ خطأ: يجب تحديد وصف المهمة")
            إرجاع 1
        نهاية
        إرجاع إضافة_مهمة(الوسائط[1])
    
    وإلا_إذا الأمر == "list"
        إرجاع عرض_المهام()
    
    وإلا_إذا الأمر == "done"
        إذا طول(الوسائط) < 2
            اطبع("❌ خطأ: يجب تحديد رقم المهمة")
            إرجاع 1
        نهاية
        رقم المعرف = إلى_رقم(الوسائط[1])
        إرجاع تعليم_مكتملة(المعرف)
    
    وإلا_إذا الأمر == "remove"
        إذا طول(الوسائط) < 2
            اطبع("❌ خطأ: يجب تحديد رقم المهمة")
            إرجاع 1
        نهاية
        رقم المعرف = إلى_رقم(الوسائط[1])
        إرجاع حذف_مهمة(المعرف)
    
    وإلا_إذا الأمر == "help"
        عرض_المساعدة()
        إرجاع 0
    
    غير_ذلك
        اطبع("❌ خطأ: أمر غير معروف: " + الأمر)
        اطبع("استخدم 'help' لعرض المساعدة")
        إرجاع 1
    نهاية
نهاية
```

الآن يمكنك كتابة برامج منظمة مع دالة رئيسية:

```sad
دالة صحيح رئيسية()
    اطبع("مرحباً يا عالم!")
    إرجاع 0
نهاية
```

**المزايا:**
- ✅ نقطة دخول واضحة للبرنامج
- ✅ تمرير وسائط سطر الأوامر
- ✅ كود خروج قابل للتحكم
- ✅ بنية منظمة ومعيارية

**مثال مع الوسائط:**

```sad
دالة صحيح رئيسية(نص[] الوسائط)
    لكل وسيط في الوسائط
        اطبع(وسيط)
    نهاية
    إرجاع 0
نهاية
```

**التشغيل:**
```bash
sad program.s arg1 arg2 arg3
```

## [2.0.0] - 2025-12-28

### ✨ إضافات

#### الدالة الرئيسية (Main Function)
- إضافة دعم الدالة الرئيسية كنقطة دخول للبرامج
- الكلمة المفتاحية الجديدة: `رئيسية` / `main`
- تمرير وسائط سطر الأوامر (command-line arguments)
- كود الخروج (exit code) من قيمة الإرجاع
- التحقق من صحة توقيع الدالة الرئيسية
- منع تعريف أكثر من دالة رئيسية
- منع تنفيذ كود خارج الدوال (في الوضع الجديد)

**قبل:**
```sad
اطبع("مرحباً")
```

**بعد:**
```sad
دالة رئيسية()
    اطبع("مرحباً")
نهاية

- تطبيق قائمة مهام (To-Do App)
