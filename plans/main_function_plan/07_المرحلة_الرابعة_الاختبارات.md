```cpp
#include "../../../include/lexer/lexer_core.h"
#include <cassert>
#include <iostream>

using namespace Sad::Lexer;

void test_main_keyword_recognition() {
    std::string code = "دالة رئيسية()";
    LexerCore lexer(code);
    auto tokens = lexer.tokenize();
    
    assert(tokens.size() >= 4);
    assert(tokens[0].getType() == TokenType::KEYWORD_FUNCTION);
    assert(tokens[1].getType() == TokenType::KEYWORD_MAIN);
    assert(tokens[2].getType() == TokenType::PAREN_LEFT);
    assert(tokens[3].getType() == TokenType::PAREN_RIGHT);
    
    std::cout << "✓ Main keyword recognition test passed" << std::endl;
}

void test_main_keyword_as_reserved_word() {
    // رئيسية يجب أن تكون كلمة محجوزة، لا معرّف
    std::string code = "رقم رئيسية = 10";
    LexerCore lexer(code);
    auto tokens = lexer.tokenize();
    
    // يجب أن يتعرف على رئيسية ككلمة مفتاحية، ليس معرّف
    bool foundMainAsKeyword = false;
    for (const auto& token : tokens) {
        if (token.getType() == TokenType::KEYWORD_MAIN) {
            foundMainAsKeyword = true;
            break;
        }
    }
    
    assert(foundMainAsKeyword);
    std::cout << "✓ Main keyword as reserved word test passed" << std::endl;
}

int main() {
    try {
        test_main_keyword_recognition();
        test_main_keyword_as_reserved_word();
        
        std::cout << "========================================" << std::endl;
        std::cout << "✅ All lexer unit tests passed!" << std::endl;
        std::cout << "========================================" << std::endl;
        return 0;
    } catch (const std::exception& e) {
        std::cerr << "❌ Test failed: " << e.what() << std::endl;
        return 1;
    }
}
```

```cpp
#include "../../../include/parser/parser_core.h"
#include "../../../include/lexer/lexer_core.h"
#include <cassert>
#include <iostream>

using namespace Sad::Parser;
using namespace Sad::Lexer;

void test_simple_main_function() {
    std::string code = 
        "دالة رئيسية()\n"
        "    اطبع(\"test\")\n"
        "نهاية";
    
    LexerCore lexer(code);
    auto tokens = lexer.tokenize();
    
    ParserCore parser(tokens);
    parser.setMainFunctionRequired(true);
    
    auto ast = parser.parseProgram();
    
    assert(parser.hasMainFunction());
    assert(ast.size() == 1);
    
    auto* funcDecl = dynamic_cast<AST::FunctionDecl*>(ast[0].get());
    assert(funcDecl != nullptr);
    assert(funcDecl->isMainFunction());
    
    std::cout << "✓ Simple main function parsing test passed" << std::endl;
}

void test_main_function_with_return_type() {
    std::string code = 
        "دالة صحيح رئيسية()\n"
        "    إرجاع 0\n"
        "نهاية";
    
    LexerCore lexer(code);
    auto tokens = lexer.tokenize();
    
    ParserCore parser(tokens);
    auto ast = parser.parseProgram();
    
    auto* funcDecl = dynamic_cast<AST::FunctionDecl*>(ast[0].get());
    assert(funcDecl->getReturnType() == Data::DataType::INTEGER);
    
    std::cout << "✓ Main with return type test passed" << std::endl;
}

void test_duplicate_main_error() {
    std::string code = 
        "دالة رئيسية()\n"
        "نهاية\n"
        "دالة رئيسية()\n"
        "نهاية";
    
    LexerCore lexer(code);
    auto tokens = lexer.tokenize();
    
    ParserCore parser(tokens);
    parser.setMainFunctionRequired(true);
    
    bool errorCaught = false;
    try {
        auto ast = parser.parseProgram();
    } catch (...) {
        errorCaught = true;
    }
    
    assert(errorCaught);
    std::cout << "✓ Duplicate main error test passed" << std::endl;
}

int main() {
```

```cpp
#include "../../../include/interpreter/core/interpreter_core.h"
#include "../../../include/parser/parser_core.h"
#include "../../../include/lexer/lexer_core.h"
#include <cassert>
#include <iostream>

using namespace Sad::Interpreter;
using namespace Sad::Parser;
using namespace Sad::Lexer;

void test_simple_main_execution() {
    std::string code = 
        "دالة صحيح رئيسية()\n"
        "    إرجاع 0\n"
        "نهاية";
    
    LexerCore lexer(code);
    auto tokens = lexer.tokenize();
    
    ParserCore parser(tokens);
    auto ast = parser.parseProgram();
    
    Interpreter interpreter;
    auto result = interpreter.executeWithMainFunction(ast, 0, nullptr);
    
    assert(result.success);
    assert(result.exitCode == 0);
    
    std::cout << "✓ Simple main execution test passed" << std::endl;
}

void test_main_exit_code() {
    std::string code = 
        "دالة صحيح رئيسية()\n"
        "    إرجاع 42\n"
        "نهاية";
    
    LexerCore lexer(code);
    auto tokens = lexer.tokenize();
    
    ParserCore parser(tokens);
    auto ast = parser.parseProgram();
    
    Interpreter interpreter;
    auto result = interpreter.executeWithMainFunction(ast, 0, nullptr);
    
    assert(result.success);
    assert(result.exitCode == 42);
    
    std::cout << "✓ Main exit code test passed" << std::endl;
}

void test_main_with_arguments() {
    std::string code = 
        "دالة صحيح رئيسية(نص[] الوسائط)\n"
        "    إرجاع طول(الوسائط)\n"
        "نهاية";
    
    LexerCore lexer(code);
    auto tokens = lexer.tokenize();
    
    ParserCore parser(tokens);
    auto ast = parser.parseProgram();

    auto result = interpreter.executeWithMainFunction(ast, argc, argv);
    
    assert(result.success);
    assert(result.exitCode == 3);  // عدد الوسائط (بدون اسم البرنامج)
    
    std::cout << "✓ Main with arguments test passed" << std::endl;
}

int main() {
    try {
        test_simple_main_execution();
        test_main_exit_code();
        test_main_with_arguments();
        
        std::cout << "========================================" << std::endl;
        std::cout << "✅ All interpreter unit tests passed!" << std::endl;
```

### 2. اختبارات التكامل (Integration Tests)

#### 2.1 اختبار التكامل الكامل

**الملف:** `tests/integration/test_main_function_integration.cpp`

```cpp
#include <iostream>
#include <string>
#include <vector>
#include <fstream>

// دالة لتنفيذ برنامج واختبار النتيجة
struct TestCase {
    std::string name;
    std::string code;
    std::vector<std::string> args;
    int expectedExitCode;
    std::vector<std::string> expectedOutput;
};

void runTest(const TestCase& test) {
    std::cout << "Running test: " << test.name << std::endl;
    
    // 1. حفظ الكود في ملف مؤقت
    std::string tempFile = "temp_test.s";
    std::ofstream out(tempFile);
    out << test.code;
    out.close();
    
    // 2. بناء سطر الأوامر
    std::string command = "sad " + tempFile;
    for (const auto& arg : test.args) {
        command += " " + arg;
    }
    
    // 3. تنفيذ البرنامج
    int exitCode = system(command.c_str());
    
    // 4. التحقق من كود الخروج
    assert(exitCode == test.expectedExitCode);
    
    // 5. تنظيف الملف المؤقت
    std::remove(tempFile.c_str());
    
    std::cout << "  ✓ Test passed" << std::endl;
}

int main() {
    std::vector<TestCase> tests = {
        {
            "Simple main function",
            "دالة صحيح رئيسية()\n"
            "    اطبع(\"Hello\")\n"
            "    إرجاع 0\n"
            "نهاية",
            {},
            0,
            {"Hello"}
        },
        {
            "Main with exit code",
            "دالة صحيح رئيسية()\n"
            "    إرجاع 42\n"
            "نهاية",
            {},
            42,
            {}
        },
        {
            "Main with arguments",
            "دالة صحيح رئيسية(نص[] الوسائط)\n"
            "    لكل وسيط في الوسائط\n"
            "        اطبع(وسيط)\n"
            "    نهاية\n"
            "    إرجاع 0\n"
            "نهاية",
            {"arg1", "arg2"},
            0,
            {"arg1", "arg2"}
        }
    };
    
    std::cout << "========================================" << std::endl;
    std::cout << "Integration Tests" << std::endl;
    std::cout << "========================================" << std::endl;
    
    for (const auto& test : tests) {
        runTest(test);
    }
    
    std::cout << "========================================" << std::endl;
    std::cout << "✅ All integration tests passed!" << std::endl;
    std::cout << "========================================" << std::endl;
    
    return 0;
}
```
