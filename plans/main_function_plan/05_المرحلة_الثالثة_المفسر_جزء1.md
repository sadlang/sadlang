### المهمة 3.1: إضافة دوال جديدة في Interpreter
**الملف:** `include/interpreter/core/interpreter_core.h`

#### إضافة دوال جديدة للواجهة العامة

```cpp
class Interpreter {
public:
    // الدوال الموجودة...
    explicit Interpreter(const InterpreterOptions& options = InterpreterOptions());
    ExecutionResult execute(const std::vector<std::unique_ptr<AST::Statement>>& program);
    ExecutionResult executeStatement(AST::Statement& statement);
    Data::Value evaluateExpression(AST::Expression& expression);
    
    // ← جديد: تنفيذ من خلال الدالة الرئيسية
    /**
     * @brief (AR) تنفيذ برنامج من خلال الدالة الرئيسية
     * @brief (EN) Execute program through main function
     * 
     * @param program (AR) البرنامج (قائمة التصريحات) / (EN) Program (list of declarations)
     * @param argc (AR) عدد الوسائط / (EN) Argument count
     * @param argv (AR) قائمة الوسائط / (EN) Argument values
     * @return (AR) نتيجة التنفيذ (مع كود الخروج) / (EN) Execution result (with exit code)
     * 
     * @details
     * هذه الدالة:
     * 1. تبحث عن الدالة الرئيسية في البرنامج
     * 2. تسجل جميع التصريحات (دوال، أصناف، متغيرات عامة)
     * 3. تحضر وسائط سطر الأوامر
     * 4. تستدعي الدالة الرئيسية
     * 5. تُرجع كود الخروج
     */
    ExecutionResult executeWithMainFunction(
        const std::vector<std::unique_ptr<AST::Statement>>& program,
        int argc = 0,
        char* argv[] = nullptr);
    
    // ← جديد: تنفيذ تلقائي (يختار الطريقة المناسبة)
    /**
     * @brief (AR) تنفيذ تلقائي - يختار بين execute و executeWithMainFunction
     * @brief (EN) Auto execute - chooses between execute and executeWithMainFunction
     * 
     * @param program البرنامج
     * @param argc عدد الوسائط (0 إذا لم يكن هناك)
     * @param argv قائمة الوسائط (nullptr إذا لم يكن هناك)
     * @return نتيجة التنفيذ
     * 
     * @details
     * - إذا وُجدت دالة رئيسية، يُستخدم executeWithMainFunction
     * - إذا لم توجد، يُستخدم execute (الوضع القديم)
     */
    ExecutionResult executeAuto(
        const std::vector<std::unique_ptr<AST::Statement>>& program,
        int argc = 0,
        char* argv[] = nullptr);

private:
    // دوال مساعدة جديدة
    
    /**
     * @brief (AR) البحث عن الدالة الرئيسية في البرنامج
     * @brief (EN) Find main function in program
     * 
     * @param program البرنامج
     * @return مؤشر للدالة الرئيسية أو nullptr
     */
    AST::FunctionDecl* findMainFunction(
        const std::vector<std::unique_ptr<AST::Statement>>& program);
    
    /**
     * @brief (AR) تسجيل جميع التصريحات (دوال، أصناف، متغيرات)
     * @brief (EN) Register all declarations (functions, classes, variables)
     * 
     * @param program البرنامج
     * 
     * @details
     * يمر على جميع الجمل ويسجل:
     * - تعريفات الدوال
     * - تعريفات الأصناف
     * - المتغيرات العامة (الثوابت)
     */
    void registerDeclarations(
        const std::vector<std::unique_ptr<AST::Statement>>& program);
    
    /**
     * @brief (AR) تحضير وسائط الدالة الرئيسية من argc/argv
     * @brief (EN) Prepare main function arguments from argc/argv
     * 
     * @param argc عدد الوسائط
     * @param argv قائمة الوسائط
     * @return مصفوفة من القيم (نص[])
     * 
     * @details
     * يحول argc/argv إلى مصفوفة نصوص:
     * - argv[0] = اسم البرنامج (يتم تجاوزه)
     * - argv[1..n] = الوسائط الفعلية
     */
    std::vector<Data::Value> prepareMainArguments(int argc, char* argv[]);
    
    /**
     * @brief (AR) استخراج كود الخروج من قيمة الإرجاع
     * @brief (EN) Extract exit code from return value
     * 
     * @param returnValue قيمة الإرجاع من الدالة الرئيسية
     * @return كود الخروج (int)
     * 
     * @details
     * - إذا كانت القيمة صحيح (int): يُستخدم كما هو
     * - إذا كانت القيمة منطقية (bool): true→0, false→1
     * - إذا كانت فراغ (void): 0
     * - غير ذلك: 0 (افتراضي)
     */
    int extractExitCode(const Data::Value& returnValue);
};
```

### المهمة 3.2: تنفيذ executeWithMainFunction
**الملف:** `src/interpreter/core/interpreter_core.cpp`

#### التنفيذ الكامل

```cpp
ExecutionResult Interpreter::executeWithMainFunction(
    const std::vector<std::unique_ptr<AST::Statement>>& program,
    int argc,
    char* argv[]) {
    
    try {
        if (options_.enableDebugMode) {
            std::cout << "(AR) تنفيذ البرنامج من خلال الدالة الرئيسية"
                     << " / (EN) Executing program through main function"
                     << std::endl;
        }
        
        // الخطوة 1: البحث عن الدالة الرئيسية
        AST::FunctionDecl* mainFunc = findMainFunction(program);
        
        if (!mainFunc) {
            std::string errorMsg = 
                "(AR) خطأ: لم يتم العثور على دالة رئيسية في البرنامج\n"
                "يجب تعريف:\n"
                "  دالة رئيسية()\n"
                "      // الأوامر\n"
                "  نهاية\n\n"
                "(EN) Error: Main function not found in program\n"
                "Must define:\n"
                "  function main()\n"
                "      // statements\n"
                "  end";
            
            return ExecutionResult(false, Data::Value(), errorMsg);
        }
        
        if (options_.enableDebugMode) {
            std::cout << "  ✓ (AR) تم العثور على الدالة الرئيسية"
                     << " / (EN) Main function found" << std::endl;
        }
        
        // الخطوة 2: تسجيل جميع التصريحات
        if (options_.enableDebugMode) {
            std::cout << "  (AR) تسجيل التصريحات..."
                     << " / (EN) Registering declarations..." << std::endl;
        }
        
        registerDeclarations(program);
        
        if (options_.enableDebugMode) {
            std::cout << "  ✓ (AR) تم تسجيل جميع التصريحات"
                     << " / (EN) All declarations registered" << std::endl;
        }
        
        // الخطوة 3: تحضير وسائط الدالة الرئيسية
        std::vector<Data::Value> mainArgs;
        
        if (mainFunc->getParameters().size() > 0) {
            // الدالة الرئيسية تتوقع وسائط
            mainArgs = prepareMainArguments(argc, argv);
            
            if (options_.enableDebugMode) {
                std::cout << "  (AR) تم تحضير " << (argc - 1) 
                         << " وسيط / (EN) Prepared " << (argc - 1) 
                         << " arguments" << std::endl;
            }
        }
        
        // الخطوة 4: استدعاء الدالة الرئيسية
        if (options_.enableDebugMode) {
            std::cout << "  (AR) استدعاء الدالة الرئيسية..."
                     << " / (EN) Calling main function..." << std::endl;
            std::cout << std::endl;
        }
        
        // استخدام callFunction من FunctionManager
        std::string mainFunctionName = mainFunc->getName();
        Data::Value returnValue;
        
        try {
            returnValue = functionManager_->callFunction(
                mainFunctionName,
                mainArgs,
                this  // تمرير المفسر للوصول إلى executeStatement
            );
        }
        catch (const std::exception& e) {
            std::string errorMsg = 
                "(AR) خطأ في تنفيذ الدالة الرئيسية: " + std::string(e.what()) +
                " / (EN) Error executing main function: " + std::string(e.what());
            
            return ExecutionResult(false, Data::Value(), errorMsg);
        }
        
        // الخطوة 5: استخراج كود الخروج
        int exitCode = extractExitCode(returnValue);
        
        if (options_.enableDebugMode) {
            std::cout << std::endl;
            std::cout << "(AR) انتهى تنفيذ الدالة الرئيسية - كود الخروج: " << exitCode
                     << " / (EN) Main function completed - exit code: " << exitCode
                     << std::endl;
        }
        
        // إنشاء نتيجة مع كود الخروج
        auto result = ExecutionResult(true, returnValue);
        result.exitCode = exitCode;
        
        return result;
    }
    catch (const ExecutionError& e) {
        std::string errorMsg = "(AR) خطأ في التنفيذ: " + std::string(e.what()) +
                              " / (EN) Execution error: " + std::string(e.what());
        return ExecutionResult(false, Data::Value(), errorMsg);
    }
    catch (const RuntimeError& e) {
        std::string errorMsg = "(AR) خطأ في وقت التشغيل: " + std::string(e.what()) +
                              " / (EN) Runtime error: " + std::string(e.what());
        return ExecutionResult(false, Data::Value(), errorMsg);
    }
    catch (const std::exception& e) {
        std::string errorMsg = "(AR) خطأ غير متوقع: " + std::string(e.what()) +
                              " / (EN) Unexpected error: " + std::string(e.what());
        return ExecutionResult(false, Data::Value(), errorMsg);
    }
}
```

### المهمة 3.3: تنفيذ findMainFunction

```cpp
AST::FunctionDecl* Interpreter::findMainFunction(
    const std::vector<std::unique_ptr<AST::Statement>>& program) {
    
    for (const auto& stmt : program) {
        // التحقق إذا كانت الجملة FunctionDecl
        if (auto* funcDecl = dynamic_cast<AST::FunctionDecl*>(stmt.get())) {
            // التحقق إذا كانت دالة رئيسية
            if (funcDecl->isMainFunction()) {
                return funcDecl;
            }
        }
    }
    
    // لم يتم العثور على دالة رئيسية
    return nullptr;
}
```

### المهمة 3.4: تنفيذ registerDeclarations

```cpp
void Interpreter::registerDeclarations(
    const std::vector<std::unique_ptr<AST::Statement>>& program) {
    
    for (const auto& stmt : program) {
        // التحقق من نوع الجملة وتسجيلها
        
        // تسجيل الدوال
        if (auto* funcDecl = dynamic_cast<AST::FunctionDecl*>(stmt.get())) {
            // لا نُنفذ الدالة، فقط نُسجلها
            // سيتم ذلك عند زيارة FunctionDecl في statementExecutor
            stmt->accept(*statementExecutor_);
        }
        // تسجيل الأصناف
        else if (auto* classDecl = dynamic_cast<AST::ClassDecl*>(stmt.get())) {
            stmt->accept(*statementExecutor_);
        }
        // تسجيل المتغيرات العامة (الثوابت)
        else if (auto* varDecl = dynamic_cast<AST::VarDecl*>(stmt.get())) {
            // فقط المتغيرات الثابتة العامة
            if (varDecl->isConst()) {
                stmt->accept(*statementExecutor_);
            }
        }
        // تسجيل Import/Export
        else if (auto* importStmt = dynamic_cast<AST::ImportStmt*>(stmt.get())) {
            stmt->accept(*statementExecutor_);
        }
        else if (auto* exportStmt = dynamic_cast<AST::ExportStmt*>(stmt.get())) {
            stmt->accept(*statementExecutor_);
        }
        // تجاهل باقي الجمل (لا يجب أن تكون موجودة خارج الدوال)
    }
}
```

```cpp
std::vector<Data::Value> Interpreter::prepareMainArguments(int argc, char* argv[]) {
    // إنشاء مصفوفة للوسائط
    auto argsArray = std::make_shared<Data::ArrayValue>();
    
    // تحويل argv إلى مصفوفة نصوص
    // نبدأ من argv[1] (نتجاوز اسم البرنامج في argv[0])
    for (int i = 1; i < argc; i++) {
        if (argv[i] != nullptr) {
            std::string arg(argv[i]);
            argsArray->push_back(Data::Value::makeString(arg));
        }
    }
    
    // إرجاع المصفوفة كمعامل وحيد
    std::vector<Data::Value> mainArgs;
    mainArgs.push_back(Data::Value::makeArray(argsArray));
    
    return mainArgs;
}
```

### المهمة 3.6: تنفيذ extractExitCode

```cpp
int Interpreter::extractExitCode(const Data::Value& returnValue) {
    // التحقق من نوع القيمة
    
    // صحيح (int) - استخدمه مباشرة
    if (returnValue.isInteger()) {
        return returnValue.toInt();
    }
    
    // منطقي (bool) - true→0, false→1
    if (returnValue.isBoolean()) {
        return returnValue.toBool() ? 0 : 1;
    }
    
    // فراغ (void) أو null - افترض نجاح (0)
    if (returnValue.isVoid() || returnValue.isNull()) {
        return 0;
    }
    
    // رقم عشري (double) - حوّله إلى صحيح
    if (returnValue.isDouble()) {
        return static_cast<int>(returnValue.toDouble());
    }
    
    // أي نوع آخر - افترض نجاح (0)
    if (options_.enableDebugMode) {
        std::cout << "(AR) تحذير: نوع إرجاع غير متوقع من الدالة الرئيسية، "
                  << "استخدام كود خروج افتراضي 0"
                  << " / (EN) Warning: Unexpected return type from main function, "
                  << "using default exit code 0"
                  << std::endl;
    }
    
    return 0;
}
```

### المهمة 3.7: تنفيذ executeAuto

```cpp
ExecutionResult Interpreter::executeAuto(
    const std::vector<std::unique_ptr<AST::Statement>>& program,
    int argc,
    char* argv[]) {
    
    // البحث عن دالة رئيسية
    AST::FunctionDecl* mainFunc = findMainFunction(program);
    
    if (mainFunc) {
        // وُجدت دالة رئيسية - استخدم الوضع الجديد
        if (options_.enableDebugMode) {
            std::cout << "(AR) تم اكتشاف دالة رئيسية - استخدام وضع التنفيذ الجديد"
                     << " / (EN) Main function detected - using new execution mode"
                     << std::endl;
        }
        
        return executeWithMainFunction(program, argc, argv);
    }
    else {
        // لم توجد دالة رئيسية - استخدم الوضع القديم
        if (options_.enableDebugMode) {
            std::cout << "(AR) لا توجد دالة رئيسية - استخدام وضع التنفيذ القديم"
                     << " / (EN) No main function - using legacy execution mode"
                     << std::endl;
        }
        
        return execute(program);
    }
}
```

```cpp
struct ExecutionResult {
    bool success;              // نجح التنفيذ أم لا
    Data::Value result;        // النتيجة (قيمة الإرجاع)
    std::string errorMessage;  // رسالة الخطأ (إن وُجد)
    int exitCode;              // ← جديد: كود الخروج
    
    // البناء الافتراضي
    ExecutionResult(bool s = true, 
                   Data::Value r = Data::Value(), 
                   std::string e = "")
        : success(s), result(r), errorMessage(e), exitCode(0) {}
};
```
