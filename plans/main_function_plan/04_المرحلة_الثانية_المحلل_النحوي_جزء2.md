### المهمة 2.5: منع الكود خارج الدوال
**الملف:** `src/parser/parser_core_impl.cpp`

#### هدف المهمة
عندما يكون البرنامج يتطلب دالة رئيسية، يجب منع تنفيذ أوامر خارج الدوال.

**المسموح:**
- تعريف دوال
- تعريف أصناف
- تصريحات متغيرات (ثوابت عامة)
- استيراد مكتبات (import)
- تصدير (export)

**غير المسموح:**
- جمل تنفيذية (if, while, for, اطبع، إلخ)

#### التعديل على parseStatement

```cpp
StmtPtr ParserCore::parseStatement() {
    // حفظ الموقع للرسائل
    Token current = peek();
    
    // إذا كنا في وضع "الدالة الرئيسية مطلوبة"
    if (!allowTopLevelCode_) {
        // التحقق من أن الجملة تصريح وليست أمر تنفيذي
        if (isExecutableStatement(current)) {
            error(current.getPosition(),
                "(AR) خطأ: لا يمكن كتابة أوامر تنفيذية خارج الدالة الرئيسية\n"
                "يجب وضع جميع الأوامر داخل:\n"
                "  دالة رئيسية()\n"
                "      // الأوامر هنا\n"
                "  نهاية\n\n"
                "(EN) Error: Cannot write executable statements outside main function\n"
                "All statements must be inside:\n"
                "  function main()\n"
                "      // statements here\n"
                "  end");
        }
    }
    
    // تحليل الجملة بشكل طبيعي
    // ...
}
```

```cpp
// في parser_core.h
class ParserCore {
private:
    /**
     * @brief (AR) يتحقق إذا كان الرمز يمثل جملة تنفيذية
     * @brief (EN) Checks if token represents executable statement
     */
    bool isExecutableStatement(const Token& token) const;
};

// في parser_core_impl.cpp
bool ParserCore::isExecutableStatement(const Token& token) const {
    TokenType type = token.getType();
    
    // الجمل التنفيذية (غير مسموحة خارج الدوال)
    return (type == TT::KEYWORD_IF ||
            type == TT::KEYWORD_WHILE ||
            type == TT::KEYWORD_FOR ||
            type == TT::KEYWORD_RETURN ||
            type == TT::KEYWORD_BREAK ||
            type == TT::KEYWORD_CONTINUE ||
            type == TT::KEYWORD_PRINT ||      // إذا كان اطبع كلمة مفتاحية
            type == TT::IDENTIFIER ||         // استدعاء دالة
            type == TT::KEYWORD_TRY ||
            type == TT::KEYWORD_THROW);
}
```

### المهمة 2.6: تحديث parseProgram
**الملف:** `src/parser/parser_core_impl.cpp`

#### إضافة التحقق النهائي من وجود الدالة الرئيسية

```cpp
std::vector<std::unique_ptr<AST::Statement>> ParserCore::parseProgram() {
    std::vector<std::unique_ptr<AST::Statement>> statements;
    
    // تحليل جميع الجمل حتى EOF
    while (!isAtEnd()) {
        try {
            // تخطي الأسطر الفارغة
            if (match(TT::NEWLINE) || match(TT::SEMICOLON)) {
                continue;
            }
            
            // تحليل جملة
            auto stmt = parseDeclaration();
            if (stmt) {
                statements.push_back(std::move(stmt));
            }
        }
        catch (const ParseError& e) {
            // معالجة الخطأ والاستمرار
            synchronize();
        }
    }
    
    // ← جديد: التحقق من وجود دالة رئيسية (إذا كان مطلوباً)
    if (!allowTopLevelCode_ && !hasMainFunction_) {
        error(Token(TT::END_OF_FILE, "", {0, 0}),
            "(AR) خطأ: البرنامج يجب أن يحتوي على دالة رئيسية\n"
            "أضف:\n"
            "  دالة رئيسية()\n"
            "      // الأوامر هنا\n"
            "  نهاية\n\n"
            "(EN) Error: Program must contain a main function\n"
            "Add:\n"
            "  function main()\n"
            "      // statements here\n"
            "  end");
    }
    
    return statements;
}
```

### المهمة 2.7: خيار التفعيل/التعطيل
**الملف:** `include/parser/parser_core.h`

#### إضافة طريقة للتحكم في السلوك

```cpp
class ParserCore {
public:
    /**
     * @brief (AR) تفعيل/تعطيل وضع "الدالة الرئيسية المطلوبة"
     * @brief (EN) Enable/disable "main function required" mode
     * 
     * @param required إذا كان true، يُطلب وجود دالة رئيسية ولا يُسمح بكود خارجها
     *                 If true, main function is required and no top-level code allowed
     * 
     * @note في الوضع القديم (legacy): required = false
     *       In legacy mode: required = false
     */
    void setMainFunctionRequired(bool required) {
        allowTopLevelCode_ = !required;
    }
    
    /**
     * @brief (AR) التحقق من وجود دالة رئيسية في البرنامج
     * @brief (EN) Check if main function exists in program
     */
    bool hasMainFunction() const {
        return hasMainFunction_;
    }
};
```

## 🔧 كود الاختبار الآلي

**الملف:** `tests/parser_tests/test_main_function.cpp`

```cpp
#include "../../include/parser/parser_core.h"
#include "../../include/lexer/lexer_core.h"
#include "test_framework.h"
#include <iostream>
#include <cassert>

using namespace Sad::Parser;
using namespace Sad::Lexer;
using namespace Sad::AST;

void testSimpleMainFunction() {
    std::cout << "Testing simple main function..." << std::endl;
    
    std::string code = 
        "دالة رئيسية()\n"
        "    اطبع(\"مرحباً\")\n"
        "نهاية";
    
    LexerCore lexer(code);
    auto tokens = lexer.tokenize();
    
    ParserCore parser(tokens);
    parser.setMainFunctionRequired(true);
    
    auto ast = parser.parseProgram();
    
    // التحقق من وجود دالة رئيسية
    assert(parser.hasMainFunction());
    assert(ast.size() == 1);
    
    // التحقق من أنها FunctionDecl
    auto* funcDecl = dynamic_cast<FunctionDecl*>(ast[0].get());
    assert(funcDecl != nullptr);
    
    // التحقق من أنها دالة رئيسية
    assert(funcDecl->isMainFunction());
    assert(funcDecl->getName() == "رئيسية");
    
    std::cout << "  ✓ Simple main function parsed correctly" << std::endl;
}

void testMainFunctionWithReturnType() {
    std::cout << "Testing main function with return type..." << std::endl;
    
    std::string code = 
        "دالة صحيح رئيسية()\n"
        "    إرجاع 0\n"
        "نهاية";
    
    LexerCore lexer(code);
    auto tokens = lexer.tokenize();
    
    ParserCore parser(tokens);
    parser.setMainFunctionRequired(true);
    
    auto ast = parser.parseProgram();
    
    auto* funcDecl = dynamic_cast<FunctionDecl*>(ast[0].get());
    assert(funcDecl != nullptr);
    assert(funcDecl->isMainFunction());
    assert(funcDecl->getReturnType() == Data::DataType::INTEGER);
    
    std::cout << "  ✓ Main with return type parsed correctly" << std::endl;
}

void testDuplicateMainError() {
    std::cout << "Testing duplicate main function error..." << std::endl;
    
    std::string code = 
        "دالة رئيسية()\n"
        "نهاية\n"
        "دالة رئيسية()\n"
        "نهاية";
    
    LexerCore lexer(code);
    auto tokens = lexer.tokenize();
    
    ParserCore parser(tokens);
    parser.setMainFunctionRequired(true);
    
    bool errorCaught = false;
    try {
        auto ast = parser.parseProgram();
    } catch (const ParseError& e) {
        errorCaught = true;
        std::cout << "  ✓ Caught expected error: " << e.what() << std::endl;
    }
    
    assert(errorCaught);
    std::cout << "  ✓ Duplicate main error detected" << std::endl;
}

void testNoMainFunctionError() {
    std::cout << "Testing missing main function error..." << std::endl;
    
    std::string code = 
        "دالة جمع(رقم أ، رقم ب)\n"
        "    إرجاع أ + ب\n"
        "نهاية";
    
    LexerCore lexer(code);
    auto tokens = lexer.tokenize();
    
    ParserCore parser(tokens);
    parser.setMainFunctionRequired(true);
    
    bool errorCaught = false;

    } catch (const ParseError& e) {
        errorCaught = true;
        std::cout << "  ✓ Caught expected error: " << e.what() << std::endl;
    }
    
    assert(errorCaught);
    std::cout << "  ✓ Missing main error detected" << std::endl;
}

void testTopLevelCodeError() {
    std::cout << "Testing top-level code error..." << std::endl;
    
    std::string code = 
        "اطبع(\"خطأ\")\n"  // كود خارج الدالة
        "دالة رئيسية()\n"
        "    اطبع(\"صحيح\")\n"
        "نهاية";
    
    LexerCore lexer(code);
    auto tokens = lexer.tokenize();
    
    ParserCore parser(tokens);
    parser.setMainFunctionRequired(true);
    
    bool errorCaught = false;
    try {
        auto ast = parser.parseProgram();
    } catch (const ParseError& e) {
        errorCaught = true;
        std::cout << "  ✓ Caught expected error: " << e.what() << std::endl;
    }
    
    assert(errorCaught);
    std::cout << "  ✓ Top-level code error detected" << std::endl;
}

void testCompleteProgram() {
    std::cout << "Testing complete program with main..." << std::endl;
    
    std::string code = 
        "دالة رقم مربع(رقم س)\n"
        "    إرجاع س * س\n"
        "نهاية\n"
        "\n"
        "دالة صحيح رئيسية()\n"
        "    رقم ن = مربع(5)\n"
        "    اطبع(ن)\n"
        "    إرجاع 0\n"
        "نهاية";
    
    LexerCore lexer(code);
    auto tokens = lexer.tokenize();
    
    ParserCore parser(tokens);
    parser.setMainFunctionRequired(true);
    
    auto ast = parser.parseProgram();
    
    assert(parser.hasMainFunction());
    assert(ast.size() == 2);  // دالتان
    
    // الدالة الأولى ليست رئيسية
    auto* func1 = dynamic_cast<FunctionDecl*>(ast[0].get());
    assert(func1 != nullptr);
    assert(!func1->isMainFunction());
    
    // الدالة الثانية رئيسية
    auto* func2 = dynamic_cast<FunctionDecl*>(ast[1].get());
    assert(func2 != nullptr);
    assert(func2->isMainFunction());
    
    std::cout << "  ✓ Complete program parsed correctly" << std::endl;
}
```
