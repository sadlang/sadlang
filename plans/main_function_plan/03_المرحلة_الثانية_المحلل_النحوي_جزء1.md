#### إضافة دالة التحقق من التوقيع

```cpp
void FunctionDecl::validateMainFunctionSignature() {
    // التحقق من اسم الدالة
    if (name_ != "رئيسية" && name_ != "main") {
        throw ParseError(
            position_,
            "اسم الدالة الرئيسية يجب أن يكون 'رئيسية' أو 'main'",
            "Main function name must be 'رئيسية' or 'main'"
        );
    }
    
    // التحقق من نوع الإرجاع
    // يجب أن يكون صحيح (int) أو عدم (void) أو فراغ
    if (returnType_ != Data::DataType::INTEGER &&
        returnType_ != Data::DataType::VOID &&
        returnType_ != Data::DataType::UNKNOWN) {
        throw ParseError(
            position_,
            "نوع إرجاع الدالة الرئيسية يجب أن يكون 'صحيح' أو 'عدم' أو 'فراغ'",
            "Main function return type must be 'int', 'void', or unspecified"
        );
    }
    
    // التحقق من المعاملات
    // يجب أن تكون إما:
    // 1. بدون معاملات: ()
    // 2. مع معامل واحد: (نص[] الوسائط) أو (string[] args)
    if (parameters_.size() > 1) {
        throw ParseError(
            position_,
            "الدالة الرئيسية يجب أن تكون بدون معاملات أو معامل واحد (نص[])",
            "Main function must have no parameters or one parameter (string[])"
        );
    }
    
    if (parameters_.size() == 1) {
        const auto& param = parameters_[0];
        
        // التحقق من أن المعامل من نوع مصفوفة نصوص
        if (param.type != Data::DataType::ARRAY) {
            throw ParseError(
                position_,
                "معامل الدالة الرئيسية يجب أن يكون من نوع مصفوفة نصوص (نص[])",
                "Main function parameter must be string array (string[])"
            );
        }
        
        // يمكن إضافة تحقق إضافي للتأكد من أن عناصر المصفوفة نصوص
    }
}
```

### المهمة 2.3: تعديل parseFunctionDecl
**الملف:** `src/parser/parser_core_impl.cpp`

#### التعديل على دالة parseFunctionDecl

```cpp
StmtPtr ParserCore::parseFunctionDecl(ExprList decorators) {
    // Spec: docs/language_spec/rules/02_functions.md
    // function_decl ::= 'دالة' [type] IDENTIFIER '(' [param_list] ')' block
    
    Data::DataType returnType = Data::DataType::UNKNOWN;
    
    // Check for optional return type before function name
    if (check(TT::TYPE_INTEGER) || check(TT::TYPE_DOUBLE) || 
        check(TT::TYPE_STRING) || check(TT::TYPE_BOOLEAN) ||
        check(TT::TYPE_ARRAY) || check(TT::TYPE_MAP)) {
        returnType = parseType();
    }
    
    // Expect function name
    Token name = consume(TT::IDENTIFIER, 
        "(AR) خطأ نحوي: بعد كلمة 'دالة' (أو بعد نوع الإرجاع) يجب أن يأتي اسم الدالة.\n"
        "مثال: دالة جمع(...) أو دالة رقم مربع(...)\n"
        "(EN) Syntax error: After 'function' keyword (or return type) expected function name.\n"
        "Example: function sum(...) or function int square(...)");
    
    // ← جديد: التحقق إذا كانت دالة رئيسية
    bool isMainFunction = false;
    if (match(TT::KEYWORD_MAIN)) {
        isMainFunction = true;
        
        // إعادة تعيين اسم الرمز
        name = Token(TT::KEYWORD_MAIN, "رئيسية", name.getPosition());
        
        // التحقق من عدم وجود دالة رئيسية أخرى
        if (hasMainFunction_) {
            error(name.getPosition(),
                "(AR) خطأ: لا يمكن تعريف أكثر من دالة رئيسية واحدة في البرنامج\n"
                "(EN) Error: Cannot define more than one main function in program");
        }
        hasMainFunction_ = true;
    }
    
    // Parse parameter list
    consume(TT::PAREN_LEFT, "...");
    auto paramObjs = parseTypedParameterList();
    consume(TT::PAREN_RIGHT, "...");
    
    // Optional return type after parameters
    if (match(TT::COLON)) {
        returnType = parseType();
    }
    
    // Parse function body
    auto body = parseBlockStmt();
    
    // إنشاء عقدة الدالة مع علامة isMainFunction
    if (!decorators.empty()) {
        return std::make_unique<FunctionDecl>(
            name.getValue(),
            std::move(paramObjs),
            returnType,
            std::move(body),
            std::move(decorators),
            false,  // isExported
            isMainFunction,  // ← جديد
            name.getPosition()
        );
    }
    
    return std::make_unique<FunctionDecl>(
        name.getValue(),
        std::move(paramObjs),
        returnType,
        std::move(body),
        isMainFunction,  // ← جديد
        name.getPosition()
    );
}
```

```cpp
class ParserCore {
private:
    // المتغيرات الموجودة...
    std::vector<Token> tokens_;
    size_t current_;
    bool hasErrors_;
    
    // ← جديد: تتبع الدالة الرئيسية
    bool hasMainFunction_;      // هل تم تعريف دالة رئيسية؟
    bool allowTopLevelCode_;    // هل نسمح بكود خارج الدوال؟
    
public:
    ParserCore();
    explicit ParserCore(const std::vector<Token>& tokens);
    
    // ← جديد: دوال للتحقق
    bool hasMainFunction() const { return hasMainFunction_; }
    void requireMainFunction(bool require) { 
        allowTopLevelCode_ = !require; 
    }
    
    // ... باقي الدوال
};
```

```cpp
// في parseFunctionDecl
Token name = consume(TT::IDENTIFIER, "...");

// التحقق المباشر للكلمة المفتاحية KEYWORD_MAIN
if (check(TT::KEYWORD_MAIN)) {
    // هذه دالة رئيسية
    isMainFunction = true;
    advance();  // تجاوز KEYWORD_MAIN
} else {
    // دالة عادية - تحقق من أن الاسم ليس "رئيسية"
    if (name.getValue() == "رئيسية" || name.getValue() == "main") {
        error(name.getPosition(),
            "(AR) خطأ: لا يمكن استخدام 'رئيسية' كاسم دالة عادية\n"
            "(EN) Error: Cannot use 'main' as regular function name");
    }
}
```
