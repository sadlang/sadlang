```cpp
int executeProgram(const std::string& filename, const std::string& code,
                  int argc, char* argv[]) {  // ← جديد: إضافة argc/argv
    DEBUG_PRINT("بدء تنفيذ البرنامج: " + filename);
    
    // تنظيف الأخطاء السابقة
    Sad::Errors::ErrorManager::getInstance().clear();
    
    try {
        std::cout << "========================================" << std::endl;
        std::cout << "تنفيذ البرنامج / Executing Program" << std::endl;
        std::cout << "الملف / File: " << filename << std::endl;
        std::cout << "========================================" << std::endl;
        std::cout << std::endl;
        
        // الخطوة 1: التحليل المعجمي
        std::cout << "[1/4] التحليل المعجمي / Lexical Analysis..." << std::endl;
        Sad::Lexer::LexerCore lexer(code);
        auto tokens = lexer.tokenize();
        
        if (Sad::Errors::ErrorManager::getInstance().hasErrors()) {
            Sad::Errors::ErrorManager::getInstance().printErrors();
            return 1;
        }
        
        std::cout << "      تم تحليل " << tokens.size() 
                  << " رمز / " << tokens.size() << " tokens analyzed" << std::endl;
        std::cout << std::endl;
        
        // الخطوة 2: التحليل النحوي
        std::cout << "[2/4] التحليل النحوي / Syntax Analysis..." << std::endl;
        
        try {
            Sad::Parser::ParserCore parser(tokens);
            
            // ← جديد: تفعيل وضع "الدالة الرئيسية مطلوبة"
            // يمكن جعل هذا قابل للتكوين عبر خيار في سطر الأوامر
            bool requireMainFunction = true;  // أو قراءة من الإعدادات
            
            if (requireMainFunction) {
                parser.setMainFunctionRequired(true);
                std::cout << "      وضع الدالة الرئيسية: مطلوبة" << std::endl;
                std::cout << "      Main function mode: required" << std::endl;
            }
            
            auto ast = parser.parseProgram();
            
            if (Sad::Errors::ErrorManager::getInstance().hasErrors()) {
                Sad::Errors::ErrorManager::getInstance().printErrors();
                return 1;
            }
            
            std::cout << "      تم تحليل " << ast.size() 
                      << " تصريح / " << ast.size() << " declarations parsed" << std::endl;
            
            // ← جديد: التحقق من وجود دالة رئيسية
            if (requireMainFunction && !parser.hasMainFunction()) {
                std::cerr << std::endl;
                std::cerr << "========================================" << std::endl;
                std::cerr << "❌ خطأ: لا توجد دالة رئيسية" << std::endl;
                std::cerr << "❌ Error: No main function found" << std::endl;
                std::cerr << "========================================" << std::endl;
                std::cerr << std::endl;
                return 1;
            }
            
            std::cout << std::endl;
            
            // الخطوة 3: إعداد المفسر
            std::cout << "[3/4] إعداد المفسر / Interpreter Setup..." << std::endl;
            Sad::Interpreter::Interpreter interpreter;
            std::cout << "      تم إنشاء المفسر" << std::endl;
            std::cout << "      Interpreter created" << std::endl;
            std::cout << std::endl;
            
            // الخطوة 4: التنفيذ
            std::cout << "[4/4] التنفيذ / Execution..." << std::endl;
            std::cout << "========================================" << std::endl;
            std::cout << std::endl;
            
            // ← جديد: استخدام executeAuto بدلاً من execute
            auto result = interpreter.executeAuto(ast, argc, argv);
            
            // فحص النتيجة
            if (!result.success || Sad::Errors::ErrorManager::getInstance().hasErrors()) {
                std::cout << std::endl;
                std::cout << "========================================" << std::endl;
                std::cout << "❌ فشل التنفيذ / Execution Failed" << std::endl;
                std::cout << "========================================" << std::endl;
                
                if (!result.errorMessage.empty()) {
                    std::cerr << "الخطأ / Error: " << result.errorMessage << std::endl;
                }
                
                if (Sad::Errors::ErrorManager::getInstance().hasErrors()) {
                    Sad::Errors::ErrorManager::getInstance().printErrors();
                }
                
                // ← جديد: إرجاع كود الخروج من المفسر
                return (result.exitCode != 0) ? result.exitCode : 1;
            }
            
            // ← جديد: إرجاع كود الخروج من الدالة الرئيسية
            return result.exitCode;
            
        } catch (const Sad::Parser::ParseError& parseEx) {
            std::cerr << std::endl;
            std::cerr << "========================================" << std::endl;
            std::cerr << "❌ خطأ في التحليل النحوي" << std::endl;
            std::cerr << "❌ Syntax Error" << std::endl;
            std::cerr << "========================================" << std::endl;
            std::cerr << "Message: " << parseEx.what() << std::endl;
            std::cerr << "========================================" << std::endl;
            std::cerr << std::endl;
            return 1;
        } catch (...) {
            std::cerr << std::endl;
            std::cerr << "========================================" << std::endl;
            std::cerr << "❌ خطأ غير معروف في التحليل النحوي" << std::endl;
            std::cerr << "❌ Unknown parsing error" << std::endl;
            std::cerr << "========================================" << std::endl;
            std::cerr << std::endl;
            return 1;
        }
        
    }
    catch (const std::exception& e) {
        std::cerr << std::endl;
        std::cerr << "========================================" << std::endl;
        std::cerr << "❌ خطأ حرج / Critical Error" << std::endl;
        std::cerr << "========================================" << std::endl;
        std::cerr << e.what() << std::endl;
        std::cerr << "========================================" << std::endl;
        std::cerr << std::endl;
        return 1;
    }
}
```

#### تعديل دالة main

```cpp
int main(int argc, char* argv[]) {
    // التحقق من المعاملات
    if (argc < 2) {
        showHelp();
        return 1;
    }
    
    // معالجة الخيارات
    std::string filename;
    
    for (int i = 1; i < argc; i++) {
        std::string arg = argv[i];
        
        if (arg == "--version" || arg == "-v") {
            showVersion();
            return 0;
        }
        else if (arg == "--help" || arg == "-h") {
            showHelp();
            return 0;
        }
        else if (arg[0] != '-') {
            // هذا اسم الملف
            filename = arg;
            break;  // باقي المعاملات للبرنامج
        }
    }
    
    if (filename.empty()) {
        std::cerr << "❌ خطأ: لم يتم تحديد ملف للتنفيذ" << std::endl;
        std::cerr << "❌ Error: No file specified for execution" << std::endl;
        showHelp();
        return 1;
    }
    
    // التحقق من وجود الملف
    if (!fileExists(filename)) {
        std::cerr << "========================================" << std::endl;
        std::cerr << "❌ خطأ: الملف غير موجود" << std::endl;
        std::cerr << "❌ Error: File not found" << std::endl;
        std::cerr << "========================================" << std::endl;
        std::cerr << "الملف / File: " << filename << std::endl;
        std::cerr << "========================================" << std::endl;
        return 1;
    }
    
    // قراءة البرنامج
    std::string code;
    if (!readProgramFile(filename, code)) {
        return 1;
    }
    
    // ← جديد: تمرير argc/argv لـ executeProgram
    // تنفيذ البرنامج مع تمرير الوسائط
    try {
        int exitCode = executeProgram(filename, code, argc, argv);
        
        if (exitCode == 0) {
            std::cout << "========================================" << std::endl;
            std::cout << "✓ تم التنفيذ بنجاح / Execution Successful" << std::endl;
            std::cout << "========================================" << std::endl;
        } else {
            std::cout << "========================================" << std::endl;
            std::cout << "⚠ انتهى التنفيذ بكود خروج: " << exitCode << std::endl;
            std::cout << "⚠ Execution ended with exit code: " << exitCode << std::endl;
            std::cout << "========================================" << std::endl;
        }
        
        return exitCode;
    }
    catch (const std::exception& e) {
        std::cerr << std::endl;
        std::cerr << "========================================" << std::endl;
        std::cerr << "❌ خطأ حرج في التنفيذ / Critical Execution Error" << std::endl;
        std::cerr << "========================================" << std::endl;
        std::cerr << e.what() << std::endl;
        std::cerr << "========================================" << std::endl;
        std::cerr << std::endl;
        return 1;
    }
}
```

```cpp
            showVersion();

            showHelp();
            return 0;
        }
        else if (arg == "--legacy-mode") {  // ← جديد
            legacyMode = true;
        }
        else if (arg == "--debug") {  // ← جديد
            debugMode = true;
        }
        else if (arg[0] != '-') {
            filename = arg;
            break;
        }
    }
    
    // ... باقي الكود
    
    // تمرير الخيارات لـ executeProgram

                                  legacyMode, debugMode);
```
