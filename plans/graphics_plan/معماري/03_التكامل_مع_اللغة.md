## 🔧 تطبيق FFI

### 1. Handle System (نظام المقابض)

```cpp
// C++ side - handle_system.h

template<typename T>
class HandleManager {
private:
    std::unordered_map<uint64_t, T*> handles;
    uint64_t nextId = 1;
    std::mutex mutex;
    
public:
    uint64_t create(T* object) {
        std::lock_guard<std::mutex> lock(mutex);
        uint64_t id = nextId++;
        handles[id] = object;
        return id;
    }
    
    T* get(uint64_t id) {
        std::lock_guard<std::mutex> lock(mutex);
        auto it = handles.find(id);
        if (it != handles.end()) {
            return it->second;
        }
        return nullptr;
    }
    
    void destroy(uint64_t id) {
        std::lock_guard<std::mutex> lock(mutex);
        auto it = handles.find(id);
        if (it != handles.end()) {
            delete it->second;
            handles.erase(it);
        }
    }
};

// مديري المقابض
HandleManager<Window> windowHandles;
HandleManager<Renderer> rendererHandles;
HandleManager<Texture> textureHandles;
HandleManager<Font> fontHandles;
// ... إلخ
```

```cpp
extern "C" {
    
// ═══════════════════════════════════════════
// Window Functions
// ═══════════════════════════════════════════

SAD_EXPORT uint64_t sad_window_create(
    const char* title, 
    int width, 
    int height
) {
    try {
        Window* window = new Window(title, width, height);
        return windowHandles.create(window);
    } catch (const std::exception& e) {
        sad_set_error(e.what());
        return 0;  // invalid handle
    }
}

SAD_EXPORT void sad_window_destroy(uint64_t handle) {
    windowHandles.destroy(handle);
}

SAD_EXPORT void sad_window_set_title(uint64_t handle, const char* title) {
    Window* window = windowHandles.get(handle);
    if (window) {
        window->setTitle(title);
    } else {
        sad_set_error("Invalid window handle");
    }
}

SAD_EXPORT int sad_window_get_width(uint64_t handle) {
    Window* window = windowHandles.get(handle);
    if (window) {
        return window->getWidth();
    }
    sad_set_error("Invalid window handle");
    return 0;
}

SAD_EXPORT bool sad_window_is_open(uint64_t handle) {
    Window* window = windowHandles.get(handle);
    if (window) {
        return window->isOpen();
    }
    return false;
}

// ═══════════════════════════════════════════
// Renderer Functions
// ═══════════════════════════════════════════

SAD_EXPORT uint64_t sad_renderer_create(uint64_t window_handle) {
    Window* window = windowHandles.get(window_handle);
    if (!window) {
        sad_set_error("Invalid window handle");
        return 0;
    }
    
    try {
        Renderer* renderer = new Renderer(window);
        return rendererHandles.create(renderer);
    } catch (const std::exception& e) {
        sad_set_error(e.what());
        return 0;
    }
}

SAD_EXPORT void sad_renderer_clear(uint64_t handle, float r, float g, float b, float a) {
    Renderer* renderer = rendererHandles.get(handle);
    if (renderer) {
        renderer->clear(Color{r, g, b, a});
    } else {
        sad_set_error("Invalid renderer handle");
    }
}

SAD_EXPORT void sad_renderer_draw_circle(
    uint64_t handle, 
    float x, float y, 
    float radius, 
    bool filled
) {
    Renderer* renderer = rendererHandles.get(handle);
    if (renderer) {
        renderer->drawCircle(x, y, radius, filled);
    } else {
        sad_set_error("Invalid renderer handle");
    }
}
```

## 🎨 High-Level Wrapper (التغليف عالي المستوى)

### نافذة (Window)

```
# نافذة.ص

استورد __داخلي.رسومات.نافذة_FFI

صنف نافذة {
    """
    نافذة رسومية
    
    مثال:
        نافذة = نافذة.جديدة("تطبيقي"، 800، 600)
        نافذة.اضبط_العنوان("عنوان جديد")
        بينما نافذة.مفتوحة() {
            # ...
        }
        نافذة.أغلق()
    """
    
    # الخصائص الخاصة
    _مقبض: رقم
    _العنوان: نص
    _العرض: رقم
    _الارتفاع: رقم
    _مفتوحة: منطقي
    _callbacks: قاموس
    
    # ═══════════════════════════════════════════
    # المنشئ
    # ═══════════════════════════════════════════
    
    دالة منشئ(العنوان: نص، العرض: رقم، الارتفاع: رقم) {
        """
        ينشئ نافذة جديدة
        
        المعاملات:
            العنوان: عنوان النافذة
            العرض: عرض النافذة بالبكسل
            الارتفاع: ارتفاع النافذة بالبكسل
            
        الأخطاء:
            يرمي خطأ إذا فشل إنشاء النافذة
        """
        
        # التحقق من الصحة
        إذا العرض <= 0 أو الارتفاع <= 0 {
            ارم خطأ("حجم النافذة يجب أن يكون موجباً")
        }
        
        # إنشاء النافذة
        هذا._مقبض = نافذة_FFI.أنشئ(العنوان، العرض، الارتفاع)
        
        إذا هذا._مقبض == 0 {
            ارم خطأ("فشل إنشاء النافذة")
        }
        
        # حفظ الخصائص
        هذا._العنوان = العنوان
        هذا._العرض = العرض
        هذا._الارتفاع = الارتفاع
        هذا._مفتوحة = صحيح
        هذا._callbacks = {}
        
        # تسجيل للتنظيف التلقائي
        __سجل_مدمر(هذا، هذا.تدمير)
    }
    
    # ═══════════════════════════════════════════
    # دوال ثابتة
    # ═══════════════════════════════════════════
    
    دالة ثابتة جديدة(العنوان: نص، العرض: رقم، الارتفاع: رقم) -> نافذة {
        """
        طريقة بديلة لإنشاء نافذة
        
        مثال:
            نافذة = نافذة.جديدة("تطبيقي"، 800، 600)
        """
        أرجع نافذة(العنوان، العرض، الارتفاع)
    }
    
    # ═══════════════════════════════════════════
    # الخصائص (Getters/Setters)
    # ═══════════════════════════════════════════
    
    دالة العنوان() -> نص {
        """يُرجع عنوان النافذة"""
        أرجع هذا._العنوان
    }
    
    دالة اضبط_العنوان(العنوان: نص) {
        """
        يضبط عنوان النافذة
        
        المعاملات:
            العنوان: العنوان الجديد
        """
        نافذة_FFI.اضبط_العنوان(هذا._مقبض، العنوان)
        هذا._العنوان = العنوان
    }
    
    دالة العرض() -> رقم {
        """يُرجع عرض النافذة بالبكسل"""
        أرجع هذا._العرض
    }
    
    دالة الارتفاع() -> رقم {
        """يُرجع ارتفاع النافذة بالبكسل"""
        أرجع هذا._الارتفاع
    }
    
    دالة الحجم() -> (رقم، رقم) {
        """يُرجع (العرض، الارتفاع)"""
        أرجع (هذا._العرض، هذا._الارتفاع)
    }
    
    دالة مفتوحة() -> منطقي {
        """يتحقق إذا كانت النافذة مفتوحة"""
        إذا !هذا._مفتوحة {
            أرجع خطأ
        }
        
        # فحص من C++
        هذا._مفتوحة = نافذة_FFI.مفتوح(هذا._مقبض)
        أرجع هذا._مفتوحة
    }
    
    # ═══════════════════════════════════════════
    # دوال الأحداث (Events)
    # ═══════════════════════════════════════════
    
    دالة عند_الإغلاق(callback: دالة) {
        """
        يسجل callback يُستدعى عند إغلاق النافذة
        
        المعاملات:
            callback: دالة بدون معاملات
            
        مثال:
            نافذة.عند_الإغلاق(دالة() {
                اطبع("النافذة تُغلق")
            })
        """
        هذا._callbacks["إغلاق"] = callback
    }
    
    دالة عند_تغيير_الحجم(callback: دالة) {
        """
        يسجل callback يُستدعى عند تغيير حجم النافذة
        
        المعاملات:
            callback: دالة(عرض: رقم، ارتفاع: رقم)
        """
        هذا._callbacks["تغيير_حجم"] = callback
    }
    
    # ═══════════════════════════════════════════
    # دوال عامة
    # ═══════════════════════════════════════════
    
    دالة أظهر() {
        """يُظهر النافذة"""
        # ... implementation
    }
    
    دالة أخفي() {
        """يخفي النافذة"""
        # ... implementation
    }
    
    دالة أغلق() {
        """يُغلق النافذة"""
        إذا هذا._مفتوحة {
            # استدعاء callback الإغلاق
            إذا "إغلاق" في هذا._callbacks {
                هذا._callbacks["إغلاق"]()
            }
            
            هذا._مفتوحة = خطأ
        }
    }
    
    # ═══════════════════════════════════════════
    # المدمر
    # ═══════════════════════════════════════════
    
    دالة تدمير() {
        """يُنظف الموارد (يُستدعى تلقائياً)"""
        إذا هذا._مقبض != 0 {
            نافذة_FFI.دمر(هذا._مقبض)
            هذا._مقبض = 0
        }
    }
}

# الصادرات
# ═══════════════════════════════════════════

صدّر نافذة
```

#### 1. Reference Counting (للموارد الكبيرة)

```cpp
// C++ side - resource.h

class Resource {
protected:
    std::atomic<int> refCount{1};
    
public:
    void addRef() {
        refCount++;
    }
    
    void release() {
        if (--refCount == 0) {
            delete this;
        }
    }
    
    int getRefCount() const {
        return refCount;
    }
    
    virtual ~Resource() = default;
};

// Texture example
class Texture : public Resource {
    // ...
};
```

## 🔒 Thread Safety

### استراتيجيات الأمان في التزامن

```cpp
// C++ side

class ThreadSafeRenderer {
private:
    std::mutex commandMutex;
    std::vector<RenderCommand> commandQueue;
    
public:
    void addCommand(RenderCommand cmd) {
        std::lock_guard<std::mutex> lock(commandMutex);
        commandQueue.push_back(cmd);
    }
    
    void flush() {
        std::vector<RenderCommand> commands;
        {
            std::lock_guard<std::mutex> lock(commandMutex);
            commands = std::move(commandQueue);
            commandQueue.clear();
        }
        
        // Execute commands (no lock needed)
        for (const auto& cmd : commands) {
            executeCommand(cmd);
        }
    }
};
```

```cpp
enum class ErrorCode {
    None = 0,
    InvalidHandle,
    FileNotFound,
    OutOfMemory,
    InvalidParameter,
    OperationFailed,
    NotImplemented
};

struct Error {
    ErrorCode code;
    std::string message;
    std::string file;
    int line;
};

thread_local Error lastError;

void setError(ErrorCode code, const std::string& message,
              const char* file, int line) {
    lastError = {code, message, file, line};
}

#define SAD_SET_ERROR(code, msg) \
    setError(code, msg, __FILE__, __LINE__)

SAD_EXPORT const char* sad_get_error_message() {
    return lastError.message.c_str();
}

SAD_EXPORT int sad_get_error_code() {
    return static_cast<int>(lastError.code);
}
```

## 📊 Performance Considerations

### Best Practices

#### 1. Minimize FFI Calls

```
# ❌ سيء - استدعاءات كثيرة
لـ i من 0 إلى 1000 {
    راسم.ارسم_نقطة(i، i)  # 1000 استدعاء FFI
}

# ✅ جيد - استدعاء واحد
نقاط = []
لـ i من 0 إلى 1000 {
    نقاط.أضف((i، i))
}
راسم.ارسم_نقاط(نقاط)  # استدعاء FFI واحد
```

#### 2. Batch Operations

```cpp
// C++ side

SAD_EXPORT void sad_renderer_draw_points_batch(
    uint64_t handle,
```
