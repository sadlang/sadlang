- **GitHub Actions** - CI/CD

## 💡 أمثلة الاستخدام

### مثال 1: نافذة ورسم بسيط

```
استورد رسومات.*

# إنشاء نافذة
نافذة = نافذة.جديدة("تطبيقي"، 800، 600)

# إنشاء راسم
راسم = راسم_2د.جديد(نافذة)

# الحلقة الرئيسية
بينما نافذة.مفتوحة() {
    # مسح الشاشة
    راسم.نظف(لون.أبيض)
    
    # رسم دائرة حمراء
    راسم.اضبط_لون(لون.أحمر)
    راسم.ارسم_دائرة(400، 300، 50، ممتلئة: صحيح)
    
    # عرض
    راسم.اعرض()
}
```

### مثال 2: واجهة مستخدم

```
استورد واجهة.*

نافذة = نافذة.جديدة("تطبيق UI"، 600، 400)

تخطيط = تخطيط_صندوق.عمودي([
    تسمية.جديد("أهلاً بك!")،
    حقل_نص.جديد("اكتب شيئاً...")،
    زر.جديد("إرسال"، عند_النقر: دالة() {
        اطبع("تم النقر!")
    })
]، مسافة: 10)

نافذة.اضبط_التخطيط(تخطيط)
نافذة.اعرض()
```

### مثال 3: لعبة بسيطة

```
استورد ألعاب.*

مشهد = مشهد.جديد("لعبتي")

# كائن اللاعب
لاعب = كائن.جديد("اللاعب")
لاعب.أضف_مكون(مكون_رسم_صورة.جديد("لاعب.png"))
لاعب.اضبط_الموقع([100، 100])

# التحكم
دالة تحديث_لاعب(كائن، دلتا) {
    إذا إدخال.مفتاح_مضغوط(مفتاح.سهم_يمين) {
        كائن.حرك(5، 0)
    }
}

لاعب.عند_التحديث(تحديث_لاعب)
مشهد.أضف_كائن(لاعب)

مشهد.شغل()
```
