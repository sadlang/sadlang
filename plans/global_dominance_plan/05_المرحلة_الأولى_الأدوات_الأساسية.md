### 2. REPL - Multi-line Input

```cpp
// repl.cpp
class REPL {
    std::string current_input_;
    int bracket_depth_ = 0;
    
public:
    void run() {
        while (true) {
            std::string prompt = (bracket_depth_ == 0) ? ">>> " : "... ";
            std::string line = readline(prompt.c_str());
            
            if (!line.empty()) {
                add_history(line.c_str());
            }
            
            current_input_ += line + "\n";
            bracket_depth_ += count_brackets(line);
            
            if (bracket_depth_ == 0 && !current_input_.empty()) {
                execute(current_input_);
                current_input_.clear();
            }
        }
    }
    
    void execute(const std::string& code) {
        try {
            // Lexer → Parser → Interpreter
            auto result = interpreter_.eval(code);
            std::cout << result.toString() << std::endl;
        } catch (const std::exception& e) {
            std::cerr << "خطأ: " << e.what() << std::endl;
        }
    }
};
```

### 3. LSP - JSON-RPC Protocol

```cpp
// lsp_server.h
class LSPServer {
    DocumentManager doc_manager_;
    DiagnosticsEngine diagnostics_;
    CompletionProvider completion_;
    
public:
    void start() {
        // Read JSON-RPC messages from stdin
        while (true) {
            auto message = read_message();
            handle_message(message);
        }
    }
    
    void handle_message(const json& message) {
        std::string method = message["method"];
        
        if (method == "textDocument/didOpen") {
            handle_did_open(message["params"]);
        } else if (method == "textDocument/didChange") {
            handle_did_change(message["params"]);
        } else if (method == "textDocument/completion") {
            handle_completion(message["params"]);
        }
        // ... المزيد من الأوامر
    }
    
    void handle_did_change(const json& params) {
        std::string uri = params["textDocument"]["uri"];
        std::string text = params["contentChanges"][0]["text"];
        
        // تحديث المستند
        doc_manager_.update(uri, text);
        
        // تشغيل diagnostics
        auto errors = diagnostics_.check(text);
        send_diagnostics(uri, errors);
    }
};
```

### 4. Build System - Task Runner

```cpp
// task_runner.h
class TaskRunner {
    struct Task {
        std::string name;
        std::string command;
        std::vector<std::string> dependencies;
        bool is_background;
    };
    
    std::unordered_map<std::string, Task> tasks_;
    
public:
    void load_tasks(const std::string& project_file) {
        // قراءة مشروع.ص
        auto config = parse_project_file(project_file);
        tasks_ = config.tasks;
    }
    
    Result<void> run(const std::string& task_name) {
        if (!tasks_.contains(task_name)) {
            return Error("Task not found: " + task_name);
        }
        
        auto& task = tasks_[task_name];
        
        // تشغيل dependencies أولاً
        for (const auto& dep : task.dependencies) {
            auto result = run(dep);
            if (!result.is_ok()) {
                return result;
            }
        }
        
        // تشغيل المهمة
        return execute_command(task.command, task.is_background);
    }
};
```

## 🚨 التحديات المتوقعة

### 1. التحدي: Dependency Hell
**الوصف:** تعارضات معقدة في الإصدارات

**الحل:**
```
1. استخدام SemVer صارم
2. Lock file لضمان reproducibility
3. رسائل خطأ واضحة عند التعارض
4. اقتراح حلول للتعارضات
```

### 2. التحدي: Cross-platform Build
**الوصف:** اختلافات بين Windows/Linux/macOS

**الحل:**
```cpp
// Platform abstraction
namespace platform {
    std::string get_cache_dir() {
        #ifdef _WIN32
            return std::getenv("LOCALAPPDATA") + "\\sad";
        #else
            return std::getenv("HOME") + "/.sad";
        #endif
    }
    
    Result<void> execute_command(const std::string& cmd) {
        #ifdef _WIN32
            return system(("cmd /c " + cmd).c_str());
        #else
            return system(cmd.c_str());
        #endif
    }
}
```

### 3. التحدي: LSP Performance
**الوصف:** بطء في الملفات الكبيرة

**الحل:**
```
1. Incremental parsing (parse only changed parts)
2. Background thread للـ diagnostics
3. Caching لنتائج parsing
4. Debouncing للتغييرات السريعة
```
