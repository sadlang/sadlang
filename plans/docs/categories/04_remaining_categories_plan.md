```s
# AND
نتيجة = 0b1100 & 0b1010  # 0b1000

# OR
نتيجة = 0b1100 | 0b1010  # 0b1110

# XOR
نتيجة = 0b1100 ^ 0b1010  # 0b0110

# NOT
نتيجة = ~0b1100

# Shift
يسار = 0b0001 << 2  # 0b0100
يمين = 0b1000 >> 2  # 0b0010
```
