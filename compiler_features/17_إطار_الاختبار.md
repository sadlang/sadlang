## ١. الوصف

دستور BeeOS (§4) يفرض **TDD إلزامي** بتغطية 90%+ للنواة و80%+ لباقي المكونات. لتحقيق هذا، المترجم يجب أن يدعم إطار اختبار **مدمج في اللغة نفسها** — `اختبر()` كلمة محجوزة.

### الفلسفة

الاختبارات تُكتب **في نفس الملف** بجانب الكود المُختبَر — ليست في مجلدات منفصلة. هذا يُسهّل صيانتها ويضمن تحديثها مع الكود.

---

## ٢. المتطلبات

### ٢.١ اختبارات الوحدة

```
// ملف: نظام_ملفات/محلي/خلية_العسل.ص

بنية خلية_عسل {
    معرّف: ع128,
    نوع: نوع_ملف,
    حجم: ع64,
    // ...
}

دالة أنشئ_خلية(نوع: نوع_ملف) → خلية_عسل {
    أعد خلية_عسل {
        معرّف: ولّد_معرّف(),
        نوع: نوع,
        حجم: 0,
    }
}

// الاختبارات — في نفس الملف!
اختبر("إنشاء خلية ملف") {
    دع خلية = أنشئ_خلية(نوع_ملف::ملف)
    تحقق(خلية.حجم == 0)
    تحقق(خلية.نوع == نوع_ملف::ملف)
    تحقق(خلية.معرّف != 0)
}

اختبر("إنشاء خلية مجلد") {
    دع خلية = أنشئ_خلية(نوع_ملف::مجلد)
    تحقق(خلية.نوع == نوع_ملف::مجلد)
}
```

### ٢.٢ تأكيدات عربية

```
تحقق(شرط)                              // يفشل إذا خطأ
تحقق_مساوٍ(فعلي, متوقع)                 // يفشل إذا مختلفان
تحقق_مختلف(فعلي, غير_متوقع)             // يفشل إذا متساويان
تحقق_خطأ(نتيجة)                        // يفشل إذا نجاح
تحقق_نجاح(نتيجة)                       // يفشل إذا فشل
تحقق_قريب(فعلي, متوقع, هامش)            // مقارنة أعداد حقيقية
تحقق_ينهار(|| { /* كود ينهار */ })      // يفشل إذا لم ينهار
```

### ٢.٣ BDD عربي

```
سيناريو("كتابة ملف جديد في BeeFS") {
    بفرض("نظام ملفات فارغ") {
        دع نظام = أنشئ_نظام_اختبار()
    }
    عندما("يكتب المستخدم ملفاً جديداً") {
        دع نتيجة = نظام.اكتب("سلام.نص", "السلام عليكم")
    }
    فإن("الملف يُحفظ بنجاح") {
        تحقق_نجاح(نتيجة)
    }
    وأن("يمكن قراءة الملف لاحقاً") {
        دع محتوى = نظام.اقرأ("سلام.نص")?
        تحقق_مساوٍ(محتوى, "السلام عليكم")
    }
}
```

### ٢.٤ محاكاة (Mocking)

```
// محاكاة قرص — لاختبار نظام الملفات بدون قرص حقيقي
دع قرص_وهمي = محاكاة_قرص(حجم: 1024 * 1024)    // 1MB
دع شبكة_وهمية = محاكاة_شبكة(عقد: 10)
دع ساعة_وهمية = محاكاة_ساعة()

ساعة_وهمية.تقدّم(ثانية(30))     // تقديم الزمن الوهمي
```

### ٢.٥ تشغيل الاختبارات

```
ص اختبر                          // تشغيل كل الاختبارات
ص اختبر نظام_ملفات/              // اختبارات مجلد محدد
ص اختبر --مرشح="كتابة"           // اختبارات تحتوي كلمة
ص اختبر --موازٍ=4                // 4 خيوط
ص اختبر --تغطية                  // مع تقرير التغطية
```
