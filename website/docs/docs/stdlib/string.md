# مكتبة string

```sad
استورد نصوص من "stdlib/string"

متغير ن = "مرحباً يا عالم"

اطبع(ن.طول())              # 14
اطبع(ن.يحتوي_على("عالم"))  # صحيح
اطبع(ن.ابدأ_بـ("مرحبا"))  # صحيح
اطبع(ن.استبدل("عالم"، "ص")) # "مرحباً يا ص"
اطبع(ن.قسّم(" "))          # ["مرحباً"، "يا"، "عالم"]
اطبع(ن.أزل_المسافات())     # بدون مسافات طرفية
اطبع(ن.أكبر())             # نص بالأحرف الكبيرة
اطبع(ن.أصغر())             # نص بالأحرف الصغيرة
```
