# مكتبة graphics

تعتمد على SDL2 + OpenGL.

```sad
استورد رسوميات من "stdlib/graphics"

متغير نافذة = رسوميات.إنشاء_نافذة("لعبة ص"، 800، 600)
متغير رسام  = نافذة.احصل_على_رسام()

بينما (نافذة.مفتوحة())
  رسام.امسح(رسوميات.أسود)
  رسام.ارسم_دائرة(400، 300، 50، رسوميات.أزرق)
  رسام.ارسم_نص("مرحباً!"، 350، 280، رسوميات.أبيض)
  نافذة.عرض()
  نافذة.معالجة_الأحداث()
نهاية
```
